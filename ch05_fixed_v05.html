03RTL 작성

아래 파일을 경로 그대로 만들고 전체 코드를 입력합니다.

ch05/rtl/fixed_quantizer.v · verilog · 원본 · RTL_SIM 통과 2026-11-18
`timescale 1ns/1ps
module fixed_quantizer #(
    parameter integer IN_WIDTH=32, OUT_WIDTH=8, SHIFT=8, ROUND=1, SATURATE=1
) (
    input wire clk, resetn, valid_i,
    input wire signed [IN_WIDTH-1:0] x_i,
    output reg signed [OUT_WIDTH-1:0] q_o,
    output reg valid_o, overflow_o
);
    wire signed [63:0] wide, rounded, scaled, lower, upper;
    wire ov;
    // (변경) 입력이 초기 0에서 변하지 않아도 조합 계산을 시작하는 연속 할당이다.
    assign wide = $signed(x_i);
    assign lower = -(64'sd1 <<< (OUT_WIDTH-1));
    assign upper = (64'sd1 <<< (OUT_WIDTH-1))-1;
    // 음수는 크기를 반올림한 뒤 부호를 복원하여 tie-away 규칙을 적용한다.
    assign rounded = (ROUND && SHIFT>0) ?
      (wide<0 ? -(((-wide)+(64'sd1 <<< (SHIFT-1))) >>> SHIFT) :
                 (wide+(64'sd1 <<< (SHIFT-1))) >>> SHIFT) : wide >>> SHIFT;
    assign ov = rounded<lower || rounded>upper;
    assign scaled = (SATURATE && rounded<lower) ? lower :
                    (SATURATE && rounded>upper) ? upper : rounded;
    always @(posedge clk) begin
        if(!resetn) begin q_o<=0; valid_o<=0; overflow_o<=0; end
        else begin
            valid_o<=valid_i;
            if(valid_i) begin q_o<=scaled[OUT_WIDTH-1:0]; overflow_o<=ov; end
        end
    end
    // synthesis translate_off
    initial if(IN_WIDTH<2 || IN_WIDTH>48 || OUT_WIDTH<2 || OUT_WIDTH>32 || SHIFT<0 || SHIFT>30)
        $fatal(1,"Unsupported quantizer parameters");
    // synthesis translate_on
endmodule

module tb_fixed_quantizer;
    parameter IW=32,OW=8,S=8;
    reg clk=0,rn=0,vi=0;
    reg signed [IW-1:0] x=0;
    wire signed [OW-1:0] q[0:3];
    wire ov[0:3],v[0:3];
    genvar g;
    generate for(g=0;g<4;g=g+1) begin: modes
        fixed_quantizer #(.IN_WIDTH(IW),.OUT_WIDTH(OW),.SHIFT(S),.ROUND(g/2),.SATURATE(g%2)) dut
          (.clk(clk),.resetn(rn),.valid_i(vi),.x_i(x),.q_o(q[g]),.valid_o(v[g]),.overflow_o(ov[g]));
    end endgenerate
    always #5 clk=~clk;
    integer f,o,n,c=0,r,valid_in,xi,eq[0:3],eo[0:3],ev,i;
    reg signed [OW-1:0] held[0:3];
    reg hv[0:3],ho[0:3];
    string vp,ap;
    initial begin
        if(!$value$plusargs("VECTORS=%s",vp)||!$value$plusargs("ACTUAL=%s",ap)) $fatal(1,"Missing paths");
        f=$fopen(vp,"r"); o=$fopen(ap,"w"); if(!f||!o) $fatal(1,"Open failed");
        @(posedge clk); #1;
        while(!$feof(f)) begin
            n=$fscanf(f,"%d %d %d %d %d %d %d %d %d %d %d %d\n",
              r,valid_in,xi,eq[0],eo[0],eq[1],eo[1],eq[2],eo[2],eq[3],eo[3],ev);
            if(n!=12) $fatal(1,"Bad row");
            @(negedge clk);
            for(i=0;i<4;i=i+1) begin held[i]=q[i]; hv[i]=v[i]; ho[i]=ov[i]; end
            rn=r; vi=valid_in; x=xi; #1;
            for(i=0;i<4;i=i+1) if(q[i]!==held[i]||v[i]!==hv[i]||ov[i]!==ho[i]) $fatal(1,"Async output");
            @(posedge clk); #1; c=c+1;
            for(i=0;i<4;i=i+1) if($signed(q[i])!==eq[i]||ov[i]!==eo[i][0]||v[i]!==ev[0])
                $fatal(1,"Mismatch cycle=%0d mode=%0d",c,i);
            $fdisplay(o,"%0d %0d %0d %0d %0d %0d %0d %0d %0d %0d",c,
              q[0],ov[0],q[1],ov[1],q[2],ov[2],q[3],ov[3],v[0]);
        end
        if(!c) $fatal(1,"Empty");
        $fclose(o); $display("RTL_SIM_PASS FIXED IW=%0d OW=%0d SHIFT=%0d cycles=%0d",IW,OW,S,c); $finish;
    end
    initial begin #10000000; $fatal(1,"Timeout"); end
endmodule
ch05/reference/fixed_reference.py · python · 원본 · RTL_SIM 통과 2026-11-18
import argparse
import random
import sys
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[2]/'ch01/reference'))
from mac_reference import wrap, compare


def quantize(x, width, shift, rounding=1, saturate=1):
    # Python 정수 나눗셈을 사용하여 RTL의 시프트 구현과 독립적으로 계산한다.
    if rounding and shift:
        mag=(abs(x)+(1<<(shift-1)))//(1<<shift)
        y=-mag if x<0 else mag
    else: y=x//(1<<shift)
    lo,hi=-(1<<(width-1)),(1<<(width-1))-1
    ov=int(y<lo or y>hi)
    return (min(hi,max(lo,y)) if saturate else wrap(y,width)),ov


def self_test():
    assert quantize(-3,8,1,0)==(-2,0)
    assert quantize(-3,8,1,1)==(-2,0)
    assert quantize(-1,8,1,0)==(-1,0)
    assert quantize(-1,8,1,1)==(-1,0)
    assert quantize(1,8,1,1)==(1,0)
    assert quantize(256,8,0,1,1)==(127,1)
    assert quantize(256,8,0,1,0)==(0,1)
    print('PYTHON_FIXED_SELF_TEST_PASS')


def generate(iw,ow,shift,out):
    lo,hi=-(1<<(iw-1)),(1<<(iw-1))-1
    half=1<<(max(shift-1,0))
    xs=[lo,hi,0,1,-1]
    for anchor in (0,1,-1,(1<<(ow-1))-1,-(1<<(ow-1))):
        for delta in (-half-1,-half,-half+1,half-1,half,half+1):
            x=anchor*(1<<shift)+delta
            if lo<=x<=hi: xs.append(x)
    rng=random.Random(305)
    xs += [rng.randint(lo,hi) for _ in range(2000)]
    state=[(0,0)]*4
    with (out/'vectors.txt').open('w') as vf,(out/'expected.txt').open('w') as ef:
        for c,x in enumerate(xs,1):
            # TB 초기 Reset 이후 첫 최솟값도 실제 수락시켜 경계 검사를 생략하지 않는다.
            rn=int(c%97!=0); vi=int(c%7!=0)
            ev=rn and vi
            for k in range(4):
                if not rn: state[k]=(0,0)
                elif vi: state[k]=quantize(x,ow,shift,k//2,k%2)
            results=[v for pair in state for v in pair]+[int(ev)]
            vf.write(' '.join(map(str,(rn,vi,x,*results)))+'\n')
            ef.write(' '.join(map(str,(c,*results)))+'\n')
    print(f'PYTHON_VECTORS IN={iw} OUT={ow} SHIFT={shift} cycles={len(xs)} seed=305')


if __name__=='__main__':
    p=argparse.ArgumentParser(); p.add_argument('--in-width',type=int,default=32)
    p.add_argument('--out-width',type=int,default=8); p.add_argument('--shift',type=int,default=8)
    p.add_argument('--out',type=Path); p.add_argument('--actual',type=Path)
    a=p.parse_args(); self_test()
    if a.actual: compare(a.out/'expected.txt',a.actual)
    else: generate(a.in_width,a.out_width,a.shift,a.out)

07코드 읽기 포인트

위치내용
Tie-away 반올림양수는 (x + (1 << (SHIFT-1))) >>> SHIFT, 음수는 크기를 반올림한 뒤 부호를 복원합니다. Python round()의 Banker's rounding과 다르므로 기준 모델도 같은 규칙을 따릅니다.
포화(Saturation)범위 초과 시 lower 또는 upper로 클리핑합니다. SATURATE=0이면 2의 보수 순환이 발생합니다. overflow_o는 SATURATE와 무관하게 오버플로 여부를 알려줍니다.
조합 연산wide·rounded·scaled는 모두 assign으로 조합 계산하고, 클록 에지에서 q_o에만 등록합니다. 이를 통해 Vivado가 연산 경로를 최적화할 수 있습니다.
