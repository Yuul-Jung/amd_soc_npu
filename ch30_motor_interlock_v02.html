`default_nettype none
// 30장 모터 인터록: AI가 낸 속도·방향 '제안'과 실제 모터 출력 사이에 AI와 무관한 안전 조건을 둔다.
// 출력 허용 = 비상정지 해제 && 무장(arm) && 사용 스위치 && 센서 상태 정상(29장 state < 4) && AI 명령이 제때 옴(감시 타이머).
// 하나라도 빠지면 그 클록에 en_o·pwm_o를 끄고 속도를 0으로 되돌린다. 비상정지는 걸쇠: 풀어도 arm_i를 다시 눌러야 한다.
// 허용 상태에서도 속도는 TICK마다 STEP 이하로만 바뀌고, 방향은 속도가 0이 된 뒤에만 바뀐다.
module motor_interlock #(
    parameter integer TICK = 1000,         // 속도 갱신 간격 (클록)
    parameter integer STEP = 8,            // 한 번에 바뀌는 속도 상한
    parameter integer WD_LIMIT = 100000    // AI 명령이 이 클록 동안 없으면 정지
) (
    input  wire       clk,
    input  wire       resetn,
    input  wire       estop_n_i,           // 비상정지 (0 = 눌림)
    input  wire       arm_i,               // 무장 요청 (속도 0, 비상정지 해제, 사용 스위치 켬일 때만)
    input  wire       enable_i,            // 사용 스위치
    input  wire [2:0] sensor_state_i,      // 29장 상태 (4 UNKNOWN, 5 FAULT → 정지)
    input  wire       ai_valid_i,          // AI 제안 갱신
    input  wire [7:0] ai_speed_i,
    input  wire       ai_dir_i,
    output wire       en_o,                // 드라이버 사용 (허용일 때만 1)
    output wire       pwm_o,
    output reg        dir_o,
    output reg  [7:0] speed_o,             // 지금 속도 (PWM 듀티 /256)
    output wire [4:0] block_o              // 막힌 이유: {감시 타이머, 센서, 사용 스위치, 무장 안 됨, 비상정지}
);
    reg armed;
    reg [31:0] wd_cnt;
    reg [31:0] tick_cnt;
    reg [7:0] pwm_cnt;
    reg [7:0] req_speed; reg req_dir;

    wire estop = !estop_n_i;
    wire wd_expired = (wd_cnt >= WD_LIMIT);
    wire sensor_bad = (sensor_state_i >= 3'd4);
    assign block_o = {wd_expired, sensor_bad, !enable_i, !armed, estop};
    wire ok = (block_o == 5'd0);
    assign en_o = ok;
    assign pwm_o = ok && (pwm_cnt < speed_o);

    wire tick = (tick_cnt == TICK - 1);
    wire [7:0] target = (req_dir != dir_o) ? 8'd0 : req_speed;      // 방향이 다르면 먼저 0까지 줄인다
    wire [7:0] up = (target - speed_o > STEP) ? speed_o + STEP : target;
    wire [7:0] dn = (speed_o - target > STEP) ? speed_o - STEP : target;

    always @(posedge clk) begin
        if (!resetn) begin
            armed <= 0; wd_cnt <= WD_LIMIT; tick_cnt <= 0; pwm_cnt <= 0;
            req_speed <= 0; req_dir <= 0; dir_o <= 0; speed_o <= 0;
        end else begin
            pwm_cnt <= pwm_cnt + 8'd1;
            tick_cnt <= tick ? 0 : tick_cnt + 1;
            if (ai_valid_i) begin wd_cnt <= 0; req_speed <= ai_speed_i; req_dir <= ai_dir_i; end
            else if (!wd_expired) wd_cnt <= wd_cnt + 1;
            if (estop) armed <= 0;
            else if (arm_i && enable_i && speed_o == 0) armed <= 1;
            if (!ok) speed_o <= 0;                                       // 막히면 즉시 0 (다시 시작하면 0부터)
            else if (tick) begin
                if (speed_o == 0 && req_dir != dir_o) dir_o <= req_dir;   // 멈춘 뒤에만 방향 전환
                else if (target > speed_o) speed_o <= up;
                else speed_o <= dn;
            end
        end
    end
endmodule

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. 시나리오 15구간 3,660클록(무장 전 명령, 무장·가속, 역방향 요청, 센서 UNKNOWN·FAULT, 비상정지와 재무장, AI 명령 끊김, 사용 스위치)을 넣고, 출력 5개를 클록마다 Python 모델과 비교합니다. Python은 안전 성질(막힘 중 PWM 없음, 안전 입력 위반 시 허용 없음, 속도 변화 ≤ STEP, 움직이는 중 방향 전환 없음)도 따로 확인합니다. 시뮬레이션은 TICK=16, WD_LIMIT=400으로 줄입니다.

ch30/reference/interlock_model.py · python · 원본 · RTL_SIM 통과 2026-11-19
#!/usr/bin/env python3
"""30장 기준 모델: motor_interlock.v를 클록 단위로 옮긴 Python 모델 + 시나리오(입력 파형) + 안전 성질 검사.
시나리오는 모터 없이 출력만 보는 시험이다 (TICK=16, STEP=8, WD_LIMIT=400으로 줄여 시뮬레이션)."""
from __future__ import annotations
import sys

TICK, STEP, WD = 16, 8, 400


def model(ins):
    s = dict(armed=0, wd=WD, tick=0, pwm=0, rs=0, rd=0, dir=0, spd=0)
    out = []
    for estop_n, arm, en, sens, av, asp, adir in ins:
        estop = not estop_n
        wd_exp = s["wd"] >= WD
        bad = sens >= 4
        block = (wd_exp << 4) | (bad << 3) | ((not en) << 2) | ((not s["armed"]) << 1) | estop
        ok = block == 0
        pwm = ok and s["pwm"] < s["spd"]
        out.append((int(ok), int(pwm), s["dir"], s["spd"], block))
        tick = s["tick"] == TICK - 1
        target = 0 if s["rd"] != s["dir"] else s["rs"]
        n = dict(s)
        n["pwm"] = (s["pwm"] + 1) & 0xFF
        n["tick"] = 0 if tick else s["tick"] + 1
        if av: n["wd"], n["rs"], n["rd"] = 0, asp, adir
        elif not wd_exp: n["wd"] = s["wd"] + 1
        if estop: n["armed"] = 0
        elif arm and en and s["spd"] == 0: n["armed"] = 1
        if not ok: n["spd"] = 0
        elif tick:
            if s["spd"] == 0 and s["rd"] != s["dir"]: n["dir"] = s["rd"]
            elif target > s["spd"]: n["spd"] = s["spd"] + STEP if target - s["spd"] > STEP else target
            else: n["spd"] = s["spd"] - STEP if s["spd"] - target > STEP else target
        s = n
    return out


def scenario():
    """(구간 길이, estop_n, arm, enable, sensor, ai 주기, 속도, 방향, 설명)"""
    seg = [(50, 1, 0, 1, 0, 0, 0, 0, "무장 전: AI가 명령해도 정지"),
           (40, 1, 0, 1, 0, 20, 200, 0, "무장 전 AI 명령 200"),
           (5, 1, 1, 1, 0, 20, 200, 0, "무장"),
           (900, 1, 0, 1, 0, 20, 200, 0, "가속 (TICK마다 +8, 200까지)"),
           (600, 1, 0, 1, 0, 20, 120, 1, "반대 방향 요청: 0까지 감속 → 방향 전환 → 120을 향해 가속"),
           (100, 1, 0, 1, 4, 20, 120, 1, "센서 UNKNOWN → 즉시 정지"),
           (300, 1, 0, 1, 0, 20, 120, 1, "센서 정상 복귀: 0부터 다시 가속"),
           (60, 0, 0, 1, 0, 20, 120, 1, "비상정지 눌림"),
           (200, 1, 0, 1, 0, 20, 120, 1, "비상정지 풀림: 무장이 풀려 계속 정지"),
           (5, 1, 1, 1, 0, 20, 120, 1, "다시 무장"),
           (300, 1, 0, 1, 0, 20, 120, 1, "다시 가속"),
           (700, 1, 0, 1, 0, 0, 120, 1, "AI 명령 끊김 → 400클록 뒤 정지"),
           (200, 1, 0, 1, 0, 20, 60, 1, "AI 복귀"),
           (100, 1, 0, 0, 0, 20, 60, 1, "사용 스위치 끔"),
           (100, 1, 0, 1, 5, 20, 60, 1, "센서 FAULT")]
    ins, notes = [], []
    for n, en_n, arm, en, sens, per, sp, d, why in seg:
        notes.append((len(ins), why))
        for k in range(n):
            av = int(per > 0 and k % per == 0)
            ins.append((en_n, arm, en, sens, av, sp, d))
    return ins, notes


def properties(ins, out):
    errs = []
    for t, ((estop_n, arm, en, sens, av, sp, d), (ok, pwm, dr, spd, blk)) in enumerate(zip(ins, out)):
        if pwm and not ok: errs.append(f"t={t} 막힘인데 PWM")
        if (not estop_n or sens >= 4 or not en) and ok: errs.append(f"t={t} 안전 입력 위반인데 허용")
        if t and abs(spd - out[t - 1][3]) > STEP and spd != 0: errs.append(f"t={t} 속도 변화 > STEP")
        if t and dr != out[t - 1][2] and out[t - 1][3] != 0: errs.append(f"t={t} 움직이는 중 방향 전환")
    return errs


if __name__ == "__main__":
    ins, notes = scenario()
    out = model(ins)
    errs = properties(ins, out)
    if errs:
        print("\n".join(errs[:5])); print("PYTHON_INTERLOCK_PROPERTY_FAIL"); sys.exit(1)
    peak = max(o[3] for o in out)
    print(f"PYTHON_INTERLOCK_PROPERTY_PASS cycles={len(ins)} max_speed={peak} reversals={sum(1 for t in range(1, len(out)) if out[t][2] != out[t-1][2])}")
    if len(sys.argv) > 2 and sys.argv[1] == "--out":
        with open(sys.argv[2], "w") as f:
            f.writelines(" ".join(map(str, i)) + "\n" for i in ins)
        with open(sys.argv[2] + ".expect", "w") as f:
            f.writelines(" ".join(map(str, o)) + "\n" for o in out)
        ends = [s for s, _ in notes[1:]] + [len(out)]
        for (start, why), end in zip(notes, ends):           # 구간의 마지막 클록 출력
            o = out[end - 1]
            print(f"SCENE t={start:4d}~{end - 1:4d} {why} → 끝에서 en={o[0]} speed={o[3]} dir={o[2]} block=0b{o[4]:05b}")

위치내용
제안과 출력 분리AI 입력(ai_*)은 요청 레지스터에 저장될 뿐 출력으로 바로 가지 않습니다. 출력 허용(en_o)은 AI와 무관한 다섯 조건의 AND이고, block_o가 막힌 이유를 비트로 보여 줍니다.

감시 타이머AI(CPU)가 멈추거나 통신이 끊겨 명령이 오지 않으면, 마지막 명령을 계속 따르지 않고 정지합니다.
보드 확인 범위이 장은 RTL과 시나리오까지입니다. 보드에서는 출력을 LED나 Pmod 핀의 로직 분석기로만 봅니다(⚠ 미작성·미검증). 실제 비상정지는 FPGA 논리만 믿지 않고 드라이버 전원을 끊는 하드웨어로 따로 둡니다.
