`default_nettype none

// DSP48E1의 25x18 signed 입력 범위에 맞춘 독립 곱셈 추론 후보.
// primitive를 직접 생성하지 않으며 실제 DSP 배치는 합성 후 확인한다.
module dsp_multiplier #(
    parameter integer A_WIDTH = 16,
    parameter integer B_WIDTH = 16
) (
    input wire clk,
    input wire resetn,
    input wire clear_i,
    input wire valid_i,
    input wire signed [A_WIDTH-1:0] a_i,
    input wire signed [B_WIDTH-1:0] b_i,
    output reg signed [A_WIDTH+B_WIDTH-1:0] product_o,
    output reg valid_o
);
    wire signed [24:0] a_dsp_w;
    wire signed [17:0] b_dsp_w;
    (* use_dsp = "yes" *) wire signed [42:0] product_full_w;

    // 작은 정수의 부호를 보존하여 DSP 곱셈 입력 폭까지 확장한다.
    assign a_dsp_w = {{(25-A_WIDTH){a_i[A_WIDTH-1]}}, a_i};
    assign b_dsp_w = {{(18-B_WIDTH){b_i[B_WIDTH-1]}}, b_i};
    assign product_full_w = $signed(a_dsp_w) * $signed(b_dsp_w);

    // P3-01과 같은 동기식 Reset > Clear > Valid 계약이다.
    // 상승 에지에서 입력을 수락하고 그 에지 직후 곱과 valid_o를 갱신한다.
    always @(posedge clk) begin
        if (!resetn) begin
            product_o <= {(A_WIDTH+B_WIDTH){1'b0}};
            valid_o <= 1'b0;
        end else if (clear_i) begin
            product_o <= {(A_WIDTH+B_WIDTH){1'b0}};
            valid_o <= 1'b0;
        end else begin
            valid_o <= valid_i;
            if (valid_i) begin
                // 곱의 전체 유효 폭은 A_WIDTH+B_WIDTH. 버리는 상위 비트는 부호 확장뿐이다.
                product_o <= product_full_w[A_WIDTH+B_WIDTH-1:0];
            end
        end
    end

    // synthesis translate_off
    initial begin
        if (A_WIDTH < 2 || A_WIDTH > 25 || B_WIDTH < 2 || B_WIDTH > 18)
            $fatal(1, "Require 2<=A_WIDTH<=25 and 2<=B_WIDTH<=18");
    end
    // synthesis translate_on
endmodule
`default_nettype wire

`timescale 1ns/1ps
`default_nettype none
module tb_dsp_multiplier;
    parameter integer A_WIDTH = 16;
    parameter integer B_WIDTH = 16;
    parameter integer ACC_WIDTH = 40;
    localparam integer MAC_WIDTH = A_WIDTH > B_WIDTH ? A_WIDTH : B_WIDTH;
    reg clk = 0;
    reg resetn = 0, clear_i = 0, valid_i = 0;
    reg signed [A_WIDTH-1:0] a_i = 0;
    reg signed [B_WIDTH-1:0] b_i = 0;
    wire signed [A_WIDTH+B_WIDTH-1:0] product_o;
    wire valid_o;
    wire signed [ACC_WIDTH-1:0] acc_o;
    wire mac_valid;
    wire signed [MAC_WIDTH-1:0] a_mac = a_i;
    wire signed [MAC_WIDTH-1:0] b_mac = b_i;
    dsp_multiplier #(.A_WIDTH(A_WIDTH), .B_WIDTH(B_WIDTH)) dut (.*);
    // 누산 로직은 복제하지 않고 P3-01 정상 모듈에 같은 입력을 연결한다.
    mac_core #(.DATA_WIDTH(MAC_WIDTH), .ACC_WIDTH(ACC_WIDTH)) mac (
        .clk(clk), .resetn(resetn), .clear_i(clear_i), .valid_i(valid_i),
        .a_i(a_mac), .b_i(b_mac), .acc_o(acc_o), .valid_o(mac_valid)
    );
    always #5 clk = ~clk;
    integer vf, af, fields, cycle = 0;
    integer rn, cl, vi, ai, bi, ev;
    reg signed [A_WIDTH+B_WIDTH-1:0] ep;
    reg signed [ACC_WIDTH-1:0] ea;
    longint signed model_product = 0, model_acc = 0, multiply_w;
    reg signed [A_WIDTH+B_WIDTH-1:0] held_product;
    reg signed [ACC_WIDTH-1:0] held_acc;
    reg held_valid, held_mac_valid;
    string vp, ap;

    initial begin
        if (ACC_WIDTH > 56 || ACC_WIDTH < 2*MAC_WIDTH)
            $fatal(1, "TB requires 2*MAC_WIDTH<=ACC_WIDTH<=56");
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap))
            $fatal(1, "Missing vector/trace argument");
        vf = $fopen(vp, "r"); af = $fopen(ap, "w");
        if (!vf || !af) $fatal(1, "Cannot open test files");
        @(posedge clk); #1;
        if (product_o !== 0 || valid_o !== 0 || acc_o !== 0 || mac_valid !== 0)
            $fatal(1, "Initial reset failed");
        while (!$feof(vf)) begin
            fields = $fscanf(vf, "%d %d %d %d %d %d %d %d\n", rn, cl, vi, ai, bi, ep, ea, ev);
            if (fields != 8) $fatal(1, "Malformed vector");
            @(negedge clk);
            held_product = product_o; held_acc = acc_o;
            held_valid = valid_o; held_mac_valid = mac_valid;
            resetn = rn; clear_i = cl; valid_i = vi; a_i = ai; b_i = bi;
            #1;
            if (product_o !== held_product || acc_o !== held_acc ||
                valid_o !== held_valid || mac_valid !== held_mac_valid)
                $fatal(1, "Output changed before rising edge cycle=%0d", cycle+1);
            @(posedge clk); #1;
            cycle = cycle + 1;
            // 64비트 정수로 먼저 확장하여 곱을 계산하고 명시적으로 누산 폭을 적용한다.
            multiply_w = $signed(a_i);
            multiply_w = multiply_w * $signed(b_i);
            if (!rn || cl) begin
                model_product = 0;
                model_acc = 0;
            end else if (vi) begin
                model_product = multiply_w;
                model_acc = model_acc + multiply_w;
                model_acc = $signed(model_acc[ACC_WIDTH-1:0]);
            end
            if (ep !== model_product[A_WIDTH+B_WIDTH-1:0] ||
                ea !== model_acc[ACC_WIDTH-1:0] || ev !== (rn && !cl && vi))
                $fatal(1, "Python/SV reference mismatch cycle=%0d", cycle);
            if (product_o !== ep || acc_o !== ea || valid_o !== ev[0] || mac_valid !== ev[0])
                $fatal(1, "DUT mismatch cycle=%0d a=%0d b=%0d product=%0d expected=%0d acc=%0d expected=%0d",
                       cycle, a_i, b_i, product_o, ep, acc_o, ea);
            $fdisplay(af, "%0d %0d %0d %0d", cycle, product_o, acc_o, valid_o);
        end
        if (!cycle) $fatal(1, "Empty test");
        $fclose(vf); $fclose(af);
        $display("RTL_SIM_PASS A=%0d B=%0d ACC=%0d cycles=%0d", A_WIDTH, B_WIDTH, ACC_WIDTH, cycle);
        $finish;
    end
    initial begin
        #10000000;
        $fatal(1, "Simulation timeout");
    end
endmodule
`default_nettype wire

#!/usr/bin/env python3
"""P3-02 signed 곱 기준 모델. P3-01의 누산/제어 모델과 trace 비교를 재사용한다."""
import argparse
import random
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2] / "ch01" / "reference"))
from mac_reference import compare, self_test, step, stimulus


def vectors(a_width, b_width, acc_width):
    # 대칭 INT16/INT8은 기존 누산 오버플로 자극도 그대로 사용한다.
    if a_width == b_width:
        yield from stimulus(a_width, acc_width)
    else:
        yield (0, 1, 1, 1, -1)
        yield (1, 0, 0, 0, 0)
    amin, amax = -(1 << (a_width - 1)), (1 << (a_width - 1)) - 1
    bmin, bmax = -(1 << (b_width - 1)), (1 << (b_width - 1)) - 1
    yield (1, 1, 1, amax, bmax)
    for a in (amin, amin + 1, -1, 0, 1, amax - 1, amax):
        for b in (bmin, bmin + 1, -1, 0, 1, bmax - 1, bmax):
            yield (1, 0, 1, a, b)
    # INT8의 65536개 입력 쌍은 전수 대조한다(제어 상태 전수 검증은 아님).
    if (a_width, b_width) == (8, 8):
        for a in range(amin, amax + 1):
            for b in range(bmin, bmax + 1):
                yield (1, 0, 1, a, b)
    rng = random.Random(a_width * 1000 + b_width)
    for _ in range(2000):
        yield (int(rng.randrange(43) != 0), int(rng.randrange(31) == 0),
               int(rng.randrange(4) != 0), rng.randint(amin, amax), rng.randint(bmin, bmax))
    yield (0, 0, 1, amin, bmax)
    yield (0, 1, 1, amin, bmin)
    yield (1, 0, 1, amax, bmin)
    yield (1, 0, 0, amin, bmax)
    yield (1, 1, 0, 1, 1)
    yield (1, 0, 0, 0, 0)


def generate(a_width, b_width, acc_width, vector_path, expected_path):
    product, acc = 0, 0
    count = 0
    with vector_path.open("w") as vf, expected_path.open("w") as ef:
        for count, row in enumerate(vectors(a_width, b_width, acc_width), 1):
            rn, clear, valid, a, b = row
            if not rn or clear:
                product = 0
            elif valid:
                product = a * b  # Python의 무제한 정수 곱, RTL 폭 변환에 의존하지 않는다.
            acc, out_valid = step(acc, *row, acc_width)
            if not -(1 << (a_width+b_width-1)) <= product < (1 << (a_width+b_width-1)):
                raise AssertionError("Product outside full signed result width")
            vf.write(" ".join(map(str, (*row, product, acc, out_valid))) + "\n")
            ef.write(f"{count} {product} {acc} {out_valid}\n")
    print(f"PYTHON_VECTORS A={a_width} B={b_width} ACC={acc_width} cycles={count}")


def product_self_test():
    assert (-32768) * (-32768) == 1073741824
    assert (-32768) * 32767 == -1073709056
    assert (-128) * (-128) == 16384
    assert (-16777216) * (-131072) == 2199023255552
    assert 16777215 * (-131072) == -2199023124480
    print("PYTHON_PRODUCT_SELF_TEST_PASS INT16,INT8,native25x18")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--a-width", type=int, default=16)
    parser.add_argument("--b-width", type=int, default=16)
    parser.add_argument("--acc-width", type=int, default=40)
    parser.add_argument("--vectors", type=Path)
    parser.add_argument("--expected", type=Path, required=True)
    parser.add_argument("--actual", type=Path)
    args = parser.parse_args()
    # TB는 P3-01 MAC 입력 폭을 두 입력 중 큰 폭으로 맞춰 비교한다.
    if not (2 <= args.a_width <= 25 and 2 <= args.b_width <= 18
            and 2 * max(args.a_width, args.b_width) <= args.acc_width <= 56):
        parser.error("Unsupported DSP input or TB accumulator width")
    if bool(args.vectors) == bool(args.actual):
        parser.error("Select exactly one of --vectors and --actual")
    product_self_test()
    self_test()
    if args.actual:
        compare(args.expected, args.actual)
    else:
        generate(args.a_width, args.b_width, args.acc_width, args.vectors, args.expected)

07코드 읽기 포인트

위치내용
DSP48E1 25×18 범위XC7A35T의 DSP48E1 A 포트는 25비트, B 포트는 18비트 signed 입력을 받습니다. INT16(±32768)은 두 포트 모두에 들어가므로 추론이 성공합니다.
상위 비트 버리기43비트 전체 곱 중 하위 A_WIDTH+B_WIDTH비트만 등록합니다. 버려지는 비트는 부호 확장 중복분으로 INT16×INT16에서 의미 있는 값이 없습니다.
교차 검증 전략테스트벤치가 mac_core를 같은 입력에 연결하고 누산 결과까지 비교합니다. 이를 통해 dsp_multiplier 단독으로 중복 모델 없이 완전한 검증이 가능합니다.
