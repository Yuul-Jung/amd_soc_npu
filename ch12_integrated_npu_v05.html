// (추가) 02장 등록 DSP 곱을 다음 에지에 누산한다. 곱을 다시 수행하지 않는다.
module npu_dsp_mac_lane #(
    parameter integer DATA_WIDTH=16,
    parameter integer ACC_WIDTH=40
)(
    input wire clk,resetn,clear_i,valid_i,
    input wire signed [DATA_WIDTH-1:0] x_i,w_i,
    output reg signed [ACC_WIDTH-1:0] acc_o
);
    wire signed [2*DATA_WIDTH-1:0] product;
    wire product_valid;
    dsp_multiplier #(.A_WIDTH(DATA_WIDTH),.B_WIDTH(DATA_WIDTH)) dsp(
        .clk(clk),.resetn(resetn),.clear_i(clear_i),
        .valid_i(valid_i),.a_i(x_i),.b_i(w_i),
        .product_o(product),.valid_o(product_valid));
    wire signed [ACC_WIDTH-1:0] product_extended =
        {{(ACC_WIDTH-2*DATA_WIDTH){product[2*DATA_WIDTH-1]}},product};
    always @(posedge clk)begin
        if(!resetn || clear_i)acc_o<=0;
        else if(product_valid)acc_o<=acc_o+product_extended;
    end
    // synthesis translate_off
    initial if(ACC_WIDTH<2*DATA_WIDTH)$fatal(1,"ACC_WIDTH too small");
    // synthesis translate_on
endmodule

// (추가) 04장 BRAM 구조 + 02장 DSP 레지스터 곱 + 4개 누산기 통합.
// 12장 최종 경로는 기존 ch07 벡터 닷의 곱 경로를 중복 사용하지 않는다.
// 메모리 쓰기는 IDLE에서만 허용하며 Reset 시 유효 비트만 지운다.
module npu_engine4(
    input wire clk,resetn,start_i,
    input wire [6:0] length_i,
    input wire x_wr_en,w_wr_en,b_wr_en,
    input wire [5:0] addr_i,
    input wire [1:0] neuron_i,
    input wire signed [15:0] data_i,
    input wire signed [39:0] bias_i,
    output wire load_ready_o,
    output reg busy_o,done_o,error_o,
    output reg [1:0] class_o,
    output reg [163:0] scores_o
);
    localparam [2:0] IDLE=0,READ=1,ISSUE=2,DRAIN1=3,DRAIN2=4;
    reg [2:0] state;
    reg [6:0] length_q;
    reg [5:0] index_q;
    (* ram_style="block" *) reg signed [15:0] xmem[0:63];
    (* ram_style="block" *) reg signed [15:0] w0[0:63];
    (* ram_style="block" *) reg signed [15:0] w1[0:63];
    (* ram_style="block" *) reg signed [15:0] w2[0:63];
    (* ram_style="block" *) reg signed [15:0] w3[0:63];
    reg [63:0] x_seen,w0_seen,w1_seen,w2_seen,w3_seen;
    reg signed [39:0] bias0,bias1,bias2,bias3;
    reg signed [15:0] x_sample,w_sample0,w_sample1,w_sample2,w_sample3;

    reg input_valid;
    integer k;
    always @* begin
        input_valid=(length_i!=0 && length_i<=64);
        for(k=0;k<64;k=k+1)
            if(k<length_i &&
               (!x_seen[k] || !w0_seen[k] || !w1_seen[k] ||
                !w2_seen[k] || !w3_seen[k]))
                input_valid=0;
    end
    wire any_write=x_wr_en||w_wr_en||b_wr_en;
    wire launch=(state==IDLE)&&start_i&&input_valid&&!any_write;
    wire mul_valid=(state==ISSUE);
    assign load_ready_o=resetn&&!busy_o&&!start_i;

    wire signed [39:0] acc0,acc1,acc2,acc3;
    npu_dsp_mac_lane lane0(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample0),.acc_o(acc0));
    npu_dsp_mac_lane lane1(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample1),.acc_o(acc1));
    npu_dsp_mac_lane lane2(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample2),.acc_o(acc2));
    npu_dsp_mac_lane lane3(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample3),.acc_o(acc3));
    // 41비트 signed: 40비트 누산값에 Bias를 더할 때 부호 확장 1비트.
    wire signed [40:0] score0=$signed({acc0[39],acc0})+$signed({bias0[39],bias0});
    wire signed [40:0] score1=$signed({acc1[39],acc1})+$signed({bias1[39],bias1});
    wire signed [40:0] score2=$signed({acc2[39],acc2})+$signed({bias2[39],bias2});
    wire signed [40:0] score3=$signed({acc3[39],acc3})+$signed({bias3[39],bias3});
    reg [1:0] best_class;
    reg signed [40:0] best_score;
    always @*begin
        best_class=0;best_score=score0;
        if(score1>best_score)begin best_score=score1;best_class=1;end
        if(score2>best_score)begin best_score=score2;best_class=2;end
        if(score3>best_score)begin best_score=score3;best_class=3;end
    end

    always @(posedge clk)begin
        if(!resetn)begin
            state<=IDLE;busy_o<=0;done_o<=0;error_o<=0;
            class_o<=0;scores_o<=0;length_q<=0;index_q<=0;
            x_seen<=0;w0_seen<=0;w1_seen<=0;w2_seen<=0;w3_seen<=0;
            bias0<=0;bias1<=0;bias2<=0;bias3<=0;
            x_sample<=0;w_sample0<=0;w_sample1<=0;w_sample2<=0;w_sample3<=0;
        end else begin
            done_o<=0;error_o<=0;
            if(load_ready_o)begin
                if(x_wr_en)begin
                    xmem[addr_i]<=data_i;x_seen[addr_i]<=1;
                end
                if(w_wr_en)case(neuron_i)
                    0:begin w0[addr_i]<=data_i;w0_seen[addr_i]<=1;end
                    1:begin w1[addr_i]<=data_i;w1_seen[addr_i]<=1;end
                    2:begin w2[addr_i]<=data_i;w2_seen[addr_i]<=1;end
                    3:begin w3[addr_i]<=data_i;w3_seen[addr_i]<=1;end
                endcase
                if(b_wr_en)case(neuron_i)
                    0:bias0<=bias_i;
                    1:bias1<=bias_i;
                    2:bias2<=bias_i;
                    3:bias3<=bias_i;
                endcase
            end else if(any_write)error_o<=1;
            case(state)
                IDLE:if(start_i)begin
                    if(!input_valid||any_write)error_o<=1;
                    else begin
                        state<=READ;length_q<=length_i;index_q<=0;busy_o<=1;
                    end
                end
                READ:begin
                    x_sample<=xmem[index_q];
                    w_sample0<=w0[index_q];w_sample1<=w1[index_q];
                    w_sample2<=w2[index_q];w_sample3<=w3[index_q];
                    state<=ISSUE;
                end
                ISSUE:begin
                    if({1'b0,index_q}+7'd1==length_q)state<=DRAIN1;
                    else begin index_q<=index_q+1'b1;state<=READ;end
                end
                DRAIN1:state<=DRAIN2;
                DRAIN2:begin
                    scores_o<={score3,score2,score1,score0};
                    class_o<=best_class;done_o<=1;busy_o<=0;state<=IDLE;
                end
                default:begin state<=IDLE;busy_o<=0;error_o<=1;end
            endcase
        end
    end
endmodule

// (추가) FND와 LED 표현 정책은 Top이 아닌 Controller에서 처리.
module npu_display_controller(
    input wire clk,resetn,
    input wire [1:0] class_i,
    input wire busy_i,done_i,error_i,protocol_error_i,
    input wire uart_framing_error_i,uart_overrun_i,
    output wire [15:0] led_o,
    output reg [6:0] seg_o,
    output reg [3:0] an_o,
    output wire dp_o
);
    reg done_latched,error_latched;
    reg [17:0] scan_count;
    reg [3:0] digit;
    wire [3:0] onehot=4'b0001<<class_i;
    always @(posedge clk)begin
        if(!resetn)begin
            done_latched<=0;error_latched<=0;scan_count<=0;
        end else begin
            scan_count<=scan_count+1'b1;
            if(done_i)done_latched<=1;
            if(error_i||protocol_error_i||uart_framing_error_i||uart_overrun_i)
                error_latched<=1;
        end
    end
    assign led_o[3:0]=onehot;
    assign led_o[12:4]=9'd0;
    assign led_o[13]=busy_i;
    assign led_o[14]=done_latched;
    assign led_o[15]=error_latched;
    assign dp_o=1'b1;
    always @*begin
        case(scan_count[17:16])
            2'd0:begin an_o=4'b1110;digit={2'b00,class_i};end
            2'd1:begin an_o=4'b1101;digit=0;end
            2'd2:begin an_o=4'b1011;digit=0;end
            default:begin an_o=4'b0111;digit=0;end
        endcase
        // Active low: segments {g,f,e,d,c,b,a} (seg[0] is a).
        case(digit)
            0:seg_o=7'b1000000;
            1:seg_o=7'b1111001;
            2:seg_o=7'b0100100;
            3:seg_o=7'b0110000;
            default:seg_o=7'b1111111;
        endcase
    end
endmodule
mini_npu/rtl/npu_reset_controller.v · verilog
// (추가) Basys3 btnC 비동기 Assertion, 동기 Release Reset.
module npu_reset_controller(
    input wire clk,btnC,
    output wire resetn
);
    reg [1:0] release_q;
    always @(posedge clk or posedge btnC)
        if(btnC)release_q<=2'b00;
        else release_q<={release_q[0],1'b1};
    assign resetn=release_q[1];
endmodule

proc worst_slack {kind} {
    set p [get_timing_paths -quiet -$kind -max_paths 1 -nworst 1]
    if {[llength $p] == 0} { return NA }
    return [format %.3f [get_property SLACK $p]]
}
proc count_ref {pattern} {
    return [llength [get_cells -quiet -hierarchical -filter "IS_PRIMITIVE && REF_NAME =~ $pattern"]]
}
set wns  [worst_slack setup]

// Empty/Full, 동시 Push/Pop, Ready/Valid, Reset 초기화, 연속 수신 손실, Overrun, Framing, 전송 중단.
// 실제 직렬 비트파형으로 RX를 구동하고 TX 핀을 독립 uart_rx로 복조해 순서·값을 비교한다.
`timescale 1ns/1ps
module tb_uart_core_fifo;
    localparam integer CLK_HZ=1000000,BAUD=100000;
    localparam integer CPB=(CLK_HZ+BAUD/2)/BAUD; // 10 clocks/bit: 시뮬레이션 단축
    reg clk=0,resetn=0,rxd=1;
    wire txd;
    wire [7:0] rx_data;
    wire rx_valid;
    reg rx_ready=0;
    reg [7:0] tx_data=0;
    reg tx_valid=0;
    wire tx_ready;
    wire fe,ov;
    reg mon_resetn=0;
    wire [7:0] mon_data;
    wire mon_valid,mon_fe,mon_ov;
    always #5 clk=~clk;

    uart_core #(.CLK_HZ(CLK_HZ),.BAUD(BAUD)) dut(.clk(clk),.resetn(resetn),
        .uart_rxd_i(rxd),.uart_txd_o(txd),
        .rx_out_data(rx_data),.rx_out_valid(rx_valid),.rx_out_ready(rx_ready),
        .tx_in_data(tx_data),.tx_in_valid(tx_valid),.tx_in_ready(tx_ready),
        .rx_framing_error_o(fe),.rx_overrun_o(ov));
    uart_rx #(.CLK_HZ(CLK_HZ),.BAUD(BAUD)) mon(.clk(clk),.resetn(mon_resetn),
        .rx_i(txd),.out_data_o(mon_data),.out_valid_o(mon_valid),
        .out_ready_i(1'b1),.framing_error_o(mon_fe),.overrun_o(mon_ov));

    // ---------------- scoreboards ----------------
    reg [7:0] txq[0:1023];
    integer tx_head=0,tx_tail=0;
    reg [7:0] rxq[0:1023];
    integer rx_head=0,rx_tail=0;
    integer tx_accepted=0,rx_delivered=0;
    // coverage
    integer cov_rx_full=0,cov_tx_full=0,cov_rx_full_pushpop=0,cov_tx_full_pushpop=0;
    integer cov_rx_pushpop=0,cov_tx_pushpop=0,cov_rx_empty_pop_attempt=0;
    integer cov_tx_stall=0,cov_rx_hold=0;
    reg [7:0] prev_rx_data;
    reg prev_rx_stall=0;
    reg [7:0] prev_tx_data;
    reg prev_tx_stall=0;

    always @(posedge clk)begin
        // TX handshake: 수락된 바이트만 기대 큐에 넣는다.
        if(tx_valid && tx_ready)begin
            txq[tx_tail%1024]=tx_data;tx_tail=tx_tail+1;tx_accepted=tx_accepted+1;
        end
        // RX handshake: 실제 수신 순서 비교
        if(rx_valid && rx_ready)begin
            if(rx_head==rx_tail)$fatal(1,"RX unexpected byte %02h",rx_data);
            if(rx_data!==rxq[rx_head%1024])
                $fatal(1,"RX order/data mismatch idx=%0d got=%02h exp=%02h",
                    rx_head,rx_data,rxq[rx_head%1024]);
            rx_head=rx_head+1;rx_delivered=rx_delivered+1;
        end
        // TX 핀 복조 비교
        if(mon_valid)begin
            if(tx_head==tx_tail)$fatal(1,"TX unexpected byte on pin %02h",mon_data);
            if(mon_data!==txq[tx_head%1024])
                $fatal(1,"TX order/data mismatch idx=%0d got=%02h exp=%02h",
                    tx_head,mon_data,txq[tx_head%1024]);
            tx_head=tx_head+1;
        end
        if(mon_fe)$fatal(1,"TX pin framing error");
        // Ready/Valid 규칙: valid && !ready 동안 rx_out_data 유지
        if(prev_rx_stall && rx_data!==prev_rx_data)$fatal(1,"RX data changed while stalled");
        prev_rx_stall=rx_valid && !rx_ready;prev_rx_data=rx_data;
        if(prev_tx_stall && (!tx_valid || tx_data!==prev_tx_data))
            $fatal(1,"TB violated TX valid/data hold");
        prev_tx_stall=tx_valid && !tx_ready;prev_tx_data=tx_data;
        // FIFO 내부 불변식
        if(resetn)begin
            if(dut.rx_count>16 || dut.tx_count>16)$fatal(1,"FIFO count overflow");
            if(dut.rx_count!=0 && dut.rx_count!=16 &&
               dut.rx_count!=((dut.rx_wptr-dut.rx_rptr)&4'hf))$fatal(1,"RX ptr/count mismatch");
            if(dut.tx_count!=0 && dut.tx_count!=16 &&
               dut.tx_count!=((dut.tx_wptr-dut.tx_rptr)&4'hf))$fatal(1,"TX ptr/count mismatch");
            if((dut.rx_count==0 || dut.rx_count==16) && dut.rx_wptr!=dut.rx_rptr)
                $fatal(1,"RX empty/full pointer mismatch");
            if((dut.tx_count==0 || dut.tx_count==16) && dut.tx_wptr!=dut.tx_rptr)
                $fatal(1,"TX empty/full pointer mismatch");
            if(rx_valid!==(dut.rx_count!=0))$fatal(1,"RX valid != !empty");
            if(dut.tx_count==16 && !dut.tx_pop && tx_ready)$fatal(1,"TX ready while full");
            if(dut.rx_count==16 && dut.rx_push && !dut.rx_pop)$fatal(1,"RX push while full");
            // coverage
            if(dut.rx_count==16)cov_rx_full=cov_rx_full+1;
            if(dut.tx_count==16)cov_tx_full=cov_tx_full+1;
            if(dut.rx_push && dut.rx_pop)cov_rx_pushpop=cov_rx_pushpop+1;
            if(dut.tx_push && dut.tx_pop)cov_tx_pushpop=cov_tx_pushpop+1;
            if(dut.rx_count==16 && dut.rx_push && dut.rx_pop)cov_rx_full_pushpop=cov_rx_full_pushpop+1;
            if(dut.tx_count==16 && dut.tx_push && dut.tx_pop)cov_tx_full_pushpop=cov_tx_full_pushpop+1;
            if(dut.rx_count==0 && rx_ready)cov_rx_empty_pop_attempt=cov_rx_empty_pop_attempt+1;
            if(tx_valid && !tx_ready)cov_tx_stall=cov_tx_stall+1;
            if(rx_valid && !rx_ready)cov_rx_hold=cov_rx_hold+1;
        end
    end

    // ---------------- serial drivers ----------------
    task automatic send_frame(input [7:0] b,input stop_bit);
        integer k;
        begin
            @(negedge clk);rxd=0;repeat(CPB)@(negedge clk);
            for(k=0;k<8;k=k+1)begin rxd=b[k];repeat(CPB)@(negedge clk);end
            rxd=stop_bit;repeat(CPB)@(negedge clk);
            rxd=1;
        end
    endtask
    task automatic send_byte(input [7:0] b);
        begin rxq[rx_tail%1024]=b;rx_tail=rx_tail+1;send_frame(b,1'b1);end
    endtask
    task automatic send_byte_dropped(input [7:0] b); // 기대 큐에 넣지 않음(overrun 대상)
        begin send_frame(b,1'b1);end
    endtask
    task automatic do_reset(input integer cycles);
        begin
            @(negedge clk);resetn=0;mon_resetn=0;rx_ready=0;tx_valid=0;
            repeat(2)@(negedge clk);
            if(tx_ready!==1'b0)$fatal(1,"tx_ready must be low in reset");
            repeat(cycles)@(negedge clk);
            if(txd!==1'b1)$fatal(1,"TXD not idle in reset");
            if(dut.rx_count!==0 || dut.tx_count!==0)$fatal(1,"FIFO count not cleared");
            if(dut.rx_wptr!==0 || dut.rx_rptr!==0 || dut.tx_wptr!==0 || dut.tx_rptr!==0)
                $fatal(1,"FIFO pointers not cleared");
            if(rx_valid!==1'b0 || fe!==1'b0 || ov!==1'b0)$fatal(1,"status not cleared");
            if(dut.rx.out_valid_o!==1'b0)$fatal(1,"PHY RX hold not cleared");
            // 스코어보드도 초기화(리셋으로 버린 데이터는 기대하지 않는다)
            tx_head=tx_tail;rx_head=rx_tail;
            resetn=1;mon_resetn=1;
            @(negedge clk);
            if(tx_ready!==1'b1)$fatal(1,"tx_ready must rise after reset");
        end
    endtask
    // TX push: negedge에서 데이터 제시, ready를 기다려 다음 posedge에서 수락.
    task automatic tx_push(input [7:0] b);
        begin
            tx_data=b;tx_valid=1;
            while(!tx_ready)@(negedge clk);
            @(negedge clk);
        end
    endtask
    task automatic wait_tx_drain;
        integer guard;
        begin
            guard=0;
            while(tx_head!=tx_tail || dut.tx_count!=0 || dut.tx.sending)begin
                @(negedge clk);guard=guard+1;
                if(guard>CPB*10*80)$fatal(1,"TX drain timeout");
            end
            repeat(CPB*2)@(negedge clk);
        end
    endtask
    task automatic wait_rx_drain;
        integer guard;
        begin
            guard=0;
            while(rx_head!=rx_tail)begin
                @(negedge clk);guard=guard+1;
                if(guard>CPB*10*80)$fatal(1,"RX drain timeout head=%0d tail=%0d",rx_head,rx_tail);
            end
        end
    endtask

    integer i,first_stall_accept,seed,t;
    reg rand_ready_en=0;
    always @(negedge clk)if(rand_ready_en)rx_ready=($random(seed)&1);

    initial begin
        seed=12345;
        // ===== T1 Reset 상태 =====
        repeat(5)@(negedge clk);
        if(tx_ready!==1'b0)$fatal(1,"T1 tx_ready high during reset");
        if(txd!==1'b1)$fatal(1,"T1 txd not idle");
        do_reset(3);
        if(rx_valid!==0)$fatal(1,"T1 rx_valid after reset");
        $display("T1_RESET_DEFAULTS_PASS");

        // ===== T2 TX Empty->Full, 정확한 용량(PHY 1 + FIFO 16), 동시 Push/Pop =====
        @(negedge clk);
        first_stall_accept=-1;
        for(i=0;i<40;i=i+1)begin
            tx_data=8'h30+i;tx_valid=1;
            if(!tx_ready && first_stall_accept<0)first_stall_accept=tx_accepted;
            while(!tx_ready)@(negedge clk);
            @(negedge clk);
        end
        tx_valid=0;
        if(first_stall_accept!=17)

        $display("T2_TX_FULL_CAPACITY_PASS capacity=%0d bytes=40 full_pushpop=%0d",
            first_stall_accept,cov_tx_full_pushpop);

        // ===== T3 valid=0일 때 데이터 변동은 무시 =====
        for(i=0;i<200;i=i+1)begin tx_data=$random(seed);tx_valid=0;@(negedge clk);end
        if(dut.tx_count!==0 || dut.tx.sending)$fatal(1,"T3 TX pushed without valid");
        // valid=0 상태에서 rx_ready만 들어도 Empty pop 없음
        rx_ready=1;repeat(50)@(negedge clk);rx_ready=0;
        if(dut.rx_count!==0 || dut.rx_rptr!==dut.rx_wptr)$fatal(1,"T3 RX pop on empty");

        rx_ready=0;
        for(i=0;i<17;i=i+1)send_byte(8'hA0+i);
        repeat(CPB*2)@(negedge clk);
        if(dut.rx_count!==16)$fatal(1,"T4 RX count %0d expected 16",dut.rx_count);
        if(dut.rx.out_valid_o!==1'b1)$fatal(1,"T4 PHY hold byte missing");

        if(fe!==1'b1)$fatal(1,"T8 framing error not flagged");
        if(rx_valid || dut.rx_count!=0)$fatal(1,"T8 bad frame was pushed");
        send_byte(8'h3C);rx_ready=1;wait_rx_drain;rx_ready=0;
        do_reset(4);
        $display("T8_FRAMING_ERROR_PASS");

        // ===== T9 TX 전송 중 Reset: FIFO/PHY 초기화, 이후 잔류 데이터 없음 =====
        @(negedge clk);
        for(i=0;i<12;i=i+1)tx_push(8'h10+i);
        tx_valid=0;
        repeat(CPB*10+CPB*4)@(negedge clk); // 2번째 프레임 중간
        if(!dut.tx.sending || dut.tx_count==0)$fatal(1,"T9 precondition");
        do_reset(5);
        @(negedge clk);
        for(i=0;i<3;i=i+1)tx_push(8'hD0+i);
        tx_valid=0;
        wait_tx_drain;
        $display("T9_RESET_DURING_TX_FLUSH_PASS");

        // ===== T10 RX 프레임 중간 Reset + 송신 중단: 반쪽 바이트 미적재 =====
        fork
            begin : partial
                @(negedge clk);rxd=0;repeat(CPB)@(negedge clk);
                for(t=0;t<4;t=t+1)begin rxd=0;repeat(CPB)@(negedge clk);end
                rxd=1; // 송신 중단(라인 idle)
            end
            begin repeat(CPB*3)@(negedge clk);@(negedge clk);resetn=0;mon_resetn=0;end
        join
        repeat(CPB*12)@(negedge clk);
        if(dut.rx_count!==0 || dut.rx.out_valid_o!==0)$fatal(1,"T10 partial frame stored");
        resetn=1;mon_resetn=1;
        repeat(CPB*12)@(negedge clk);
        if(rx_valid || fe || ov)$fatal(1,"T10 garbage after reset");
        send_byte(8'h99);rx_ready=1;wait_rx_drain;rx_ready=0;
        $display("T10_RESET_DURING_RX_PASS");

        // ===== T11 (관측) Reset 없는 송신 중단: UART 특성상 남은 비트가 1로 읽힌다 =====
        @(negedge clk);rxd=0;repeat(CPB)@(negedge clk);
        for(t=0;t<4;t=t+1)begin rxd=0;repeat(CPB)@(negedge clk);end
        rxd=1;
        rxq[rx_tail%1024]=8'hF0;rx_tail=rx_tail+1; // 하위 4비트 0, 상위 4비트 idle=1
        repeat(CPB*8)@(negedge clk);
        rx_ready=1;wait_rx_drain;rx_ready=0;
        if(fe)$fatal(1,"T11 unexpected framing");
        $display("T11_ABORTED_FRAME_OBSERVED byte=F0 (8N1 has no truncation detection)");

        if(cov_rx_hold==0 || cov_tx_stall==0 || cov_rx_empty_pop_attempt==0)
            $fatal(1,"coverage hole hold=%0d stall=%0d empty=%0d",
                cov_rx_hold,cov_tx_stall,cov_rx_empty_pop_attempt);
        $display("COVERAGE rx_full=%0d tx_full=%0d rx_full_pushpop=%0d tx_full_pushpop=%0d rx_pushpop=%0d tx_pushpop=%0d rx_hold=%0d tx_stall=%0d",
            cov_rx_full,cov_tx_full,cov_rx_full_pushpop,cov_tx_full_pushpop,
            cov_rx_pushpop,cov_tx_pushpop,cov_rx_hold,cov_tx_stall);
        $display("UART_FIFO_ADVERSARIAL_SIM_PASS tests=11");
        $finish;
    end
    initial begin #50000000;$fatal(1,"UART FIFO TB TIMEOUT");end

// (추가) 11·12장 통합 Top 적대적 UART 시험(실제 RX 핀 구동, TX 핀 복조).
// 잘못된 명령·주소·길이, CR/LF 무시, 응답 전송 중 다음 명령 수신(FIFO 버퍼링),
// 연속 명령 폭주, 응답 전송 중 Reset, 명령 중단(Timeout 부재 관측), Framing error.
`timescale 1ns/1ps
module tb_npu_uart_adversarial;
    localparam integer CPB=868;
    reg clk=0,btnC=1,uart_rxd=1;
    wire uart_txd,dp;
    wire [15:0] led;
    wire [6:0] seg;
    wire [3:0] an;
    wire [7:0] byte_out;
    wire byte_valid,mfe,mov;
    always #5 clk=~clk;
    basys3_npu_top dut(.clk(clk),.btnC(btnC),.uart_rxd(uart_rxd),.uart_txd(uart_txd),
        .led(led),.seg(seg),.an(an),.dp(dp));
    uart_rx mon(.clk(clk),.resetn(!btnC),.rx_i(uart_txd),
        .out_data_o(byte_out),.out_valid_o(byte_valid),
        .out_ready_i(1'b1),.framing_error_o(mfe),.overrun_o(mov));

    // 응답 기록: kind 1=I 2=OK 3=ERR 4=C(class)
    reg [7:0] chars[0:63];
    integer nbytes=0,nresp=0;
    integer kind[0:255];
    integer cls[0:255];
    integer i,j,k,base;
    function automatic [7:0] hexchar(input [3:0] v);
        hexchar=(v<10)?("0"+v):("A"+(v-10));
    endfunction
    function automatic signed [40:0] golden(input integer s,input integer n);
        integer x0,x1,w0,w1;
        begin
            x0=(s&2)?-1:1;x1=(s&1)?-1:1;w0=(n&2)?-1:1;w1=(n&1)?-1:1;
            golden=x0*w0+x1*w1;
        end
    endfunction
    function automatic [7:0] gold_hex(input signed [40:0] val,input integer pos);
        reg [43:0] ext;
        begin
            ext={{3{val[40]}},val}; // 부호 비트만 3회 확장(44비트 = 11 hex)
            gold_hex=hexchar((ext>>((10-pos)*4))&4'hf);
        end
    endfunction
    always @(posedge clk)if(byte_valid)begin
        if(nbytes>=63)$fatal(1,"TX line overflow");
        if(byte_out!=8'h0a)begin chars[nbytes]=byte_out;nbytes=nbytes+1;end
        else begin
            if(nbytes==2 && chars[0]=="I" && chars[1]==8'h0d)kind[nresp]=1;
            else if(nbytes==3 && chars[0]=="O" && chars[1]=="K" && chars[2]==8'h0d)kind[nresp]=2;
            else if(nbytes==4 && chars[0]=="E" && chars[1]=="R" && chars[2]=="R" &&
                    chars[3]==8'h0d)kind[nresp]=3;
            else if(nbytes==51 && chars[0]=="C" && chars[2]==":" && chars[50]==8'h0d)begin
                kind[nresp]=4;cls[nresp]=chars[1]-"0";
                // 프리셋 표본(Bias=0) 점수는 실제 TX 복조 문자로 비교
                for(i=0;i<4;i=i+1)for(j=0;j<11;j=j+1)
                    if(chars[3+i*12+j]!==gold_hex(golden(cls[nresp],i),j))
                        $fatal(1,"Score mismatch resp=%0d lane=%0d",nresp,i);
            end else $fatal(1,"Malformed UART response len=%0d first=%02h",nbytes,chars[0]);
            nresp=nresp+1;nbytes=0;
        end
    end

    task automatic send_frame(input [7:0] b,input stop_bit);
        integer n;
        begin
            @(negedge clk);uart_rxd=0;repeat(CPB)@(negedge clk);
            for(n=0;n<8;n=n+1)begin uart_rxd=b[n];repeat(CPB)@(negedge clk);end
            uart_rxd=stop_bit;repeat(CPB)@(negedge clk);uart_rxd=1;
        end
    endtask
    task automatic send_byte(input [7:0] b);send_frame(b,1'b1);endtask
    task automatic wait_resp(input integer n);
        integer guard;
        begin
            guard=0;
            while(nresp<n)begin
                @(negedge clk);guard=guard+1;
                if(guard>CPB*10*400)$fatal(1,"Response timeout want=%0d have=%0d",n,nresp);
            end
        end
    endtask
    task automatic expect_kind(input integer idx,input integer kd);
        if(kind[idx]!==kd)$fatal(1,"Response %0d kind=%0d expected %0d",idx,kind[idx],kd);
    endtask
    task automatic expect_err(input integer idx);expect_kind(idx,3);endtask
    task automatic quiet(input integer frames); // frames 바이트 시간 동안 새 응답 없음
        integer before_n;
        begin
            before_n=nresp;repeat(CPB*10*frames)@(negedge clk);
            if(nresp!=before_n || nbytes!=0)$fatal(1,"Unexpected response during quiet");
        end
    endtask
    task automatic press_reset;
        begin
            @(negedge clk);btnC=1;repeat(100)@(negedge clk);btnC=0;
            nbytes=0;repeat(10)@(negedge clk);
        end
    endtask

    initial begin
        repeat(10)@(negedge clk);btnC=0;repeat(10)@(negedge clk);

        // A1 상태 질의 + CR/LF 무시
        send_byte(8'h0d);send_byte(8'h0a);send_byte("S");
        wait_resp(1);expect_kind(0,1);
        quiet(3);
        $display("A1_STATUS_CRLF_IGNORED_PASS");

        // A2 잘못된 opcode
        base=nresp;
        send_byte("Z");wait_resp(base+1);expect_err(base);
        send_byte("x");wait_resp(base+2);expect_err(base+1);
        send_byte(8'hff);wait_resp(base+3);expect_err(base+2);
        send_byte(8'h00);wait_resp(base+4);expect_err(base+3);
        $display("A2_INVALID_OPCODE_ERR_PASS count=4");

        // A3 잘못된 주소·뉴런·길이 + 경계 유효값
        base=nresp;
        send_byte("X");send_byte(64);send_byte(0);send_byte(1);wait_resp(base+1);expect_err(base);
        send_byte("X");send_byte(63);send_byte(0);send_byte(1);wait_resp(base+2);expect_kind(base+1,2);
        send_byte("W");send_byte(4);send_byte(0);send_byte(0);send_byte(1);wait_resp(base+3);expect_err(base+2);
        send_byte("W");send_byte(0);send_byte(64);send_byte(0);send_byte(1);wait_resp(base+4);expect_err(base+3);
        send_byte("W");send_byte(3);send_byte(63);send_byte(0);send_byte(1);wait_resp(base+5);expect_kind(base+4,2);
        send_byte("B");send_byte(4);for(k=0;k<5;k=k+1)send_byte(0);wait_resp(base+6);expect_err(base+5);
        send_byte("R");send_byte(0);wait_resp(base+7);expect_err(base+6);
        send_byte("R");send_byte(65);wait_resp(base+8);expect_err(base+7);
        send_byte("R");send_byte(8'h80);wait_resp(base+9);expect_err(base+8);
        send_byte("R");send_byte(64);wait_resp(base+10);expect_err(base+9); // 미적재 주소
        if(led[15]!==1'b1)$fatal(1,"Error LED15 not latched");
        $display("A3_INVALID_FIELDS_ERR_PASS err=8 ok_boundary=2");

        // A4 응답 전송 중 다음 명령 수신: '0'~'3' 연속 송신(stop-and-wait 위반) -> FIFO 보관, 순서 보존
        base=nresp;
        send_byte("0");send_byte("1");send_byte("2");send_byte("3");
        wait_resp(base+4);
        for(k=0;k<4;k=k+1)begin
            expect_kind(base+k,4);
            if(cls[base+k]!=k)$fatal(1,"A4 class order k=%0d got=%0d",k,cls[base+k]);
        end
        if(dut.ov||dut.fe)$fatal(1,"A4 overrun/framing");
        $display("A4_COMMAND_DURING_RESPONSE_PASS ordered=4");

        // A5 'S' 20개 연속 폭주: 응답 3바이트 > 입력 1바이트, FIFO 내에서 무손실
        base=nresp;
        for(k=0;k<20;k=k+1)send_byte("S");
        wait_resp(base+20);
        for(k=0;k<20;k=k+1)expect_kind(base+k,1);
        if(dut.ov)$fatal(1,"A5 overrun within FIFO capacity");
        $display("A5_BURST_20_STATUS_NO_LOSS_PASS");

        // A6 응답 송신 중 Reset: TX 즉시 idle, 잔류 응답 없음, 이후 정상
        send_byte("1");
        while(nbytes<10)@(negedge clk);
        press_reset;
        if(uart_txd!==1'b1)$fatal(1,"A6 TX not idle after reset");
        base=nresp;
        quiet(60); // 남은 42바이트가 새어 나오면 실패
        send_byte("S");wait_resp(base+1);expect_kind(base,1);
        send_byte("2");wait_resp(base+2);expect_kind(base+1,4);
        if(cls[base+1]!=2)$fatal(1,"A6 class after reset");
        if(led[15]!==1'b0)$fatal(1,"A6 error LED not cleared by reset");
        $display("A6_RESET_DURING_RESPONSE_PASS");

        // A7 명령 중단(Timeout 관측): 'B'+2바이트 후 중단 -> 응답 없음,
        //    다음 'S'는 페이로드로 소비됨(설계에 RX Timeout 없음). Reset으로만 복구.
        base=nresp;
        send_byte("B");send_byte(1);send_byte(0);
        quiet(50);
        send_byte("S");
        quiet(10);
        if(dut.commands.state!==3'd1)$fatal(1,"A7 expected controller stuck in RECV");
        press_reset;
        send_byte("S");wait_resp(base+1);expect_kind(base,1);
        $display("A7_ABORTED_COMMAND_NO_TIMEOUT_OBSERVED recovered_by_reset=1");

        // A8 Framing error: stop=0 바이트 무시, fe/LED15 래치, 명령 처리 지속
        base=nresp;
        send_frame("S",1'b0);
        quiet(3);
        if(dut.fe!==1'b1 || led[15]!==1'b1)$fatal(1,"A8 framing error not latched");
        send_byte("S");wait_resp(base+1);expect_kind(base,1);
        press_reset;
        $display("A8_FRAMING_ERROR_PASS");

        if(mfe||mov)$fatal(1,"TX monitor framing/overrun");
        $display("NPU_UART_ADVERSARIAL_SIM_PASS tests=8 responses=%0d",nresp);
        $finish;
    end
    initial begin #400000000;$fatal(1,"Adversarial TB global timeout");end
endmodule
