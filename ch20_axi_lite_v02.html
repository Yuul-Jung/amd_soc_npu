03RTL 작성

아래 파일을 경로 그대로 만들고 전체 코드를 입력합니다.

ch20/rtl/npu_axi_lite.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 20장 AXI4-Lite 슬레이브: 19장 연산 코어(NPE=4)를 CPU가 레지스터로 다루게 한다.
// 쓰기 주소(AW)와 쓰기 데이터(W)는 순서와 관계없이 따로 받아 두고, 둘이 모이면 한 번 쓰고 B 응답을 낸다.
// 읽기(AR)는 다음 클록에 R 응답. B·R은 READY가 올 때까지 VALID·데이터를 유지한다 (역압력).
//
// 레지스터 (바이트 주소, 32비트)
//   0x00 CTRL    W  bit0 START(1 쓰면 시작), bit1 CLR(DONE·ERROR 지움)
//   0x04 STATUS  R  bit0 BUSY, bit1 DONE(유지), bit2 ERROR(유지)
//   0x08 CLASS   R  [3:0] 결과 클래스
//   0x0C CYCLES  R  [15:0] 마지막 추론 클록 수
//   0x10 X_ADDR  RW [9:0] 다음 입력 주소 (X_DATA 쓰기마다 +1)
//   0x14 X_DATA  W  [7:0] 입력 값 → x[X_ADDR]
//   0x18 W_ADDR  RW [19:16] 클래스, [9:0] 원소 주소 (W_DATA 쓰기마다 원소 +1)
//   0x1C W_DATA  W  [7:0] 가중치 → w[클래스][원소]
//   0x20 B_SEL   RW [3:0] Bias 클래스 (B_DATA 쓰기마다 +1)
//   0x24 B_DATA  W  [31:0] Bias → b[B_SEL]
//   0x28 ID      R  0x4E505531 ("NPU1")
//   0x2C LENGTH  R  784 (입력 원소 수)
//   0x40~0x64 SCORE0~9 R 점수 (signed 32비트)
// WSTRB: 바이트 단위로 적용. X_DATA·W_DATA는 바이트 0(WSTRB[0])이 켜져 있을 때만 쓴다.
module npu_axi_lite #(
    parameter integer ADDR_WIDTH = 8
) (
    input  wire                  s_axi_aclk,
    input  wire                  s_axi_aresetn,
    input  wire [ADDR_WIDTH-1:0] s_axi_awaddr,
    input  wire                  s_axi_awvalid,
    output wire                  s_axi_awready,
    input  wire           [31:0] s_axi_wdata,
    input  wire            [3:0] s_axi_wstrb,
    input  wire                  s_axi_wvalid,
    output wire                  s_axi_wready,
    output wire            [1:0] s_axi_bresp,
    output reg                   s_axi_bvalid,
    input  wire                  s_axi_bready,
    input  wire [ADDR_WIDTH-1:0] s_axi_araddr,
    input  wire                  s_axi_arvalid,
    output wire                  s_axi_arready,
    output reg            [31:0] s_axi_rdata,
    output wire            [1:0] s_axi_rresp,
    output reg                   s_axi_rvalid,
    input  wire                  s_axi_rready,
    output wire                  irq_o                    // DONE 또는 ERROR (레벨)
);
    wire clk = s_axi_aclk, rn = s_axi_aresetn;
    assign s_axi_bresp = 2'b00;                           // 항상 OKAY
    assign s_axi_rresp = 2'b00;

    // ── 쓰기: AW·W를 따로 받아 둔다 (한 번에 하나씩)
    reg aw_full, w_full;
    reg [ADDR_WIDTH-1:0] aw_q;
    reg [31:0] wd_q; reg [3:0] ws_q;
    assign s_axi_awready = !aw_full && !s_axi_bvalid;
    assign s_axi_wready  = !w_full && !s_axi_bvalid;
    wire do_write = aw_full && w_full && !s_axi_bvalid;

    // ── 읽기: 다음 클록에 응답, RREADY까지 유지
    assign s_axi_arready = !s_axi_rvalid;

    // ── 레지스터
    reg [9:0] x_addr; reg [3:0] w_cls; reg [9:0] w_addr; reg [3:0] b_sel;
    reg done_st, err_st;
    reg start_p, x_we, w_we, b_we;
    reg [7:0] x_d, w_d; reg [31:0] b_d; reg [9:0] x_a, w_a; reg [3:0] w_n, b_n;

    wire busy, done, error; wire [3:0] cls; wire [319:0] scores; wire [15:0] cycles;
    npu_tiled_784_npe #(.NPE(4), .ACC_WIDTH(32)) core (
        .clk(clk), .resetn(rn), .x_wr_en_i(x_we), .x_addr_i(x_a), .x_data_i(x_d),
        .w_wr_en_i(w_we), .w_neuron_i(w_n), .w_addr_i(w_a), .w_data_i(w_d),
        .b_wr_en_i(b_we), .b_neuron_i(b_n), .b_data_i(b_d), .start_i(start_p), .busy_o(busy),
        .scores_o(scores), .done_o(done), .error_o(error), .class_o(cls), .cycles_o(cycles));
    assign irq_o = done_st | err_st;

    // 바이트 단위 WSTRB 적용
    function [31:0] merge(input [31:0] old, input [31:0] nw, input [3:0] st);
        integer k;
        begin
            for (k = 0; k < 4; k = k + 1) merge[k*8 +: 8] = st[k] ? nw[k*8 +: 8] : old[k*8 +: 8];
        end
    endfunction
    wire [5:0] wreg = aw_q[7:2];
    wire [31:0] x_addr_m = merge({22'd0, x_addr}, wd_q, ws_q);
    wire [31:0] w_addr_m = merge({12'd0, w_cls, 6'd0, w_addr}, wd_q, ws_q);
    wire [31:0] b_sel_m  = merge({28'd0, b_sel}, wd_q, ws_q);

    always @(posedge clk) begin
        if (!rn) begin
            aw_full <= 0; w_full <= 0; s_axi_bvalid <= 0; aw_q <= 0; wd_q <= 0; ws_q <= 0;
            x_addr <= 0; w_cls <= 0; w_addr <= 0; b_sel <= 0; done_st <= 0; err_st <= 0;
            start_p <= 0; x_we <= 0; w_we <= 0; b_we <= 0;
            x_d <= 0; w_d <= 0; b_d <= 0; x_a <= 0; w_a <= 0; w_n <= 0; b_n <= 0;
        end else begin
            start_p <= 0; x_we <= 0; w_we <= 0; b_we <= 0;
            if (s_axi_awvalid && s_axi_awready) begin aw_q <= s_axi_awaddr; aw_full <= 1; end
            if (s_axi_wvalid && s_axi_wready) begin wd_q <= s_axi_wdata; ws_q <= s_axi_wstrb; w_full <= 1; end
            if (s_axi_bvalid && s_axi_bready) s_axi_bvalid <= 0;
            if (done) done_st <= 1;
            if (error) err_st <= 1;
            if (do_write) begin
                aw_full <= 0; w_full <= 0; s_axi_bvalid <= 1;
                case (wreg)
                    6'h00: begin
                        if (ws_q[0] && wd_q[0] && !busy) begin start_p <= 1; done_st <= 0; err_st <= 0; end
                        if (ws_q[0] && wd_q[1]) begin done_st <= 0; err_st <= 0; end
                    end
                    6'h04: x_addr <= x_addr_m[9:0];
                    6'h05: if (ws_q[0]) begin x_we <= 1; x_a <= x_addr; x_d <= wd_q[7:0]; x_addr <= x_addr + 10'd1; end
                    6'h06: begin w_cls <= w_addr_m[19:16]; w_addr <= w_addr_m[9:0]; end
                    6'h07: if (ws_q[0]) begin w_we <= 1; w_n <= w_cls; w_a <= w_addr; w_d <= wd_q[7:0]; w_addr <= w_addr + 10'd1; end
                    6'h08: b_sel <= b_sel_m[3:0];
                    6'h09: begin b_we <= 1; b_n <= b_sel; b_d <= merge(32'd0, wd_q, ws_q); b_sel <= b_sel + 4'd1; end
                    default: ;                                // 읽기 전용·없는 주소: 무시 (OKAY)
                endcase
            end
        end
    end

    // 읽기 응답
    wire [5:0] rreg = s_axi_araddr[7:2];
    reg [31:0] rmux;
    always @(*) begin
        case (rreg)
            6'h01: rmux = {29'd0, err_st, done_st, busy};
            6'h02: rmux = {28'd0, cls};
            6'h03: rmux = {16'd0, cycles};
            6'h04: rmux = {22'd0, x_addr};
            6'h06: rmux = {12'd0, w_cls, 6'd0, w_addr};
            6'h08: rmux = {28'd0, b_sel};
            6'h0A: rmux = 32'h4E505531;
            6'h0B: rmux = 32'd784;
            default: rmux = (rreg >= 6'h10 && rreg <= 6'h19) ? scores[(rreg - 6'h10)*32 +: 32] : 32'd0;
        endcase
    end
    always @(posedge clk) begin
        if (!rn) begin s_axi_rvalid <= 0; s_axi_rdata <= 0; end
        else begin
            if (s_axi_arvalid && s_axi_arready) begin s_axi_rvalid <= 1; s_axi_rdata <= rmux; end
            else if (s_axi_rvalid && s_axi_rready) s_axi_rvalid <= 0;
        end
    end
endmodule
`default_nettype wire

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. TB의 AXI 마스터 태스크가 쓰기마다 순서(AW·W 동시, AW 먼저, W 먼저)를 돌려 가며 쓰고, BREADY·RREADY를 0~2클록 무작위로 늦춥니다. 감시기는 VALID 유지·데이터 안정·응답 수를 매 클록 확인합니다. 모델 7,840개와 입력 784개를 모두 레지스터 쓰기로 적재해 2장을 추론합니다.

ch20/tb/tb_npu_axi_lite.sv · systemverilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
// 20장 TB: AXI4-Lite 마스터 태스크로 레지스터를 읽고 쓴다.
//  · 쓰기 순서 3가지(AW·W 동시, AW 먼저, W 먼저)를 번갈아 쓰고, BREADY·RREADY를 무작위로 늦춘다 (역압력)
//  · 감시기: B·R은 READY 전까지 VALID·데이터 유지, B 응답 수 ≤ AW·W 수
//  · WSTRB 부분 쓰기, ID·LENGTH, 실제 MNIST INT8 모델로 추론 2장 (점수·클래스는 Python 비교)
module tb_npu_axi_lite;
    reg clk=0, rn=0;
    reg [7:0] awaddr=0, araddr=0; reg awvalid=0, wvalid=0, bready=0, arvalid=0, rready=0;
    reg [31:0] wdata=0; reg [3:0] wstrb=0;
    wire awready, wready, bvalid, arready, rvalid, irq; wire [1:0] bresp, rresp; wire [31:0] rdata;
    npu_axi_lite dut(.s_axi_aclk(clk), .s_axi_aresetn(rn), .s_axi_awaddr(awaddr), .s_axi_awvalid(awvalid),
        .s_axi_awready(awready), .s_axi_wdata(wdata), .s_axi_wstrb(wstrb), .s_axi_wvalid(wvalid), .s_axi_wready(wready),
        .s_axi_bresp(bresp), .s_axi_bvalid(bvalid), .s_axi_bready(bready), .s_axi_araddr(araddr), .s_axi_arvalid(arvalid),
        .s_axi_arready(arready), .s_axi_rdata(rdata), .s_axi_rresp(rresp), .s_axi_rvalid(rvalid), .s_axi_rready(rready),
        .irq_o(irq));
    always #5 clk = ~clk;

    // ── 감시기
    integer n_aw = 0, n_w = 0, n_b = 0, n_r = 0, viol = 0;
    reg pb = 0, pr = 0; reg [31:0] prd = 0;
    always @(posedge clk) if (rn) begin
        if (awvalid && awready) n_aw = n_aw + 1;
        if (wvalid && wready) n_w = n_w + 1;
        if (bvalid && bready) n_b = n_b + 1;
        if (rvalid && rready) n_r = n_r + 1;
        if (pb && !bvalid) begin viol = viol + 1; $display("VIOLATION BVALID가 BREADY 전에 내려감"); end
        if (pr && (!rvalid || rdata !== prd)) begin viol = viol + 1; $display("VIOLATION R이 RREADY 전에 바뀜"); end
        if (n_b > n_aw || n_b > n_w) begin viol = viol + 1; $display("VIOLATION AW·W 없이 B 응답"); end
        if (bresp !== 2'b00 || rresp !== 2'b00) begin viol = viol + 1; $display("VIOLATION 응답 코드"); end
        pb = bvalid && !bready; pr = rvalid && !rready; prd = rdata;
    end

    reg [15:0] lfsr = 16'h2020;
    function integer rnd(input integer m); begin
        lfsr = {lfsr[14:0], lfsr[15] ^ lfsr[13] ^ lfsr[12] ^ lfsr[10]}; rnd = lfsr % m; end endfunction

    integer mode = 0, modes [0:2];
    // AW·W 채널을 각자 독립된 프로세스로 구동: 한쪽을 gap클록 늦게 올리고, 각자 핸드셰이크 즉시 내린다.
    task drive_aw(input [7:0] a, input integer delay); begin
        repeat (delay) @(negedge clk);
        @(negedge clk); awaddr = a; awvalid = 1;
        @(posedge clk); while (!awready) @(posedge clk);
        @(negedge clk); awvalid = 0;
    end endtask
    task drive_w(input [31:0] d, input [3:0] s, input integer delay); begin
        repeat (delay) @(negedge clk);
        @(negedge clk); wdata = d; wstrb = s; wvalid = 1;
        @(posedge clk); while (!wready) @(posedge clk);
        @(negedge clk); wvalid = 0;
    end endtask
    task axi_write(input [7:0] a, input [31:0] d, input [3:0] s); integer gap; begin
        gap = 1 + rnd(3);
        fork
            drive_aw(a, mode == 2 ? gap : 0);      // mode 2: W 먼저
            drive_w(d, s, mode == 1 ? gap : 0);    // mode 1: AW 먼저
        join
        modes[mode] = modes[mode] + 1; mode = (mode + 1) % 3;
        while (!bvalid) @(negedge clk);
        repeat (rnd(3)) @(negedge clk);                    // BREADY 늦추기
        bready = 1; @(negedge clk); bready = 0;
    end endtask

    task axi_read(input [7:0] a, output [31:0] d); begin
        @(negedge clk); araddr = a; arvalid = 1;
        while (!(arvalid && arready)) @(posedge clk);
        @(negedge clk); arvalid = 0;
        while (!rvalid) @(negedge clk);
        repeat (rnd(3)) @(negedge clk);                    // RREADY 늦추기
        d = rdata; rready = 1; @(negedge clk); rready = 0;
    end endtask

    integer f, o, k, c, v, n, label, cases = 0, ncases, polls;
    reg [31:0] r; string vp, ap;

    initial begin
        modes[0] = 0; modes[1] = 0; modes[2] = 0;
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap) || !$value$plusargs("NCASES=%d", ncases))
            $fatal(1, "인자");
        f = $fopen(vp, "r"); o = $fopen(ap, "w"); if (!f || !o) $fatal(1, "파일 열기");
        repeat (3) @(negedge clk); rn = 1;
        axi_read(8'h28, r); if (r !== 32'h4E505531) $fatal(1, "ID %h", r);
        axi_read(8'h2C, r); if (r !== 784) $fatal(1, "LENGTH %0d", r);
        // WSTRB: 바이트 0만 → 0x055, 바이트 1만 → 0x355, X_DATA를 WSTRB 0으로 쓰면 무시
        axi_write(8'h10, 32'hFFFF_FF55, 4'b0001); axi_write(8'h10, 32'hFFFF_0300, 4'b0010);
        axi_read(8'h10, r); if (r !== 32'h355) $fatal(1, "WSTRB X_ADDR %h", r);
        axi_write(8'h14, 32'h7, 4'b0000); axi_read(8'h10, r); if (r !== 32'h355) $fatal(1, "WSTRB 0 X_DATA가 주소를 바꿈");
        // 미해결 쓰기 2개: AW1·AW2를 먼저 연달아 보내고 W1·W2를 뒤에 보낸다 → 슬레이브는 B1 전까지 AW2를 받지 않아야 함
        fork
            begin drive_aw(8'h10, 0); drive_aw(8'h20, 0); end
            begin drive_w(32'h12, 4'b1111, 3); drive_w(32'h5, 4'b0001, 0); end
            begin repeat (2) begin while (!bvalid) @(negedge clk); repeat (4) @(negedge clk); bready = 1; @(negedge clk); bready = 0; end end
        join
        axi_read(8'h10, r); if (r !== 32'h12) $fatal(1, "미해결 쓰기 1 X_ADDR %h", r);
        axi_read(8'h20, r); if (r !== 32'h5) $fatal(1, "미해결 쓰기 2 B_SEL %h", r);
        // 모델 적재: 클래스마다 W_ADDR 설정 후 W_DATA 784번, B_SEL=0 후 B_DATA 10번
        for (c = 0; c < 10; c = c + 1) begin
            axi_write(8'h18, c << 16, 4'b1111);
            for (k = 0; k < 784; k = k + 1) begin if ($fscanf(f, "%d", v) != 1) $fatal(1, "w"); axi_write(8'h1C, v, 4'b0001); end
        end
        axi_read(8'h18, r); if (r !== {12'd0, 4'd9, 6'd0, 10'd784}) $fatal(1, "W_ADDR 자동 증가 %h", r);
        axi_write(8'h20, 0, 4'b0001);
        for (c = 0; c < 10; c = c + 1) begin if ($fscanf(f, "%d", v) != 1) $fatal(1, "b"); axi_write(8'h24, v, 4'b1111); end
        while (cases < ncases && $fscanf(f, "%d %d", n, label) == 2) begin
            axi_write(8'h10, 0, 4'b1111);
            for (k = 0; k < 784; k = k + 1) begin if ($fscanf(f, "%d", v) != 1) $fatal(1, "x"); axi_write(8'h14, v, 4'b0001); end
            axi_write(8'h00, 1, 4'b0001);                         // START
            polls = 0;
            do begin axi_read(8'h04, r); polls = polls + 1; end while (!r[1] && !r[2]);
            if (r[2] || !irq) $fatal(1, "case %0d: ERROR 또는 irq 없음", n);
            $fwrite(o, "%0d", n);
            for (c = 0; c < 10; c = c + 1) begin axi_read(8'h40 + 4 * c, r); $fwrite(o, " %0d", $signed(r)); end
            axi_read(8'h08, r); $fwrite(o, " %0d\n", r);
            axi_read(8'h0C, r); if (r !== 7185) $fatal(1, "CYCLES %0d", r);
            axi_write(8'h00, 2, 4'b0001); axi_read(8'h04, r);     // CLR → DONE 0
            if (r[1] || irq) $fatal(1, "CLR 후 DONE 남음");
            cases = cases + 1;
        end
        if (viol != 0) $fatal(1, "AXI 규칙 위반 %0d", viol);
        $display("AXI_PROTOCOL_PASS writes=%0d reads=%0d aw_first=%0d w_first=%0d same=%0d violations=0", n_b, n_r, modes[1], modes[2], modes[0]);
        $display("RTL_SIM_PASS NPU_AXI_LITE cases=%0d wstrb=3 outstanding=2 id=NPU1 cycles=7185", cases);
        $fclose(o); $finish;
    end
    initial begin #900000000; $fatal(1, "시간 초과"); end
endmodule

06결과 판정

확인 항목의미
AXI_PROTOCOL_PASS writes=9440 … violations=0쓰기 9,440번(순서 3가지 각 3,146번)과 읽기 3,623번에서 VALID가 READY 전에 내려가거나 데이터가 바뀐 적 없음
RTL_SIM_PASS NPU_AXI_LITE cases=2 wstrb=3 outstanding=2ID·LENGTH, WSTRB 부분 쓰기 3종, AW 두 개를 W보다 먼저 보낸 미해결 쓰기, CLR 후 DONE·irq 0
PYTHON_RTL_COMPARE_PASS npe=4 cases=2 scores=20레지스터로 읽은 점수 20개와 클래스가 Python과 같음

실패 시: TB의 $fatal 메시지에 표시된 cycle 번호와 기대값/실제값을 확인하고, 03절 코드와 한 줄씩 대조합니다. 하나라도 실패하면 다음 장으로 넘어가지 않습니다.

07코드 읽기 포인트

위치내용
레지스터 맵0x00 CTRL(START·CLR) · 0x04 STATUS(BUSY·DONE·ERROR) · 0x08 CLASS · 0x0C CYCLES · 0x10/0x14 X_ADDR/X_DATA · 0x18/0x1C W_ADDR/W_DATA · 0x20/0x24 B_SEL/B_DATA · 0x28 ID=0x4E505531 · 0x2C LENGTH=784 · 0x40~0x64 SCORE0~9. 22장 드라이버가 이 표를 그대로 씁니다.
AW·W 독립aw_full·w_full로 따로 받아 두고, 둘 다 찼을 때 한 번 씁니다. B 응답 중이거나 이미 받아 둔 채널은 READY를 내립니다. 그래서 AW를 두 번 먼저 보내도 둘째는 첫 B가 끝날 때까지 기다립니다.
역압력BVALID·RVALID는 READY가 올 때까지 유지되고 RDATA도 바뀌지 않습니다. 감시기가 이것을 매 클록 확인합니다.
WSTRB주소 레지스터는 켜진 바이트만 바꿉니다(merge). X_DATA·W_DATA는 바이트 0이 꺼져 있으면 쓰지 않고 주소도 늘리지 않습니다.
자동 증가X_DATA·W_DATA·B_DATA를 쓸 때마다 주소가 1씩 늘어 784개를 주소 재설정 없이 연속으로 씁니다. 연산 중 쓰기는 19장 코어가 무시합니다.
인터럽트irq_o = DONE | ERROR(레벨)입니다. 21장 블록 설계에서 인터럽트 제어기에 연결할 수 있게 내 두었습니다. 22장 드라이버는 STATUS 폴링을 기본으로 씁니다.
