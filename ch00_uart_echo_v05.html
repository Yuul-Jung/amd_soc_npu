// (추가) 8N1 수신. RX 2FF, start midpoint/bit center, hold/overrun.
module uart_rx #(
    parameter integer CLK_HZ=100000000, BAUD=115200
)(
    input wire clk,resetn,rx_i,
    output reg [7:0] out_data_o,
    output reg out_valid_o,
    input wire out_ready_i,
    output reg framing_error_o,overrun_o
);
    localparam integer CPB=(CLK_HZ+BAUD/2)/BAUD;
    localparam integer HALF=CPB/2;
    localparam [1:0] IDLE=0,START=1,DATA=2,STOP=3;
    reg [1:0] state;
    reg rx_meta,rx_sync;
    reg [31:0] count_q;
    reg [2:0] bit_q;
    reg [7:0] shift_q;
    always @(posedge clk)begin
        if(!resetn)begin
            state<=IDLE;rx_meta<=1;rx_sync<=1;count_q<=0;bit_q<=0;shift_q<=0;
            out_data_o<=0;out_valid_o<=0;framing_error_o<=0;overrun_o<=0;
        end else begin
            rx_meta<=rx_i;rx_sync<=rx_meta;
            if(out_valid_o && out_ready_i)out_valid_o<=0;
            case(state)
                IDLE:if(!rx_sync)begin count_q<=HALF-1;state<=START;end
                START:if(count_q==0)begin
                    if(!rx_sync)begin count_q<=CPB-1;bit_q<=0;state<=DATA;end
                    else state<=IDLE;
                end else count_q<=count_q-1'b1;
                DATA:if(count_q==0)begin
                    shift_q[bit_q]<=rx_sync;count_q<=CPB-1;
                    if(bit_q==7)state<=STOP;
                    else bit_q<=bit_q+1'b1;
                end else count_q<=count_q-1'b1;
                STOP:if(count_q==0)begin
                    state<=IDLE;
                    if(!rx_sync)framing_error_o<=1;
                    else if(!out_valid_o || out_ready_i)begin
                        out_data_o<=shift_q;out_valid_o<=1;
                    end else overrun_o<=1;
                end else count_q<=count_q-1'b1;
                default:state<=IDLE;
            endcase
        end
    end
    // synthesis translate_off
    initial if(CPB<4)$fatal(1,"Unsupported UART clock/baud ratio");
    // synthesis translate_on
endmodule

// (추가) 8N1 송신. valid/ready handshake, 10bit start/data/stop.
module uart_tx #(
    parameter integer CLK_HZ=100000000, BAUD=115200
)(
    input wire clk,resetn,
    input wire [7:0] in_data_i,
    input wire in_valid_i,
    output wire in_ready_o,
    output reg tx_o
);
    localparam integer CPB=(CLK_HZ+BAUD/2)/BAUD;
    reg sending;
    reg [31:0] tick_q;
    reg [3:0] bit_q;
    reg [9:0] frame_q;
    assign in_ready_o=resetn && !sending;
    always @(posedge clk)begin
        if(!resetn)begin
            sending<=0;tick_q<=0;bit_q<=0;frame_q<=10'h3ff;tx_o<=1;
        end else if(!sending)begin
            tx_o<=1;
            if(in_valid_i)begin
                sending<=1;tick_q<=CPB-1;bit_q<=0;
                frame_q<={1'b1,in_data_i,1'b0};tx_o<=0;
            end
        end else if(tick_q!=0)tick_q<=tick_q-1'b1;
        else begin
            tick_q<=CPB-1;frame_q<={1'b1,frame_q[9:1]};
            if(bit_q==9)begin sending<=0;tx_o<=1;end
            else begin bit_q<=bit_q+1'b1;tx_o<=frame_q[1];end
        end
    end
    // synthesis translate_off
    initial if(CPB<4)$fatal(1,"Unsupported UART clock/baud ratio");
    // synthesis translate_on
endmodule
common/rtl/uart_core.v · verilog · 원본 · RTL_SIM 통과 2026-11-18
// (변경) 8N1 PHY 내부에 깊이 16 RX/TX FIFO를 넣어 byte stream ready/valid를 보존.
// 1바이트 UART 수신 레지스터와 FIFO는 별개. Controller는 stop-and-wait 정책을 유지.
module uart_core #(
    parameter integer CLK_HZ=100000000,BAUD=115200
)(
    input wire clk,resetn,uart_rxd_i,
    output wire uart_txd_o,
    output wire [7:0] rx_out_data,
    output wire rx_out_valid,
    input wire rx_out_ready,
    input wire [7:0] tx_in_data,
    input wire tx_in_valid,
    output wire tx_in_ready,
    output wire rx_framing_error_o,rx_overrun_o
);
    wire [7:0] phy_rx_data;
    wire phy_rx_valid,phy_rx_ready;
    wire [7:0] phy_tx_data;
    wire phy_tx_valid,phy_tx_ready;
    uart_rx #(.CLK_HZ(CLK_HZ),.BAUD(BAUD)) rx(
        .clk(clk),.resetn(resetn),.rx_i(uart_rxd_i),
        .out_data_o(phy_rx_data),.out_valid_o(phy_rx_valid),
        .out_ready_i(phy_rx_ready),.framing_error_o(rx_framing_error_o),
        .overrun_o(rx_overrun_o));
    uart_tx #(.CLK_HZ(CLK_HZ),.BAUD(BAUD)) tx(
        .clk(clk),.resetn(resetn),.in_data_i(phy_tx_data),
        .in_valid_i(phy_tx_valid),.in_ready_o(phy_tx_ready),.tx_o(uart_txd_o));

    reg [7:0] rx_fifo[0:15];
    reg [3:0] rx_wptr,rx_rptr;
    reg [4:0] rx_count;
    wire rx_pop=rx_out_valid && rx_out_ready;
    wire rx_push=phy_rx_valid && phy_rx_ready;
    assign rx_out_valid=(rx_count!=0);
    assign rx_out_data=rx_fifo[rx_rptr];
    assign phy_rx_ready=resetn && ((rx_count<16) || rx_pop);

    reg [7:0] tx_fifo[0:15];
    reg [3:0] tx_wptr,tx_rptr;
    reg [4:0] tx_count;
    wire tx_pop=phy_tx_ready && phy_tx_valid;
    wire tx_push=tx_in_valid && tx_in_ready;
    assign tx_in_ready=resetn && ((tx_count<16) || tx_pop);
    assign phy_tx_valid=(tx_count!=0);
    assign phy_tx_data=tx_fifo[tx_rptr];

    always @(posedge clk)begin
        if(!resetn)begin
            rx_wptr<=0;rx_rptr<=0;rx_count<=0;
            tx_wptr<=0;tx_rptr<=0;tx_count<=0;
        end else begin
            if(rx_push)begin
                rx_fifo[rx_wptr]<=phy_rx_data;
                rx_wptr<=rx_wptr+1'b1;
            end
            if(rx_pop)rx_rptr<=rx_rptr+1'b1;
            case({rx_push,rx_pop})
                2'b10:rx_count<=rx_count+1'b1;
                2'b01:rx_count<=rx_count-1'b1;
                default:begin end
            endcase
            if(tx_push)begin
                tx_fifo[tx_wptr]<=tx_in_data;
                tx_wptr<=tx_wptr+1'b1;
            end
            if(tx_pop)tx_rptr<=tx_rptr+1'b1;
            case({tx_push,tx_pop})
                2'b10:tx_count<=tx_count+1'b1;
                2'b01:tx_count<=tx_count-1'b1;
                default:begin end
            endcase
        end
    end
endmodule

테스트벤치는 출력을 자동 판정합니다. 기준 모델(Python)은 시험 벡터와 정답을 만듭니다.

ch00/tb/tb_uart_echo.sv · systemverilog · 원본 · RTL_SIM 통과 2026-11-18
// (추가) UART 실핀 비트파형 8N1 입력 및 TX 재복조 자동 판정.
`timescale 1ns/1ps
module tb_uart_echo;
    localparam integer CPB=868;
    reg clk=0,resetn=0,rxd=1;
    wire txd;
    wire [7:0] decoded;
    wire decoded_valid,fe,ov;
    integer observed=0;
    always #5 clk=~clk;
    basys3_uart_echo_top dut(.clk(clk),.resetn(resetn),.uart_rxd(rxd),.uart_txd(txd));
    uart_rx monitor(.clk(clk),.resetn(resetn),.rx_i(txd),
        .out_data_o(decoded),.out_valid_o(decoded_valid),
        .out_ready_i(1'b1),.framing_error_o(fe),.overrun_o(ov));
    task automatic send_byte(input [7:0] b);
        integer k;
        begin
            @(negedge clk);rxd=0;repeat(CPB)@(negedge clk);
            for(k=0;k<8;k=k+1)begin
                rxd=b[k];repeat(CPB)@(negedge clk);
            end
            rxd=1;repeat(CPB)@(negedge clk);
        end
    endtask
    always @(posedge clk)if(decoded_valid)begin
        if(observed==0 && decoded!==8'h41)$fatal(1,"Echo A mismatch");
        if(observed==1 && decoded!==8'h35)$fatal(1,"Echo 5 mismatch");
        observed=observed+1;
    end
    initial begin
        repeat(8)@(negedge clk);resetn=1;
        send_byte("A");send_byte("5");
        wait(observed==2);
        if(fe||ov)$fatal(1,"Echo framing/overrun");
        $display("UART_ECHO_RTL_SIM_PASS decoded=2");$finish;
    end
    initial begin #2000000;$fatal(1,"UART Echo TIMEOUT");end
endmodule

07코드 읽기 포인트

위치내용
CPB 계산CPB=(CLK_HZ+BAUD/2)/BAUD는 반올림 나누기입니다. 100 MHz/115200 bps = 868.056 → 868 클록/비트. CPB<4이면 $fatal으로 합성 전에 차단합니다.
RX 2FF 동기화rx_meta←rx_i; rx_sync←rx_meta로 2단계 플립플롭을 거쳐 메타스태빌리티를 억제합니다. 이후 FSM은 rx_sync만 참조합니다.
Start bit midpointIDLE에서 falling edge 감지 후 HALF-1 클록 대기합니다. START에서 중앙에서 rx_sync를 재확인해 글리치를 걸러내고 DATA로 진입합니다.
FIFO 포인터rx_wptr/rx_rptr는 4비트 자연 wrap-around입니다. rx_count는 5비트(0~16)로 Full(16)/Empty(0) 구분합니다.
Echo controller조합 로직만으로 RX→TX를 그대로 연결합니다. rx_ready_o=tx_ready_i가 성립하면 반드시 backpressure가 전파됩니다.

proc worst_slack {kind} {
    set p [get_timing_paths -quiet -$kind -max_paths 1 -nworst 1]
    if {[llength $p] == 0} { return NA }
    return [format %.3f [get_property SLACK $p]]
}
proc count_ref {pattern} {
    return [llength [get_cells -quiet -hierarchical -filter "IS_PRIMITIVE && REF_NAME =~ $pattern"]]
}

foreach d $designs {
    lassign $d ch top files generics
    if {[llength $selected] > 0 && [lsearch -exact $selected $ch] < 0} { continue }
    set gargs {}
    foreach gv $generics { lappend gargs -generic $gv }
    set tag_g [expr {[llength $generics] ? "_[join $generics _]" : ""}]
    set cdir [file join $outdir $ch$tag_g]
    file mkdir $cdir
    set rc [catch {
        create_project -in_memory -part $part
        foreach f $files { read_verilog [file join $root $f] }
        set xdc [file join $cdir ooc.xdc]
        set x [open $xdc w]
        puts $x "create_clock -name clk -period $period \[get_ports -quiet {clk s_axi_aclk}\]"
        close $x
        read_xdc $xdc
        synth_design -top $top -part $part -mode out_of_context {*}$gargs
        report_utilization -file [file join $cdir synth_util.rpt]
        opt_design
        place_design
        route_design
        report_utilization -file [file join $cdir route_util.rpt]
        report_timing_summary -max_paths 10 -file [file join $cdir timing_route.rpt]
        set wns [worst_slack setup]
        set whs [worst_slack hold]
        set res "LUT=[count_ref LUT*] FF=[count_ref FD*] DSP48E1=[count_ref DSP48E1] RAMB=[count_ref RAMB*]"
        set ok [expr {($wns eq "NA" || $wns >= 0) && ($whs eq "NA" || $whs >= 0)}]
        set tag [expr {$ok ? "SYNTH_TIMING_PASS" : "SYNTH_TIMING_FAIL"}]
        set line "$tag $ch $top[expr {[llength $generics] ? " [join $generics ,]" : ""}] WNS=$wns WHS=$whs $res"
        close_project
    } err]
    if {$rc} {
        set line "SYNTH_TIMING_FAIL $ch $top ERROR=[string map {\n { }} $err]"
        catch {close_project}
    }
    if {[string match SYNTH_TIMING_FAIL* $line]} { incr fail }
    puts $line
    puts $fh $line
    flush $fh
}
close $fh
puts "OOC_SUMMARY: $summary"

tools/run_vivado_all.sh · bash · 원본 · ⚠ 미검증 (Vivado 미실행)
#!/usr/bin/env bash
# VS Code(Vivado 설치 PC)에서 00~13장 합성·타이밍·타이밍 시뮬레이션을 한 번에 실행하고
# 각 장 validation/ 에 증거 로그를 남긴다. 보드 확인(BOARD_PASS)과는 별개다.
#   source /tools/Xilinx/Vivado/<버전>/settings64.sh
#   bash tools/run_vivado_all.sh            # 전체
#   bash tools/run_vivado_all.sh ooc        # 00~10·13장 OOC만
#   bash tools/run_vivado_all.sh impl       # 12장 구현 + 타이밍 시뮬레이션만
set -Eeuo pipefail
root="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/.." && pwd)"
cd "$root"
command -v vivado >/dev/null 2>&1 || { echo "VIVADO_NOT_RUN: vivado not found (source settings64.sh)" >&2; exit 2; }
what="${1:-all}"
day="$(date +%Y%m%d)"
logs="$root/build/vivado/logs"; mkdir -p "$logs"
clean() { sed -e "s#$root#<repo>#g" -e "s#$HOME#~#g" "$1"; }
status=0

if [ "$what" = all ] || [ "$what" = ooc ]; then
    vivado -mode batch -nojournal -log "$logs/ooc.log" -source tools/vivado/ooc_timing.tcl || status=1
    seen=" "
    while read -r tag ch top gen rest; do
        case "$tag" in SYNTH_TIMING_*) ;; *) continue ;; esac
        # 매개변수 줄(예: NPE=1)은 build/vivado/chNN_NPE=1/ 에 보고서가 있고, 같은 장 증거 파일에 이어 쓴다.
        dir=$ch; case "$gen" in WNS=*) ;; *=*) dir="${ch}_${gen//,/_}" ;; esac
        out="$ch/validation/vivado_ooc_$day.txt"
        case "$seen" in *" $ch "*) ;; *) head -1 build/vivado/ooc_summary.txt | clean /dev/stdin > "$out"; seen="$seen$ch " ;; esac
        { echo "$tag $ch $top $gen $rest"; echo;
          sed -n '/Design Timing Summary/,/^$/p;/WNS(ns)/,+3p' "build/vivado/$dir/timing_route.rpt" 2>/dev/null || true; } \
          | clean /dev/stdin >> "$out"
    done < build/vivado/ooc_summary.txt
    cat build/vivado/ooc_summary.txt
fi

if [ "$what" = all ] || [ "$what" = impl ]; then
    if vivado -mode batch -nojournal -log "$logs/impl.log" -source mini_npu/scripts/vivado_impl.tcl; then
        bash mini_npu/scripts/run_timing_sim.sh > "$logs/timesim.log" 2>&1 || status=1
    else
        status=1
    fi
    { cat mini_npu/build/impl/impl_summary.txt 2>/dev/null || true; echo;
      sed -n '/WNS(ns)/,+3p' mini_npu/build/impl/timing_route.rpt 2>/dev/null || true; echo;
      if [ -f mini_npu/build/impl/basys3_npu_top.bit ]; then sha256sum mini_npu/build/impl/basys3_npu_top.bit; fi;
      echo; echo "## run_timing_sim.sh"; grep -E '_PASS|_FAIL|NOT_RUN|Error|FATAL' "$logs/timesim.log" 2>/dev/null || true; } \
      | clean /dev/stdin > ch12/validation/vivado_impl_timesim_$day.txt
    cat ch12/validation/vivado_impl_timesim_$day.txt
fi

[ $status -eq 0 ] && echo "VIVADO_ALL_PASS ($what)" || echo "VIVADO_ALL_FAIL ($what) — build/vivado/logs 확인"
exit $status
