`timescale 1ns/1ps
`default_nettype none
// 29장 센서 감시기: 가상 센서 4채널 × 16샘플(12비트)을 받아 유효성(범위·고착·순서)을 먼저 보고,
// 고장이 없을 때만 15장 Dense 64→4로 상태를 분류한다. 점수 1·2위 차가 MARGIN보다 작으면 UNKNOWN.
// 상태: 0 NORMAL, 1 HOT, 2 VIBRATION, 3 DRIFT, 4 UNKNOWN, 5 FAULT (고장이 AI보다 우선)
module sensor_guard_ai #(
    parameter integer MIN_RAW = 100,
    parameter integer MAX_RAW = 4000,
    parameter integer STUCK_N = 8,           // 같은 채널이 같은 값을 연속 STUCK_N번 → 고착
    parameter integer MARGIN  = 3000
) (
    input  wire         clk,
    input  wire         resetn,
    input  wire         s_valid_i,           // 샘플 하나 (채널 0,1,2,3 순서로 16번 = 창 1개)
    input  wire   [1:0] s_ch_i,
    input  wire  [11:0] s_raw_i,
    input  wire         w_wr_en_i,           // 15장 Dense 모델 쓰기 (클래스 0~3, 원소 0~63)
    input  wire   [3:0] w_neuron_i,
    input  wire   [5:0] w_addr_i,
    input  wire   [7:0] w_data_i,
    input  wire         b_wr_en_i,
    input  wire  [31:0] b_data_i,
    output reg    [2:0] state_o,
    output reg    [2:0] fault_o,             // bit0 범위, bit1 고착, bit2 순서
    output wire [127:0] scores_o,
    output reg          done_o,              // 창 하나 판정 끝 1클록 펄스
    output wire         busy_o
);
    localparam COLLECT=0, CLASSIFY=1, DECIDE=2;
    reg [1:0] st;
    reg dense_started;                       // CLASSIFY에서 Dense를 한 번만 시작
    reg [5:0] idx;
    reg signed [7:0] x_q [0:63];
    reg [11:0] prev [0:3];
    reg [3:0]  run  [0:3];
    integer i;
    assign busy_o = (st == CLASSIFY || st == DECIDE);

    // ── 샘플 하나 판정 (채널은 '와야 할' 채널 idx[1:0] 기준으로 센다 — 순서가 틀려도 위치로 비교)
    wire [1:0] ch = idx[1:0];
    wire bad_order = (s_ch_i != ch);
    wire bad_range = (s_raw_i < MIN_RAW) || (s_raw_i > MAX_RAW);
    wire same = (idx >= 4) && (s_raw_i == prev[ch]);
    wire [3:0] run_n = same ? run[ch] + 4'd1 : 4'd1;
    wire signed [12:0] centered = $signed({1'b0, s_raw_i}) - 13'sd2048;
    wire signed [12:0] shifted = centered >>> 4;
    wire signed [7:0] scaled = (shifted > 127) ? 8'sd127 : (shifted < -128) ? -8'sd128 : shifted[7:0];

    // ── 15장 Dense 64→4 (입력 = 창 버퍼, 1클록 지연)
    wire x_en; wire [5:0] x_idx; reg signed [7:0] x_d; reg x_v;
    wire d_done, d_err;
    npu_dense #(.N_IN(64), .N_OUT(4), .ACC_WIDTH(32)) dense (
        .clk(clk), .resetn(resetn), .model_clear_i(1'b0),
        .w_wr_en_i(w_wr_en_i), .w_neuron_i(w_neuron_i), .w_addr_i(w_addr_i), .w_data_i(w_data_i),
        .b_wr_en_i(b_wr_en_i), .b_neuron_i(w_neuron_i), .b_data_i(b_data_i), .model_sum_o(), .model_err_o(),
        .x_rd_en_o(x_en), .x_rd_idx_o(x_idx), .x_data_i(x_d), .x_valid_i(x_v),
        .start_i(st == CLASSIFY && !dense_started && fault_o == 0), .busy_o(), .done_o(d_done), .error_o(d_err), .scores_o(scores_o));
    always @(posedge clk) begin x_v <= resetn && x_en; if (x_en) x_d <= x_q[x_idx]; end

    // ── 1·2위 찾기 (4개, 같으면 낮은 번호가 위)
    wire signed [31:0] s0 = scores_o[0 +: 32], s1 = scores_o[32 +: 32], s2 = scores_o[64 +: 32], s3 = scores_o[96 +: 32];
    wire [1:0] b01 = (s1 > s0) ? 2'd1 : 2'd0;  wire signed [31:0] v01 = (s1 > s0) ? s1 : s0;  wire signed [31:0] l01 = (s1 > s0) ? s0 : s1;
    wire [1:0] b23 = (s3 > s2) ? 2'd3 : 2'd2;  wire signed [31:0] v23 = (s3 > s2) ? s3 : s2;  wire signed [31:0] l23 = (s3 > s2) ? s2 : s3;
    wire top23 = (v23 > v01);
    wire [1:0] best = top23 ? b23 : b01;
    wire signed [31:0] bestv = top23 ? v23 : v01;
    wire signed [31:0] second = top23 ? ((v01 > l23) ? v01 : l23) : ((v23 > l01) ? v23 : l01);

    always @(posedge clk) begin
        if (!resetn) begin
            st <= COLLECT; idx <= 0; state_o <= 0; fault_o <= 0; done_o <= 0; dense_started <= 0;
            for (i = 0; i < 4; i = i + 1) begin prev[i] <= 0; run[i] <= 0; end
        end else begin
            done_o <= 0;
            case (st)
                COLLECT: if (s_valid_i) begin
                    x_q[idx] <= scaled;
                    prev[ch] <= s_raw_i; run[ch] <= run_n;
                    fault_o <= (idx == 0 ? 3'd0 : fault_o) | {bad_order, run_n >= STUCK_N, bad_range};
                    idx <= idx + 6'd1;
                    if (idx == 63) st <= CLASSIFY;
                end
                CLASSIFY: begin
                    if (fault_o != 0) begin state_o <= 3'd5; done_o <= 1; st <= COLLECT; end   // 고장: 분류하지 않음
                    else begin
                        dense_started <= 1;
                        if (d_err) begin state_o <= 3'd5; fault_o <= 3'd4; done_o <= 1; dense_started <= 0; st <= COLLECT; end
                        else if (d_done) begin dense_started <= 0; st <= DECIDE; end
                    end
                end
                DECIDE: begin
                    state_o <= (bestv - second >= MARGIN) ? {1'b0, best} : 3'd4;
                    done_o <= 1; st <= COLLECT;
                end
                default: st <= COLLECT;
            endcase
        end
    end
endmodule

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. 가상 센서 창 12개: 상태별 2개씩, 점수 차가 작은 섞인 경우 1개, 범위 밖 값(50), 채널 3이 같은 값 9번(고착), 채널 순서 뒤바뀜. 샘플 사이에 빈 클록을 섞습니다. 합성 시험 400개 정확도는 100%였는데, 합성 데이터가 쉬워서이며 실제 센서 성능을 뜻하지 않습니다.

    wire [2:0] state, fault; wire [127:0] scores; wire done, busy;
    sensor_guard_ai dut(.clk(clk), .resetn(rn), .s_valid_i(sv), .s_ch_i(sch), .s_raw_i(sraw), .w_wr_en_i(wwe),
        .w_neuron_i(wn), .w_addr_i(wa), .w_data_i(wd), .b_wr_en_i(bwe), .b_data_i(bd),

    integer f, o, n, k, c, v, r, cases = 0; reg [8*16-1:0] name; string vp, ap;
    initial begin
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap)) $fatal(1, "경로 인자");
        f = $fopen(vp, "r"); o = $fopen(ap, "w"); if (!f || !o) $fatal(1, "파일 열기");
        repeat (3) @(negedge clk); rn = 1;
        for (c = 0; c < 4; c = c + 1) for (k = 0; k < 64; k = k + 1) begin
            if ($fscanf(f, "%d", v) != 1) $fatal(1, "w"); @(negedge clk); wwe = 1; wn = c; wa = k; wd = v; end
        @(negedge clk); wwe = 0;
        for (c = 0; c < 4; c = c + 1) begin if ($fscanf(f, "%d", v) != 1) $fatal(1, "b"); @(negedge clk); bwe = 1; wn = c; bd = v; end
        @(negedge clk); bwe = 0;

            for (k = 0; k < 64; k = k + 1) begin

#!/usr/bin/env python3
"""29장 기준 모델: 가상 센서 4채널 × 16샘플(12비트) → 유효성 검사(범위·고착·순서) → INT8 스케일 → Dense 64→4 → 판정.

상태: 0 NORMAL, 1 HOT(채널0 상승), 2 VIBRATION(채널1 진동), 3 DRIFT(채널2 천천히 벗어남), 4 UNKNOWN(점수 차 < MARGIN), 5 FAULT.
모델은 합성 데이터(이 파일이 만든 가상 센서 값)로 학습한 소프트맥스 회귀이며, 실제 센서 데이터가 아니다.
--train: 학습·양자화해 model/sensor_int8.txt 저장 (numpy 필요) / --out·--actual: RTL 시험 벡터·비교 (표준 라이브러리)."""
from __future__ import annotations
import argparse
import math
import random
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "host_tools"))
from p3_integer_reference import dot_signed               # noqa: E402

MODEL = ROOT / "ch29" / "model" / "sensor_int8.txt"
MIN_RAW, MAX_RAW, STUCK_N, MARGIN = 100, 4000, 8, 3000
NAMES = ["NORMAL", "HOT", "VIBRATION", "DRIFT", "UNKNOWN", "FAULT"]


def scenario(kind, rng, mix=None):
    """16×4 원시값 (t 우선: 인덱스 = t*4 + ch)."""
    out = []
    for t in range(16):
        v = [2048 + rng.randint(-40, 40) for _ in range(4)]
        if kind == 1: v[0] += 60 * t                          # 온도 상승
        if kind == 2: v[1] += int(700 * math.sin(t * 1.9))     # 진동
        if kind == 3: v[2] -= 25 * t * t // 4                  # 느린 이탈 (가속)
        if mix: v[0] += int(mix[0] * 60 * t); v[1] += int(mix[1] * 700 * math.sin(t * 1.9))
        out += v
    return out


def scale(raw):
    return max(-128, min(127, (raw - 2048) >> 4))


def guard(raw, chans=None):
    """고장 비트: bit0 범위, bit1 고착(같은 채널 같은 값 8번 연속), bit2 순서."""
    fault = 0
    chans = chans or [i % 4 for i in range(64)]
    if any(c != i % 4 for i, c in enumerate(chans)): fault |= 4
    if any(not MIN_RAW <= r <= MAX_RAW for r in raw): fault |= 1
    for ch in range(4):
        run = 1
        for t in range(1, 16):
            run = run + 1 if raw[t * 4 + ch] == raw[(t - 1) * 4 + ch] else 1
            if run >= STUCK_N: fault |= 2
    return fault


def decide(raw, w, b, chans=None):
    f = guard(raw, chans)
    if f: return 5, f, [0] * 4
    x = [scale(r) for r in raw]
    s = [dot_signed(x, row) + bb for row, bb in zip(w, b)]
    order = sorted(range(4), key=lambda i: (-s[i], i))
    return (order[0] if s[order[0]] - s[order[1]] >= MARGIN else 4), 0, s


def train():
    import numpy as np
    rng = random.Random(29)
    X, Y = [], []
    for _ in range(2000):
        k = rng.randrange(4)
        X.append([scale(r) / 128 for r in scenario(k, rng)]); Y.append(k)
    X, Y = np.array(X, np.float32), np.array(Y)
    W = np.zeros((64, 4), np.float32); B = np.zeros(4, np.float32); oh = np.eye(4, dtype=np.float32)[Y]
    for _ in range(300):
        z = X @ W + B; z -= z.max(1, keepdims=True); p = np.exp(z); p /= p.sum(1, keepdims=True)
        g = (p - oh) / len(X); W -= 2.0 * X.T @ g; B -= 2.0 * g.sum(0)
    s = float(np.abs(W).max()) / 127
    wq = np.clip(np.rint(W / s), -128, 127).astype(int).T
    bq = np.rint(B / (s / 128)).astype(int)                 # x = scale/128 단위 → 정수 점수 단위
    MODEL.parent.mkdir(exist_ok=True)
    MODEL.write_text("\n".join(" ".join(map(str, r)) for r in wq) + "\n" + " ".join(map(str, bq)) + "\n")
    rng = random.Random(2929); right = 0
    w, b = load()
    for _ in range(400):
        k = rng.randrange(4); st, _, _ = decide(scenario(k, rng), w, b); right += st == k
    print(f"SENSOR_MODEL_TRAINED synthetic_test_accuracy={right / 4:.1f}% (합성 데이터 400개, UNKNOWN은 틀림으로 셈)")


def load():
    rows = [list(map(int, l.split())) for l in MODEL.read_text().splitlines()]
    return rows[:4], rows[4]


def cases():
    rng = random.Random(299)
    out = [(f"{NAMES[k]}_{i}", scenario(k, rng), None) for k in range(4) for i in range(2)]
    w, b = load()
    for _ in range(500):                                     # 점수 차가 작은 섞인 경우를 하나 찾는다
        raw = scenario(0, rng, mix=(rng.random(), rng.random()))
        if decide(raw, w, b)[0] == 4:
            out.append(("ambiguous", raw, None)); break
    raw = scenario(0, rng); raw[17] = 50; out.append(("range_low", raw, None))
    raw = scenario(0, rng)
    for t in range(4, 13): raw[t * 4 + 3] = 2100
    out.append(("stuck_ch3", raw, None))
    ch = [i % 4 for i in range(64)]; ch[30], ch[31] = ch[31], ch[30]
    out.append(("order", scenario(0, rng), ch))
    return out


def main():
    ap = argparse.ArgumentParser(); ap.add_argument("--train", action="store_true"); ap.add_argument("--out"); ap.add_argument("--actual")
    a = ap.parse_args()
    if a.train:
        train(); return
    w, b = load(); cs = cases()
    got = [decide(r, w, b, c)[0] for _, r, c in cs]
    assert set(got) == {0, 1, 2, 3, 4, 5}, got
    print(f"PYTHON_SENSOR_SELF_TEST_PASS cases={len(cs)} states={'/'.join(NAMES[s] for s in sorted(set(got)))}")
    if a.out and not a.actual:
        with open(a.out, "w") as f:
            f.writelines(" ".join(map(str, row)) + "\n" for row in w)
            f.write(" ".join(map(str, b)) + "\n")
            for n, (name, raw, ch) in enumerate(cs):
                ch = ch or [i % 4 for i in range(64)]
                f.write(f"{n} {name} " + " ".join(f"{c} {r}" for c, r in zip(ch, raw)) + "\n")
    if a.actual:
        rows = {int(p[0]): list(map(int, p[1:])) for p in (l.split() for l in open(a.actual))}
        for n, (name, raw, ch) in enumerate(cs):
            st, f, s = decide(raw, w, b, ch)
            if rows.get(n) != [st, f] + s:
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL case={name} rtl={rows.get(n)} py={[st, f] + s}")
            print(f"STATE {name:12s} → {NAMES[st]}" + (f" fault=0b{f:03b}" if f else f" margin={sorted(s)[-1] - sorted(s)[-2]}"))
        print(f"PYTHON_RTL_COMPARE_PASS cases={len(cs)}")


if __name__ == "__main__":
    main()

07코드 읽기 포인트

위치내용
고장이 먼저범위(100~4000), 고착(같은 채널 같은 값 8번 연속), 순서(채널 0·1·2·3 반복)를 샘플마다 보고, 하나라도 있으면 FAULT입니다. 고장 난 값으로 AI를 돌리면 그럴듯한 오답이 나오기 때문입니다.
위치 기준 비교고착은 '와야 할' 채널(idx[1:0]) 기준으로 셉니다. 순서가 틀려도 같은 위치끼리 비교하므로 Python 기준 모델과 결과가 같습니다.
스케일(raw − 2048) >> 4를 −128~127로 포화합니다. 12비트 가운데(2048)를 0으로 두는 부호 있는 INT8입니다.
UNKNOWN1·2위 차가 MARGIN(3,000)보다 작으면 분류하지 않습니다. 경보 장치에서 애매한 답을 확정처럼 보이지 않게 하는 장치입니다.
보드 연결은 다음 단계이 장은 RTL과 가상 벡터까지입니다. SoC에 붙이려면 20장처럼 AXI 래퍼를 두고 27장 LED·7세그먼트로 상태를 보이게 합니다(⚠ 미작성·미검증).
