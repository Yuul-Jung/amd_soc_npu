00실습 개요

MAX_LEN·ready/valid·done 정렬를 중심으로 실제 RTL의 포트·입출력 타이밍과 계산 결과를 연결하여 학습합니다.

02전체 RTL 소스

ch07/rtl/vector_dot_core.v · 전체 RTL
`timescale 1ns/1ps
module vector_dot_core #(parameter integer DATA_WIDTH=16,ACC_WIDTH=40,MAX_LEN=256)(
    input wire clk,resetn,start_i,valid_i,
    input wire [8:0] length_i,
    input wire signed [DATA_WIDTH-1:0] a_i,b_i,
    output wire ready_o,
    output reg busy_o,done_o,error_o,
    output reg signed [ACC_WIDTH-1:0] result_o
);
    reg [8:0] accepted,length_q;
    wire signed [ACC_WIDTH-1:0] acc;
    wire mac_valid;
    wire launch=start_i && !busy_o;
    assign ready_o=busy_o && accepted<length_q;
    // 정상 MAC을 그대로 사용하고 트랜잭션 경계에서만 Clear한다.
    mac_core #(.DATA_WIDTH(DATA_WIDTH),.ACC_WIDTH(ACC_WIDTH)) mac
      (.clk(clk),.resetn(resetn),.clear_i(launch),.valid_i(valid_i && ready_o),
       .a_i(a_i),.b_i(b_i),.acc_o(acc),.valid_o(mac_valid));
    always @(posedge clk) begin
        if(!resetn) begin
            busy_o<=0;done_o<=0;error_o<=0;result_o<=0;accepted<=0;length_q<=0;
        end else begin
            done_o<=0;error_o<=0;
            if(launch) begin
                accepted<=0;length_q<=length_i;
                if(length_i>MAX_LEN) error_o<=1;
                else if(length_i==0) begin result_o<=0;done_o<=1;end
                else busy_o<=1;
            end else if(busy_o) begin
                if(valid_i && ready_o) accepted<=accepted+1'b1;
                // MAC 출력은 직전 에지에 등록되었으므로 마지막 수락 다음 에지에 읽는다.
                if(mac_valid && accepted==length_q) begin
                    result_o<=acc;done_o<=1;busy_o<=0;
                end
            end
        end
    end
    // synthesis translate_off
    initial if(MAX_LEN<1 || MAX_LEN>256) $fatal(1,"MAX_LEN range");
    // synthesis translate_on
endmodule

03전체 테스트벤치

ch07/tb/tb_vector_dot.sv · 전체 테스트벤치
`timescale 1ns/1ps
module tb_vector_dot;
    parameter DW=16,AW=40;
    reg clk=0,rn=0,start=0,vi=0;
    reg [8:0] len=0;
    reg signed [DW-1:0] a=0,b=0;
    wire ready,busy,done,error;
    wire signed [AW-1:0] result;
    vector_dot_core #(.DATA_WIDTH(DW),.ACC_WIDTH(AW)) dut
      (.clk(clk),.resetn(rn),.start_i(start),.valid_i(vi),.length_i(len),.a_i(a),.b_i(b),
       .ready_o(ready),.busy_o(busy),.done_o(done),.error_o(error),.result_o(result));
    always #5 clk=~clk;
    integer f,o,n,x,y,c=0,k,fields;
    reg signed [AW-1:0] expected;
    string vp,ap;
    initial begin
        if(!$value$plusargs("VECTORS=%s",vp)||!$value$plusargs("ACTUAL=%s",ap))$fatal(1,"Paths");
        f=$fopen(vp,"r");o=$fopen(ap,"w");if(!f||!o)$fatal(1,"Open");
        repeat(2)@(negedge clk);rn=1;start=1;len=257;
        @(posedge clk);#1;if(!error||busy||done)$fatal(1,"Length guard");
        @(negedge clk);start=0;
        // 계류 계산을 Reset으로 취소하고 새 벡터를 시작한다.
        @(negedge clk);start=1;len=4;
        @(negedge clk);start=0;vi=1;a=3;b=4;
        @(negedge clk);rn=0;
        @(posedge clk);#1;if(busy||done||ready||result!==0)$fatal(1,"Reset cancel");
        @(negedge clk);rn=1;vi=0;
        while(!$feof(f))begin
            fields=$fscanf(f,"%d %d\n",n,expected);if(fields!=2)$fatal(1,"Header");
            @(negedge clk);start=1;len=n;vi=0;
            @(posedge clk);#1;
            if(n==0)begin if(!done||busy||result!==0)$fatal(1,"Zero");end
            else if(!busy||done||!ready)$fatal(1,"Start");
            @(negedge clk);start=0;
            for(k=0;k<n;k=k+1)begin
                if($fscanf(f,"%d %d\n",x,y)!=2)$fatal(1,"Pair");
                @(negedge clk);a=x;b=y;vi=1;start=(k==0);
                if(!ready)$fatal(1,"Ready");
                @(posedge clk);#1;if(done||!busy)$fatal(1,"Early done");
                if(k<n-1 && k%3==0)begin
                    @(negedge clk);vi=0;start=0;
                    @(posedge clk);#1;if(done||!busy)$fatal(1,"Bubble");
                end
            end
            if(n>0)begin
                @(negedge clk);vi=0;start=0;
                @(posedge clk);#1;if(!done||busy||result!==expected)$fatal(1,"Result case=%0d",c+1);
            end
            c=c+1;$fdisplay(o,"%0d %0d",c,result);
            @(negedge clk);start=0;vi=0;
            @(posedge clk);#1;if(done||busy||error)$fatal(1,"Pulse");
        end
        if(!c)$fatal(1,"Empty");$fclose(o);$display("RTL_SIM_PASS DOT INT%0d/%0d cases=%0d",DW,AW,c);$finish;
    end
    initial begin #10000000;$fatal(1,"Timeout");end
endmodule
