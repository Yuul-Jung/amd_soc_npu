제09장 · Dense 4뉴런 레이어

dense_layer4는 08장 pe_array4에 편향 덧셈·ReLU·양자화를 추가하여 완전한 Dense 레이어를 구성합니다.

Bias+ReLUfixed_quantizer 적용편향 등록

00개요

dense_layer4는 start 시 편향을 biases_q에 등록한 뒤 pe_array4에 위임합니다. PE done 직후 각 뉴런마다 편향을 더하고, 음수이면 0으로 클리핑(ReLU), 마지막으로 fixed_quantizer를 통과시킵니다.

범위이 장은 4뉴런 레이어 하나를 다룹니다. BRAM 연결과 FSM은 10장 NPU Core에서 추가합니다.

01모듈 구조

PE done 직후 Bias+ReLU를 조합 계산하고, 다음 클록에 Quantizer 출력이 등록됩니다.

02RTL 전체 소스

ch09/rtl/dense_layer4.v · verilog
`timescale 1ns/1ps
module dense_layer4 #(parameter integer DATA_WIDTH=16,ACC_WIDTH=40,SHIFT=0)(
    input wire clk,resetn,start_i,valid_i,
    input wire [8:0] length_i,
    input wire signed [DATA_WIDTH-1:0] x_i,
    input wire [4*DATA_WIDTH-1:0] weights_i,
    input wire [4*ACC_WIDTH-1:0] biases_i,
    output wire ready_o,
    output reg busy_o,done_o,error_o,
    output reg [4*DATA_WIDTH-1:0] result_o,
    output reg [3:0] overflow_o
);
    wire launch=start_i && !busy_o && length_i<=64;
    reg [4*ACC_WIDTH-1:0] biases_q;
    wire [3:0] ready_pe;
    wire pe_done;
    wire [4*ACC_WIDTH-1:0] pe_result;
    wire [4*DATA_WIDTH-1:0] quantized;
    wire [3:0] qv,qov;
    assign ready_o=busy_o && (&ready_pe);
    pe_array4 #(.DATA_WIDTH(DATA_WIDTH),.ACC_WIDTH(ACC_WIDTH)) pe
      (.clk(clk),.resetn(resetn),.start_i(launch),.length_i(length_i),
       .valid_i({4{valid_i && ready_o}}),.a_i({4{x_i}}),.b_i(weights_i),
       .ready_o(ready_pe),.busy_o(),.done_o(pe_done),.error_o(),.pe_results_o(pe_result),.sum_o());
    genvar g;
    generate for(g=0;g<4;g=g+1)begin: neurons
        wire signed [ACC_WIDTH-1:0] dot_value=pe_result[g*ACC_WIDTH+:ACC_WIDTH];
        wire signed [ACC_WIDTH-1:0] bias_value=biases_q[g*ACC_WIDTH+:ACC_WIDTH];
        wire signed [ACC_WIDTH:0] biased={dot_value[ACC_WIDTH-1],dot_value}+{bias_value[ACC_WIDTH-1],bias_value};
        wire signed [ACC_WIDTH:0] activated=biased<0 ? 0 : biased;
        fixed_quantizer #(.IN_WIDTH(ACC_WIDTH+1),.OUT_WIDTH(DATA_WIDTH),.SHIFT(SHIFT),.ROUND(1),.SATURATE(1)) q
          (.clk(clk),.resetn(resetn),.valid_i(pe_done && busy_o),.x_i(activated),
           .q_o(quantized[g*DATA_WIDTH+:DATA_WIDTH]),.valid_o(qv[g]),.overflow_o(qov[g]));
    end endgenerate
    always @(posedge clk)begin
        if(!resetn)begin busy_o<=0;done_o<=0;error_o<=0;result_o<=0;overflow_o<=0;biases_q<=0;end
        else begin
            done_o<=0;error_o<=0;
            if(start_i && !busy_o)begin
                if(length_i>64)error_o<=1;
                else begin busy_o<=1;biases_q<=biases_i;end
            end else if(busy_o && (&qv))begin
                result_o<=quantized;overflow_o<=qov;busy_o<=0;done_o<=1;
            end
        end
    end
endmodule

03자동 테스트벤치 전체 소스

ch09/tb/tb_dense_layer4.sv · systemverilog
`timescale 1ns/1ps
module tb_dense_layer4;
    parameter DW=16,AW=40;
    reg clk=0,rn=0,start=0,vi=0;
    reg [8:0] len=0;
    reg signed [DW-1:0] x=0;
    reg [4*DW-1:0] weights=0;
    reg [4*AW-1:0] biases=0;
    wire ready,busy,done,error;wire [4*DW-1:0] result;wire[3:0]overflow;
    dense_layer4 #(.DATA_WIDTH(DW),.ACC_WIDTH(AW)) dut(.clk(clk),.resetn(rn),.start_i(start),.valid_i(vi),
      .length_i(len),.x_i(x),.weights_i(weights),.biases_i(biases),.ready_o(ready),.busy_o(busy),
      .done_o(done),.error_o(error),.result_o(result),.overflow_o(overflow));
    always #5 clk=~clk;
    integer f,o,n,c=0,k,i,t,xi,ws[0:3],em;
    reg signed [AW-1:0] bs[0:3];integer expected[0:3];
    string vp,ap;
    initial begin
        if(!$value$plusargs("VECTORS=%s",vp)||!$value$plusargs("ACTUAL=%s",ap))$fatal(1,"Paths");
        f=$fopen(vp,"r");o=$fopen(ap,"w");if(!f||!o)$fatal(1,"Open");
        repeat(2)@(negedge clk);rn=1;start=1;len=65;
        @(posedge clk);#1;if(!error||busy)$fatal(1,"Length guard");
        @(negedge clk);start=0;
        @(negedge clk);start=1;len=4;
        @(negedge clk);start=0;vi=1;x=3;
        @(negedge clk);rn=0;
        @(posedge clk);#1;if(busy||done||ready)$fatal(1,"Reset");
        @(negedge clk);rn=1;vi=0;
        while(!$feof(f))begin
            if($fscanf(f,"%d %d %d %d %d %d %d %d %d %d\n",n,bs[0],bs[1],bs[2],bs[3],
              expected[0],expected[1],expected[2],expected[3],em)!=10)$fatal(1,"Header");
            @(negedge clk);len=n;start=1;vi=0;
            for(i=0;i<4;i=i+1)biases[i*AW+:AW]=bs[i];
            @(posedge clk);#1;if(!busy||done)$fatal(1,"Start");
            @(negedge clk);start=0;biases=0; // 저장된 Bias가 외부 변경에 영향받지 않아야 한다.
            for(k=0;k<n;k=k+1)begin
                if($fscanf(f,"%d %d %d %d %d\n",xi,ws[0],ws[1],ws[2],ws[3])!=5)$fatal(1,"Pair");
                @(negedge clk);if(!ready)$fatal(1,"Ready");
                x=xi;vi=1;for(i=0;i<4;i=i+1)weights[i*DW+:DW]=ws[i];
                @(posedge clk);#1;if(done)$fatal(1,"Early");
                if(k%3==0 && k<n-1)begin @(negedge clk);vi=0;@(posedge clk);#1;if(done)$fatal(1,"Bubble");end
            end
            @(negedge clk);vi=0;t=0;
            while(!done)begin @(posedge clk);#1;t=t+1;if(t>20)$fatal(1,"Completion timeout");end
            if(busy||overflow!==em[3:0])$fatal(1,"Flags");
            for(i=0;i<4;i=i+1)if($signed(result[i*DW+:DW])!==expected[i])$fatal(1,"Neuron %0d case%0d",i,c+1);
            c=c+1;$fdisplay(o,"%0d %0d %0d %0d %0d %0d",c,$signed(result[0*DW+:DW]),
              $signed(result[1*DW+:DW]),$signed(result[2*DW+:DW]),$signed(result[3*DW+:DW]),overflow);
            @(negedge clk);start=0;vi=0;@(posedge clk);#1;if(done||busy)$fatal(1,"Pulse");
        end
        if(!c)$fatal(1,"Empty");$fclose(o);$display("RTL_SIM_PASS DENSE4 INT%0d/%0d cases=%0d",DW,AW,c);$finish;
    end
    initial begin #10000000;$fatal(1,"Timeout");end
endmodule

05실행 결과 확인

테스트벤치의 자동 판정에서 오류가 발생하지 않고 RTL_SIM_PASS DENSE4 INT16/40 cases=46가 출력되어야 합니다. 시뮬레이션 결과는 합성·실물 보드 확인과 구분합니다.

편향=0 조건에서 ReLU 출력을 확인합니다. 음수 누산 → ReLU로 0, 양수 누산 → 양자화 후 출력. overflow_o는 양자화 범위 초과 여부입니다. 실행 스크립트는 같은 장에서 두 번째 정밀도 설정도 함께 돌립니다(INT8/32 cases=46 추가 통과). 또한 Python 기준 모델 자체 시험과 RTL 출력 ↔ Python 기준값 비교(PYTHON_RTL_COMPARE_PASS)를 같이 판정합니다.

06핵심 코드 풀이

편향 등록: start 에지에서 biases_i를 biases_q에 등록합니다. PE 처리 중 외부 편향 값이 변해도 이 값이 보존됩니다.

ReLU: activated = biased < 0 ? 0 : biased는 조합 로직입니다. 부호 확장된 (ACC_WIDTH+1)비트 덧셈 결과를 조건부로 클리핑합니다.

Quantizer 완료 추적: fixed_quantizer는 1클록 레이턴시입니다. 4개 모두의 qv[g]가 1이 되는 에지에서 result_o를 등록하고 done을 발생합니다.
