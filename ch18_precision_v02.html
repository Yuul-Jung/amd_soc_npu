`timescale 1ns/1ps
`default_nettype none
// 18장 데이터 폭 매개변수 784→10: 17장과 같은 13타일·코어 밖 누산·Bias 한 번 구조를 DATA_WIDTH(8 또는 16)로 일반화한다.
// 15장 코어는 8비트로 고정이므로, 08장 4PE를 직접 써서 타일(13)·그룹(3) 반복을 이 모듈 하나에 둔다.
// INT16 최악: |내적| ≤ 784·32640·32768 (41비트) + Bias → ACC_WIDTH 48 사용.
module npu_tiled_784_p #(
    parameter integer DATA_WIDTH = 8,
    parameter integer ACC_WIDTH = 32
) (
    input  wire                      clk,
    input  wire                      resetn,
    input  wire                      x_wr_en_i,
    input  wire                [9:0] x_addr_i,
    input  wire     [DATA_WIDTH-1:0] x_data_i,
    input  wire                      w_wr_en_i,
    input  wire                [3:0] w_neuron_i,
    input  wire                [9:0] w_addr_i,
    input  wire     [DATA_WIDTH-1:0] w_data_i,
    input  wire                      b_wr_en_i,
    input  wire                [3:0] b_neuron_i,
    input  wire      [ACC_WIDTH-1:0] b_data_i,
    input  wire                      start_i,
    output wire                      busy_o,
    output wire [10*ACC_WIDTH-1:0]   scores_o,
    output reg                       done_o,
    output reg                       error_o,
    output wire                [3:0] class_o,
    output reg                [15:0] cycles_o
);
    localparam integer DW = DATA_WIDTH;
    localparam IDLE=0, LAUNCH=1, READ=2, WAIT_READ=3, SEND=4, WAIT_PE=5, BIAS=6, AM_GO=7, AM_WAIT=8;
    reg [3:0] state;
    reg [3:0] tile_q;                                     // 0~12
    reg [1:0] group_q;                                    // 0~2
    reg [5:0] idx_q;
    reg signed [ACC_WIDTH-1:0] acc_q [0:9];
    reg signed [ACC_WIDTH-1:0] score_q [0:9];
    reg signed [ACC_WIDTH-1:0] bias_q [0:9];
    integer i;

    wire [6:0] len = (tile_q == 4'd12) ? 7'd16 : 7'd64;
    wire [9:0] addr = {tile_q, 6'd0} + idx_q;
    assign busy_o = (state != IDLE);

    wire signed [DW-1:0] x_rd; wire x_rv;
    bram_sync_buffer #(.DATA_WIDTH(DW), .ADDR_WIDTH(10)) x_ram (
        .clk(clk), .resetn(resetn), .wr_en(x_wr_en_i && !busy_o), .wr_addr(x_addr_i), .wr_data(x_data_i),
        .rd_en(state == READ), .rd_addr(addr), .rd_data(x_rd), .rd_valid(x_rv));
    wire [4*DW-1:0] w_rd, w_bus; wire [3:0] w_rv;
    genvar g;
    generate for (g = 0; g < 4; g = g + 1) begin : lane
        bram_sync_buffer #(.DATA_WIDTH(DW), .ADDR_WIDTH(12)) w_ram (
            .clk(clk), .resetn(resetn),
            .wr_en(w_wr_en_i && !busy_o && w_neuron_i <= 9 && w_neuron_i[1:0] == g),
            .wr_addr({w_neuron_i[3:2], w_addr_i}), .wr_data(w_data_i),
            .rd_en(state == READ), .rd_addr({group_q, addr}), .rd_data(w_rd[g*DW +: DW]), .rd_valid(w_rv[g]));
        assign w_bus[g*DW +: DW] = (group_q == 2'd2 && g >= 2) ? {DW{1'b0}} : w_rd[g*DW +: DW];
    end endgenerate
    generate for (g = 0; g < 10; g = g + 1) begin : o
        assign scores_o[g*ACC_WIDTH +: ACC_WIDTH] = score_q[g];
    end endgenerate

    wire [3:0] pe_ready; wire pe_done; wire [4*ACC_WIDTH-1:0] pe_res;
    pe_array4 #(.DATA_WIDTH(DW), .ACC_WIDTH(ACC_WIDTH)) pe (
        .clk(clk), .resetn(resetn), .start_i(state == LAUNCH), .length_i({2'b00, len}),
        .valid_i({4{state == SEND}}), .a_i({4{x_rd}}), .b_i(w_bus),
        .ready_o(pe_ready), .busy_o(), .done_o(pe_done), .error_o(), .pe_results_o(pe_res), .sum_o());

    wire am_done;
    npu_argmax #(.N(10), .W(ACC_WIDTH)) argmax (
        .clk(clk), .resetn(resetn), .start_i(state == AM_GO), .scores_i(scores_o),
        .busy_o(), .done_o(am_done), .class_o(class_o));

    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE; tile_q <= 0; group_q <= 0; idx_q <= 0; done_o <= 0; error_o <= 0; cycles_o <= 0;
            for (i = 0; i < 10; i = i + 1) begin acc_q[i] <= 0; score_q[i] <= 0; bias_q[i] <= 0; end
        end else begin
            done_o <= 0; error_o <= 0;
            if (busy_o) cycles_o <= cycles_o + 16'd1;
            if (b_wr_en_i && !busy_o && b_neuron_i <= 9) bias_q[b_neuron_i] <= b_data_i;
            case (state)
                IDLE: if (start_i) begin
                    tile_q <= 0; group_q <= 0; idx_q <= 0; cycles_o <= 0; state <= LAUNCH;
                    for (i = 0; i < 10; i = i + 1) acc_q[i] <= 0;
                end
                LAUNCH: state <= READ;
                READ: state <= WAIT_READ;
                WAIT_READ: if (x_rv && (&w_rv)) state <= SEND;
                    else begin state <= IDLE; error_o <= 1; end
                SEND: if (&pe_ready) begin
                    if (idx_q == len - 7'd1) state <= WAIT_PE;
                    else begin idx_q <= idx_q + 6'd1; state <= READ; end
                end
                WAIT_PE: if (pe_done) begin                    // 코어 밖 누산: 그룹의 뉴런 4개(마지막 그룹 2개)
                    for (i = 0; i < 10; i = i + 1)
                        if (i / 4 == group_q) acc_q[i] <= acc_q[i] + $signed(pe_res[(i % 4)*ACC_WIDTH +: ACC_WIDTH]);
                    idx_q <= 0;
                    if (group_q != 2'd2) begin group_q <= group_q + 2'd1; state <= LAUNCH; end
                    else if (tile_q != 4'd12) begin group_q <= 0; tile_q <= tile_q + 4'd1; state <= LAUNCH; end
                    else state <= BIAS;
                end
                BIAS: begin
                    for (i = 0; i < 10; i = i + 1) score_q[i] <= acc_q[i] + bias_q[i];
                    state <= AM_GO;
                end
                AM_GO: state <= AM_WAIT;
                AM_WAIT: if (am_done) begin state <= IDLE; done_o <= 1; end
                default: begin state <= IDLE; error_o <= 1; end
            endcase
        end
    end
endmodule

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. precision_pc.py --export가 실제 MNIST 학습 집합 60,000장으로 소프트맥스 회귀를 학습(시드 18, 8회)하고 INT8·INT16으로 양자화해 ch18/model/에 정수 모델과 시험 집합 앞 20장을 씁니다. TB는 DW=8(누산 32비트)과 DW=16(48비트)으로 두 번 컴파일하고, DW=8에서는 17장 모듈을 같은 데이터로 함께 돌려 점수가 같은지 봅니다.

ch18/tb/tb_npu_precision.sv · systemverilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
// 18장 TB: DW(8·16)·AW를 바꿔 같은 실제 MNIST 정수 모델과 시험 20장을 돌린다.
// DW=8이면 17장 npu_tiled_784를 같은 데이터로 함께 돌려 점수가 같은지도 본다.
module tb_npu_precision;
    parameter integer DW = 8, AW = 32;
    reg clk=0, rn=0, xwe=0, wwe=0, bwe=0, start=0;
    reg [9:0] xa=0, wa=0; reg [DW-1:0] xd=0, wd=0; reg [3:0] wn=0, bn=0; reg [AW-1:0] bd=0;
    wire busy, done, err; wire [3:0] cls; wire [10*AW-1:0] scores; wire [15:0] cycles;
    npu_tiled_784_p #(.DATA_WIDTH(DW), .ACC_WIDTH(AW)) dut(.clk(clk), .resetn(rn), .x_wr_en_i(xwe), .x_addr_i(xa),
        .x_data_i(xd), .w_wr_en_i(wwe), .w_neuron_i(wn), .w_addr_i(wa), .w_data_i(wd), .b_wr_en_i(bwe), .b_neuron_i(bn),
        .b_data_i(bd), .start_i(start), .busy_o(busy), .scores_o(scores), .done_o(done), .error_o(err), .class_o(cls),
        .cycles_o(cycles));
    always #5 clk = ~clk;

    // DW=8: 17장 모듈과 교차 확인
    wire [10*AW-1:0] ref_scores; wire ref_done; wire [3:0] ref_cls;
    generate if (DW == 8) begin : cross_ch17
        npu_tiled_784 #(.ACC_WIDTH(AW)) ch17 (.clk(clk), .resetn(rn), .x_wr_en_i(xwe), .x_addr_i(xa), .x_data_i(xd),
            .w_wr_en_i(wwe), .w_neuron_i(wn), .w_addr_i(wa), .w_data_i(wd), .b_wr_en_i(bwe), .b_neuron_i(bn), .b_data_i(bd),
            .start_i(start), .busy_o(), .tile_done_o(), .tile_o(), .tile_done_num_o(), .tile_len_o(), .acc_o(),
            .scores_o(ref_scores), .done_o(ref_done), .error_o(), .class_o(ref_cls));
    end endgenerate

    integer f, o, k, c, v, n, label, cases = 0, cross_n = 0, cyc = 0;
    string vp, ap;

    initial begin
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap)) $fatal(1, "경로 인자");
        f = $fopen(vp, "r"); o = $fopen(ap, "w"); if (!f || !o) $fatal(1, "파일 열기");
        repeat (3) @(negedge clk); rn = 1;
        for (c = 0; c < 10; c = c + 1) for (k = 0; k < 784; k = k + 1) begin
            if ($fscanf(f, "%d", v) != 1) $fatal(1, "w"); @(negedge clk); wwe = 1; wn = c; wa = k; wd = v; end
        @(negedge clk); wwe = 0;
        for (c = 0; c < 10; c = c + 1) begin
            if ($fscanf(f, "%d", v) != 1) $fatal(1, "b"); @(negedge clk); bwe = 1; bn = c; bd = v; end
        @(negedge clk); bwe = 0;
        while ($fscanf(f, "%d %d", n, label) == 2) begin
            for (k = 0; k < 784; k = k + 1) begin
                if ($fscanf(f, "%d", v) != 1) $fatal(1, "x"); @(negedge clk); xwe = 1; xa = k; xd = v; end
            @(negedge clk); xwe = 0; start = 1; @(negedge clk); start = 0;
            while (!done) begin @(negedge clk); if (err) $fatal(1, "case %0d: error_o", n); end
            if (DW == 8) begin
                while (!ref_done) @(negedge clk);
                if (ref_scores !== scores || ref_cls !== cls) $fatal(1, "case %0d: 17장과 점수 다름", n);
                cross_n = cross_n + 1;
            end
            $fwrite(o, "%0d", n);
            for (c = 0; c < 10; c = c + 1) $fwrite(o, " %0d", $signed(scores[c*AW +: AW]));
            $fwrite(o, " %0d\n", cls);
            cases = cases + 1; cyc = cycles;
        end
        $display("RTL_SIM_PASS PRECISION int%0d acc=%0d cases=%0d cross_ch17=%0d cycles_per_inference=%0d", DW, AW, cases, cross_n, cyc);
        $fclose(o); $finish;
    end
    initial begin #500000000; $fatal(1, "시간 초과"); end
endmodule

#!/usr/bin/env python3
"""18장 PC 정밀도 비교: 같은 784→10 선형 모델을 Float32, INT8, INT16으로 실제 MNIST 시험 집합 10,000장에서 평가한다.

학습: 소프트맥스 회귀(1층), 미니배치 SGD, 시드 고정. 학습한 Float32 가중치를 정수로 양자화한다.
  INT8 : 입력 x = min(127, (p+1)>>1) (13·17장 계약), 가중치 = round(w / s8),  s8  = max|w| / 127
  INT16: 입력 x = p × 128 (0~32640),               가중치 = round(w / s16), s16 = max|w| / 32767
  Bias : 정수 점수 단위로 바꿔 반올림 (INT16은 40비트 이상 필요할 수 있음 → 실제 비트 수를 출력)
결과: 정확도, 혼동행렬, 정수 경로와 Float32의 클래스 일치율, 최악 누산 비트 수.
--export: 정수 모델과 RTL 시험용 앞 20장을 ch18/model/에 쓴다 (RTL 시뮬레이션은 이 파일만 읽음).
"""
from __future__ import annotations
import argparse
import sys
from pathlib import Path

import numpy as np

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "host_tools"))
from mnist_idx import images, labels          # noqa: E402

MODEL = ROOT / "ch18" / "model"
RTL_CASES = 20


def load(split):
    x = np.frombuffer(b"".join(images(split)), dtype=np.uint8).reshape(-1, 784)
    return x, np.array(labels(split), dtype=np.int64)


def train(p, y, epochs=8, lr=0.5, batch=128, seed=18):
    rng = np.random.default_rng(seed)
    x = p.astype(np.float32) / 255.0
    w = np.zeros((784, 10), np.float32); b = np.zeros(10, np.float32)
    onehot = np.eye(10, dtype=np.float32)[y]
    for _ in range(epochs):
        order = rng.permutation(len(x))
        for s in range(0, len(x), batch):
            i = order[s:s + batch]
            z = x[i] @ w + b
            z -= z.max(1, keepdims=True)
            pr = np.exp(z); pr /= pr.sum(1, keepdims=True)
            g = (pr - onehot[i]) / len(i)
            w -= lr * x[i].T @ g; b -= lr * g.sum(0)
    return w, b


def quantize(w, b, bits):
    qmax = (1 << (bits - 1)) - 1
    s_w = float(np.abs(w).max()) / qmax
    if bits == 8:
        s_x = 2.0 / 255.0                       # x8 ≈ p/2 → p/255 = x8 · 2/255
    else:
        s_x = 1.0 / (255.0 * 128.0)             # x16 = p·128 → p/255 = x16 / (255·128)
    wq = np.clip(np.rint(w / s_w), -qmax - 1, qmax).astype(np.int64)
    bq = np.rint(b / (s_x * s_w)).astype(np.int64)
    return wq.T, bq                              # (10, 784), (10,)


def to_x(p, bits):
    p = p.astype(np.int64)
    return np.minimum(127, (p + 1) >> 1) if bits == 8 else p * 128


def argmax_lowest(s):
    return np.argmax(s, axis=1)                  # numpy argmax는 동점이면 앞 번호 → argmax_lowest와 같음


def confusion(y, pred):
    m = np.zeros((10, 10), np.int64)
    np.add.at(m, (y, pred), 1)
    return m


def bits_needed(v):
    return int(abs(int(v))).bit_length() + 1    # 부호 포함


def main():
    ap = argparse.ArgumentParser(); ap.add_argument("--export", action="store_true")
    a = ap.parse_args()
    ptr, ytr = load("train"); pte, yte = load("test")
    w, b = train(ptr, ytr)
    pred = {"float32": argmax_lowest((pte.astype(np.float32) / 255.0) @ w + b)}
    q = {}
    for bits in (8, 16):
        wq, bq = quantize(w, b, bits)
        s = to_x(pte, bits) @ wq.T + bq
        q[bits] = (wq, bq, s)
        pred[f"int{bits}"] = argmax_lowest(s)
    print(f"MNIST test={len(yte)} train={len(ytr)} model=softmax_784x10 epochs=8 seed=18")
    for k, pr in pred.items():
        acc = (pr == yte).mean()
        print(f"ACCURACY {k} {acc * 100:.2f}% ({int((pr == yte).sum())}/{len(yte)})")
    for bits in (8, 16):
        print(f"AGREE int{bits}_vs_float32 {(pred[f'int{bits}'] == pred['float32']).mean() * 100:.2f}%")
    for k, pr in pred.items():
        print(f"CONFUSION {k} (행 = 정답 0~9, 열 = 예측 0~9)")
        for r, row in enumerate(confusion(yte, pr)):
            print(f"  {r}: " + " ".join(f"{v:4d}" for v in row))
    for bits in (8, 16):
        wq, bq, s = q[bits]
        xmax = 127 if bits == 8 else 255 * 128
        worst_dot = 784 * xmax * (1 << (bits - 1))
        print(f"RANGE int{bits} max|w|={int(np.abs(wq).max())} bias_bits={bits_needed(np.abs(bq).max())} "
              f"worst_dot_bits={bits_needed(worst_dot)} worst_total_bits={bits_needed(worst_dot + int(np.abs(bq).max()))} "
              f"observed_score_bits={bits_needed(np.abs(s).max())}")
    if a.export:
        MODEL.mkdir(exist_ok=True)
        for bits in (8, 16):
            wq, bq, s = q[bits]
            with open(MODEL / f"model_int{bits}.txt", "w") as f:

                f.write(" ".join(map(str, bq)) + "\n")

                for k in range(RTL_CASES):
                    f.write(f"{k} {yte[k]} " + " ".join(map(str, to_x(pte[k:k + 1], bits)[0])) + "\n")
        print(f"EXPORT model_int8 model_int16 cases={RTL_CASES}")
    ok = (pred["int8"] == yte).mean() > 0.85 and (pred["int16"] == pred["float32"]).mean() > 0.99
    print("PC_PRECISION_PASS" if ok else "PC_PRECISION_FAIL")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""18장 RTL 비교용 기준 모델: ch18/model/의 정수 모델(INT8·INT16)과 MNIST 시험 앞 20장으로 정수 점수·클래스를 계산한다.

모델 파일은 precision_pc.py --export가 실제 MNIST로 학습·양자화해 만든 것이며, RTL 시뮬레이션은 이 파일만 읽는다(내려받기 없음).
"""
from __future__ import annotations
import argparse
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "host_tools"))
from p3_integer_reference import argmax_lowest, dot_tiled_64  # noqa: E402

MODEL = ROOT / "ch18" / "model"


def load(bits):
    rows = [list(map(int, ln.split())) for ln in open(MODEL / f"model_int{bits}.txt")]
    cases = []
    for ln in open(MODEL / f"cases_int{bits}.txt"):
        p = list(map(int, ln.split()))
        cases.append((p[0], p[1], p[2:]))
    return rows[:10], rows[10], cases


def scores(x, w, b):
    return [dot_tiled_64(x, row) + bb for row, bb in zip(w, b)]


def main():
    ap = argparse.ArgumentParser(); ap.add_argument("--bits", type=int, required=True)
    ap.add_argument("--out"); ap.add_argument("--actual")
    a = ap.parse_args()
    w, b, cases = load(a.bits)
    lim = (1 << (a.bits - 1))
    assert len(w) == 10 and all(len(r) == 784 and all(-lim <= v < lim for v in r) for r in w)
    assert all(len(x) == 784 and all(0 <= v < lim for v in x) for _, _, x in cases)
    if a.out:
        with open(a.out, "w") as f:
            f.writelines(" ".join(map(str, r)) + "\n" for r in w)
            f.write(" ".join(map(str, b)) + "\n")
            for k, label, x in cases:
                f.write(f"{k} {label} " + " ".join(map(str, x)) + "\n")
        print(f"PYTHON_PRECISION_VECTORS int{a.bits} cases={len(cases)}")
    if a.actual:
        got = {}
        for ln in open(a.actual):
            try:
                p = list(map(int, ln.split()))
            except ValueError:
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL 정수가 아닌 RTL 출력: {ln[:80]}")
            got[p[0]] = p[1:]
        right = 0
        for k, label, x in cases:
            s = scores(x, w, b)
            if got.get(k) != s + [argmax_lowest(s)]:
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL int{a.bits} case={k} rtl={got.get(k)} py={s}")
            right += argmax_lowest(s) == label
        print(f"PYTHON_RTL_COMPARE_PASS int{a.bits} cases={len(cases)} scores={10 * len(cases)} correct={right}/{len(cases)}")


if __name__ == "__main__":
    main()

07코드 읽기 포인트

위치내용
양자화 식INT8: 입력 min(127, (p+1)>>1)(13·17장 계약), 가중치 round(w / s), s = max|w| / 127. INT16: 입력 p × 128, s = max|w| / 32767. Bias는 정수 점수 단위(입력 스케일 × 가중치 스케일)로 바꿔 반올림합니다.
누산 폭INT16 최악: 784 × 32640 × 32768 → 41비트, 40비트 Bias를 더하면 부호 포함 42비트입니다. 이 모델에서 실제로 나온 점수는 36비트였지만, 32비트로는 넘치므로 RTL은 48비트를 씁니다. INT8은 최악 25비트라 32비트로 충분합니다.
15장 코어를 쓰지 않는 이유15장 npu_dense_core는 8비트로 고정입니다. 이 장은 08장 4PE를 직접 써서 타일·그룹 반복을 한 FSM에 두고 DATA_WIDTH를 매개변수로 뺐습니다. DW=8 결과가 17장과 같다는 교차 확인으로 같은 계산임을 보입니다.
PC와 RTL의 역할정확도·혼동행렬은 PC에서 10,000장 전부로 계산합니다. RTL은 앞 20장으로 정수 결과가 비트 단위로 같은지만 확인합니다. 정확도는 RTL 시뮬레이션으로 주장하지 않습니다.
사이클784→10 한 번에 7,263클록(INT8·INT16 같음, 시뮬레이션). 19장에서 PE 수에 따른 사이클 비교의 출발점입니다.
