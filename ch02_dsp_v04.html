제02장 · DSP48E1 타깃 곱셈

dsp_multiplier는 DSP48E1의 25×18 입력 범위에 맞춰 INT16 곱셈을 별도 레지스터로 분리합니다. 01장 MAC 누산기와 같은 Reset·Clear·Valid 계약을 따릅니다.

DSP48E1 25×18부호 확장mac_core 교차 검증

00개요

dsp_multiplier는 XC7A35T의 DSP48E1에 맞는 25비트·18비트 부호 있는 입력으로 INT16을 확장한 뒤 클록 1회에 32비트 곱을 출력합니다. 테스트벤치는 01장 mac_core를 같은 입력에 연결하여 누산 결과까지 교차 검증합니다.

범위이 장은 순수 곱셈 레지스터만 다룹니다. 누산은 01장 mac_core, 파이프라인은 03장을 참조합니다.

01모듈 구조

a_i→25b, b_i→18b 부호 확장 후 곱 → product_o[A_WIDTH+B_WIDTH-1:0] 등록.

02RTL 전체 소스

ch02/rtl/dsp_multiplier.v · verilog
`timescale 1ns/1ps
`default_nettype none

// DSP48E1의 25x18 signed 입력 범위에 맞춘 독립 곱셈 추론 후보.
// primitive를 직접 생성하지 않으며 실제 DSP 배치는 합성 후 확인한다.
module dsp_multiplier #(
    parameter integer A_WIDTH = 16,
    parameter integer B_WIDTH = 16
) (
    input wire clk,
    input wire resetn,
    input wire clear_i,
    input wire valid_i,
    input wire signed [A_WIDTH-1:0] a_i,
    input wire signed [B_WIDTH-1:0] b_i,
    output reg signed [A_WIDTH+B_WIDTH-1:0] product_o,
    output reg valid_o
);
    wire signed [24:0] a_dsp_w;
    wire signed [17:0] b_dsp_w;
    (* use_dsp = "yes" *) wire signed [42:0] product_full_w;

    // 작은 정수의 부호를 보존하여 DSP 곱셈 입력 폭까지 확장한다.
    assign a_dsp_w = {{(25-A_WIDTH){a_i[A_WIDTH-1]}}, a_i};
    assign b_dsp_w = {{(18-B_WIDTH){b_i[B_WIDTH-1]}}, b_i};
    assign product_full_w = $signed(a_dsp_w) * $signed(b_dsp_w);

    // P3-01과 같은 동기식 Reset > Clear > Valid 계약이다.
    // 상승 에지에서 입력을 수락하고 그 에지 직후 곱과 valid_o를 갱신한다.
    always @(posedge clk) begin
        if (!resetn) begin
            product_o <= {(A_WIDTH+B_WIDTH){1'b0}};
            valid_o <= 1'b0;
        end else if (clear_i) begin
            product_o <= {(A_WIDTH+B_WIDTH){1'b0}};
            valid_o <= 1'b0;
        end else begin
            valid_o <= valid_i;
            if (valid_i) begin
                // 곱의 전체 유효 폭은 A_WIDTH+B_WIDTH. 버리는 상위 비트는 부호 확장뿐이다.
                product_o <= product_full_w[A_WIDTH+B_WIDTH-1:0];
            end
        end
    end

    // synthesis translate_off
    initial begin
        if (A_WIDTH < 2 || A_WIDTH > 25 || B_WIDTH < 2 || B_WIDTH > 18)
            $fatal(1, "Require 2<=A_WIDTH<=25 and 2<=B_WIDTH<=18");
    end
    // synthesis translate_on
endmodule
`default_nettype wire

03자동 테스트벤치 전체 소스

ch02/tb/tb_dsp_multiplier.sv · systemverilog
`timescale 1ns/1ps
`default_nettype none
module tb_dsp_multiplier;
    parameter integer A_WIDTH = 16;
    parameter integer B_WIDTH = 16;
    parameter integer ACC_WIDTH = 40;
    localparam integer MAC_WIDTH = A_WIDTH > B_WIDTH ? A_WIDTH : B_WIDTH;
    reg clk = 0;
    reg resetn = 0, clear_i = 0, valid_i = 0;
    reg signed [A_WIDTH-1:0] a_i = 0;
    reg signed [B_WIDTH-1:0] b_i = 0;
    wire signed [A_WIDTH+B_WIDTH-1:0] product_o;
    wire valid_o;
    wire signed [ACC_WIDTH-1:0] acc_o;
    wire mac_valid;
    wire signed [MAC_WIDTH-1:0] a_mac = a_i;
    wire signed [MAC_WIDTH-1:0] b_mac = b_i;
    dsp_multiplier #(.A_WIDTH(A_WIDTH), .B_WIDTH(B_WIDTH)) dut (.*);
    // 누산 로직은 복제하지 않고 P3-01 정상 모듈에 같은 입력을 연결한다.
    mac_core #(.DATA_WIDTH(MAC_WIDTH), .ACC_WIDTH(ACC_WIDTH)) mac (
        .clk(clk), .resetn(resetn), .clear_i(clear_i), .valid_i(valid_i),
        .a_i(a_mac), .b_i(b_mac), .acc_o(acc_o), .valid_o(mac_valid)
    );
    always #5 clk = ~clk;
    integer vf, af, fields, cycle = 0;
    integer rn, cl, vi, ai, bi, ev;
    reg signed [A_WIDTH+B_WIDTH-1:0] ep;
    reg signed [ACC_WIDTH-1:0] ea;
    longint signed model_product = 0, model_acc = 0, multiply_w;
    reg signed [A_WIDTH+B_WIDTH-1:0] held_product;
    reg signed [ACC_WIDTH-1:0] held_acc;
    reg held_valid, held_mac_valid;
    string vp, ap;

    initial begin
        if (ACC_WIDTH > 56 || ACC_WIDTH < 2*MAC_WIDTH)
            $fatal(1, "TB requires 2*MAC_WIDTH<=ACC_WIDTH<=56");
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap))
            $fatal(1, "Missing vector/trace argument");
        vf = $fopen(vp, "r"); af = $fopen(ap, "w");
        if (!vf || !af) $fatal(1, "Cannot open test files");
        @(posedge clk); #1;
        if (product_o !== 0 || valid_o !== 0 || acc_o !== 0 || mac_valid !== 0)
            $fatal(1, "Initial reset failed");
        while (!$feof(vf)) begin
            fields = $fscanf(vf, "%d %d %d %d %d %d %d %d\n", rn, cl, vi, ai, bi, ep, ea, ev);
            if (fields != 8) $fatal(1, "Malformed vector");
            @(negedge clk);
            held_product = product_o; held_acc = acc_o;
            held_valid = valid_o; held_mac_valid = mac_valid;
            resetn = rn; clear_i = cl; valid_i = vi; a_i = ai; b_i = bi;
            #1;
            if (product_o !== held_product || acc_o !== held_acc ||
                valid_o !== held_valid || mac_valid !== held_mac_valid)
                $fatal(1, "Output changed before rising edge cycle=%0d", cycle+1);
            @(posedge clk); #1;
            cycle = cycle + 1;
            // 64비트 정수로 먼저 확장하여 곱을 계산하고 명시적으로 누산 폭을 적용한다.
            multiply_w = $signed(a_i);
            multiply_w = multiply_w * $signed(b_i);
            if (!rn || cl) begin
                model_product = 0;
                model_acc = 0;
            end else if (vi) begin
                model_product = multiply_w;
                model_acc = model_acc + multiply_w;
                model_acc = $signed(model_acc[ACC_WIDTH-1:0]);
            end
            if (ep !== model_product[A_WIDTH+B_WIDTH-1:0] ||
                ea !== model_acc[ACC_WIDTH-1:0] || ev !== (rn && !cl && vi))
                $fatal(1, "Python/SV reference mismatch cycle=%0d", cycle);
            if (product_o !== ep || acc_o !== ea || valid_o !== ev[0] || mac_valid !== ev[0])
                $fatal(1, "DUT mismatch cycle=%0d a=%0d b=%0d product=%0d expected=%0d acc=%0d expected=%0d",
                       cycle, a_i, b_i, product_o, ep, acc_o, ea);
            $fdisplay(af, "%0d %0d %0d %0d", cycle, product_o, acc_o, valid_o);
        end
        if (!cycle) $fatal(1, "Empty test");
        $fclose(vf); $fclose(af);
        $display("RTL_SIM_PASS A=%0d B=%0d ACC=%0d cycles=%0d", A_WIDTH, B_WIDTH, ACC_WIDTH, cycle);
        $finish;
    end
    initial begin
        #10000000;
        $fatal(1, "Simulation timeout");
    end
endmodule
`default_nettype wire

05실행 결과 확인

테스트벤치의 자동 판정에서 오류가 발생하지 않고 RTL_SIM_PASS A=16 B=16 ACC=40 cycles=37360가 출력되어야 합니다. 시뮬레이션 결과는 합성·실물 보드 확인과 구분합니다.

dsp_multiplier 출력과 mac_core 누산값을 같은 클록에서 비교하여 두 경로가 동일하게 동작함을 확인합니다. Reset·Clear 직후 product_o=0, valid_o=0이 필요합니다. 실행 스크립트는 같은 장에서 두 번째 정밀도 설정도 함께 돌립니다(A=8 B=8 ACC=32 추가 통과). 또한 Python 기준 모델 자체 시험과 RTL 출력 ↔ Python 기준값 비교(PYTHON_RTL_COMPARE_PASS)를 같이 판정합니다.

06핵심 코드 풀이

DSP48E1 25×18 범위: XC7A35T의 DSP48E1 A 포트는 25비트, B 포트는 18비트 signed 입력을 받습니다. INT16(±32768)은 두 포트 모두에 들어가므로 추론이 성공합니다.

상위 비트 버리기: 43비트 전체 곱 중 하위 A_WIDTH+B_WIDTH비트만 등록합니다. 버려지는 비트는 부호 확장 중복분으로 INT16×INT16에서 의미 있는 값이 없습니다.

교차 검증 전략: 테스트벤치가 mac_core를 같은 입력에 연결하고 누산 결과까지 비교합니다. 이를 통해 dsp_multiplier 단독으로 중복 모델 없이 완전한 검증이 가능합니다.
