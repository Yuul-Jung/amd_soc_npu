ch19/rtl/npu_tiled_784_npe.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 19장 PE 수 매개변수 784→10 (INT8): 07장 내적 코어를 NPE개(1·2·4·8) 나란히 두고,
// 클래스 10개를 NPE개씩 그룹 ceil(10/NPE)번에 나눠 13타일을 돈다. 차선마다 가중치 RAM 1개(읽기 포트 1개)를 둔다.
// 사이클 = 그룹 수 × 타일 실행 시간 + 마무리. NPE=4이면 18장 DW=8과 점수가 같고, 08장 pe_array4의 완료 레지스터가 없어 그룹마다 2클록 빠르다.
module npu_tiled_784_npe #(
    parameter integer NPE = 4,                       // 1, 2, 4, 8
    parameter integer ACC_WIDTH = 32
) (
    input  wire                      clk,
    input  wire                      resetn,
    input  wire                      x_wr_en_i,
    input  wire                [9:0] x_addr_i,
    input  wire                [7:0] x_data_i,
    input  wire                      w_wr_en_i,
    input  wire                [3:0] w_neuron_i,
    input  wire                [9:0] w_addr_i,
    input  wire                [7:0] w_data_i,
    input  wire                      b_wr_en_i,
    input  wire                [3:0] b_neuron_i,
    input  wire      [ACC_WIDTH-1:0] b_data_i,
    input  wire                      start_i,
    output wire                      busy_o,
    output wire [10*ACC_WIDTH-1:0]   scores_o,
    output reg                       done_o,
    output reg                       error_o,
    output wire                [3:0] class_o,
    output reg                [15:0] cycles_o
);
    localparam integer LG = (NPE == 1) ? 0 : $clog2(NPE);
    localparam integer GROUPS = (10 + NPE - 1) / NPE;
    localparam integer GW = (GROUPS <= 1) ? 1 : $clog2(GROUPS);
    localparam IDLE=0, LAUNCH=1, READ=2, WAIT_READ=3, SEND=4, WAIT_PE=5, BIAS=6, AM_GO=7, AM_WAIT=8;
    reg [3:0] state;
    reg [3:0] tile_q;
    reg [GW-1:0] group_q;
    reg [5:0] idx_q;
    reg [NPE-1:0] finished;
    reg signed [ACC_WIDTH-1:0] acc_q [0:9];
    reg signed [ACC_WIDTH-1:0] score_q [0:9];
    reg signed [ACC_WIDTH-1:0] bias_q [0:9];
    integer i;

    wire [6:0] len = (tile_q == 4'd12) ? 7'd16 : 7'd64;
    wire [9:0] addr = {tile_q, 6'd0} + idx_q;
    assign busy_o = (state != IDLE);

    wire signed [7:0] x_rd; wire x_rv;
    bram_sync_buffer #(.DATA_WIDTH(8), .ADDR_WIDTH(10)) x_ram (
        .clk(clk), .resetn(resetn), .wr_en(x_wr_en_i && !busy_o), .wr_addr(x_addr_i), .wr_data(x_data_i),
        .rd_en(state == READ), .rd_addr(addr), .rd_data(x_rd), .rd_valid(x_rv));

    // 클래스 n → 그룹 n / NPE, 차선 n % NPE (NPE는 2의 거듭제곱이라 시프트·마스크)
    wire [3:0] w_lane  = (NPE == 1) ? 4'd0 : (w_neuron_i & (NPE - 1));
    wire [3:0] w_group = w_neuron_i >> LG;
    wire [NPE-1:0] ready, lane_done, w_rv;
    wire [NPE*ACC_WIDTH-1:0] res;
    genvar g;
    generate for (g = 0; g < NPE; g = g + 1) begin : lane
        wire signed [7:0] w_rd;
        bram_sync_buffer #(.DATA_WIDTH(8), .ADDR_WIDTH(GW + 10)) w_ram (
            .clk(clk), .resetn(resetn),
            .wr_en(w_wr_en_i && !busy_o && w_neuron_i <= 9 && w_lane == g),
            .wr_addr({w_group[GW-1:0], w_addr_i}), .wr_data(w_data_i),
            .rd_en(state == READ), .rd_addr({group_q, addr}), .rd_data(w_rd), .rd_valid(w_rv[g]));
        wire signed [7:0] w_use = (group_q * NPE + g >= 10) ? 8'sd0 : w_rd;   // 클래스 10 이상 차선은 0
        wire [ACC_WIDTH-1:0] r;
        vector_dot_core #(.DATA_WIDTH(8), .ACC_WIDTH(ACC_WIDTH)) dot (
            .clk(clk), .resetn(resetn), .start_i(state == LAUNCH), .valid_i(state == SEND), .length_i({2'b00, len}),
            .a_i(x_rd), .b_i(w_use), .ready_o(ready[g]), .busy_o(), .done_o(lane_done[g]), .error_o(), .result_o(r));
        assign res[g*ACC_WIDTH +: ACC_WIDTH] = r;
    end endgenerate
    generate for (g = 0; g < 10; g = g + 1) begin : o
        assign scores_o[g*ACC_WIDTH +: ACC_WIDTH] = score_q[g];
    end endgenerate

    wire am_done;
    npu_argmax #(.N(10), .W(ACC_WIDTH)) argmax (
        .clk(clk), .resetn(resetn), .start_i(state == AM_GO), .scores_i(scores_o),
        .busy_o(), .done_o(am_done), .class_o(class_o));

    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE; tile_q <= 0; group_q <= 0; idx_q <= 0; finished <= 0;
            done_o <= 0; error_o <= 0; cycles_o <= 0;
            for (i = 0; i < 10; i = i + 1) begin acc_q[i] <= 0; score_q[i] <= 0; bias_q[i] <= 0; end
        end else begin
            done_o <= 0; error_o <= 0;
            if (busy_o) cycles_o <= cycles_o + 16'd1;
            if (b_wr_en_i && !busy_o && b_neuron_i <= 9) bias_q[b_neuron_i] <= b_data_i;
            if (state == LAUNCH) finished <= 0;
            else finished <= finished | lane_done;
            case (state)
                IDLE: if (start_i) begin
                    tile_q <= 0; group_q <= 0; idx_q <= 0; cycles_o <= 0; state <= LAUNCH;
                    for (i = 0; i < 10; i = i + 1) acc_q[i] <= 0;
                end
                LAUNCH: state <= READ;
                READ: state <= WAIT_READ;
                WAIT_READ: if (x_rv && (&w_rv)) state <= SEND;
                    else begin state <= IDLE; error_o <= 1; end
                SEND: if (&ready) begin
                    if (idx_q == len - 7'd1) state <= WAIT_PE;
                    else begin idx_q <= idx_q + 6'd1; state <= READ; end
                end
                WAIT_PE: if (&(finished | lane_done)) begin   // 모든 차선 완료 (같은 길이라 같은 클록)
                    for (i = 0; i < 10; i = i + 1)
                        if (i / NPE == group_q) acc_q[i] <= acc_q[i] + $signed(res[(i % NPE)*ACC_WIDTH +: ACC_WIDTH]);
                    idx_q <= 0;
                    if (group_q != GROUPS - 1) begin group_q <= group_q + 1'b1; state <= LAUNCH; end
                    else if (tile_q != 4'd12) begin group_q <= 0; tile_q <= tile_q + 4'd1; state <= LAUNCH; end
                    else state <= BIAS;
                end
                BIAS: begin
                    for (i = 0; i < 10; i = i + 1) score_q[i] <= acc_q[i] + bias_q[i];
                    state <= AM_GO;
                end
                AM_GO: state <= AM_WAIT;
                AM_WAIT: if (am_done) begin state <= IDLE; done_o <= 1; end
                default: begin state <= IDLE; error_o <= 1; end
            endcase
        end
    end
endmodule

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. 같은 TB를 -P tb_npu_npe.NPE=1·2·4·8로 네 번 컴파일합니다. NPE=4에서는 18장 모듈을 함께 돌려 점수가 같은지 보고 18장 사이클도 출력합니다. 마지막에 측정 사이클 네 개를 사이클 식과 맞춰 봅니다.

ch19/tb/tb_npu_npe.sv · systemverilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
// 19장 TB: NPE(1·2·4·8)를 바꿔 18장 INT8 실제 MNIST 모델·시험 5장을 돌리고 점수·클래스·사이클을 돌려준다.
// NPE=4이면 18장 npu_tiled_784_p(DW=8)를 함께 돌려 점수가 같은지 보고, 18장 사이클도 함께 출력한다.
module tb_npu_npe;
    parameter integer NPE = 4;
    localparam AW = 32;
    reg clk=0, rn=0, xwe=0, wwe=0, bwe=0, start=0;
    reg [9:0] xa=0, wa=0; reg [7:0] xd=0, wd=0; reg [3:0] wn=0, bn=0; reg [AW-1:0] bd=0;
    wire busy, done, err; wire [3:0] cls; wire [10*AW-1:0] scores; wire [15:0] cycles;
    npu_tiled_784_npe #(.NPE(NPE), .ACC_WIDTH(AW)) dut(.clk(clk), .resetn(rn), .x_wr_en_i(xwe), .x_addr_i(xa),
        .x_data_i(xd), .w_wr_en_i(wwe), .w_neuron_i(wn), .w_addr_i(wa), .w_data_i(wd), .b_wr_en_i(bwe), .b_neuron_i(bn),
        .b_data_i(bd), .start_i(start), .busy_o(busy), .scores_o(scores), .done_o(done), .error_o(err), .class_o(cls),
        .cycles_o(cycles));
    always #5 clk = ~clk;

    wire [10*AW-1:0] ref_scores; wire ref_done; wire [15:0] ref_cycles;
    generate if (NPE == 4) begin : same_as_ch18
        npu_tiled_784_p #(.DATA_WIDTH(8), .ACC_WIDTH(AW)) ch18 (.clk(clk), .resetn(rn), .x_wr_en_i(xwe), .x_addr_i(xa),
            .x_data_i(xd), .w_wr_en_i(wwe), .w_neuron_i(wn), .w_addr_i(wa), .w_data_i(wd), .b_wr_en_i(bwe),
            .b_neuron_i(bn), .b_data_i(bd), .start_i(start), .busy_o(), .scores_o(ref_scores), .done_o(ref_done),
            .error_o(), .class_o(), .cycles_o(ref_cycles));
    end endgenerate

    integer f, o, k, c, v, n, label, cases = 0, same = 0, cyc = 0, ref_cyc = 0;
    reg ref_seen = 0;
    always @(posedge clk) if (start) ref_seen <= 0; else if (ref_done) ref_seen <= 1;
    string vp, ap;

    initial begin
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap)) $fatal(1, "경로 인자");
        f = $fopen(vp, "r"); o = $fopen(ap, "w"); if (!f || !o) $fatal(1, "파일 열기");
        repeat (3) @(negedge clk); rn = 1;
        for (c = 0; c < 10; c = c + 1) for (k = 0; k < 784; k = k + 1) begin
            if ($fscanf(f, "%d", v) != 1) $fatal(1, "w"); @(negedge clk); wwe = 1; wn = c; wa = k; wd = v; end
        @(negedge clk); wwe = 0;
        for (c = 0; c < 10; c = c + 1) begin
            if ($fscanf(f, "%d", v) != 1) $fatal(1, "b"); @(negedge clk); bwe = 1; bn = c; bd = v; end
        @(negedge clk); bwe = 0;
        while ($fscanf(f, "%d %d", n, label) == 2) begin
            for (k = 0; k < 784; k = k + 1) begin
                if ($fscanf(f, "%d", v) != 1) $fatal(1, "x"); @(negedge clk); xwe = 1; xa = k; xd = v; end
            @(negedge clk); xwe = 0; start = 1; @(negedge clk); start = 0;
            while (!done) begin @(negedge clk); if (err) $fatal(1, "case %0d: error_o", n); end
            if (NPE == 4) begin
                while (!ref_seen) @(negedge clk);
                if (ref_scores !== scores) $fatal(1, "case %0d: 18장과 점수 다름", n);
                same = same + 1; ref_cyc = ref_cycles;
            end
            $fwrite(o, "%0d", n);
            for (c = 0; c < 10; c = c + 1) $fwrite(o, " %0d", $signed(scores[c*AW +: AW]));
            $fwrite(o, " %0d\n", cls);
            cases = cases + 1; cyc = cycles;
        end
        if (NPE == 4) $display("CH18_DW8_CYCLES %0d (pe_array4의 완료 레지스터 때문에 그룹마다 2클록 더 걸림)", ref_cyc);
        $display("RTL_SIM_PASS NPE=%0d cases=%0d same_as_ch18=%0d cycles=%0d", NPE, cases, same, cyc);
        $fclose(o); $finish;
    end
    initial begin #900000000; $fatal(1, "시간 초과"); end
endmodule

#!/usr/bin/env python3
"""19장 기준 모델: 18장 INT8 실제 MNIST 모델로 1·2·4·8PE 결과가 같은지 보고, 측정 사이클을 식으로 설명한다.

사이클 식 (파형에서 읽은 구조): 원소 1개 = READ·WAIT_READ·SEND 3클록, 타일·그룹 1회 = 3·len + A, 마무리 = B
  cycles(NPE) = G · Σ_t (3·len_t + A) + B,  G = ceil(10 / NPE),  len = 64 × 12 + 16
A·B는 4PE·8PE 측정값 두 개로 구하고, 1PE·2PE 측정값을 예측해 맞는지 확인한다.
"""
from __future__ import annotations
import argparse
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "ch18" / "reference"))
from precision_rtl import load, scores                   # noqa: E402
from p3_integer_reference import argmax_lowest            # noqa: E402

CASES = 5
LENS = [64] * 12 + [16]


def groups(npe):
    return -(-10 // npe)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--out"); ap.add_argument("--npe", type=int); ap.add_argument("--actual")
    ap.add_argument("--cycles", nargs="*", help="NPE=cycles 목록 (사이클 식 확인)")
    a = ap.parse_args()
    w, b, cases = load(8)
    cases = cases[:CASES]
    if a.out:
        with open(a.out, "w") as f:
            f.writelines(" ".join(map(str, r)) + "\n" for r in w)
            f.write(" ".join(map(str, b)) + "\n")
            for k, label, x in cases:
                f.write(f"{k} {label} " + " ".join(map(str, x)) + "\n")
    if a.actual:
        seen = 0
        for ln in open(a.actual):
            try:
                p = list(map(int, ln.split()))
            except ValueError:
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL 정수가 아닌 RTL 출력: {ln[:80]}")
            k = p[0]; s = scores(cases[k][2], w, b)
            if p[1:] != s + [argmax_lowest(s)]:
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL npe={a.npe} case={k}")
            seen += 1
        if seen == 0:
            raise SystemExit("PYTHON_RTL_COMPARE_FAIL RTL 결과 없음")
        print(f"PYTHON_RTL_COMPARE_PASS npe={a.npe} cases={seen} scores={10 * seen}")
    if a.cycles:
        meas = {int(k): int(v) for k, v in (c.split("=") for c in a.cycles)}
        base = sum(3 * n for n in LENS)                   # 그룹 1회의 원소 클록 합
        # 4PE·8PE로 A·B 구하기: c = G·(base + 13·A) + B
        g4, g8 = groups(4), groups(8)
        a13 = (meas[4] - meas[8]) // (g4 - g8) - base     # 13·A
        A, B = a13 // 13, meas[4] - g4 * (base + a13)
        assert a13 % 13 == 0, a13
        for npe, c in sorted(meas.items()):
            pred = groups(npe) * (base + 13 * A) + B
            print(f"CYCLES npe={npe} groups={groups(npe)} measured={c} formula={pred} speedup_vs_1pe={meas[1] / c:.2f}")
            if pred != c:
                raise SystemExit(f"CYCLE_MODEL_FAIL npe={npe}")
        print(f"CYCLE_MODEL_PASS A={A} B={B} element=3clk")


if __name__ == "__main__":
    main()

07코드 읽기 포인트

위치내용
사이클 식cycles = G · Σ(3·len + 3) + 12, G = ceil(10/NPE), len = 64×12 + 16. 원소 1개에 READ·WAIT_READ·SEND 3클록, 타일·그룹마다 LAUNCH 등 3클록, 끝에 Bias·argmax 12클록입니다. 파형에서 상태 길이를 세면 같은 식이 나옵니다.
8PE가 5배인 이유클래스 10개를 8개씩 나누면 그룹이 2개(8 + 2)이고, 둘째 그룹은 8차선 중 2개만 씁니다. 사이클은 그룹 수에 비례하므로 10/2 = 5배가 한계입니다. 10PE면 그룹 1개가 됩니다.
원소당 3클록BRAM 읽기 지연 1클록을 기다린 뒤 보내므로 원소마다 3클록이 듭니다. 읽기를 미리 내는 파이프라인으로 1클록에 가깝게 줄일 수 있지만, 이 장은 PE 수의 효과만 봅니다.
18장과의 차이NPE=4는 18장과 점수가 같지만 그룹마다 2클록 빠릅니다(7185 vs 7263, 차이 78 = 39그룹 × 2). 08장 pe_array4의 완료 레지스터 단계를 거치지 않고 07장 코어를 직접 쓰기 때문입니다.
BRAM 포트·자원차선마다 가중치 RAM 1개(읽기 포트 1개)를 둡니다. PE를 늘리면 DSP와 RAM 수가 함께 늘어납니다. 실제 DSP·BRAM·LUT·FF 숫자는 Vivado OOC 보고서(NPE=1·4·8 세 줄)로만 기록합니다 — 미실행.
