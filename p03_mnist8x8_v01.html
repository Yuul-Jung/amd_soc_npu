// 3장 디버그 콘솔: 8×8 MNIST 서브젝트를 UART 명령으로 확인한다. 1·2장과 같은 형식 (1바이트 명령 + 바이너리).
//
//   명령 페이로드                      응답                                         서브젝트 (STAGE)
//   S    없음                          "I"                                          모두
//   F    n (1~80) + 화소 n바이트        "F:OK" / "F:SHORT" (n<64) / "F:LONG" (n>64)   3-1 (1)  n번째 바이트에 last
//   p    addr                          "P:" 2자리 (프레임 없으면 "ERR")              3-1 (1)
//   W    층 class addr v(int8)          "OK"                                         3-2 (2)  층 0 = 64→10, 1 = 64→8, 2 = 8→10
//   B    층 class b2 b1 b0 (int24)      "OK"                                         3-2 (2)
//   R    없음                          "R" 클래스 ":" 점수 7자리 ×10                3-2 (2)  64→10
//   H    없음                          "H" 클래스 ":" 점수 7자리 ×8                 3-3 (3)  64→8
//   G    없음                          "G" 클래스 ":" 점수 7자리 ×10                3-3 (3)  8→10 (프레임 앞 8원소)
//   Z    shift (0~31)                  "OK"                                         3-4 (4)
//   M    없음                          "M" 클래스 ":" 은닉 2자리 ×8 " " 점수 ×10     3-4 (4)  MLP
//   그 밖 (CR·LF 제외) "?",  범위 밖 "ERR",  코어 오류(프레임 없음 등) "ERR". 끝은 모두 \r\n
// 층 1은 3-3 64→8 코어와 3-4 MLP 1층에, 층 2는 3-3 8→10 코어와 MLP 2층에 함께 쓴다.
`timescale 1ns/1ps
`default_nettype none
module p03_console #(
    parameter integer STAGE = 4
) (
    input  wire               clk,
    input  wire               resetn,
    input  wire [7:0]         rx_data_i,
    input  wire               rx_valid_i,
    output wire               rx_ready_o,
    output reg  [7:0]         tx_data_o,
    output reg                tx_valid_o,
    input  wire               tx_ready_i,
    // 묶음으로
    output wire [7:0]         s_data_o,
    output wire               s_valid_o,
    output wire               s_last_o,
    input  wire               s_ready_i,
    input  wire               frame_valid_i,
    output reg                frame_ack_o,
    input  wire               err_short_i,
    input  wire               err_long_i,
    output reg                px_rd_o,
    output reg  [5:0]         px_addr_o,
    input  wire signed [7:0]  px_data_i,
    input  wire               px_valid_i,
    output reg                w_wr_o,
    output reg                b_wr_o,
    output reg  [1:0]         layer_o,
    output reg  [3:0]         class_o,
    output reg  [5:0]         addr_o,
    output reg  signed [7:0]  w_o,
    output reg  signed [23:0] b_o,
    output reg  [4:0]         shift_o,
    output reg  [1:0]         run_sel_o,
    output reg                run_o,
    input  wire               run_done_i,
    input  wire               run_err_i,
    input  wire [3:0]         run_class_i,
    input  wire [249:0]       run_scores_i,
    input  wire [63:0]        run_hidden_i,
    // 표시
    output reg  [15:0]        view_o,
    output reg  [3:0]         cls_o,
    output reg                cls_ok_o,
    output reg                fshort_o,
    output reg                flong_o
);
    localparam [3:0] S_IDLE = 4'd0, S_RECV = 4'd1, S_ISSUE = 4'd2, S_ACK = 4'd3, S_STREAM = 4'd4,
                     S_FWAIT = 4'd5, S_PX = 4'd6, S_RUN = 4'd7, S_LOAD = 4'd8, S_SEND = 4'd9;
    localparam integer MSG_MAX = 104;

    reg [3:0] state;
    reg [7:0] opc;
    reg [7:0] pl [0:4];
    reg [2:0] need, got;
    reg [6:0] n_left;
    reg       e_short, e_long;
    reg [1:0] fw;
    reg [7:0] msg [0:MSG_MAX-1];
    reg [6:0] msg_len, idx;
    integer   k, p;

    function [7:0] hex(input [3:0] n);
        hex = (n < 4'd10) ? (8'h30 + {4'd0, n}) : (8'h37 + {4'd0, n});
    endfunction

    function [2:0] plen(input [7:0] c);
        case (c)
            "S":     plen = 3'd0;
            "F":     plen = 3'd1;
            "p":     plen = 3'd1;
            "W":     plen = (STAGE >= 2) ? 3'd4 : 3'd7;
            "B":     plen = (STAGE >= 2) ? 3'd5 : 3'd7;
            "R":     plen = (STAGE >= 2) ? 3'd0 : 3'd7;
            "H":     plen = (STAGE >= 3) ? 3'd0 : 3'd7;
            "G":     plen = (STAGE >= 3) ? 3'd0 : 3'd7;
            "Z":     plen = (STAGE >= 4) ? 3'd1 : 3'd7;
            "M":     plen = (STAGE >= 4) ? 3'd0 : 3'd7;
            default: plen = 3'd7;
        endcase
    endfunction

    // 층별 범위: 층 0 class<10·addr<64, 층 1 class<8·addr<64, 층 2 class<10·addr<8. 3-2 빌드에는 층 0만
    wire [7:0] lay = pl[0], cl = pl[1], ad = pl[2];
    wire layer_ok = (lay == 8'd0) || (lay <= 8'd2 && STAGE >= 3);
    wire wb_ok = layer_ok && (lay == 8'd1 ? cl < 8'd8 : cl < 8'd10);
    wire fields_ok =
        (opc == "F") ? (pl[0] >= 8'd1 && pl[0] <= 8'd80) :
        (opc == "p") ? (pl[0] < 8'd64) :
        (opc == "W") ? (wb_ok && (lay == 8'd2 ? ad < 8'd8 : ad < 8'd64)) :
        (opc == "B") ? wb_ok :
        (opc == "Z") ? (pl[0] < 8'd32) : 1'b1;

    // 프레임 바이트는 UART에서 묶음으로 바로 흘린다 (묶음의 ready로 흐름 제어)
    assign s_data_o   = rx_data_i;
    assign s_valid_o  = (state == S_STREAM) && rx_valid_i;
    assign s_last_o   = (n_left == 7'd1);
    assign rx_ready_o = (state == S_IDLE) || (state == S_RECV) || (state == S_STREAM && s_ready_i);
    wire   rx_take    = rx_valid_i && rx_ready_o;

    task put_err;
        begin
            msg[0] <= "E"; msg[1] <= "R"; msg[2] <= "R"; msg[3] <= 8'h0D; msg[4] <= 8'h0A; msg_len <= 7'd5;
        end
    endtask

    always @(posedge clk) begin
        if (!resetn) begin
            state <= S_IDLE; opc <= 8'd0; need <= 3'd0; got <= 3'd0; n_left <= 7'd0;
            e_short <= 1'b0; e_long <= 1'b0; fw <= 2'd0; msg_len <= 7'd0; idx <= 7'd0;
            tx_data_o <= 8'd0; tx_valid_o <= 1'b0; frame_ack_o <= 1'b0; px_rd_o <= 1'b0; px_addr_o <= 6'd0;
            w_wr_o <= 1'b0; b_wr_o <= 1'b0; layer_o <= 2'd0; class_o <= 4'd0; addr_o <= 6'd0;
            w_o <= 8'sd0; b_o <= 24'sd0; shift_o <= 5'd0; run_sel_o <= 2'd0; run_o <= 1'b0;
            view_o <= 16'd0; cls_o <= 4'd0; cls_ok_o <= 1'b0; fshort_o <= 1'b0; flong_o <= 1'b0;
        end else begin
            frame_ack_o <= 1'b0; px_rd_o <= 1'b0; w_wr_o <= 1'b0; b_wr_o <= 1'b0; run_o <= 1'b0;
            case (state)
                S_IDLE: if (rx_take && rx_data_i != 8'h0D && rx_data_i != 8'h0A) begin
                    opc <= rx_data_i; got <= 3'd0; need <= plen(rx_data_i);
                    if (plen(rx_data_i) == 3'd7) begin
                        msg[0] <= "?"; msg[1] <= 8'h0D; msg[2] <= 8'h0A; msg_len <= 7'd3; state <= S_LOAD;
                    end else if (plen(rx_data_i) == 3'd0) begin
                        state <= S_ISSUE;
                    end else begin
                        state <= S_RECV;
                    end
                end
                S_RECV: if (rx_take) begin
                    pl[got] <= rx_data_i;
                    if (got + 3'd1 == need) state <= S_ISSUE;
                    else got <= got + 3'd1;
                end
                S_ISSUE: begin
                    if (!fields_ok) begin
                        put_err; state <= S_LOAD;
                    end else case (opc)
                        "S": begin msg[0] <= "I"; msg[1] <= 8'h0D; msg[2] <= 8'h0A; msg_len <= 7'd3; state <= S_LOAD; end
                        "F": begin                // 들고 있는 프레임은 먼저 놓는다
                            n_left <= pl[0][6:0]; e_short <= 1'b0; e_long <= 1'b0;
                            frame_ack_o <= frame_valid_i; state <= S_ACK;
                        end
                        "p": begin
                            if (!frame_valid_i) begin put_err; state <= S_LOAD; end
                            else begin px_rd_o <= 1'b1; px_addr_o <= pl[0][5:0]; state <= S_PX; end
                        end
                        "W", "B": begin
                            layer_o <= pl[0][1:0]; class_o <= pl[1][3:0];
                            if (opc == "W") begin addr_o <= pl[2][5:0]; w_o <= pl[3]; w_wr_o <= 1'b1; end
                            else begin b_o <= {pl[2], pl[3], pl[4]}; b_wr_o <= 1'b1; end
                            msg[0] <= "O"; msg[1] <= "K"; msg[2] <= 8'h0D; msg[3] <= 8'h0A; msg_len <= 7'd4; state <= S_LOAD;
                        end
                        "Z": begin
                            shift_o <= pl[0][4:0];
                            msg[0] <= "O"; msg[1] <= "K"; msg[2] <= 8'h0D; msg[3] <= 8'h0A; msg_len <= 7'd4; state <= S_LOAD;
                        end
                        default: begin            // R H G M
                            run_sel_o <= (opc == "R") ? 2'd0 : (opc == "H") ? 2'd1 : (opc == "G") ? 2'd2 : 2'd3;
                            run_o <= 1'b1; state <= S_RUN;
                        end
                    endcase
                end
                S_ACK: state <= S_STREAM;         // ack가 들어간 다음 클록부터 FILL
                S_STREAM: begin
                    if (err_short_i) e_short <= 1'b1;
                    if (err_long_i)  e_long  <= 1'b1;
                    if (rx_take) begin
                        if (n_left == 7'd1) begin fw <= 2'd0; state <= S_FWAIT; end
                        n_left <= n_left - 7'd1;
                    end
                end
                S_FWAIT: begin                    // 마지막 바이트 뒤 판정이 나올 때까지 2클록
                    if (err_short_i) e_short <= 1'b1;
                    if (err_long_i)  e_long  <= 1'b1;
                    fw <= fw + 2'd1;
                    if (fw == 2'd2) begin
                        msg[0] <= "F"; msg[1] <= ":";
                        if (frame_valid_i && !e_long) begin
                            msg[2] <= "O"; msg[3] <= "K"; msg[4] <= 8'h0D; msg[5] <= 8'h0A; msg_len <= 7'd6;
                        end else if (e_long) begin
                            msg[2] <= "L"; msg[3] <= "O"; msg[4] <= "N"; msg[5] <= "G"; msg[6] <= 8'h0D; msg[7] <= 8'h0A; msg_len <= 7'd8;
                        end else begin
                            msg[2] <= "S"; msg[3] <= "H"; msg[4] <= "O"; msg[5] <= "R"; msg[6] <= "T"; msg[7] <= 8'h0D; msg[8] <= 8'h0A; msg_len <= 7'd9;
                        end
                        fshort_o <= !frame_valid_i && !e_long;
                        flong_o  <= e_long;
                        state <= S_LOAD;
                    end
                end
                S_PX: if (px_valid_i) begin
                    msg[0] <= "P"; msg[1] <= ":"; msg[2] <= hex(px_data_i[7:4]); msg[3] <= hex(px_data_i[3:0]);
                    msg[4] <= 8'h0D; msg[5] <= 8'h0A; msg_len <= 7'd6;
                    view_o <= {{8{px_data_i[7]}}, px_data_i};
                    state <= S_LOAD;
                end
                S_RUN: if (run_done_i) begin
                    if (run_err_i) begin
                        put_err;
                    end else begin
                        msg[0] <= opc; msg[1] <= hex(run_class_i); msg[2] <= ":";
                        view_o <= {12'd0, run_class_i}; cls_o <= run_class_i; cls_ok_o <= 1'b1;
                        if (opc == "M") begin
                            for (k = 0; k < 8; k = k + 1) begin
                                msg[3 + 2*k] <= hex(run_hidden_i[8*k + 4 +: 4]);
                                msg[4 + 2*k] <= hex(run_hidden_i[8*k +: 4]);
                            end
                            msg[19] <= " ";
                            for (p = 0; p < 10; p = p + 1) begin
                                for (k = 0; k < 7; k = k + 1)
                                    msg[20 + 8*p + k] <= hex((k == 0) ? {{3{run_scores_i[25*p + 24]}}, run_scores_i[25*p + 24]}
                                                                     : run_scores_i[25*p + 4*(6-k) +: 4]);
                                if (p < 9) msg[27 + 8*p] <= ",";
                            end
                            msg[99] <= 8'h0D; msg[100] <= 8'h0A; msg_len <= 7'd101;
                        end else begin
                            for (p = 0; p < 10; p = p + 1) begin
                                for (k = 0; k < 7; k = k + 1)
                                    msg[3 + 8*p + k] <= hex((k == 0) ? {{3{run_scores_i[25*p + 24]}}, run_scores_i[25*p + 24]}
                                                                    : run_scores_i[25*p + 4*(6-k) +: 4]);
                                if (p < 9) msg[10 + 8*p] <= ",";
                            end
                            if (opc == "H") begin         // 8개
                                msg[66] <= 8'h0D; msg[67] <= 8'h0A; msg_len <= 7'd68;
                            end else begin
                                msg[82] <= 8'h0D; msg[83] <= 8'h0A; msg_len <= 7'd84;
                            end
                        end
                    end
                    state <= S_LOAD;
                end
                S_LOAD: begin
                    tx_data_o <= msg[0]; tx_valid_o <= 1'b1; idx <= 7'd0; state <= S_SEND;
                end
                S_SEND: if (tx_valid_o && tx_ready_i) begin
                    if (idx + 7'd1 == msg_len) begin
                        tx_valid_o <= 1'b0; state <= S_IDLE;
                    end else begin
                        tx_data_o <= msg[idx + 7'd1]; idx <= idx + 7'd1;
                    end
                end

"""3장 8×8 MNIST 모델 학습·INT8 양자화·정수 평가 (PC, numpy).

  python3 p03/model/train_p03.py            # 학습 → p03/model/p03_models.json, 평가 줄 출력

모델 두 개 (같은 입력):
  선형 64→10        : 3-2 서브젝트 (p3_group10_core 64→10). 점수 = x·W + b
  MLP 64→8→10       : 3-4 서브젝트 (p3_mlp_64_8_10). 은닉 = sat_int8(round_shift_away(ReLU(x·W1 + b1), shift))

import numpy as np

ROOT = Path(__file__).resolve().parents[2]

    sat_int8, round_shift_away, linear_64x10, mlp_64_8_10, argmax_lowest)

SEED = 303

def to8x8(img784):
    """28×28 화소 → 8×8 INT8 (0..127). 정수 연산만."""
    a = np.frombuffer(img784, dtype=np.uint8).reshape(28, 28)[2:26, 2:26].astype(np.int64)
    s = a.reshape(8, 3, 8, 3).sum(axis=(1, 3))
    avg = (s + 4) // 9
    return [sat_int8(round_shift_away(int(v), 1)) for v in avg.reshape(64)]


def load(split):
    xs = np.array([to8x8(im) for im in images(split)], dtype=np.int64)
    return xs, np.array(labels(split), dtype=np.int64)


def sat(v, bits):
    lo, hi = -(1 << (bits - 1)), (1 << (bits - 1)) - 1
    return max(lo, min(hi, v))


def rnd(v):
    """절반은 0에서 먼 쪽 (Python round의 짝수 반올림을 쓰지 않는다)."""
    return int(np.sign(v) * np.floor(abs(v) + 0.5))


def qweights(w):
    s = float(np.abs(w).max()) / 127.0
    return [[sat(rnd(v / s), 8) for v in row] for row in w], s


def softmax_ce(logits, y):
    z = logits - logits.max(axis=1, keepdims=True)
    p = np.exp(z)
    p /= p.sum(axis=1, keepdims=True)
    g = p.copy()
    g[np.arange(len(y)), y] -= 1
    return g / len(y)


def adam_train(params, grad_fn, x, y, epochs, lr, rng):
    m = [np.zeros_like(p) for p in params]
    v = [np.zeros_like(p) for p in params]
    t = 0
    for _ in range(epochs):
        order = rng.permutation(len(x))
        for i in range(0, len(x), 128):
            idx = order[i:i + 128]
            grads = grad_fn(params, x[idx], y[idx])
            t += 1
            for k, (p, g) in enumerate(zip(params, grads)):
                m[k] = 0.9 * m[k] + 0.1 * g
                v[k] = 0.999 * v[k] + 0.001 * g * g
                p -= lr * (m[k] / (1 - 0.9 ** t)) / (np.sqrt(v[k] / (1 - 0.999 ** t)) + 1e-8)
    return params


def main():
    rng = np.random.default_rng(SEED)
    xtr, ytr = load("train")
    xte, yte = load("test")
    ftr, fte = xtr.astype(np.float32) / 127.0, xte.astype(np.float32) / 127.0

    # ── 선형 64→10 ──
    W = rng.normal(0, 0.05, (64, 10)).astype(np.float32)
    b = np.zeros(10, np.float32)

    def g_lin(ps, x, y):
        W, b = ps
        g = softmax_ce(x @ W + b, y)
        return [x.T @ g, g.sum(axis=0)]

    W, b = adam_train([W, b], g_lin, ftr, ytr, 8, 0.01, rng)
    f_lin = float(((fte @ W + b).argmax(axis=1) == yte).mean())
    wq, sw = qweights(W.T)                                          # wq[class][i]
    bq = [sat(rnd(float(v) / ((1 / 127.0) * sw)), 24) for v in b]

    # ── MLP 64→8→10 ──
    W1 = rng.normal(0, 0.2, (64, 8)).astype(np.float32)
    b1 = np.zeros(8, np.float32)
    W2 = rng.normal(0, 0.3, (8, 10)).astype(np.float32)
    b2 = np.zeros(10, np.float32)

    def g_mlp(ps, x, y):
        W1, b1, W2, b2 = ps
        h_pre = x @ W1 + b1
        h = np.maximum(h_pre, 0)
        g = softmax_ce(h @ W2 + b2, y)
        gh = (g @ W2.T) * (h_pre > 0)
        return [x.T @ gh, gh.sum(axis=0), h.T @ g, g.sum(axis=0)]

    W1, b1, W2, b2 = adam_train([W1, b1, W2, b2], g_mlp, ftr, ytr, 15, 0.005, rng)
    h_te = np.maximum(fte @ W1 + b1, 0)
    f_mlp = float(((h_te @ W2 + b2).argmax(axis=1) == yte).mean())

    w1q, s1 = qweights(W1.T)                                        # w1q[hidden][i]
    b1q = [sat(rnd(float(v) / ((1 / 127.0) * s1)), 24) for v in b1]
    # 은닉 shift: 학습 데이터 은닉 누산의 99.9 백분위가 127 안에 들어오는 가장 작은 shift
    acc = np.maximum(xtr[:5000] @ np.array(w1q).T + np.array(b1q), 0)
    p999 = float(np.percentile(acc, 99.9))
    shift = next(s for s in range(32) if p999 / (1 << s) <= 127)
    h_unit = (1 / 127.0) * s1 * (1 << shift)                        # 은닉 정수 1의 실수 크기
    w2q, s2 = qweights(W2.T)                                        # w2q[class][hidden]
    b2q = [sat(rnd(float(v) / (h_unit * s2)), 24) for v in b2]

    # ── 정수 평가 (RTL 규칙 함수) ──
    q_lin = sum(argmax_lowest(linear_64x10(list(x), wq, bq)) == y for x, y in zip(xte.tolist(), yte))

    n = len(yte)

        "preprocess": "28x28 -> center 24x24 -> 3x3 mean (sum+4)//9 -> sat_int8(round_shift_away(p,1)), row-major 64",

    print(f"P03_MODEL_WRITTEN {OUT.relative_to(ROOT)}")
    return 0

// 유효 프레임은 frame_ack_i까지 유지되며, 그동안 s_ready_o=0(Busy)으로 다음 입력을 막는다.
module p3_frame64_buffer(
    input wire clk,
    input wire resetn,
    input wire [7:0] s_data_i,
    input wire s_valid_i,
    input wire s_last_i,
    output wire s_ready_o,
    output wire frame_valid_o,
    input wire frame_ack_i,
    input wire rd_en_i,
    input wire [5:0] rd_addr_i,
    output reg signed [7:0] rd_data_o,
    output reg rd_valid_o,
    output reg err_short_o,
    output reg err_long_o
);
    localparam [1:0] FILL=2'd0, DROP=2'd1, FULL=2'd2;
    reg [1:0] state;
    reg [5:0] count_q;
    reg signed [7:0] mem [0:63];

    assign s_ready_o = resetn && (state != FULL);
    assign frame_valid_o = (state == FULL);
    wire accept = s_valid_i && s_ready_o;

    // 저장 배열은 Reset 없이 동기 쓰기·동기 읽기(1클록 지연)로 메모리 추론을 허용한다.
    always @(posedge clk) begin
        if (accept && state == FILL)
            mem[count_q] <= s_data_i;
        if (rd_en_i)
            rd_data_o <= mem[rd_addr_i];
    end

    always @(posedge clk) begin
        if (!resetn) begin
            state <= FILL;
            count_q <= 6'd0;
            rd_valid_o <= 1'b0;
            err_short_o <= 1'b0;
            err_long_o <= 1'b0;
        end else begin
            err_short_o <= 1'b0;
            err_long_o <= 1'b0;
            // 읽기 데이터는 유효 프레임 보관 중 요청한 경우에만 유효로 표시한다.
            rd_valid_o <= rd_en_i && (state == FULL);
            case (state)
                FILL: if (accept) begin
                    if (s_last_i) begin
                        count_q <= 6'd0;
                        if (count_q == 6'd63)
                            state <= FULL;
                        else
                            err_short_o <= 1'b1;
                    end else if (count_q == 6'd63) begin
                        // 64번째에 last가 없으면 긴 프레임: 다음 last까지 버리고 재동기화.
                        count_q <= 6'd0;
                        err_long_o <= 1'b1;
                        state <= DROP;
                    end else begin
                        count_q <= count_q + 6'd1;
                    end
                end
                DROP: if (accept && s_last_i)
                    state <= FILL;
                FULL: if (frame_ack_i)
                    state <= FILL;
                default: state <= FILL;
            endcase
        end
    end
endmodule

def quantize_pixel_u8(pixel):
    """8비트 무부호 밝기(0..255)를 signed INT8 입력(0..127)으로 변환한다."""
    if not 0 <= pixel <= 255:
        raise ValueError("pixel must be 0..255")
    return sat_int8(round_shift_away(pixel, 1))


def glyph_frame(digit):
    """획=255, 획 옆=128(회색), 배경=0 밝기를 양자화해 64원소 행 우선 벡터로 만든다."""
    rows = GLYPHS[digit]
    assert len(rows) == 8 and all(len(r) == 8 for r in rows)
    out = []
    for r in range(8):
        for c in range(8):
            if rows[r][c] == "#":
                p = 255
            elif (c > 0 and rows[r][c - 1] == "#") or (c < 7 and rows[r][c + 1] == "#"):
                p = 128
            else:
                p = 0
            out.append(quantize_pixel_u8(p))
    return out


def model(stream):
    """RTL 계약의 기준 상태기계. 수락될 프레임 목록과 오류 횟수를 반환한다."""
    state, buf, frames, short, long_ = "FILL", [], [], 0, 0
    drop_next = False
    for item in stream:
        kind = item[0]
        if kind == "R":
            state, buf = "FILL", []
            continue
        if kind == "Q":
            drop_next = True
            continue
        _, value, last = item
        if state == "FILL":
            buf.append(value)
            if last:
                if len(buf) == 64:
                    if drop_next:
                        drop_next = False  # FULL 상태에서 Reset되어 출력되지 않음
                    else:
                        frames.append(buf)
                else:
                    short += 1
                buf = []
            elif len(buf) == 64:
                long_ += 1
                buf, state = [], "DROP"
        elif state == "DROP" and last:
            state = "FILL"
    return frames, short, long_


def frame_items(values, last_at=None):
    last_at = len(values) if last_at is None else last_at
    return [("D", v, i + 1 == last_at) for i, v in enumerate(values)]

from __future__ import annotations


def sat_int8(value: int) -> int:
    """부호 있는 INT8 범위 [-128, 127]로 포화한다."""
    return max(-128, min(127, value))


def round_shift_away(value: int, shift: int) -> int:
    """2**shift로 나눈 뒤 절반은 0에서 먼 쪽으로 반올림한다."""
    if shift < 0:
        raise ValueError("shift must be >= 0")
    if shift == 0:
        return value
    magnitude = (abs(value) + (1 << (shift - 1))) >> shift
    return -magnitude if value < 0 else magnitude


def dot_signed(x: list[int], w: list[int]) -> int:
    if len(x) != len(w):
        raise ValueError("vector lengths must match")
    return sum(a * b for a, b in zip(x, w))


def dot_tiled_64(x: list[int], w: list[int]) -> int:
    """최대 64원소씩 처리해 부분 누산을 결합한다."""
    if len(x) != len(w):
        raise ValueError("vector lengths must match")
    return sum(dot_signed(x[i:i + 64], w[i:i + 64])
               for i in range(0, len(x), 64))


def argmax_lowest(scores: list[int]) -> int:
    if not scores:
        raise ValueError("scores must not be empty")
    return max(range(len(scores)), key=lambda i: (scores[i], -i))


def linear_64x10(x: list[int], w: list[list[int]], b: list[int]) -> list[int]:
    if len(x) != 64 or len(w) != 10 or len(b) != 10:
        raise ValueError("require 64 inputs, 10 neurons, 10 biases")
    if any(len(row) != 64 for row in w):
        raise ValueError("each weight row must contain 64 values")
    return [dot_signed(x, row) + bias for row, bias in zip(w, b)]


def mlp_64_8_10(x: list[int], w1: list[list[int]], b1: list[int],
                w2: list[list[int]], b2: list[int], shift: int) -> list[int]:

        raise ValueError("first layer shape must be 64 -> 8")
    if any(len(row) != 64 for row in w1):
        raise ValueError("first layer weight width must be 64")
    if len(w2) != 10 or len(b2) != 10 or any(len(row) != 8 for row in w2):
        raise ValueError("second layer shape must be 8 -> 10")
    hidden = [sat_int8(round_shift_away(max(0, dot_signed(x, w) + b), shift))
              for w, b in zip(w1, b1)]
    return [dot_signed(hidden, w) + b for w, b in zip(w2, b2)]

`timescale 1ns/1ps
// (추가) 14장: 4PE를 3그룹(4+4+2)으로 재사용하는 64입력·10분류 정수 Dense 코어.
// class = 4*group + lane. group 2의 lane 2·3은 무효이므로 valid를 막아 누산하지 않고 점수에도 쓰지 않는다.
// x는 13장 p3_frame64_buffer 읽기 포트(1클록 지연)에서 읽는다.
// 가중치는 lane별 bank(bank l = class l, l+4, l+8)에 저장해 매 클록 4개를 병렬로 읽는다.
// score[c] = sum_i x[i]*W[c][i] + bias[c] (signed 25비트, 정수 기준모델 linear_64x10과 동일)
// class_o = 최대 점수 class, 동점이면 낮은 번호(argmax_lowest)
// (변경) 15장 Dense 계층 재사용을 위해 입력 길이·출력 수를 파라미터로 둔다. 기본값 64→10이 14장 동작이다.
// (변경) 18장 정밀도 비교를 위해 데이터·누산·Bias·점수 폭을 파라미터로 둔다.
//   기본값 DATA_W 8 / ACC_W 24 / BIAS_W 24 / SCORE_W 25가 14장 INT8 동작이다.
//   INT16 예: DATA_W 16 / ACC_W 38(64항 최악 2^36) / SCORE_W ≥ max(ACC_W, BIAS_W) + 1.
module p3_group10_core #(
    parameter integer IN_LEN = 64,
    parameter integer OUT_N = 10,
    parameter integer DATA_W = 8,
    parameter integer ACC_W = 24,
    parameter integer BIAS_W = 24,
    parameter integer SCORE_W = 25
)(
    input wire clk,
    input wire resetn,
    input wire start_i,
    input wire x_valid_i,
    output wire x_rd_en_o,
    output wire [5:0] x_rd_addr_o,
    input wire signed [DATA_W-1:0] x_rd_data_i,
    input wire w_wr_en_i,
    input wire b_wr_en_i,
    input wire [3:0] wr_class_i,
    input wire [5:0] wr_addr_i,
    input wire signed [DATA_W-1:0] w_data_i,
    input wire signed [BIAS_W-1:0] b_data_i,
    output wire load_ready_o,
    output reg busy_o,
    output reg done_o,
    output reg error_o,
    output reg [3:0] class_o,
    output wire [OUT_N*SCORE_W-1:0] scores_o
);
    localparam [2:0] IDLE=3'd0, CLEAR=3'd1, RUN=3'd2, DRAIN=3'd3, CAPT=3'd4, FINISH=3'd5;
    reg [2:0] state;
    reg [1:0] group_q;
    reg [5:0] idx_q;
    reg drain_q;
    reg issue_q;

    // lane별 가중치 bank: 주소 = group*64 + index
    reg signed [DATA_W-1:0] bank0 [0:191];
    reg signed [DATA_W-1:0] bank1 [0:191];
    reg signed [DATA_W-1:0] bank2 [0:191];
    reg signed [DATA_W-1:0] bank3 [0:191];
    reg signed [DATA_W-1:0] w_q0, w_q1, w_q2, w_q3;
    reg signed [BIAS_W-1:0] bias [0:OUT_N-1];
    reg signed [SCORE_W-1:0] score_q [0:OUT_N-1];
    // (변경) 마지막 그룹 번호: 출력 수를 4개 차선으로 나눈 그룹 수 - 1
    localparam [1:0] LAST_GROUP = (OUT_N - 1) / 4;

    assign load_ready_o = resetn && !busy_o;
    assign x_rd_en_o = (state == RUN);
    assign x_rd_addr_o = idx_q;

    // (변경) class = 4*group + lane < OUT_N 인 차선만 유효(10분류: group 2는 lane 0·1만)
    wire [3:0] lane_en = {({group_q, 2'd3} < OUT_N), ({group_q, 2'd2} < OUT_N),
                          ({group_q, 2'd1} < OUT_N), ({group_q, 2'd0} < OUT_N)};
    wire lane_clear = (state == CLEAR);
    wire signed [ACC_W-1:0] acc0, acc1, acc2, acc3;
    npu_dsp_mac_lane #(.DATA_WIDTH(DATA_W), .ACC_WIDTH(ACC_W)) lane0(.clk(clk), .resetn(resetn),
        .clear_i(lane_clear), .valid_i(issue_q && lane_en[0]),
        .x_i(x_rd_data_i), .w_i(w_q0), .acc_o(acc0));
    npu_dsp_mac_lane #(.DATA_WIDTH(DATA_W), .ACC_WIDTH(ACC_W)) lane1(.clk(clk), .resetn(resetn),
        .clear_i(lane_clear), .valid_i(issue_q && lane_en[1]),
        .x_i(x_rd_data_i), .w_i(w_q1), .acc_o(acc1));
    npu_dsp_mac_lane #(.DATA_WIDTH(DATA_W), .ACC_WIDTH(ACC_W)) lane2(.clk(clk), .resetn(resetn),
        .clear_i(lane_clear), .valid_i(issue_q && lane_en[2]),
        .x_i(x_rd_data_i), .w_i(w_q2), .acc_o(acc2));
    npu_dsp_mac_lane #(.DATA_WIDTH(DATA_W), .ACC_WIDTH(ACC_W)) lane3(.clk(clk), .resetn(resetn),
        .clear_i(lane_clear), .valid_i(issue_q && lane_en[3]),
        .x_i(x_rd_data_i), .w_i(w_q3), .acc_o(acc3));

    // 쓰기 수락: Busy가 아니고 class 0..9일 때만
    wire wr_any = w_wr_en_i || b_wr_en_i;
    wire wr_ok = load_ready_o && (wr_class_i < OUT_N);
    wire [7:0] wr_bank_addr = {wr_class_i[3:2], wr_addr_i};
    wire [7:0] rd_bank_addr = {group_q, idx_q};

    // 가중치 bank: Reset 없는 동기 쓰기·동기 읽기(BRAM 추론 후보, 실제 매핑은 합성으로 확인)
    always @(posedge clk) begin
        if (w_wr_en_i && wr_ok) begin
            case (wr_class_i[1:0])
                2'd0: bank0[wr_bank_addr] <= w_data_i;
                2'd1: bank1[wr_bank_addr] <= w_data_i;
                2'd2: bank2[wr_bank_addr] <= w_data_i;
                2'd3: bank3[wr_bank_addr] <= w_data_i;
            endcase
        end
        if (state == RUN) begin
            w_q0 <= bank0[rd_bank_addr];
            w_q1 <= bank1[rd_bank_addr];
            w_q2 <= bank2[rd_bank_addr];
            w_q3 <= bank3[rd_bank_addr];
        end
    end

    // (변경) 19장 타이밍 결함 수정: 10개 점수를 한 클록에 직렬 비교(9단)하던 argmax를 누적형으로 바꾼다.
    //   CAPT에서 lane 쌍(0·1, 2·3)의 승자를 레지스터에 저장하고(1단), 다음 CLEAR에서 두 승자와
    //   누적 최대값(best_s_q)을 비교해 합친다(2단). 마지막 그룹은 FINISH에서 합친다.
    //   class_o·done 시점과 동점 규칙(낮은 번호)은 그대로.
    integer k;
    reg have_best_q;
    reg [3:0] best_c_q;
    reg signed [SCORE_W-1:0] best_s_q;
    reg signed [SCORE_W-1:0] s01, s23;   // 직전 CAPT 그룹의 lane 쌍 승자
    reg [3:0] c01, c23;
    reg v23_q;                           // lane 2 유효(lane 0은 항상 유효)
    // lane 쌍 비교(CAPT): 높은 번호는 엄격히 클 때만 이긴다(동점 = 낮은 번호)
    wire signed [SCORE_W-1:0] ns0 = add_bias(acc0, bias[{group_q, 2'd0}]);
    wire signed [SCORE_W-1:0] ns1 = add_bias(acc1, bias[{group_q, 2'd1}]);
    wire signed [SCORE_W-1:0] ns2 = add_bias(acc2, bias[{group_q, 2'd2}]);
    wire signed [SCORE_W-1:0] ns3 = add_bias(acc3, bias[{group_q, 2'd3}]);
    wire take1 = lane_en[1] && (ns1 > ns0);
    wire take3 = lane_en[3] && (ns3 > ns2);
    wire take23 = v23_q && (s23 > s01);
    wire signed [SCORE_W-1:0] grp_s = take23 ? s23 : s01;
    wire [3:0] grp_c = take23 ? c23 : c01;
    // 누적값은 앞 그룹(낮은 번호)이므로 그룹 최대가 엄격히 클 때만 바꾼다
    wire take_grp = !have_best_q || (grp_s > best_s_q);
    wire [3:0] best_class = take_grp ? grp_c : best_c_q;

    genvar g;
    generate
        for (g = 0; g < OUT_N; g = g + 1) begin : pack
            assign scores_o[g*SCORE_W +: SCORE_W] = score_q[g];
        end
    endgenerate

    // 누산값(ACC_W)과 Bias(BIAS_W)를 SCORE_W로 부호 확장해 더한다(기본 24+24 → 25비트).
    function signed [SCORE_W-1:0] add_bias(input signed [ACC_W-1:0] acc, input signed [BIAS_W-1:0] b);
        add_bias = acc + b;
    endfunction

    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE;
            group_q <= 2'd0;
            idx_q <= 6'd0;
            drain_q <= 1'b0;
            issue_q <= 1'b0;
            busy_o <= 1'b0;
            done_o <= 1'b0;
            error_o <= 1'b0;
            class_o <= 4'd0;
            for (k = 0; k < OUT_N; k = k + 1) begin
                bias[k] <= {BIAS_W{1'b0}};
                score_q[k] <= {SCORE_W{1'b0}};
            end
        end else begin
            done_o <= 1'b0;
            error_o <= 1'b0;
            issue_q <= (state == RUN);
            // Busy 중 쓰기, 범위 밖 class 쓰기는 무시하고 오류 펄스
            if (wr_any && !wr_ok)
                error_o <= 1'b1;
            if (b_wr_en_i && wr_ok)
                bias[wr_class_i] <= b_data_i;
            case (state)
                IDLE: if (start_i) begin
                    if (!x_valid_i || wr_any) begin
                        error_o <= 1'b1;
                    end else begin
                        busy_o <= 1'b1;
                        group_q <= 2'd0;
                        have_best_q <= 1'b0;
                        state <= CLEAR;
                    end
                end
                CLEAR: begin
                    // 직전 그룹(group_q - 1) 점수를 누적 최대값에 합친다
                    if (group_q != 2'd0) begin
                        have_best_q <= 1'b1;
                        if (take_grp) begin
                            best_s_q <= grp_s;
                            best_c_q <= grp_c;
                        end
                    end
                    idx_q <= 6'd0;
                    state <= RUN;
                end
                RUN: begin
                    if (idx_q == IN_LEN - 1) begin
                        drain_q <= 1'b0;
                        state <= DRAIN;
                    end
                    idx_q <= idx_q + 6'd1;
                end
                // 마지막 곱: issue_q 다음 에지에서 DSP 등록, 그다음 에지에서 누산 완료
                DRAIN: begin
                    drain_q <= 1'b1;
                    if (drain_q)
                        state <= CAPT;
                end
                CAPT: begin
                    if (lane_en[0]) score_q[{group_q, 2'd0}] <= add_bias(acc0, bias[{group_q, 2'd0}]);
                    if (lane_en[1]) score_q[{group_q, 2'd1}] <= add_bias(acc1, bias[{group_q, 2'd1}]);
                    if (lane_en[2]) score_q[{group_q, 2'd2}] <= add_bias(acc2, bias[{group_q, 2'd2}]);
                    if (lane_en[3]) score_q[{group_q, 2'd3}] <= add_bias(acc3, bias[{group_q, 2'd3}]);
                    // argmax 1단: lane 쌍 승자. 무효 lane은 lane_en으로 제외되므로 값은 상관없다
                    s01 <= take1 ? ns1 : ns0;
                    c01 <= take1 ? {group_q, 2'd1} : {group_q, 2'd0};
                    s23 <= take3 ? ns3 : ns2;
                    c23 <= take3 ? {group_q, 2'd3} : {group_q, 2'd2};
                    v23_q <= lane_en[2];
                    if (group_q == LAST_GROUP) begin
                        state <= FINISH;
                    end else begin
                        group_q <= group_q + 2'd1;
                        state <= CLEAR;
                    end
                end
                FINISH: begin
                    class_o <= best_class;
                    done_o <= 1'b1;
                    busy_o <= 1'b0;
                    state <= IDLE;
                end
                default: begin
                    state <= IDLE;
                    busy_o <= 1'b0;
                end
            endcase
            // Busy 중 start는 무시하고 오류 표시
            if (start_i && state != IDLE)
                error_o <= 1'b1;
        end
    end
    // synthesis translate_off
    initial begin
        if (IN_LEN < 1 || IN_LEN > 64 || OUT_N < 1 || OUT_N > 12)
            $fatal(1, "Require 1<=IN_LEN<=64 and 1<=OUT_N<=12");
        if (SCORE_W < ((ACC_W > BIAS_W) ? ACC_W : BIAS_W) + 1)
            $fatal(1, "Require SCORE_W >= max(ACC_W, BIAS_W) + 1");
    end
    // synthesis translate_on
endmodule

// (추가) 14장 p3_group10_core 자동 판정 테스트벤치.
// 13장 p3_frame64_buffer 실제 RTL과 연결: 스트림 → 프레임 버퍼 → 4PE 3그룹 → 10점수·Class.
// Python 벡터(가중치·Bias·입력)를 쓰기 포트로 적재하고 결과를 actual에 기록한다(Python이 비트 단위 비교).
// 추가 판정: 프레임 없는 start 거부, Busy 중 쓰기·start 거부, 범위 밖 class 거부, Busy 중 Reset 후 재실행,
// done 1클록 펄스, 처리 클록 수 일정, x 읽기는 프레임 보관 중에만.
`timescale 1ns/1ps
module tb_p3_group10_core;
    reg clk = 0, resetn = 0;
    always #5 clk = ~clk;

    reg [7:0] s_data = 0;
    reg s_valid = 0, s_last = 0, frame_ack = 0;
    wire s_ready, frame_valid, x_rd_valid, err_short, err_long;
    wire x_rd_en;
    wire [5:0] x_rd_addr;
    wire signed [7:0] x_rd_data;
    reg start = 0, w_wr = 0, b_wr = 0;
    reg [3:0] wr_class = 0;
    reg [5:0] wr_addr = 0;
    reg signed [7:0] w_data = 0;
    reg signed [23:0] b_data = 0;
    wire load_ready, busy, done, error;
    wire [3:0] cls;
    wire [249:0] scores;

    p3_frame64_buffer buffer(
        .clk(clk), .resetn(resetn),
        .s_data_i(s_data), .s_valid_i(s_valid), .s_last_i(s_last), .s_ready_o(s_ready),
        .frame_valid_o(frame_valid), .frame_ack_i(frame_ack),
        .rd_en_i(x_rd_en), .rd_addr_i(x_rd_addr), .rd_data_o(x_rd_data), .rd_valid_o(x_rd_valid),
        .err_short_o(err_short), .err_long_o(err_long));
    p3_group10_core core(
        .clk(clk), .resetn(resetn), .start_i(start), .x_valid_i(frame_valid),
        .x_rd_en_o(x_rd_en), .x_rd_addr_o(x_rd_addr), .x_rd_data_i(x_rd_data),
        .w_wr_en_i(w_wr), .b_wr_en_i(b_wr), .wr_class_i(wr_class), .wr_addr_i(wr_addr),
        .w_data_i(w_data), .b_data_i(b_data), .load_ready_o(load_ready),
        .busy_o(busy), .done_o(done), .error_o(error), .class_o(cls), .scores_o(scores));

    // 상시 규칙 검사
    integer n_error = 0, n_done = 0;
    reg prev_done = 0;
    always @(posedge clk) begin
        if (resetn) begin
            if (x_rd_en && !frame_valid) $fatal(1, "core read x without held frame");
            if (done && prev_done) $fatal(1, "done longer than 1 clock");
            if (load_ready === busy) $fatal(1, "load_ready must be !busy");
            if (err_short || err_long) $fatal(1, "unexpected frame error");
            if (error) n_error = n_error + 1;
            if (done) n_done = n_done + 1;
        end
        prev_done = done;
    end

    integer x [0:63];
    integer w [0:9][0:63];
    integer b [0:9];
    integer tmp, vf, af, rc, ncase, cidx, i, c, cyc, cyc_ref, err_before;
    reg [8*256-1:0] vpath, apath;

    task automatic load_model;
        begin
            for (c = 0; c < 10; c = c + 1)
                for (i = 0; i < 64; i = i + 1) begin
                    @(negedge clk);
                    w_wr = 1; wr_class = c; wr_addr = i; w_data = w[c][i];
                end
            for (c = 0; c < 10; c = c + 1) begin
                @(negedge clk);
                w_wr = 0; b_wr = 1; wr_class = c; b_data = b[c];
            end
            @(negedge clk);
            w_wr = 0; b_wr = 0;
        end
    endtask
    task automatic send_frame;
        integer k;
        begin
            for (k = 0; k < 64; k = k + 1) begin
                @(negedge clk);
                s_data = x[k]; s_valid = 1; s_last = (k == 63);
                while (!s_ready) @(negedge clk);
            end
            @(negedge clk);
            s_valid = 0; s_last = 0;
            if (!frame_valid) $fatal(1, "frame not held after 64 bytes");
        end
    endtask
    task automatic pulse_start;
        begin
            @(negedge clk);
            start = 1;
            @(negedge clk);
            start = 0;
        end
    endtask
    task automatic expect_error(input [8*40-1:0] what);
        begin
            @(negedge clk);
            if (n_error != err_before + 1) $fatal(1, "missing error pulse: %0s", what);
            err_before = n_error;
        end
    endtask

    initial begin
        if (!$value$plusargs("VECTORS=%s", vpath)) $fatal(1, "+VECTORS required");
        if (!$value$plusargs("ACTUAL=%s", apath)) $fatal(1, "+ACTUAL required");
        vf = $fopen(vpath, "r");
        af = $fopen(apath, "w");
        rc = $fscanf(vf, "%d", ncase);
        repeat (4) @(negedge clk);
        resetn = 1;
        @(negedge clk);
        cyc_ref = -1;
        err_before = 0;

        // 프레임 없이 start → 거부
        pulse_start;
        if (busy) $fatal(1, "started without frame");
        expect_error("start without frame");
        // 범위 밖 class 쓰기 → 거부
        @(negedge clk); w_wr = 1; wr_class = 10; wr_addr = 0; w_data = 5;
        @(negedge clk); w_wr = 0;
        expect_error("class 10 write");

        for (cidx = 0; cidx < ncase; cidx = cidx + 1) begin
            // Icarus는 배열 원소에 직접 $fscanf할 수 없어 임시 변수로 읽는다.
            for (i = 0; i < 64; i = i + 1) begin rc = $fscanf(vf, "%d", tmp); x[i] = tmp; end
            for (c = 0; c < 10; c = c + 1)
                for (i = 0; i < 64; i = i + 1) begin rc = $fscanf(vf, "%d", tmp); w[c][i] = tmp; end
            for (c = 0; c < 10; c = c + 1) begin rc = $fscanf(vf, "%d", tmp); b[c] = tmp; end
            load_model;
            send_frame;

            if (cidx == 1) begin
                // Busy 중 Reset: 프레임·Bias 소실 → 재적재 후 재실행
                pulse_start;
                repeat (50) @(negedge clk);

"""14장: 4PE 3그룹(4+4+2) 10분류 코어 기준모델·시험 벡터 생성·RTL 결과 비교.

import argparse
import random
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[3]))
from part03.host_tools.p3_integer_reference import linear_64x10, argmax_lowest  # noqa: E402

BIAS_MIN, BIAS_MAX = -(1 << 23), (1 << 23) - 1


def group_schedule():
    """(group, lane, class) 목록. 유효 차선만 포함한다."""
    return [(g, l, 4 * g + l) for g in range(3) for l in range(4) if 4 * g + l < 10]


def expected(x, w, b):
    for v in x + [e for row in w for e in row]:
        if not -128 <= v <= 127:
            raise ValueError("x/w must be INT8")
    for v in b:
        if not BIAS_MIN <= v <= BIAS_MAX:
            raise ValueError("bias must be signed 24-bit")
    scores = linear_64x10(x, w, b)
    for s in scores:
        assert -(1 << 24) <= s < (1 << 24), "score exceeds 25-bit contract"
    return scores, argmax_lowest(scores)


def onehot(c, i):
    return int(c == i)


def build_cases():
    rng = random.Random(1414)
    ramp = [i - 32 for i in range(64)]
    cases = []
    # ① class 매핑: W[c][c]=1 → score[c] = x[c] = c-32 (+bias c*1000)
    cases.append((ramp, [[onehot(c, i) for i in range(64)] for c in range(10)],
                  [c * 1000 for c in range(10)]))
    # ② 모든 점수 음수: 무효 차선(점수 0)이 섞이면 class가 틀어진다
    cases.append(([1] * 64, [[-1 - c] * 64 for c in range(10)], [0] * 10))

    w = [[0] * 64 for _ in range(10)]

    cases.append(([0] * 64, [[0] * 64 for _ in range(10)], b))

    w = [[-128] * 64 if c % 2 == 0 else [127] * 64 for c in range(10)]
    b = [BIAS_MAX, BIAS_MIN, 0, 0, BIAS_MAX, BIAS_MIN, -1, 1, BIAS_MAX, BIAS_MIN]
    cases.append(([-128] * 64, w, b))
    cases.append(([127] * 64, w, b))
    # ⑥ 무작위 30개

    z = [[0] * 64 for _ in range(10)]

        b = [-100 - c for c in range(10)]

def selftest():
    sched = group_schedule()
    assert [c for _, _, c in sched] == list(range(10))
    assert [len([s for s in sched if s[0] == g]) for g in range(3)] == [4, 4, 2]
    s, c = expected([1] * 64, [[-1 - k] * 64 for k in range(10)], [0] * 10)
    assert c == 0 and max(s) < 0

    assert c == 3
    try:
        expected([0] * 64, [[0] * 64] * 10, [1 << 23] + [0] * 9)
    except ValueError:
        pass
    else:
        raise AssertionError("bias range check missing")

    selftest()
    cases = build_cases()
    with (out / "vectors.txt").open("w") as vf, (out / "expected.txt").open("w") as ef:

#!/usr/bin/env python3
"""15장: 정수 Dense 계층(64→10, 64→8, 8→10) 기준모델·시험 명령 생성·RTL 결과 비교.

점수 = dot_signed(x, W[c]) + bias[c], Class = argmax_lowest (host_tools 규칙 그대로).
64→10은 linear_64x10 결과와 동일함을 함께 확인한다.
명령: W c a v (가중치 쓰기) / B c v (Bias 쓰기) / F x... (프레임 입력 후 추론)
모델은 명령 사이에 유지된다(부분 갱신·다중 프레임 검증).
"""
import argparse
import random
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[3]))
from part03.host_tools.p3_integer_reference import (  # noqa: E402
    dot_signed, linear_64x10, argmax_lowest,
)

BIAS_MIN, BIAS_MAX = -(1 << 23), (1 << 23) - 1


def dense(x, w, b):
    scores = [dot_signed(x, row) + bias for row, bias in zip(w, b)]
    if len(x) == 64 and len(w) == 10:
        assert scores == linear_64x10(x, w, b)
    for s in scores:
        assert -(1 << 24) <= s < (1 << 24), "score exceeds 25-bit contract"
    return scores


def build_ops(n_in, n_out, seed):
    rng = random.Random(seed)
    i8 = lambda: rng.randint(-128, 127)  # noqa: E731
    ops = []

    def full_model(wfun, bfun):
        for c in range(n_out):
            for a in range(n_in):
                ops.append(("W", c, a, wfun(c, a)))
        for c in range(n_out):
            ops.append(("B", c, bfun(c)))

    def frame(xs=None):
        ops.append(("F", xs if xs is not None else [i8() for _ in range(n_in)]))

    # ① 무작위 모델 1회 적재 후 4프레임(모델 유지)
    full_model(lambda c, a: i8(), lambda c: rng.randint(BIAS_MIN, BIAS_MAX))
    for _ in range(4):
        frame()
    # ② 부분 갱신: 가중치 5개 + Bias 1개, 같은 주소 2회 쓰기(마지막 값 유효)
    for _ in range(5):
        ops.append(("W", rng.randrange(n_out), rng.randrange(n_in), i8()))
    ops.append(("W", n_out - 1, n_in - 1, 77))
    ops.append(("W", n_out - 1, n_in - 1, -77))
    ops.append(("B", rng.randrange(n_out), rng.randint(BIAS_MIN, BIAS_MAX)))
    frame()
    frame()
    # ③ Bias만 전부 갱신
    for c in range(n_out):
        ops.append(("B", c, rng.randint(-1000, 1000)))
    frame()
    # ④ 극값: W=-128, x=-128 / x=127, Bias 극값 교대
    full_model(lambda c, a: -128, lambda c: BIAS_MAX if c % 2 else BIAS_MIN)
    frame([-128] * n_in)
    frame([127] * n_in)
    # ⑤ 부호 혼합 누산: 짝수 class W=+127, 홀수 class W=-128, x 부호 교대
    full_model(lambda c, a: 127 if c % 2 == 0 else -128, lambda c: 0)
    frame([127 if a % 2 == 0 else -128 for a in range(n_in)])
    # ⑥ 새 무작위 모델 5개 × 2프레임
    for _ in range(5):
        full_model(lambda c, a: i8(), lambda c: rng.randint(BIAS_MIN, BIAS_MAX))
        frame()
        frame()
    return ops


def run_model(ops, n_in, n_out):
    w = [[None] * n_in for _ in range(n_out)]
    b = [0] * n_out  # RTL Bias는 Reset 시 0
    out = []
    for op in ops:
        if op[0] == "W":
            _, c, a, v = op
            w[c][a] = v
        elif op[0] == "B":
            _, c, v = op
            b[c] = v
        else:
            assert all(v is not None for row in w for v in row), "weights not fully loaded"
            scores = dense(op[1], w, b)
            out.append((scores, argmax_lowest(scores)))
    return out


def selftest():
    assert dense([1, -2], [[3, 4], [-1, -1]], [10, -10]) == [5, -9]
    x = [i - 32 for i in range(64)]
    w = [[int(j == i) for j in range(64)] for i in range(10)]
    assert dense(x, w, [0] * 10) == [-32 + i for i in range(10)]
    ops = [("W", 0, 0, 2), ("W", 1, 0, 3), ("F", [5]), ("W", 1, 0, -3), ("B", 0, 1), ("F", [5])]
    assert run_model(ops, 1, 2) == [([10, 15], 1), ([11, -15], 0)]


def generate(out, n_in, n_out):
    selftest()
    ops = build_ops(n_in, n_out, seed=1500 + n_in * 16 + n_out)
    results = run_model(ops, n_in, n_out)
    with (out / "ops.txt").open("w") as f:
        for op in ops:
            if op[0] == "W":
                f.write(f"W {op[1]} {op[2]} {op[3]}\n")
            elif op[0] == "B":
                f.write(f"B {op[1]} 0 {op[2]}\n")
            else:
                padded = op[1] + [0] * (64 - n_in)  # 13장 프레임은 항상 64원소
                f.write("F 0 0 0 " + " ".join(map(str, padded)) + "\n")
        f.write("E 0 0 0\n")
    with (out / "expected.txt").open("w") as f:
        for scores, cls in results:
            f.write(" ".join(map(str, scores)) + f" C{cls}\n")
    print(f"PYTHON_DENSE_SELF_TEST_PASS shape={n_in}x{n_out} ops={len(ops)} frames={len(results)}")


def compare(out, actual):
    exp = (out / "expected.txt").read_text().splitlines()
    got = actual.read_text().splitlines()
    if exp != got:
        for i, (e, g) in enumerate(zip(exp, got)):
            if e != g:
                print(f"FAIL frame {i}:\n  expected={e}\n  got     ={g}")
                break
        print(f"frames expected={len(exp)} got={len(got)}")
        raise SystemExit("PYTHON_RTL_COMPARE_FAIL")
    print(f"PYTHON_RTL_COMPARE_PASS frames={len(exp)} bit_exact=1")

// (추가) 16장: 은닉층 출력 재양자화 = sat_int8(round_shift_away(max(0, score), shift)).
// ReLU 이후 값은 0 이상이므로 절반 반올림은 (v + 2^(shift-1)) >> shift 와 같다(shift=0이면 그대로).
// 조합 회로. 입력 signed 25비트 점수, 출력 signed INT8(0..127).
module p3_relu_requant(
    input wire signed [24:0] score_i,
    input wire [4:0] shift_i,
    output wire signed [7:0] q_o
);
    // ReLU: 음수는 0
    wire [24:0] relu = score_i[24] ? 25'd0 : score_i;
    // 반올림 가산값: shift=0이면 0
    wire [32:0] half = (shift_i == 5'd0) ? 33'd0 : (33'd1 << (shift_i - 5'd1));
    wire [32:0] rounded = ({8'd0, relu} + half) >> shift_i;
    // INT8 상한 포화(ReLU 이후이므로 하한 -128은 발생하지 않음)
    assign q_o = (rounded > 33'd127) ? 8'sd127 : $signed(rounded[7:0]);
endmodule

// (추가) 16장: 64→8→10 정수 MLP. 14장 코어를 두 계층으로 재사용한다.

    input wire resetn,
    input wire start_i,

    output wire x_rd_en_o,
    output wire [5:0] x_rd_addr_o,

    input wire w_wr_en_i,
    input wire b_wr_en_i,
    input wire [3:0] wr_class_i,
    input wire [5:0] wr_addr_i,
    input wire signed [7:0] w_data_i,
    input wire signed [23:0] b_data_i,
    output wire load_ready_o,
    output reg busy_o,
    output reg done_o,
    output wire error_o,
    output reg [3:0] class_o,
    output wire [249:0] scores_o,
    output wire [63:0] hidden_o
);

    reg [4:0] shift_q;
    reg start1_q, start2_q, error_q;

        .busy_o(), .done_o(done2), .error_o(err2), .class_o(class2), .scores_o(scores_o));

    genvar g;
    generate
        for (g = 0; g < 8; g = g + 1) begin : requant
            p3_relu_requant rq(.score_i(scores1[g*25 +: 25]), .shift_i(shift_q), .q_o(q[g]));
            assign hidden_o[g*8 +: 8] = hidden[g];
        end
    endgenerate

    // 2층 x 읽기: 13장 버퍼와 같은 1클록 지연 계약
    always @(posedge clk)
        if (h_rd_en)
            hid_rd_data <= hidden[h_rd_addr[2:0]];

    assign error_o = error_q || err1 || err2;

    integer k;
    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE;
            shift_q <= 5'd0;
            start1_q <= 1'b0;
            start2_q <= 1'b0;
            error_q <= 1'b0;
            busy_o <= 1'b0;
            done_o <= 1'b0;
            class_o <= 4'd0;
            for (k = 0; k < 8; k = k + 1)
                hidden[k] <= 8'sd0;
        end else begin
            start1_q <= 1'b0;
            start2_q <= 1'b0;
            done_o <= 1'b0;
            error_q <= 1'b0;
            // 동작 중 쓰기·start, 프레임 없는 start는 무시하고 오류 펄스
            if (wr_any && !load_ready_o)
                error_q <= 1'b1;
            case (state)
                IDLE: if (start_i) begin
                    if (!x_valid_i || wr_any) begin
                        error_q <= 1'b1;
                    end else begin
                        shift_q <= shift_i;
                        busy_o <= 1'b1;
                        start1_q <= 1'b1;
                        state <= L1;
                    end
                end
                L1: if (done1) begin
                    // 1층 점수가 확정된 클록에 재양자화 결과를 은닉 레지스터로 저장
                    for (k = 0; k < 8; k = k + 1)
                        hidden[k] <= q[k];
                    state <= L2_START;
                end
                L2_START: begin
                    start2_q <= 1'b1;
                    state <= L2;
                end
                L2: if (done2) begin
                    class_o <= class2;
                    done_o <= 1'b1;
                    busy_o <= 1'b0;
                    state <= IDLE;
                end
                default: begin
                    state <= IDLE;
                    busy_o <= 1'b0;
                end
            endcase
            if (start_i && state != IDLE)
                error_q <= 1'b1;
        end
    end
endmodule

#!/usr/bin/env python3
"""16장: 64→8→10 정수 MLP 기준모델·시험 명령 생성·RTL 결과 비교.

최종 점수는 host_tools의 mlp_64_8_10(x, w1, b1, w2, b2, shift), Class는 argmax_lowest.
진단용 은닉값은 같은 규칙 sat_int8(round_shift_away(max(0, dot_signed + b1), shift))을 함수 조합으로 계산해
mlp_64_8_10 출력과 일치하는지 교차 확인한다(계산 규칙 변경 없음).
명령: W L c a v / B L c v (L=0 1층, L=1 2층) / S shift / F x[64]
"""
import argparse
import random
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[3]))
from part03.host_tools.p3_integer_reference import (  # noqa: E402
    sat_int8, round_shift_away, dot_signed, argmax_lowest, mlp_64_8_10,
)

BIAS_MIN, BIAS_MAX = -(1 << 23), (1 << 23) - 1


def hidden_of(x, w1, b1, shift):
    return [sat_int8(round_shift_away(max(0, dot_signed(x, w) + b), shift)) for w, b in zip(w1, b1)]


def infer(x, w1, b1, w2, b2, shift):
    scores = mlp_64_8_10(x, w1, b1, w2, b2, shift)
    h = hidden_of(x, w1, b1, shift)
    assert scores == [dot_signed(h, w) + b for w, b in zip(w2, b2)]
    for s in scores:
        assert -(1 << 24) <= s < (1 << 24)
    return h, scores, argmax_lowest(scores)


def build_ops():
    rng = random.Random(1616)
    i8 = lambda: rng.randint(-128, 127)  # noqa: E731
    ops = []

    def load(w1f, b1f, w2f, b2f):
        for c in range(8):
            for a in range(64):
                ops.append(("W", 0, c, a, w1f(c, a)))
            ops.append(("B", 0, c, b1f(c)))
        for c in range(10):
            for a in range(8):
                ops.append(("W", 1, c, a, w2f(c, a)))
            ops.append(("B", 1, c, b2f(c)))

    def rand_x():
        return [i8() for _ in range(64)]

    # ① 무작위 모델, 여러 shift(0=포화 다수, 큰 shift=0 다수)
    load(lambda c, a: i8(), lambda c: rng.randint(-50000, 50000),
         lambda c, a: i8(), lambda c: rng.randint(-5000, 5000))
    # 첫 프레임을 같은 입력으로 2회: TB가 첫 프레임 2층 동작 중 1층 쓰기를 시도하므로
    # 차단이 실패하면 두 번째 결과가 달라진다(shift 12: 포화·0 아닌 은닉값 다수).
    x0 = rand_x()
    ops.append(("S", 12))
    ops.append(("F", x0))
    ops.append(("F", x0))
    for s in (0, 1, 4, 8, 14, 16, 20, 24, 31):
        ops.append(("S", s))
        ops.append(("F", rand_x()))
    # ② ReLU 경계: 은닉 c의 1층 점수를 Bias로 정확히 지정(-1, 0, 1, 254, 255, 256, 257, 큰 음수)
    #    W1=0이므로 점수 = Bias. shift=1: 1→1(0.5 반올림), 255→128→127 포화, 254→127
    load(lambda c, a: 0, lambda c: [-1, 0, 1, 254, 255, 256, 257, -(1 << 23)][c],
         lambda c, a: int(a == c % 8), lambda c: c)
    ops.append(("S", 1))
    ops.append(("F", rand_x()))
    ops.append(("S", 0))
    ops.append(("F", rand_x()))
    # ③ 2층 단독 최대·동점: 은닉 고정(W1=0, B1=200, shift=3 → 25), 2층 class 4·9 동점 → 4
    load(lambda c, a: 0, lambda c: 200,
         lambda c, a: 1 if c in (4, 9) else 0, lambda c: 0)
    ops.append(("S", 3))
    ops.append(("F", rand_x()))
    # ④ 전부 음수 출력: 은닉 모두 0 → 점수 = B2(전부 음수, class 6 최대)
    load(lambda c, a: -1, lambda c: -10, lambda c, a: i8(),
         lambda c: -100 + (90 if c == 6 else 0) - c)
    ops.append(("S", 0))
    ops.append(("F", [127] * 64))
    # ⑤ 무작위 모델 4개 × shift 무작위 3프레임
    for _ in range(4):
        load(lambda c, a: i8(), lambda c: rng.randint(BIAS_MIN, BIAS_MAX),
             lambda c, a: i8(), lambda c: rng.randint(BIAS_MIN, BIAS_MAX))
        for _ in range(3):
            ops.append(("S", rng.randint(6, 18)))
            ops.append(("F", rand_x()))
    return ops


def run_model(ops):
    w1 = [[None] * 64 for _ in range(8)]
    w2 = [[None] * 8 for _ in range(10)]
    b1, b2 = [0] * 8, [0] * 10
    shift, out = 0, []
    for op in ops:
        if op[0] == "W":
            _, layer, c, a, v = op
            (w1 if layer == 0 else w2)[c][a] = v
        elif op[0] == "B":
            _, layer, c, v = op
            (b1 if layer == 0 else b2)[c] = v
        elif op[0] == "S":
            shift = op[1]
        else:
            out.append(infer(op[1], w1, b1, w2, b2, shift))
    return out


def selftest():
    x = [0] * 64
    w1 = [[0] * 64 for _ in range(8)]
    b1 = [-1, 0, 1, 254, 255, 256, 257, -(1 << 23)]
    assert hidden_of(x, w1, b1, 1) == [0, 0, 1, 127, 127, 127, 127, 0]
    assert hidden_of(x, w1, [3, 5, 6, 7, 2, 1, 0, 9], 2) == [1, 1, 2, 2, 1, 0, 0, 2]
    assert hidden_of(x, w1, [200] * 8, 3) == [25] * 8


def generate(out):
    selftest()
    ops = build_ops()
    results = run_model(ops)
    with (out / "ops.txt").open("w") as f:
        for op in ops:
            if op[0] == "W":
                f.write(f"W {op[1]} {op[2]} {op[3]} {op[4]}\n")
            elif op[0] == "B":
                f.write(f"B {op[1]} {op[2]} 0 {op[3]}\n")
            elif op[0] == "S":
                f.write(f"S 0 0 0 {op[1]}\n")
            else:
                f.write("F 0 0 0 0 " + " ".join(map(str, op[1])) + "\n")
        f.write("E 0 0 0 0\n")
    with (out / "expected.txt").open("w") as f:
        for h, scores, cls in results:
            f.write("H " + " ".join(map(str, h)) + " S " + " ".join(map(str, scores)) + f" C{cls}\n")
    print(f"PYTHON_MLP_SELF_TEST_PASS ops={len(ops)} frames={len(results)} seed=1616")


def compare(out, actual):
    exp = (out / "expected.txt").read_text().splitlines()
    got = actual.read_text().splitlines()
    if exp != got:
        for i, (e, g) in enumerate(zip(exp, got)):
            if e != g:
                print(f"FAIL frame {i}:\n  expected={e}\n  got     ={g}")
                break
        print(f"frames expected={len(exp)} got={len(got)}")
        raise SystemExit("PYTHON_RTL_COMPARE_FAIL")
    print(f"PYTHON_RTL_COMPARE_PASS frames={len(exp)} hidden=8 scores=10 class=1 bit_exact=1")


if __name__ == "__main__":
    p = argparse.ArgumentParser()
    p.add_argument("--out", type=Path, required=True)
    p.add_argument("--actual", type=Path)
    a = p.parse_args()
    if a.actual:
        compare(a.out, a.actual)
    else:
        generate(a.out)

CRLF = "\r\n"
NEED = {"S": 0, "F": 1, "p": 1, "W": 4, "B": 5, "R": 0, "H": 0, "G": 0, "Z": 1, "M": 0}
MIN_STAGE = {"W": 2, "B": 2, "R": 2, "H": 3, "G": 3, "Z": 4, "M": 4}
SHAPE = {0: (10, 64), 1: (8, 64), 2: (10, 8)}          # 층 → (class 수, 입력 수)


def h(v, n):
    return format(v & ((1 << (4 * n)) - 1), "0%dX" % n)


class Board:
    def __init__(self, stage):
        self.stage = stage
        self.frame = None                    # 들고 있는 프레임 (64 INT8)
        self.w = {L: [[None] * n_in for _ in range(n_out)] for L, (n_out, n_in) in SHAPE.items()}
        self.b = {L: [0] * n_out for L, (n_out, _) in SHAPE.items()}
        self.shift = 0
        self.view, self.cls, self.cls_ok, self.fshort, self.flong = 0, 0, False, False, False

    def step(self, data):
        c = chr(data[0])
        if c in "\r\n":
            return ""
        if c not in NEED or self.stage < MIN_STAGE.get(c, 1):
            return "?" + CRLF
        p = data[1:1 + NEED[c]]
        if c == "S":
            return "I" + CRLF
        if c == "F":
            n = p[0]
            if not 1 <= n <= 80:
                return "ERR" + CRLF
            px = [struct.unpack("b", bytes([v]))[0] for v in data[2:]]
            assert len(px) == n
            self.frame = px if n == 64 else None
            self.fshort, self.flong = n < 64, n > 64
            return "F:" + ("OK" if n == 64 else "SHORT" if n < 64 else "LONG") + CRLF
        if c == "p":
            if p[0] >= 64 or self.frame is None:
                return "ERR" + CRLF
            v = self.frame[p[0]]
            self.view = v & 0xFFFF
            return "P:" + h(v, 2) + CRLF
        if c in "WB":
            L, cl = p[0], p[1]
            if L > 2 or (L > 0 and self.stage < 3) or cl >= SHAPE[L][0]:
                return "ERR" + CRLF
            if c == "W":
                if p[2] >= SHAPE[L][1]:
                    return "ERR" + CRLF
                self.w[L][cl][p[2]] = struct.unpack("b", bytes([p[3]]))[0]
            else:
                v = int.from_bytes(bytes(p[2:5]), "big")
                self.b[L][cl] = v - (1 << 24) if v >> 23 else v
            return "OK" + CRLF
        if c == "Z":
            if p[0] >= 32:
                return "ERR" + CRLF
            self.shift = p[0]
            return "OK" + CRLF
        if self.frame is None:                # 프레임 없는 start → 코어 error_o
            return "ERR" + CRLF
        x = self.frame
        if c == "M":
            hid, scores, cls = infer(x, self.w[1], self.b[1], self.w[2], self.b[2], self.shift)
            body = "".join(h(v, 2) for v in hid) + " " + ",".join(h(s, 7) for s in scores)
        else:
            L = {"R": 0, "H": 1, "G": 2}[c]
            xin = x[:SHAPE[L][1]]
            for row in self.w[L]:
                assert None not in row, "안 쓴 가중치"
            scores = dense(xin, self.w[L], self.b[L])
            cls = argmax_lowest(scores)
            body = ",".join(h(s, 7) for s in scores)
        self.view, self.cls, self.cls_ok = cls, cls, True
        return c + "%X:" % cls + body + CRLF

    def led(self):
        onehot = (1 << self.cls) if self.cls_ok else 0
        return onehot | (int(self.frame is not None) << 15) | (int(self.fshort) << 14) | (int(self.flong) << 13)


def frame(px):
    return [ord("F"), len(px)] + [v & 0xFF for v in px]


def W(L, c, a, v):
    return [ord("W"), L, c, a, v & 0xFF]


def B(L, c, v):
    return [ord("B"), L, c] + list((v & 0xFFFFFF).to_bytes(3, "big"))

    out = [W(L, c, i, v) for c, row in enumerate(w) for i, v in enumerate(row)]
    return out + [B(L, c, v) for c, v in enumerate(b)]
