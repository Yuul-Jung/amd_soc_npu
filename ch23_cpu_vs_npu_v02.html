/* 23장 CPU 기준 추론: NPU와 같은 정수 계산(784×10 내적 + Bias, 동점은 낮은 번호)을 C로 한다. */

#define CPU_INFER_H
#include <stdint.h>

ch23/sw/cpu_infer.c · c · 원본 · ⚠ 미검증
/* 23장 CPU 기준 추론 (RV32IMC: 곱은 MUL, 누산은 32비트 — |점수| < 2^25이므로 넘치지 않음) */
#include "cpu_infer.h"

uint32_t cpu_infer(const int8_t w[10][784], const int32_t b[10], const uint8_t x[784], int32_t score[10])
{
    uint32_t best = 0;
    for (uint32_t c = 0; c < 10; c++) {
        int32_t acc = 0;
        for (uint32_t i = 0; i < 784; i++)
            acc += (int32_t)w[c][i] * (int32_t)x[i];
        score[c] = acc + b[c];
        if (score[c] > score[best]) best = c;     /* 같으면 앞 번호 유지 */
    }
    return best;
}
ch23/sw/axi_timer.h · c · 원본 · ⚠ 미검증
/* 23장 AXI Timer 0번 카운터를 자유 증가 카운터로 쓴다 (레지스터 직접 접근, PG079 레지스터 표). */
#ifndef AXI_TIMER_H
#define AXI_TIMER_H
#include <stdint.h>
#include "xil_io.h"

#ifndef TIMER_BASE
/* ⚠ xparameters.h의 AXI Timer 기준 주소로 바꾼다 (제1부 11장 SoC의 값). 0x41C00000은 Vivado 기본값이며 미확인. */
#define TIMER_BASE 0x41C00000u
#endif
#define TMR_TCSR0 0x00u     /* 제어·상태 */
#define TMR_TLR0  0x04u     /* 적재 값 */
#define TMR_TCR0  0x08u     /* 카운터 */
#define TCSR_LOAD 0x20u     /* TLR 값을 카운터에 적재 */
#define TCSR_ENT  0x80u     /* 카운터 동작 (UDT0=0: 위로 셈) */

static inline void timer_start(void)
{
    Xil_Out32(TIMER_BASE + TMR_TLR0, 0);
    Xil_Out32(TIMER_BASE + TMR_TCSR0, TCSR_LOAD);   /* 0 적재 */
    Xil_Out32(TIMER_BASE + TMR_TCSR0, TCSR_ENT);    /* 위로 세기 시작 */
}
static inline uint32_t timer_now(void) { return Xil_In32(TIMER_BASE + TMR_TCR0); }

#endif
ch23/sw/bench.c · c · 원본 · ⚠ 미검증
/* 23장 CPU·NPU 시간 비교: 같은 모델·같은 5장으로 CPU 추론, NPU 입력 적재, NPU 실행을 AXI Timer로 잰다.
 * 출력은 CSV 한 줄씩: image,label,cpu_class,npu_class,cpu_clk,npu_load_clk,npu_run_clk,npu_core_clk */
#include "cpu_infer.h"
#include "npu_drv.h"
#include "axi_timer.h"
#include "model_int8.h"
#include "xil_printf.h"

int main(void)
{
    int32_t s[10];
    npu_result_t r;
    xil_printf("=== ch23 CPU vs NPU (MNIST INT8 784x10, 100 MHz) ===\r\n");
    if (npu_init() != NPU_OK) { xil_printf("[FAIL] NPU ID\r\n"); return 1; }
    npu_load_model(MODEL_W, MODEL_B);
    timer_start();
    xil_printf("image,label,cpu_class,npu_class,cpu_clk,npu_load_clk,npu_run_clk,npu_core_clk\r\n");
    int same = 0;
    for (int n = 0; n < TEST_N; n++) {
        uint32_t t0 = timer_now();
        uint32_t cc = cpu_infer(MODEL_W, MODEL_B, TEST_X[n], s);
        uint32_t t1 = timer_now();
        if (npu_load_input(TEST_X[n]) != NPU_OK) return 1;
        uint32_t t2 = timer_now();
        if (npu_run(&r, 100000u) != NPU_OK) return 1;
        uint32_t t3 = timer_now();
        int eq = (cc == r.cls);
        for (int k = 0; k < 10; k++) eq &= (s[k] == r.score[k]);
        same += eq;
        xil_printf("%d,%d,%d,%d,%d,%d,%d,%d\r\n", n, (int)TEST_LABEL[n], (int)cc, (int)r.cls,
                   (int)(t1 - t0), (int)(t2 - t1), (int)(t3 - t2), (int)r.cycles);
    }
    xil_printf("[%s] CPU and NPU scores equal %d/%d\r\n", same == TEST_N ? "PASS" : "FAIL", same, TEST_N);
    return 0;
}

ch23/reference/estimate.py · python · 원본 · ⚠ 미검증
#!/usr/bin/env python3
"""23장 추정: cpu_infer.c를 RV32IMC -O2로 컴파일해 안쪽 반복의 명령 수를 세고, CPU 클록을 CPI 가정별로 추정한다.
NPU 쪽은 19·22장 시뮬레이션 값(코어 7,185클록)과 AXI 쓰기 횟수(입력 784번)를 쓴다.
보드에서 bench.c로 잰 값이 나오기 전까지 모든 CPU 숫자는 '추정'이다."""
from __future__ import annotations
import re
import subprocess
import sys
import tempfile
from pathlib import Path

HERE = Path(__file__).resolve().parents[1]


def inner_loop_len(obj):
    dis = subprocess.run(["llvm-objdump", "-d", "--no-show-raw-insn", obj], capture_output=True, text=True, check=True).stdout
    lines = [ln for ln in dis.splitlines() if re.match(r"\s+[0-9a-f]+:\s", ln)]
    addr = [int(ln.split(":")[0], 16) for ln in lines]
    best = None
    for k, ln in enumerate(lines):                       # 뒤로 가는 분기 = 반복의 끝, 가장 짧은 반복 = 안쪽 반복
        m = re.search(r"\s(b\w+)\s.*?0x([0-9a-f]+)\s*<", ln)
        if m:
            tgt = int(m.group(2), 16)
            if tgt < addr[k]:
                body = [x for x in lines if tgt <= int(x.split(":")[0], 16) <= addr[k]]
                if best is None or len(body) < len(best):
                    best = body
    return best


def main():
    with tempfile.TemporaryDirectory() as d:
        obj = f"{d}/cpu_infer.elf"         # 분기 목적지를 풀기 위해 링크까지 한다 (진입점 = cpu_infer)
        subprocess.run(["clang", "--target=riscv32-unknown-elf", "-march=rv32imc", "-O2", "-nostdlib", "-fuse-ld=lld",
                        "-Wl,--entry=cpu_infer", str(HERE / "sw" / "cpu_infer.c"), "-I", str(HERE / "sw"), "-o", obj], check=True)
        body = inner_loop_len(obj)
    n = len(body)
    ops = ", ".join(re.split(r"\s+", b.split(":", 1)[1].strip())[0] for b in body)
    macs = 784 * 10
    print(f"CPU_INNER_LOOP instructions={n} ({ops})")
    for cpi in (1.0, 1.5, 2.0):
        print(f"ESTIMATE cpu_clk cpi={cpi} ≈ {int(macs * n * cpi):,} (= 7,840 MAC × {n}명령 × CPI, 바깥 반복 제외)")
    print("MEASURED_SIM npu_core_clk=7185 (19장 NPE=4) · npu_load = AXI 쓰기 784번 + 제어 (버스 클록은 보드 측정)")
    print(f"ESTIMATE speedup_core_only cpi=1 ≈ {macs * n / 7185:.1f}x (입력 적재 시간 제외)")
    print("CPU_ESTIMATE_DONE (보드 측정 전 추정)")


if __name__ == "__main__":
    main()

06결과 판정

확인 항목의미
RV32IMC_COMPILE_PASS files=2cpu_infer.c·bench.c가 RISC-V 32비트로 경고 없이 컴파일됨
CPU_REF_PASS images=20 scores=200C CPU 추론 점수 200개·클래스가 Python 정수 기준과 같음 → 보드에서 CPU·NPU 점수가 같아야 함
CPU_INNER_LOOP instructions=7 (lb, lbu, mul, add, addi, addi, bne)곱셈-누산 1번에 명령 7개 → CPU ≥ 54,880클록(CPI 1 가정, 추정)
ESTIMATE speedup_core_only cpi=1 ≈ 7.6xNPU 코어 7,185클록 대비 추정 배율. 입력 적재 시간을 빼고 본 값이며 보드 측정 전까지 추정

실패 시: TB의 $fatal 메시지에 표시된 cycle 번호와 기대값/실제값을 확인하고, 03절 코드와 한 줄씩 대조합니다. 하나라도 실패하면 다음 장으로 넘어가지 않습니다.

07코드 읽기 포인트

위치내용
무엇을 따로 재나CPU 추론, NPU 입력 적재(AXI 쓰기 784번), NPU 실행(START~결과 읽기)을 각각 잽니다. 코어만 보면 NPU가 빠르지만, CPU가 입력을 레지스터로 한 바이트씩 쓰는 시간이 코어 시간과 비슷하거나 더 클 수 있습니다.
AXI Timer 직접 접근axi_timer.h는 드라이버 라이브러리 없이 TCSR0·TLR0·TCR0만 씁니다. 0을 적재하고 위로 세게 한 뒤 전후 값을 뺍니다. 32비트이므로 100 MHz에서 약 42초까지 넘치지 않습니다.
추정의 근거estimate.py는 실제 RV32IMC 기계어에서 가장 짧은 반복(안쪽 반복)을 찾아 명령 수를 셉니다. MicroBlaze V의 곱셈·적재 지연에 따라 CPI가 1보다 커지므로 1~2 범위로만 적습니다.
같은 결과 확인bench.c는 그림마다 CPU 점수 10개와 NPU 점수 10개를 비교해 마지막 줄에 일치 수를 찍습니다. 시간 비교는 두 결과가 같을 때만 의미가 있습니다.
