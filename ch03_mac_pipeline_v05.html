01구조도

구현 도식: 포트 이름·폭과 하위 인스턴스는 ch03/rtl/mac_pipeline_core.v 선언에서 자동으로 가져왔습니다.

방향포트폭구분설명
inputclk1클록시스템 클록 (상승 에지에서 모든 레지스터 갱신)
inputresetn1리셋동기 리셋, 0이면 초기화 (Active-Low)
inputclear_i1데이터·제어곱·누산 두 단계 모두 0으로 초기화
inputvalid_i1데이터·제어이번 클록 입력을 파이프라인에 넣음
inputa_i[DATA_WIDTH-1:0]데이터·제어signed 피연산자 A
inputb_i[DATA_WIDTH-1:0]데이터·제어signed 피연산자 B
outputacc_o[ACC_WIDTH-1:0]데이터·제어누산 결과 (입력보다 1클록 늦게 반영)

// 곱 레지스터와 누산 레지스터를 분리한 독립 MAC. k 입력 → k+1 출력.
(* use_dsp = "yes" *)
module mac_pipeline_core #(
    parameter integer DATA_WIDTH = 16,
    parameter integer ACC_WIDTH = 40
) (
    input wire clk,
    input wire resetn,
    input wire clear_i,
    input wire valid_i,
    input wire signed [DATA_WIDTH-1:0] a_i,
    input wire signed [DATA_WIDTH-1:0] b_i,
    output reg signed [ACC_WIDTH-1:0] acc_o,
    output reg valid_o
);
    localparam integer PRODUCT_WIDTH = 2 * DATA_WIDTH;
    wire signed [PRODUCT_WIDTH-1:0] product_w;
    reg signed [PRODUCT_WIDTH-1:0] product_q;
    reg product_valid_q;
    wire signed [ACC_WIDTH-1:0] product_ext_w;

    assign product_w = $signed(a_i) * $signed(b_i);
    assign product_ext_w = {{(ACC_WIDTH-PRODUCT_WIDTH){product_q[PRODUCT_WIDTH-1]}}, product_q};

    always @(posedge clk) begin
        if (!resetn) begin
            product_q <= {PRODUCT_WIDTH{1'b0}};
            product_valid_q <= 1'b0;
            acc_o <= {ACC_WIDTH{1'b0}};
            valid_o <= 1'b0;
        end else if (clear_i) begin
            // 현재 입력과 미누산 곱을 모두 버려 지연 Valid가 남지 않게 한다.
            product_q <= {PRODUCT_WIDTH{1'b0}};
            product_valid_q <= 1'b0;
            acc_o <= {ACC_WIDTH{1'b0}};
            valid_o <= 1'b0;
        end else begin
            // 1단계: 곱과 같은 에지의 Valid를 함께 저장한다.
            product_valid_q <= valid_i;
            if (valid_i) begin
                product_q <= product_w;
            end
            // 2단계: 비차단 대입은 이전 에지에 저장된 곱과 Valid를 참조한다.
            valid_o <= product_valid_q;
            if (product_valid_q) begin
                // 부호 확장 후 더하며 범위 초과는 P3-01과 같은 2의 보수 순환이다.
                acc_o <= acc_o + product_ext_w;
            end
        end
    end

    // synthesis translate_off
    initial begin
        if (DATA_WIDTH < 2 || ACC_WIDTH < PRODUCT_WIDTH)
            $fatal(1, "Require DATA_WIDTH>=2 and ACC_WIDTH>=2*DATA_WIDTH");
    end
    // synthesis translate_on
endmodule
`default_nettype wire

module tb_mac_pipeline_core;
    parameter integer DATA_WIDTH = 16;
    parameter integer ACC_WIDTH = 40;
    reg clk = 0;
    reg resetn = 0, clear_i = 0, valid_i = 0;
    reg signed [DATA_WIDTH-1:0] a_i = 0, b_i = 0;
    wire signed [ACC_WIDTH-1:0] acc_o;
    wire valid_o;
    mac_pipeline_core #(.DATA_WIDTH(DATA_WIDTH), .ACC_WIDTH(ACC_WIDTH)) dut (.*);
    always #5 clk = ~clk;
    integer vf, af, fields, cycle = 0;
    integer rn, cl, vi, ai, bi, expected_pv, expected_v;
    reg signed [2*DATA_WIDTH-1:0] expected_product;
    reg signed [ACC_WIDTH-1:0] expected_acc;
    longint signed previous_product = 0, model_acc = 0, multiply_w;
    reg previous_valid = 0, model_valid;
    reg signed [ACC_WIDTH-1:0] held_acc;
    reg signed [2*DATA_WIDTH-1:0] held_product;
    reg held_valid, held_pv;
    string vp, ap;

    initial begin
        if (DATA_WIDTH > 16 || ACC_WIDTH > 40 || ACC_WIDTH < 2*DATA_WIDTH)
            $fatal(1, "Unsupported TB width");
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap))
            $fatal(1, "Missing vector/trace argument");
        vf = $fopen(vp, "r"); af = $fopen(ap, "w");
        if (!vf || !af) $fatal(1, "Cannot open test files");
        @(posedge clk); #1;
        if (acc_o !== 0 || valid_o !== 0 || dut.product_q !== 0 || dut.product_valid_q !== 0)
            $fatal(1, "Initial reset failed");
        while (!$feof(vf)) begin
            fields = $fscanf(vf, "%d %d %d %d %d %d %d %d %d\n",
                             rn, cl, vi, ai, bi, expected_product, expected_pv, expected_acc, expected_v);
            if (fields != 9) $fatal(1, "Malformed vector");
            @(negedge clk);
            held_acc = acc_o; held_valid = valid_o;
            held_product = dut.product_q; held_pv = dut.product_valid_q;
            resetn = rn; clear_i = cl; valid_i = vi; a_i = ai; b_i = bi;
            #1;
            if (acc_o !== held_acc || valid_o !== held_valid ||
                dut.product_q !== held_product || dut.product_valid_q !== held_pv)
                $fatal(1, "Output changed before rising edge cycle=%0d", cycle+1);
            @(posedge clk); #1;
            cycle = cycle + 1;
            // 현재 valid_i가 아니라 이전 에지의 곱 유효 상태로 누산 여부를 결정한다.
            model_valid = rn && !cl && previous_valid;
            if (!rn || cl) begin
                model_acc = 0;
                previous_product = 0;
                previous_valid = 0;
            end else begin
                if (previous_valid) begin
                    model_acc = model_acc + previous_product;
                    model_acc = $signed(model_acc[ACC_WIDTH-1:0]);
                end
                previous_valid = vi;
                if (vi) begin
                    multiply_w = $signed(a_i);
                    previous_product = multiply_w * $signed(b_i);
                end
            end
            if (expected_acc !== model_acc[ACC_WIDTH-1:0] || expected_v[0] !== model_valid ||
                expected_product !== previous_product[2*DATA_WIDTH-1:0] || expected_pv[0] !== previous_valid)
                $fatal(1, "Python/SV reference mismatch cycle=%0d", cycle);
            if (acc_o !== expected_acc || valid_o !== model_valid ||
                dut.product_q !== expected_product || dut.product_valid_q !== previous_valid)
                $fatal(1, "DUT mismatch cycle=%0d acc=%0d expected=%0d valid=%0d expected=%0d",
                       cycle, acc_o, expected_acc, valid_o, model_valid);
            $fdisplay(af, "%0d %0d %0d %0d %0d", cycle, dut.product_q,
                      dut.product_valid_q, acc_o, valid_o);
        end
        if (!cycle || previous_valid) $fatal(1, "Empty test or pipeline not drained");
        $fclose(vf); $fclose(af);
        $display("RTL_SIM_PASS INT%0d/%0d cycles=%0d", DATA_WIDTH, ACC_WIDTH, cycle);
        $finish;
    end
    initial begin
        #10000000;
        $fatal(1, "Simulation timeout");
    end
endmodule

#!/usr/bin/env python3
"""P3-03 대기 큐 모델: 입력 수락 에지와 출력 에지를 분리한다."""
import argparse
import csv
import random
import sys
from collections import deque
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2] / "ch01" / "reference"))
from mac_reference import compare, wrap


class PipelineReference:
    def __init__(self, acc_width):
        self.acc_width = acc_width
        self.pending = deque()
        self.product = 0
        self.acc = 0
        self.edge = 0

    def tick(self, resetn, clear, valid, a, b):
        self.edge += 1
        source_edge = 0
        out_valid = 0
        if not resetn or clear:
            self.pending.clear()
            self.product = 0
            self.acc = 0
        else:
            # 먼저 이전 입력을 배출한 다음 현재 입력을 큐에 넣는다.
            if self.pending:
                product, source_edge = self.pending.popleft()
                if source_edge != self.edge - 1:
                    raise AssertionError("Pipeline latency must be one edge")
                self.acc = wrap(self.acc + product, self.acc_width)
                out_valid = 1
            if valid:
                self.product = a * b
                self.pending.append((self.product, self.edge))
        return self.product, int(bool(self.pending)), self.acc, out_valid, source_edge


def self_test():
    for aw in (32, 40):
        m = PipelineReference(aw)
        assert m.tick(0, 0, 1, 3, 4) == (0, 0, 0, 0, 0)
        assert m.tick(1, 0, 1, 3, 4) == (12, 1, 0, 0, 0)
        assert m.tick(1, 0, 1, 3, -4) == (-12, 1, 12, 1, 2)
        assert m.tick(1, 0, 0, 9, 9) == (-12, 0, 0, 1, 3)
        assert m.tick(1, 0, 0, 9, 9) == (-12, 0, 0, 0, 0)
        assert m.tick(1, 0, 1, -3, -4) == (12, 1, 0, 0, 0)
        assert m.tick(1, 1, 1, 9, 9) == (0, 0, 0, 0, 0)
        assert m.tick(1, 0, 0, 9, 9) == (0, 0, 0, 0, 0)
        m.tick(1, 0, 1, 9, 9)
        assert m.tick(0, 1, 1, 9, 9) == (0, 0, 0, 0, 0)
        assert m.tick(1, 0, 0, 9, 9) == (0, 0, 0, 0, 0)
    print("PYTHON_PIPELINE_SELF_TEST_PASS ACC=32,40")


def stimulus(dw, aw):
    lo, hi = -(1 << (dw - 1)), (1 << (dw - 1)) - 1
    yield (0, 0, 0, 0, 0)
    for a, b in ((3, 4), (3, -4), (-3, -4)):
        yield (1, 0, 1, a, b)
    yield (1, 0, 0, hi, hi)
    yield (1, 0, 0, lo, lo)
    for a, b in ((0, lo), (lo, lo), (lo, hi), (hi, lo), (hi, hi)):
        yield (1, 0, 1, a, b)
    yield (1, 0, 0, 0, 0)
    yield (1, 1, 0, 0, 0)
    yield (1, 0, 1, 7, 9)
    yield (1, 1, 1, hi, hi)  # 계류 곱과 동시 입력을 모두 취소
    yield (1, 0, 0, 1, 1)
    yield (1, 0, 0, 1, 1)
    yield (1, 0, 1, -2, 3)   # Clear 직후 정상 회복과 단발 Valid
    yield (1, 0, 0, 1, 1)
    yield (1, 0, 0, 1, 1)
    yield (1, 0, 1, 6, 7)
    yield (0, 1, 1, hi, hi)  # Reset > Clear > Valid 및 계류 곱 취소
    yield (1, 0, 0, 1, 1)
    yield (1, 0, 1, 6, 7)
    yield (0, 0, 1, hi, hi)  # Reset 단독으로도 파이프라인 취소
    yield (1, 0, 0, 1, 1)
    yield (1, 0, 0, 1, 1)
    yield (1, 1, 0, 0, 0)
    # 실제 입력을 누적해 양의 오버플로로 min 도달 후 min-1=max를 시험한다.
    for _ in range(1 << (aw - 2 * dw + 1)):
        yield (1, 0, 1, lo, lo)
    yield (1, 0, 0, 0, 0)  # 마지막 곱 배출 → 누산 min
    yield (1, 0, 1, -1, 1)
    yield (1, 0, 0, 0, 0)  # min-1 배출 → max
    yield (1, 0, 0, 0, 0)
    rng = random.Random(30300 + dw)
    for _ in range(2000):
        yield (int(rng.randrange(43) != 0), int(rng.randrange(29) == 0),
               int(rng.randrange(4) != 0), rng.randint(lo, hi), rng.randint(lo, hi))
    yield (1, 0, 0, 0, 0)
    yield (1, 0, 0, 0, 0)


def generate(dw, aw, vector_path, expected_path, timing_path):
    model = PipelineReference(aw)
    counts = dict(clear_pending=0, reset_pending=0, bubble_drain=0,
                  positive_wrap=0, negative_wrap=0)
    with vector_path.open("w") as vf, expected_path.open("w") as ef, timing_path.open("w") as tf:
        timing = csv.writer(tf, lineterminator="\n")
        timing.writerow(("edge", "resetn", "clear_i", "valid_i", "accepted_edge", "a_i", "b_i", "product_q",
                         "product_valid_q", "output_source_edge", "output_edge", "acc_o", "valid_o"))
        for edge, row in enumerate(stimulus(dw, aw), 1):
            rn, clear, valid, a, b = row
            if model.pending:
                if not rn:
                    counts["reset_pending"] += 1
                elif clear:
                    counts["clear_pending"] += 1
                else:
                    raw = model.acc + model.pending[0][0]
                    counts["positive_wrap"] += int(raw >= (1 << (aw-1)))
                    counts["negative_wrap"] += int(raw < -(1 << (aw-1)))
                    counts["bubble_drain"] += int(not valid)
            pq, pv, acc, out_valid, source = model.tick(*row)
            vf.write(" ".join(map(str, (*row, pq, pv, acc, out_valid))) + "\n")
            ef.write(f"{edge} {pq} {pv} {acc} {out_valid}\n")
            if edge <= 28:
                timing.writerow((edge, rn, clear, valid, edge if rn and not clear and valid else "-", a, b,
                                 pq, pv, source or "-", edge if out_valid else "-", acc, out_valid))
    if model.pending or not all(counts.values()):
        raise AssertionError(f"Missing directed coverage or final drain: {counts}")
    print(f"PYTHON_VECTORS INT{dw}/{aw} cycles={edge} seed={30300+dw}")
    print("SCENARIO_COUNTS " + " ".join(f"{key}={value}" for key, value in counts.items()))


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--data-width", type=int, default=16)
    parser.add_argument("--acc-width", type=int, default=40)
    parser.add_argument("--vectors", type=Path)
    parser.add_argument("--expected", type=Path, required=True)
    parser.add_argument("--actual", type=Path)
    parser.add_argument("--timing", type=Path)
    args = parser.parse_args()
    if (args.data_width, args.acc_width) not in ((16, 40), (8, 32)):
        parser.error("This test generator supports 16/40 and 8/32")
    if bool(args.vectors) == bool(args.actual) or (args.vectors and not args.timing):
        parser.error("Use --actual or --vectors with --timing")
    self_test()
    if args.actual:
        compare(args.expected, args.actual)
    else:
        generate(args.data_width, args.acc_width, args.vectors, args.expected, args.timing)

07코드 읽기 포인트

위치내용
2단계 레이턴시k개 valid 입력 → k+1 클록 후 출력. 마지막 valid_i=1 에지에서 곱이 product_q에 등록되고, 그 다음 에지에서 acc_o에 반영됩니다.
clear_i 대기 없음clear_i가 1이면 두 파이프라인 단계가 동시에 0으로 초기화됩니다. 이미 product_q에 있던 값도 버려져 "지연 Valid"가 남지 않습니다.
DSP P 레지스터 활용DSP48E1의 P 출력 레지스터가 product_q에 대응합니다. 합성기가 이 구조를 감지하면 외부 플립플롭 없이 DSP 내부 레지스터만으로 1단계가 구현됩니다.
