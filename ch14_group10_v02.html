제14장 · 4PE 4+4+2 그룹 실행과 10분류 argmax

08장 4PE를 4+4+2 세 그룹으로 세 번 돌려 13장 입력 64원소에 대한 10개 점수(내적 + Bias)를 모으고, 동점이면 낮은 번호를 고르는 argmax로 클래스 0~9를 냅니다.

00이 장의 위치

목표: 08장 4PE를 4+4+2 세 그룹으로 세 번 돌려 13장 입력 64원소에 대한 10개 점수(내적 + Bias)를 모으고, 동점이면 낮은 번호를 고르는 argmax로 클래스 0~9를 냅니다.

03RTL 작성

아래 파일을 경로 그대로 만들고 전체 코드를 입력합니다.

ch14/rtl/npu10_group_scheduler.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 14장 그룹 스케줄러: 08장 4PE를 4+4+2 세 그룹으로 세 번 돌려 10개 점수(내적 + Bias)를 모으고,
// 동점이면 낮은 번호를 고르는 argmax로 클래스 0~9를 낸다.
// 입력 64원소는 13장 적재기, 가중치는 차선(PE)별 04장 BRAM 4개 (주소 = 그룹*64 + 원소).
// 클래스 c → 그룹 c/4, 차선 c%4. 마지막 그룹의 차선 2·3은 클래스가 없으므로 가중치 0을 넣고 결과를 버린다.
module npu10_group_scheduler #(
    parameter integer ACC_WIDTH = 32
) (
    input  wire                      clk,
    input  wire                      resetn,
    input  wire                      load_start_i,
    input  wire                      in_valid_i,
    input  wire                [7:0] in_data_i,
    input  wire                      in_last_i,
    output wire                      x_loaded_o,
    output wire                      x_len_err_o,
    output wire                      x_range_err_o,
    input  wire                      w_wr_en_i,
    input  wire                [3:0] w_class_i,
    input  wire                [5:0] w_addr_i,
    input  wire                [7:0] w_data_i,
    input  wire                      b_wr_en_i,
    input  wire                [3:0] b_class_i,
    input  wire      [ACC_WIDTH-1:0] b_data_i,
    input  wire                      start_i,
    output reg                       busy_o,
    output reg                       done_o,
    output reg                       error_o,
    output reg                 [3:0] class_o,
    output wire [10*ACC_WIDTH-1:0]   scores_o,
    output reg                [15:0] cycles_o
);
    localparam IDLE=0, LAUNCH=1, READ=2, WAIT_READ=3, SEND=4, WAIT_PE=5, ARGMAX=6;
    reg [2:0] state;
    reg [1:0] group_q;                 // 0: 클래스 0~3, 1: 4~7, 2: 8~9
    reg [5:0] idx_q;                   // 원소 번호 0~63
    reg [3:0] k_q;                     // argmax 비교 번호 1~9
    reg signed [ACC_WIDTH-1:0] bias_q  [0:9];
    reg signed [ACC_WIDTH-1:0] score_q [0:9];
    integer i;

    // ── 13장 입력 적재기 (64원소, 읽기는 이 모듈이 담당)
    wire signed [7:0] x_data;
    wire x_valid;
    npu10_input_loader loader (
        .clk(clk), .resetn(resetn), .load_start_i(load_start_i), .in_valid_i(in_valid_i),
        .in_data_i(in_data_i), .in_last_i(in_last_i), .rd_en_i(state == READ), .rd_addr_i(idx_q),
        .rd_data_o(x_data), .rd_valid_o(x_valid), .loading_o(), .loaded_o(x_loaded_o),
        .len_err_o(x_len_err_o), .range_err_o(x_range_err_o), .count_o());

    // ── 가중치 BRAM 4개 (차선별 8비트×256, 그룹 3개 × 64원소 사용)
    wire w_ok = w_wr_en_i && !busy_o && (w_class_i <= 4'd9);
    wire [31:0] w_bus;                 // 4차선 × 8비트, PE 입력
    wire [3:0]  w_valid;
    genvar g;
    generate for (g = 0; g < 4; g = g + 1) begin : lane
        wire signed [7:0] w_rd;
        bram_sync_buffer #(.DATA_WIDTH(8), .ADDR_WIDTH(8)) weight_ram (
            .clk(clk), .resetn(resetn),
            .wr_en(w_ok && w_class_i[1:0] == g), .wr_addr({w_class_i[3:2], w_addr_i}), .wr_data(w_data_i),
            .rd_en(state == READ), .rd_addr({group_q, idx_q}),
            .rd_data(w_rd), .rd_valid(w_valid[g]));
        // 그룹 2의 차선 2·3은 클래스 10·11이 없으므로 0을 곱한다.
        assign w_bus[g*8 +: 8] = (group_q == 2'd2 && g >= 2) ? 8'sd0 : w_rd;
        assign scores_o[g*ACC_WIDTH +: ACC_WIDTH] = score_q[g];
        assign scores_o[(g+4)*ACC_WIDTH +: ACC_WIDTH] = score_q[g+4];
    end endgenerate
    assign scores_o[8*ACC_WIDTH +: 2*ACC_WIDTH] = {score_q[9], score_q[8]};

    // ── 08장 4PE (같은 입력 x를 4차선에 나눠 주고 차선별 가중치와 곱해 누산)
    wire [3:0] pe_ready;
    wire pe_done;
    wire [4*ACC_WIDTH-1:0] pe_res;
    pe_array4 #(.DATA_WIDTH(8), .ACC_WIDTH(ACC_WIDTH)) pe (
        .clk(clk), .resetn(resetn), .start_i(state == LAUNCH), .length_i(9'd64),
        .valid_i({4{state == SEND}}), .a_i({4{x_data}}), .b_i(w_bus),
        .ready_o(pe_ready), .busy_o(), .done_o(pe_done), .error_o(), .pe_results_o(pe_res), .sum_o());

    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE; busy_o <= 0; done_o <= 0; error_o <= 0; class_o <= 0; cycles_o <= 0;
            group_q <= 0; idx_q <= 0; k_q <= 0;
            for (i = 0; i < 10; i = i + 1) begin bias_q[i] <= 0; score_q[i] <= 0; end
        end else begin
            done_o <= 0; error_o <= 0;
            if (b_wr_en_i && !busy_o && b_class_i <= 4'd9) bias_q[b_class_i] <= b_data_i;
            if (busy_o) cycles_o <= cycles_o + 16'd1;
            case (state)
                IDLE: if (start_i) begin
                    if (!x_loaded_o) error_o <= 1;           // 64원소가 없으면 시작하지 않는다
                    else begin
                        busy_o <= 1; cycles_o <= 0; group_q <= 0; idx_q <= 0; state <= LAUNCH;
                    end
                end
                LAUNCH: state <= READ;                          // 4PE 누산기 비우고 길이 64 등록
                READ: state <= WAIT_READ;                       // 입력·가중치 BRAM 읽기 요청
                WAIT_READ: if (x_valid && (&w_valid)) state <= SEND;
                    else begin state <= IDLE; busy_o <= 0; error_o <= 1; end
                SEND: if (&pe_ready) begin
                    if (idx_q == 6'd63) state <= WAIT_PE;
                    else begin idx_q <= idx_q + 6'd1; state <= READ; end
                end
                WAIT_PE: if (pe_done) begin
                    for (i = 0; i < 10; i = i + 1)
                        if (i / 4 == group_q) score_q[i] <= $signed(pe_res[(i % 4)*ACC_WIDTH +: ACC_WIDTH]) + bias_q[i];
                    idx_q <= 0;
                    if (group_q == 2'd2) begin class_o <= 0; k_q <= 1; state <= ARGMAX; end
                    else begin group_q <= group_q + 2'd1; state <= LAUNCH; end
                end
                ARGMAX: begin                                   // 한 클록에 하나씩 비교, 같으면 앞 번호 유지
                    if (score_q[k_q] > score_q[class_o]) class_o <= k_q;
                    if (k_q == 4'd9) begin state <= IDLE; busy_o <= 0; done_o <= 1; end
                    else k_q <= k_q + 4'd1;
                end
                default: begin state <= IDLE; busy_o <= 0; error_o <= 1; end
            endcase
        end
    end
endmodule

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. 기준 모델(Python)이 24개 사례를 만듭니다: 형판 가중치로 숫자 0~9를 한 번씩, 무작위 가중치·Bias 10개, 전부 동점, 8·9 동점, 점수 전부 음수, 9번 차선 승리. 형판 가중치는 숫자별 특징 그대로이고 Bias = −|특징|²/2인 최근접 형판 분류이며, 학습한 모델이 아닙니다.

ch14/tb/tb_npu10_group_scheduler.sv · systemverilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
// 14장 TB: 24개 사례(형판 0~9, 무작위 10, 동점·음수·9번 차선)를 적재·실행하고 점수·클래스를 파일로 돌려준다.
module tb_npu10_group_scheduler;
    localparam AW = 32;
    reg clk=0, rn=0, ls=0, iv=0, il=0, wwe=0, bwe=0, start=0;
    reg [7:0] id=0, wd=0; reg [3:0] wc=0, bc=0; reg [5:0] wa=0; reg [AW-1:0] bd=0;
    wire xl, xle, xre, busy, done, err; wire [3:0] cls; wire [10*AW-1:0] scores; wire [15:0] cycles;
    npu10_group_scheduler #(.ACC_WIDTH(AW)) dut(.clk(clk), .resetn(rn), .load_start_i(ls), .in_valid_i(iv),
        .in_data_i(id), .in_last_i(il), .x_loaded_o(xl), .x_len_err_o(xle), .x_range_err_o(xre),
        .w_wr_en_i(wwe), .w_class_i(wc), .w_addr_i(wa), .w_data_i(wd), .b_wr_en_i(bwe), .b_class_i(bc),
        .b_data_i(bd), .start_i(start), .busy_o(busy), .done_o(done), .error_o(err), .class_o(cls),
        .scores_o(scores), .cycles_o(cycles));
    always #5 clk = ~clk;

    integer f, o, n, expc, k, c, v, cases = 0, guards = 0;
    integer x [0:63]; integer w [0:9][0:63]; integer b [0:9];
    reg [8*24-1:0] name; string vp, ap;

    task load_x; integer s; begin
        @(negedge clk); ls = 1; @(negedge clk); ls = 0;
        for (s = 0; s < 64; s = s + 1) begin @(negedge clk); iv = 1; id = x[s]; il = (s == 63); end
        @(negedge clk); iv = 0; il = 0;
        if (!xl) $fatal(1, "입력 적재 실패");
    end endtask

    task load_model; integer cc, s; begin
        for (cc = 0; cc < 10; cc = cc + 1) begin
            for (s = 0; s < 64; s = s + 1) begin @(negedge clk); wwe = 1; wc = cc; wa = s; wd = w[cc][s]; end
            @(negedge clk); wwe = 0; bwe = 1; bc = cc; bd = b[cc];
        end
        @(negedge clk); bwe = 0;
        // 클래스 10·11 쓰기는 무시돼야 한다 (그룹 2 차선 2·3 주소)
        @(negedge clk); wwe = 1; wc = 10; wa = 0; wd = 8'h7f; @(negedge clk); wc = 11; @(negedge clk); wwe = 0;
    end endtask

    task run(output integer got_cycles); begin
        @(negedge clk); start = 1; @(negedge clk); start = 0;
        if (!busy) $fatal(1, "start 후 busy 아님");
        // 실행 중 가중치·Bias 쓰기는 무시돼야 한다
        @(negedge clk); wwe = 1; wc = 8; wa = 5; wd = 8'h55; bwe = 1; bc = 9; bd = 32'h1234; @(negedge clk); wwe = 0; bwe = 0;
        while (!done) begin @(negedge clk); if (err) $fatal(1, "실행 중 error_o"); end
        if (busy) $fatal(1, "done 후 busy");
        got_cycles = cycles;
    end endtask

    initial begin
        integer cyc;
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap)) $fatal(1, "경로 인자");
        f = $fopen(vp, "r"); o = $fopen(ap, "w"); if (!f || !o) $fatal(1, "파일 열기");
        repeat (3) @(negedge clk); rn = 1;
        // 입력 적재 전 start → error_o, busy 아님
        @(negedge clk); start = 1; #1; @(posedge clk); #1;
        if (!err || busy) $fatal(1, "입력 없이 시작함"); guards = guards + 1;
        @(negedge clk); start = 0;
        while ($fscanf(f, "%d %d %s", n, expc, name) == 3) begin
            for (k = 0; k < 64; k = k + 1) if ($fscanf(f, "%d", x[k]) != 1) $fatal(1, "x 형식");
            for (c = 0; c < 10; c = c + 1) for (k = 0; k < 64; k = k + 1) begin if ($fscanf(f, "%d", v) != 1) $fatal(1, "w 형식"); w[c][k] = v; end
            for (c = 0; c < 10; c = c + 1) if ($fscanf(f, "%d", b[c]) != 1) $fatal(1, "b 형식");
            load_x; load_model; run(cyc);
            if (cls != expc) $fatal(1, "%s: class=%0d 기대 %0d", name, cls, expc);
            $fwrite(o, "%0d %0d", n, cls);
            for (c = 0; c < 10; c = c + 1) $fwrite(o, " %0d", $signed(scores[c*AW +: AW]));
            $fwrite(o, " %0d\n", cyc);
            cases = cases + 1;
        end
        // 실행 도중 새 입력 적재 시작 → 입력을 읽을 수 없으므로 error_o 후 IDLE
        load_x; @(negedge clk); start = 1; @(negedge clk); start = 0;
        repeat (40) @(negedge clk); ls = 1; @(negedge clk); ls = 0;
        k = 0; while (!err && k < 20) begin @(negedge clk); k = k + 1; end
        repeat (2) @(negedge clk);
        if (k >= 20 || busy) $fatal(1, "실행 중 적재 시작이 오류로 끝나지 않음"); guards = guards + 1;
        $display("RTL_SIM_PASS GROUP10 cases=%0d scores=%0d guards=%0d cycles_per_inference=%0d", cases, 10 * cases, guards, cyc);
        $fclose(o); $finish;
    end
    initial begin #50000000; $fatal(1, "시간 초과"); end
endmodule

#!/usr/bin/env python3
"""14장 기준 모델: 64원소 × 10클래스 점수 = 내적 + Bias, 동점이면 낮은 번호 argmax.

전처리는 13장 features(), 점수·argmax는 host_tools/p3_integer_reference.py를 그대로 가져다 쓴다.
"""
from __future__ import annotations
import argparse
import random
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT / "ch13" / "reference"), str(ROOT / "host_tools")]
from mnist8x8_reference import draw_digit, features          # noqa: E402  13장 입력 계약
from p3_integer_reference import argmax_lowest, linear_64x10  # noqa: E402  공통 정수 기준

BIAS_LIMIT = 1 << 24       # |Bias| 상한: 점수가 32비트 안에 남게 한다 (|내적| ≤ 64·127·128)


def template_weights(xs: list[list[int]]) -> tuple[list[list[int]], list[int]]:
    """최근접 형판 분류: |x - t|²가 가장 작은 t는 x·t - |t|²/2가 가장 크다.
    가중치 = 숫자별 특징(0~127, INT8 범위), Bias = -|t|²/2 (학습 아님, 형태 대조용)."""
    return [list(x) for x in xs], [-(sum(v * v for v in x) // 2) for x in xs]


def cases() -> list[tuple[str, list[int], list[list[int]], list[int]]]:
    xs = [features(draw_digit(d)) for d in range(10)]
    tw, tb = template_weights(xs)
    out = [(f"template_{d}", xs[d], tw, tb) for d in range(10)]
    rng = random.Random(1414)
    for k in range(10):                                   # 무작위 가중치·Bias (부호 섞임)
        w = [[rng.randint(-128, 127) for _ in range(64)] for _ in range(10)]
        b = [rng.randint(-BIAS_LIMIT, BIAS_LIMIT) for _ in range(10)]
        out.append((f"random_{k}", xs[k], w, b))
    out.append(("tie_all", xs[3], [[1] * 64] * 10, [5] * 10))            # 전부 동점 → 0
    out.append(("tie_8_9", xs[1], [[0] * 64] * 10, [0] * 8 + [9, 9]))     # 8·9 동점 → 8
    out.append(("all_negative", xs[8], [[-128] * 64] * 10, [-10 * c for c in range(10)][::-1]))
    out.append(("lane9_wins", [127] * 64, [[-1] * 64] * 9 + [[127] * 64], [0] * 10))
    return out


def selftest() -> list:
    cs = cases()
    for name, x, w, b in cs:
        s = linear_64x10(x, w, b)
        assert all(-(1 << 31) <= v < (1 << 31) for v in s), name
    got = [argmax_lowest(linear_64x10(x, w, b)) for _, x, w, b in cs[:10]]
    assert got == list(range(10)), got                     # 형태 대조로 0~9가 한 번씩 나온다
    by = {n: argmax_lowest(linear_64x10(x, w, b)) for n, x, w, b in cs}
    assert by["tie_all"] == 0 and by["tie_8_9"] == 8 and by["all_negative"] == 9 and by["lane9_wins"] == 9
    print(f"PYTHON_GROUP10_SELF_TEST_PASS cases={len(cs)} template_classes=0..9 ties=2")
    return cs


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", help="vectors.txt를 쓸 폴더")
    ap.add_argument("--actual", help="RTL 결과 actual.txt (Python 값과 비교)")
    a = ap.parse_args()
    cs = selftest()
    if a.out and not a.actual:
        with open(Path(a.out) / "vectors.txt", "w") as f:
            for n, (name, x, w, b) in enumerate(cs):
                s = linear_64x10(x, w, b)
                f.write(f"{n} {argmax_lowest(s)} {name}\n")
                f.write(" ".join(map(str, x)) + "\n")
                for row in w:
                    f.write(" ".join(map(str, row)) + "\n")
                f.write(" ".join(map(str, b)) + "\n")
    if a.actual:
        rows = {}
        for ln in open(a.actual):
            p = ln.split()
            rows[int(p[0])] = (int(p[1]), [int(v) for v in p[2:12]], int(p[12]))
        for n, (name, x, w, b) in enumerate(cs):
            s = linear_64x10(x, w, b)
            if rows.get(n, (None, None, 0))[:2] != (argmax_lowest(s), s):
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL case={name} rtl={rows.get(n)} python={argmax_lowest(s)} {s}")
        cyc = sorted({r[2] for r in rows.values()})
        print(f"PYTHON_RTL_COMPARE_PASS cases={len(cs)} scores={10 * len(cs)} cycles_per_inference={','.join(map(str, cyc))}")


if __name__ == "__main__":
    main()

07코드 읽기 포인트

위치내용
클래스 → 그룹·차선클래스 c는 그룹 c/4, 차선 c%4입니다. 가중치 BRAM 주소는 {w_class_i[3:2], w_addr_i} = 그룹×64 + 원소입니다.
마지막 2차선 무효그룹 2의 차선 2·3은 클래스 10·11이 없습니다. 가중치 0을 곱하고(w_bus), 점수 저장 루프의 i / 4 == group_q 조건(i < 10)으로 결과를 버립니다. 클래스 10·11 쓰기는 w_class_i <= 9로 막습니다.
점수 = 내적 + BiasReLU·양자화 없이 32비트 그대로 비교합니다. |내적| ≤ 64×127×128 ≈ 220이고 기준 모델이 |Bias| < 224로 제한하므로 32비트에서 넘치지 않습니다. 09장 dense_layer4는 ReLU가 음수 점수를 0으로 만들어 argmax에 쓰지 않습니다.
argmax 동점 규칙ARGMAX 상태에서 1~9를 한 클록에 하나씩 score_q[k_q] > score_q[class_o](엄격 비교)로 봅니다. 같으면 앞 번호가 남아 Python argmax_lowest와 일치합니다.
실행 중 쓰기 무시busy_o인 동안 가중치·Bias 쓰기를 무시합니다. TB는 실행 중 클래스 8·9에 쓰기를 넣어 결과가 바뀌지 않는지 확인합니다.
