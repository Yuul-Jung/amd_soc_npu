제04장 · BRAM 동기 버퍼

bram_sync_buffer는 Artix-7 Block RAM에 추론되는 간단한 동기 읽기·쓰기 버퍼입니다. 10장 NPU Core가 입력 벡터와 가중치를 저장하는 데 사용합니다.

Block RAM 추론쓰기 억제·Reset초기화 미검증 보호

00개요

bram_sync_buffer는 (* ram_style="block" *)으로 Artix-7 BRAM에 추론됩니다. Reset 중에는 쓰기를 억제하여 초기화 충돌을 방지합니다. 배열 초기화 루프는 두지 않으며 rd_valid로 읽기 완료를 알립니다.

범위이 장은 단독 BRAM 버퍼만 다룹니다. 10장 NPU Core에서 입력·가중치 메모리로 2개씩 사용합니다.

01모듈 구조

쓰기와 읽기는 독립 포트. 같은 주소에 동시 쓰기·읽기를 하면 충돌(COLLISION) 오류로 처리합니다.

02RTL 전체 소스

ch04/rtl/bram_sync_buffer.v · verilog
`timescale 1ns/1ps
`default_nettype none
module bram_sync_buffer #(
    parameter integer DATA_WIDTH=16, ADDR_WIDTH=8
) (
    input wire clk, resetn, wr_en, rd_en,
    input wire [ADDR_WIDTH-1:0] wr_addr, rd_addr,
    input wire signed [DATA_WIDTH-1:0] wr_data,
    output reg signed [DATA_WIDTH-1:0] rd_data,
    output reg rd_valid
);
    (* ram_style="block" *) reg signed [DATA_WIDTH-1:0] mem [0:(1<<ADDR_WIDTH)-1];
    // 배열에는 Reset 루프를 두지 않는다. Reset 중 쓰기는 억제한다.
    always @(posedge clk) begin
        if (resetn && wr_en) mem[wr_addr] <= wr_data;
        if (!resetn) begin
            rd_data <= 0;
            rd_valid <= 0;
        end else begin
            rd_valid <= rd_en;
            if (rd_en) rd_data <= mem[rd_addr];
        end
    end
endmodule
`default_nettype wire

03자동 테스트벤치 전체 소스

ch04/tb/tb_bram_sync_buffer.sv · systemverilog
`timescale 1ns/1ps
module tb_bram_sync_buffer;
    parameter W=16, A=8;
    reg clk=0, resetn=0;
    reg wr[0:1], rd[0:1];
    reg [A-1:0] wa[0:1], ra[0:1];
    reg signed [W-1:0] wd[0:1];
    wire signed [W-1:0] data[0:1];
    wire valid[0:1];
    reg written[0:1][0:(1<<A)-1];
    genvar g;
    generate for(g=0;g<2;g=g+1) begin: buffers
        bram_sync_buffer #(.DATA_WIDTH(W),.ADDR_WIDTH(A)) dut
          (.clk(clk),.resetn(resetn),.wr_en(wr[g]),.rd_en(rd[g]),
           .wr_addr(wa[g]),.rd_addr(ra[g]),.wr_data(wd[g]),.rd_data(data[g]),.rd_valid(valid[g]));
    end endgenerate
    always #5 clk=~clk;
    integer f,o,n,c=0,i,j;
    integer rn,w0,wa0,wd0,r0,ra0,w1,wa1,wd1,r1,ra1,e0,v0,e1,v1;
    reg signed [W-1:0] held[0:1];
    reg hv[0:1];
    string vp,ap;
    // 금지 입력을 시험 환경에서 검출하고 RTL 충돌 동작을 명세로 확정하지 않는다.
    always @(posedge clk) if(resetn) begin
        for(integer k=0;k<2;k=k+1) begin
            if(wr[k] && rd[k] && wa[k]==ra[k]) $fatal(1,"COLLISION");
            if(rd[k] && !written[k][ra[k]]) $fatal(1,"UNINITIALIZED_READ");
            if(wr[k]) written[k][wa[k]]=1;
        end
    end
    initial begin
        for(i=0;i<2;i=i+1) begin
            wr[i]=0; rd[i]=0; wa[i]=0; ra[i]=0; wd[i]=0;
            for(j=0;j<(1<<A);j=j+1) written[i][j]=0;
        end
        if($test$plusargs("COLLISION_TEST")) begin
            @(negedge clk); resetn=1; wr[0]=1; rd[0]=1;
            #20; $fatal(1,"Missing collision guard");
        end else if($test$plusargs("UNINITIALIZED_TEST")) begin
            @(negedge clk); resetn=1; rd[0]=1;
            #20; $fatal(1,"Missing uninitialized guard");
        end else begin
            if(!$value$plusargs("VECTORS=%s",vp)||!$value$plusargs("ACTUAL=%s",ap)) $fatal(1,"Missing paths");
            f=$fopen(vp,"r"); o=$fopen(ap,"w"); if(!f||!o) $fatal(1,"Open failed");
            @(posedge clk); #1;
            while(!$feof(f)) begin
                n=$fscanf(f,"%d %d %d %d %d %d %d %d %d %d %d %d %d %d %d\n",
                  rn,w0,wa0,wd0,r0,ra0,w1,wa1,wd1,r1,ra1,e0,v0,e1,v1);
                if(n!=15) $fatal(1,"Bad row");
                @(negedge clk);
                for(i=0;i<2;i=i+1) begin held[i]=data[i]; hv[i]=valid[i]; end
                resetn=rn; wr[0]=w0; wa[0]=wa0; wd[0]=wd0; rd[0]=r0; ra[0]=ra0;
                wr[1]=w1; wa[1]=wa1; wd[1]=wd1; rd[1]=r1; ra[1]=ra1;
                #1;
                for(i=0;i<2;i=i+1) if(data[i]!==held[i]||valid[i]!==hv[i]) $fatal(1,"Async output");
                @(posedge clk); #1; c=c+1;
                if($signed(data[0])!==e0||valid[0]!==v0[0]||$signed(data[1])!==e1||valid[1]!==v1[0])
                    $fatal(1,"Mismatch cycle=%0d",c);
                $fdisplay(o,"%0d %0d %0d %0d %0d",c,data[0],valid[0],data[1],valid[1]);
            end
            if(c==0) $fatal(1,"Empty");
            $fclose(o); $display("RTL_SIM_PASS BRAM W=%0d A=%0d cycles=%0d",W,A,c); $finish;
        end
    end
    initial begin #10000000; $fatal(1,"Timeout"); end
endmodule

05실행 결과 확인

테스트벤치의 자동 판정에서 오류가 발생하지 않고 RTL_SIM_PASS BRAM W=16 A=8 cycles=1016가 출력되어야 합니다. 시뮬레이션 결과는 합성·실물 보드 확인과 구분합니다.

두 인스턴스를 병렬로 시험합니다. 같은 주소 동시 쓰기·읽기(COLLISION)와 초기화 전 읽기(UNINITIALIZED_READ)는 테스트벤치가 금지 입력으로 검출하며, RTL의 충돌 동작을 명세로 확정하지 않습니다. +COLLISION_TEST·+UNINITIALIZED_TEST 모드는 이 감시기가 실제로 동작하는지 확인합니다.

06핵심 코드 풀이

Block RAM 추론: (* ram_style="block" *) 힌트로 Vivado가 LUT 대신 BRAM36을 사용하도록 유도합니다. 배열 선언과 순수 동기 접근 패턴이 함께 필요합니다.

Reset 중 쓰기 억제: if (resetn && wr_en) mem[wr_addr] <= wr_data;. Reset이 아닌 동안에만 쓰기가 이루어져 초기화 직후 의도치 않은 메모리 변경을 방지합니다.

배열 초기화 없음: BRAM에는 Reset 루프를 두지 않습니다. 초기화 전 읽기는 테스트벤치가 written[][] 비트맵으로 추적하여 차단합니다. 실물 보드에서는 Vivado가 초기화 비트스트림을 채울 수 있습니다.
