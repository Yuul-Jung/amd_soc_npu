제2장 · 4뉴런 NPU 보드

BRAM→양자화→내적→4PE→Dense→Core를 쌓아 보드 4뉴런 추론. 서브젝트 8개마다 확인.

00 장 목표·진행 위치

장 시연 (보드): PC가 적재한 입력·가중치로 보드가 4뉴런 추론을 하고 UART에 점수, LED·7세그먼트에 클래스를 띄운다. 누적 npu_core와 12장 엔진을 sw로 바꿔 같은 클래스를 확인한다.

편성: 2일 · 서브젝트 8개 · 서브젝트마다 보드 확인 1번 + 장 시연.

  1. 1장UART 계산기
  2. 2장4뉴런 NPU 보드
  3. 3장8×8 MNIST (CPU 없음)
  4. 4장784 가속기 설계 공간
  5. 5장MicroBlaze V SoC
  6. 6장응용 확장
  7. 7장종합과제
  1. 2-1BRAM 동기 버퍼 (구 04장 · 1일 오전)
    X·W로 쓴 값을 x·w로 되읽으면 같은 값(X:0003, X:FFFE, W:0005). 주소 64는 ERR.    시뮬 통과 · 보드 ⚠ 미검증
  2. 2-2고정소수점 양자화 (구 05장 · 1일 오전)
    32비트 → 8비트: 768 → 03, 384(=1.5) → 02(반올림), 65536 → 7F와 넘침 O:1.    시뮬 통과 · 보드 ⚠ 미검증
  3. 2-3벡터 내적 (구 07장 · 1일 오후)
    X=(1,2,3,4)·W0=(5,6,7,8) → D:0000000046(70). W0[3]을 −8로 바꾸면 6. 길이 0은 ERR.    시뮬 통과 · 보드 ⚠ 미검증
  4. 2-44PE 병렬 어레이 (구 08장 · 1일 오후)
    같은 X에 W0~W3 네 내적을 한 번에: P: 6, 10, −10(…FFF6), 8.    시뮬 통과 · 보드 ⚠ 미검증
  5. 2-5Dense 4뉴런 레이어 (구 09장 · 2일 오전)
    Bias를 더하고 ReLU·16비트 포화: L:0006,0000,005A,7FFF O:8 (음수 → 0, 65544 → 7FFF와 넘침).    시뮬 통과 · 보드 ⚠ 미검증
  6. 2-64뉴런 NPU Core (구 10장 · 2일 오전)
    npu_core가 자체 BRAM에서 읽어 같은 Dense를 한다: N 4가 L 4와 같은 답. 길이 2면 0011,0000,0061,7FFF.    시뮬 통과 · 보드 ⚠ 미검증
  7. 2-7UART 모델 적재 (구 11장 · 2일 오후)
    sw2를 올리면 11장 명령: 고정 그룹 0~3 → C0~C3. 내리고 x 0 → X:FFFF (11장이 쓴 값을 콘솔이 읽음).    시뮬 통과 · 보드 ⚠ 미검증
  8. 2-8통합 NPU Top (구 12장 · 2일 오후)
    sw3을 올리면 12장 엔진: 같은 그룹에서 음수 점수(…FFE, ReLU 없음)가 보이고 클래스는 같다. LED0~3에 클래스, LED14 끝남, 7세그먼트에 클래스.    시뮬 통과 · 보드 ⚠ 미검증

// 2장 디버그 콘솔: 서브젝트 모듈을 UART 명령 하나씩으로 확인한다. 11장 명령과 같은 형식
// (1바이트 명령 + 바이너리 페이로드, 빅엔디언, stop-and-wait). X·W·B·S는 11장과 바이트까지 같다.
//
//   명령 페이로드              응답                                   서브젝트 (STAGE)
//   S    없음                  "I"                                    모두
//   X    addr hi lo            "OK"                                   2-1 (1)
//   W    n addr hi lo          "OK"                                   2-1 (1)
//   x    addr                  "X:" 16진수 4자리                       2-1 (1)
//   w    n addr                "W:" 16진수 4자리                       2-1 (1)
//   Q    int32 4바이트          "Q:" 2자리 " O:" 넘침 1자리             2-2 (2)  (32→8비트, SHIFT 8, 반올림, 포화)
//   D    len                   "D:" 10자리                            2-3 (3)  X·W0
//   P    len                   "P:" 10자리 ×4 (쉼표)                   2-4 (4)  X·W0~W3
//   B    n b4..b0 (signed40)    "OK"                                   2-5 (5)
//   L    len                   "L:" 4자리 ×4 " O:" 넘침 1자리          2-5 (5)  Dense(+Bias, ReLU, 16비트)
//   N    len                   "N:" 4자리 ×4 " O:" 넘침 1자리          2-6 (6)  npu_core (BRAM + Dense)
//   그 밖 (CR·LF 제외)          "?"         필드 범위 밖           "ERR"   (응답 끝은 모두 \r\n)
// len은 1~64, addr 0~63, n 0~3. view_o는 마지막 결과의 아래 16비트 (LED·7세그먼트용).
`timescale 1ns/1ps
`default_nettype none
module p02_console #(
    parameter integer STAGE = 8
) (
    input  wire               clk,
    input  wire               resetn,
    input  wire [7:0]         rx_data_i,
    input  wire               rx_valid_i,
    output wire               rx_ready_o,
    output reg  [7:0]         tx_data_o,
    output reg                tx_valid_o,
    input  wire               tx_ready_i,
    // 묶음으로
    output reg  [3:0]         op_o,
    output reg                op_valid_o,
    output reg  [5:0]         addr_o,
    output reg  [1:0]         neuron_o,
    output reg  signed [15:0] data_o,
    output reg  signed [39:0] bias_o,
    output reg  signed [31:0] qin_o,
    output reg  [6:0]         len_o,
    input  wire               op_done_i,
    input  wire               op_error_i,
    input  wire [159:0]       res_i,
    input  wire [3:0]         flags_i,
    output reg  [15:0]        view_o
);
    localparam [3:0] OP_XWR = 4'd1, OP_WWR = 4'd2, OP_BWR = 4'd3, OP_XRD = 4'd4, OP_WRD = 4'd5,
                     OP_QNT = 4'd6, OP_DOT = 4'd7, OP_PE4 = 4'd8, OP_DENSE = 4'd9, OP_CORE = 4'd10;
    localparam [2:0] S_IDLE = 3'd0, S_RECV = 3'd1, S_ISSUE = 3'd2, S_WAIT = 3'd3, S_LOAD = 3'd4, S_SEND = 3'd5;
    localparam integer MSG_MAX = 48;

    reg [2:0] state;
    reg [7:0] opc;
    reg [7:0] pl [0:5];
    reg [2:0] need, got;
    reg [7:0] msg [0:MSG_MAX-1];
    reg [5:0] msg_len, idx;
    integer   i, k, p;

    function [7:0] hex(input [3:0] n);
        hex = (n < 4'd10) ? (8'h30 + {4'd0, n}) : (8'h37 + {4'd0, n});
    endfunction

    // 명령 → 페이로드 수. 이 STAGE에 없는 명령은 7 (= 모름)
    function [2:0] plen(input [7:0] c);
        case (c)
            "S":      plen = 3'd0;
            "X":      plen = 3'd3;
            "W":      plen = 3'd4;
            "x":      plen = 3'd1;
            "w":      plen = 3'd2;
            "Q":      plen = (STAGE >= 2) ? 3'd4 : 3'd7;
            "D":      plen = (STAGE >= 3) ? 3'd1 : 3'd7;
            "P":      plen = (STAGE >= 4) ? 3'd1 : 3'd7;
            "B":      plen = (STAGE >= 5) ? 3'd6 : 3'd7;
            "L":      plen = (STAGE >= 5) ? 3'd1 : 3'd7;
            "N":      plen = (STAGE >= 6) ? 3'd1 : 3'd7;
            default:  plen = 3'd7;
        endcase
    endfunction

    wire fields_ok =
        (opc == "X" || opc == "x") ? (pl[0] < 8'd64) :
        (opc == "W" || opc == "w") ? (pl[0] < 8'd4 && pl[1] < 8'd64) :
        (opc == "B")               ? (pl[0] < 8'd4) :
        (opc == "D" || opc == "P" || opc == "L" || opc == "N") ? (pl[0] >= 8'd1 && pl[0] <= 8'd64) : 1'b1;

    assign rx_ready_o = (state == S_IDLE) || (state == S_RECV);
    wire rx_take = rx_valid_i && rx_ready_o;

    always @(posedge clk) begin
        if (!resetn) begin
            state <= S_IDLE; opc <= 8'd0; need <= 3'd0; got <= 3'd0; msg_len <= 6'd0; idx <= 6'd0;
            tx_data_o <= 8'd0; tx_valid_o <= 1'b0; op_o <= 4'd0; op_valid_o <= 1'b0;
            addr_o <= 6'd0; neuron_o <= 2'd0; data_o <= 16'sd0; bias_o <= 40'sd0; qin_o <= 32'sd0; len_o <= 7'd0;
            view_o <= 16'd0;
        end else begin
            op_valid_o <= 1'b0;
            case (state)
                S_IDLE: if (rx_take && rx_data_i != 8'h0D && rx_data_i != 8'h0A) begin
                    opc <= rx_data_i; got <= 3'd0; need <= plen(rx_data_i);
                    if (plen(rx_data_i) == 3'd7) begin
                        msg[0] <= "?"; msg[1] <= 8'h0D; msg[2] <= 8'h0A; msg_len <= 6'd3; state <= S_LOAD;
                    end else if (rx_data_i == "S") begin
                        msg[0] <= "I"; msg[1] <= 8'h0D; msg[2] <= 8'h0A; msg_len <= 6'd3; state <= S_LOAD;
                    end else begin
                        state <= S_RECV;
                    end
                end
                S_RECV: if (rx_take) begin
                    pl[got] <= rx_data_i;
                    if (got + 3'd1 == need) state <= S_ISSUE;
                    else got <= got + 3'd1;
                end
                S_ISSUE: begin
                    if (!fields_ok) begin
                        msg[0] <= "E"; msg[1] <= "R"; msg[2] <= "R"; msg[3] <= 8'h0D; msg[4] <= 8'h0A;
                        msg_len <= 6'd5; state <= S_LOAD;
                    end else begin
                        case (opc)
                            "X": begin op_o <= OP_XWR; addr_o <= pl[0][5:0]; data_o <= {pl[1], pl[2]}; end
                            "W": begin op_o <= OP_WWR; neuron_o <= pl[0][1:0]; addr_o <= pl[1][5:0]; data_o <= {pl[2], pl[3]}; end
                            "x": begin op_o <= OP_XRD; addr_o <= pl[0][5:0]; end
                            "w": begin op_o <= OP_WRD; neuron_o <= pl[0][1:0]; addr_o <= pl[1][5:0]; end
                            "Q": begin op_o <= OP_QNT; qin_o <= {pl[0], pl[1], pl[2], pl[3]}; end
                            "D": begin op_o <= OP_DOT; len_o <= pl[0][6:0]; end
                            "P": begin op_o <= OP_PE4; len_o <= pl[0][6:0]; end
                            "B": begin op_o <= OP_BWR; neuron_o <= pl[0][1:0]; bias_o <= {pl[1], pl[2], pl[3], pl[4], pl[5]}; end
                            "L": begin op_o <= OP_DENSE; len_o <= pl[0][6:0]; end
                            default: begin op_o <= OP_CORE; len_o <= pl[0][6:0]; end   // "N"
                        endcase
                        op_valid_o <= 1'b1;
                        state <= S_WAIT;
                    end
                end
                S_WAIT: if (op_done_i) begin
                    if (op_error_i) begin
                        msg[0] <= "E"; msg[1] <= "R"; msg[2] <= "R"; msg[3] <= 8'h0D; msg[4] <= 8'h0A; msg_len <= 6'd5;
                    end else begin
                        case (opc)
                            "X", "W", "B": begin
                                msg[0] <= "O"; msg[1] <= "K"; msg[2] <= 8'h0D; msg[3] <= 8'h0A; msg_len <= 6'd4;
                            end
                            "x", "w": begin
                                msg[0] <= (opc == "x") ? "X" : "W"; msg[1] <= ":";
                                for (k = 0; k < 4; k = k + 1) msg[2 + k] <= hex(res_i[4*(3-k) +: 4]);
                                msg[6] <= 8'h0D; msg[7] <= 8'h0A; msg_len <= 6'd8;
                                view_o <= res_i[15:0];
                            end
                            "Q": begin
                                msg[0] <= "Q"; msg[1] <= ":"; msg[2] <= hex(res_i[7:4]); msg[3] <= hex(res_i[3:0]);
                                msg[4] <= " "; msg[5] <= "O"; msg[6] <= ":"; msg[7] <= hex(flags_i);
                                msg[8] <= 8'h0D; msg[9] <= 8'h0A; msg_len <= 6'd10;
                                view_o <= {{8{res_i[7]}}, res_i[7:0]};
                            end
                            "D": begin
                                msg[0] <= "D"; msg[1] <= ":";
                                for (k = 0; k < 10; k = k + 1) msg[2 + k] <= hex(res_i[4*(9-k) +: 4]);
                                msg[12] <= 8'h0D; msg[13] <= 8'h0A; msg_len <= 6'd14;
                                view_o <= res_i[15:0];
                            end
                            "P": begin
                                msg[0] <= "P"; msg[1] <= ":";
                                for (p = 0; p < 4; p = p + 1) begin
                                    for (k = 0; k < 10; k = k + 1) msg[2 + 11*p + k] <= hex(res_i[40*p + 4*(9-k) +: 4]);
                                    if (p < 3) msg[12 + 11*p] <= ",";
                                end
                                msg[45] <= 8'h0D; msg[46] <= 8'h0A; msg_len <= 6'd47;
                                view_o <= res_i[15:0];
                            end
                            default: begin                       // "L", "N"
                                msg[0] <= opc; msg[1] <= ":";
                                for (p = 0; p < 4; p = p + 1) begin
                                    for (k = 0; k < 4; k = k + 1) msg[2 + 5*p + k] <= hex(res_i[16*p + 4*(3-k) +: 4]);
                                    if (p < 3) msg[6 + 5*p] <= ",";
                                end
                                msg[21] <= " "; msg[22] <= "O"; msg[23] <= ":"; msg[24] <= hex(flags_i);
                                msg[25] <= 8'h0D; msg[26] <= 8'h0A; msg_len <= 6'd27;
                                view_o <= res_i[15:0];
                            end
                        endcase
                    end
                    state <= S_LOAD;
                end
                S_LOAD: begin
                    tx_data_o <= msg[0]; tx_valid_o <= 1'b1; idx <= 6'd0; state <= S_SEND;
                end
                S_SEND: if (tx_valid_o && tx_ready_i) begin
                    if (idx + 6'd1 == msg_len) begin
                        tx_valid_o <= 1'b0; state <= S_IDLE;
                    end else begin
                        tx_data_o <= msg[idx + 6'd1]; idx <= idx + 6'd1;
                    end
                end
                default: state <= S_IDLE;
            endcase
        end
    end
endmodule

//   2-1 (STAGE>=1) bram_sync_buffer 5개: X[64], W0~W3[64] (구 04장)       → 명령 x·w 되읽기
//   2-2 (STAGE>=2) fixed_quantizer 32→8비트, SHIFT 8 (구 05장)             → 명령 Q
//   2-3 (STAGE>=3) vector_dot_core: X·W0 (구 07장, 안에 1장 mac_core)       → 명령 D
//   2-4 (STAGE>=4) pe_array4: X·W0~W3 (구 08장, 안에 07장 내적 4개)          → 명령 P
//   2-5 (STAGE>=5) dense_layer4: + Bias, ReLU, 16비트 양자화 (구 09장)       → 명령 L
//   2-6 (STAGE>=6) npu_core: 자체 BRAM + Dense (구 10장)                    → 명령 N
//   2-7 (STAGE>=7) 엔진 포트: 11장 명령 해석기가 npu_core를 12장 엔진처럼 쓴다 (점수 41비트 확장, argmax)
// D·P·L은 묶음의 BRAM에서 한 원소씩 읽어 각 모듈의 ready/valid로 넣는다. npu_core는 자체 BRAM을 쓰므로
// X·W 쓰기는 묶음 BRAM과 npu_core에 같이 들어간다. Bias는 묶음 레지스터 4개.
`timescale 1ns/1ps
`default_nettype none
module p02_bank #(
    parameter integer STAGE = 8
) (
    input  wire               clk,
    input  wire               resetn,
    // 디버그 콘솔 쪽
    input  wire [3:0]         op_i,          // OP_* (아래)
    input  wire               op_valid_i,    // 1클록
    input  wire [5:0]         addr_i,
    input  wire [1:0]         neuron_i,
    input  wire signed [15:0] data_i,
    input  wire signed [39:0] bias_i,
    input  wire signed [31:0] qin_i,
    input  wire [6:0]         len_i,
    output reg                op_done_o,     // 1클록. 이 클록에 res_o·flags_o 유효
    output reg                op_error_o,
    output reg  [159:0]       res_o,
    output reg  [3:0]         flags_o,
    // 엔진 포트 (11장 명령 해석기, STAGE>=7). owner = 1이면 메모리 쓰기·코어 실행을 엔진 포트가 가진다
    input  wire               owner_engine_i,
    input  wire               e_start_i,
    input  wire [6:0]         e_length_i,
    input  wire               e_x_wr_i,
    input  wire               e_w_wr_i,
    input  wire               e_b_wr_i,
    input  wire [5:0]         e_addr_i,
    input  wire [1:0]         e_neuron_i,
    input  wire signed [15:0] e_data_i,
    input  wire signed [39:0] e_bias_i,
    output wire               e_load_ready_o,
    output wire               e_busy_o,
    output wire               e_done_o,
    output wire               e_error_o,
    output wire [1:0]         e_class_o,
    output wire [163:0]       e_scores_o
);
    localparam [3:0] OP_XWR = 4'd1, OP_WWR = 4'd2, OP_BWR = 4'd3, OP_XRD = 4'd4, OP_WRD = 4'd5,
                     OP_QNT = 4'd6, OP_DOT = 4'd7, OP_PE4 = 4'd8, OP_DENSE = 4'd9, OP_CORE = 4'd10;

    // ---------------- 쓰기 경로: 콘솔 또는 엔진 포트 ----------------
    wire               eng = (STAGE >= 7) && owner_engine_i;
    wire               x_we = eng ? e_x_wr_i : (op_valid_i && op_i == OP_XWR);
    wire               w_we = eng ? e_w_wr_i : (op_valid_i && op_i == OP_WWR);
    wire               b_we = eng ? e_b_wr_i : (op_valid_i && op_i == OP_BWR);
    wire [5:0]         w_addr = eng ? e_addr_i : addr_i;
    wire [1:0]         w_neur = eng ? e_neuron_i : neuron_i;
    wire signed [15:0] w_data = eng ? e_data_i : data_i;
    wire signed [39:0] w_bias = eng ? e_bias_i : bias_i;

    reg  [159:0] biases;                          // {b3,b2,b1,b0}
    always @(posedge clk) begin
        if (!resetn)  biases <= 160'd0;
        else if (b_we) biases[w_neur*40 +: 40] <= w_bias;
    end

    // ---------------- 2-1: BRAM 5개 ----------------
    reg         rd_en;
    reg  [5:0]  rd_addr;
    wire signed [15:0] x_rd;
    wire [63:0] w_rd;
    wire        x_rv;
    wire [3:0]  w_rv;
    bram_sync_buffer #(.DATA_WIDTH(16), .ADDR_WIDTH(6)) u_xmem (
        .clk(clk), .resetn(resetn), .wr_en(x_we), .rd_en(rd_en), .wr_addr(w_addr), .rd_addr(rd_addr),
        .wr_data(w_data), .rd_data(x_rd), .rd_valid(x_rv));
    genvar g;
    generate for (g = 0; g < 4; g = g + 1) begin : g_wmem
        bram_sync_buffer #(.DATA_WIDTH(16), .ADDR_WIDTH(6)) u_wmem (
            .clk(clk), .resetn(resetn), .wr_en(w_we && w_neur == g), .rd_en(rd_en), .wr_addr(w_addr),
            .rd_addr(rd_addr), .wr_data(w_data), .rd_data(w_rd[g*16 +: 16]), .rd_valid(w_rv[g]));
    end endgenerate

    // ---------------- 2-2: 양자화기 ----------------
    reg               q_valid;
    wire signed [7:0] q_out;
    wire              q_ov, q_done;
    generate if (STAGE >= 2) begin : g_quant
        fixed_quantizer #(.IN_WIDTH(32), .OUT_WIDTH(8), .SHIFT(8), .ROUND(1), .SATURATE(1)) u_q (
            .clk(clk), .resetn(resetn), .valid_i(q_valid), .x_i(qin_i), .q_o(q_out), .valid_o(q_done), .overflow_o(q_ov));
    end else begin : g_no_quant
        assign q_out = 8'sd0; assign q_ov = 1'b0; assign q_done = 1'b0;
    end endgenerate

    // ---------------- 2-3·2-4·2-5: 한 원소씩 넣는 순서기 ----------------
    reg  [3:0]  seq_op;                           // OP_DOT / OP_PE4 / OP_DENSE
    reg         start;                            // 1클록
    reg  [8:0]  length;
    reg         push;                             // 원소 하나를 내놓은 상태
    reg  signed [15:0] x_q;
    reg  [63:0] w_q;

    wire        dot_ready, dot_done, dot_err;
    wire signed [39:0] dot_res;
    generate if (STAGE >= 3) begin : g_dot
        vector_dot_core #(.DATA_WIDTH(16), .ACC_WIDTH(40)) u_dot (
            .clk(clk), .resetn(resetn), .start_i(start && seq_op == OP_DOT), .valid_i(push && seq_op == OP_DOT),
            .length_i(length), .a_i(x_q), .b_i(w_q[15:0]), .ready_o(dot_ready), .busy_o(), .done_o(dot_done),
            .error_o(dot_err), .result_o(dot_res));
    end else begin : g_no_dot
        assign dot_ready = 1'b0; assign dot_done = 1'b0; assign dot_err = 1'b0; assign dot_res = 40'sd0;
    end endgenerate

    wire [3:0]   pe_ready;
    wire         pe_done, pe_err;
    wire [159:0] pe_res;
    generate if (STAGE >= 4) begin : g_pe
        pe_array4 #(.DATA_WIDTH(16), .ACC_WIDTH(40)) u_pe (
            .clk(clk), .resetn(resetn), .start_i(start && seq_op == OP_PE4), .length_i(length),
            .valid_i({4{push && seq_op == OP_PE4 && (&pe_ready)}}), .a_i({4{x_q}}), .b_i(w_q),
            .ready_o(pe_ready), .busy_o(), .done_o(pe_done), .error_o(pe_err), .pe_results_o(pe_res), .sum_o());
    end else begin : g_no_pe
        assign pe_ready = 4'd0; assign pe_done = 1'b0; assign pe_err = 1'b0; assign pe_res = 160'd0;
    end endgenerate

    wire        dn_ready, dn_done, dn_err;
    wire [63:0] dn_res;
    wire [3:0]  dn_ov;
    generate if (STAGE >= 5) begin : g_dense
        dense_layer4 #(.DATA_WIDTH(16), .ACC_WIDTH(40), .SHIFT(0)) u_dense (
            .clk(clk), .resetn(resetn), .start_i(start && seq_op == OP_DENSE), .valid_i(push && seq_op == OP_DENSE),
            .length_i(length), .x_i(x_q), .weights_i(w_q), .biases_i(biases), .ready_o(dn_ready), .busy_o(),
            .done_o(dn_done), .error_o(dn_err), .result_o(dn_res), .overflow_o(dn_ov));
    end else begin : g_no_dense
        assign dn_ready = 1'b0; assign dn_done = 1'b0; assign dn_err = 1'b0; assign dn_res = 64'd0; assign dn_ov = 4'd0;
    end endgenerate

    wire seq_ready = (seq_op == OP_DOT) ? dot_ready : (seq_op == OP_PE4) ? (&pe_ready) : dn_ready;
    wire seq_done  = (seq_op == OP_DOT) ? dot_done  : (seq_op == OP_PE4) ? pe_done      : dn_done;
    wire seq_err   = (seq_op == OP_DOT) ? dot_err   : (seq_op == OP_PE4) ? pe_err       : dn_err;

    // ---------------- 2-6: npu_core (자체 BRAM) ----------------
    reg         core_start_c;                     // 콘솔이 시작
    reg         eng_run;                          // 지금 실행이 엔진 포트 것
    wire        core_start = core_start_c || (eng && e_start_i);
    wire [8:0]  core_len = (eng && e_start_i) ? {2'b00, e_length_i} : {2'b00, len_i};
    wire        core_load_ready, core_busy, core_done, core_err;
    wire [63:0] core_res;
    wire [3:0]  core_ov;
    generate if (STAGE >= 6) begin : g_core
        npu_core #(.DATA_WIDTH(16), .ACC_WIDTH(40)) u_core (
            .clk(clk), .resetn(resetn), .start_i(core_start), .length_i(core_len), .biases_i(biases),
            .x_wr_en(x_we), .w_wr_en(w_we), .x_wr_addr(w_addr), .w_wr_addr(w_addr), .w_neuron(w_neur),
            .x_wr_data(w_data), .w_wr_data(w_data), .load_ready_o(core_load_ready), .busy_o(core_busy),
            .done_o(core_done), .error_o(core_err), .result_o(core_res), .overflow_o(core_ov));
    end else begin : g_no_core
        assign core_load_ready = resetn; assign core_busy = 1'b0; assign core_done = 1'b0;
        assign core_err = 1'b0; assign core_res = 64'd0; assign core_ov = 4'd0;
    end endgenerate

    // ---------------- 2-7: 엔진 포트 (npu_core를 12장 엔진 모양으로) ----------------
    generate if (STAGE >= 7) begin : g_eng
        wire signed [15:0] r0 = core_res[15:0],  r1 = core_res[31:16];
        wire signed [15:0] r2 = core_res[47:32], r3 = core_res[63:48];
        reg  [1:0] best;
        reg  signed [15:0] best_v;
        always @(*) begin                         // 12장 엔진과 같은 규칙: 큰 값, 같으면 앞 번호
            best = 2'd0; best_v = r0;
            if (r1 > best_v) begin best = 2'd1; best_v = r1; end
            if (r2 > best_v) begin best = 2'd2; best_v = r2; end
            if (r3 > best_v) begin best = 2'd3; best_v = r3; end
        end
        assign e_load_ready_o = core_load_ready && !e_start_i;
        assign e_busy_o   = core_busy && eng_run;
        assign e_done_o   = core_done && eng_run;
        assign e_error_o  = core_err && eng_run;
        assign e_class_o  = best;
        assign e_scores_o = {{{25{r3[15]}}, r3}, {{25{r2[15]}}, r2}, {{25{r1[15]}}, r1}, {{25{r0[15]}}, r0}};
    end else begin : g_no_eng
        assign e_load_ready_o = 1'b0; assign e_busy_o = 1'b0; assign e_done_o = 1'b0; assign e_error_o = 1'b0;
        assign e_class_o = 2'd0; assign e_scores_o = 164'd0;
    end endgenerate

    // ---------------- 콘솔 명령 처리 ----------------
    localparam [2:0] S_IDLE = 3'd0, S_RDW = 3'd1, S_QW = 3'd2, S_SRD = 3'd3, S_SRDW = 3'd4,
                     S_PUSH = 3'd5, S_SWAIT = 3'd6, S_CWAIT = 3'd7;
    reg [2:0] state;
    reg [3:0] cur_op;
    reg [5:0] idx;

    function stage_ok(input [3:0] op);
        stage_ok = (op <= OP_WRD) ||
                   (op == OP_QNT && STAGE >= 2) || (op == OP_DOT && STAGE >= 3) ||
                   (op == OP_PE4 && STAGE >= 4) || (op == OP_DENSE && STAGE >= 5) || (op == OP_CORE && STAGE >= 6);
    endfunction

    always @(posedge clk) begin
        if (!resetn) begin
            state <= S_IDLE; cur_op <= 4'd0; idx <= 6'd0;
            op_done_o <= 1'b0; op_error_o <= 1'b0; res_o <= 160'd0; flags_o <= 4'd0;
            rd_en <= 1'b0; rd_addr <= 6'd0; q_valid <= 1'b0;
            seq_op <= 4'd0; start <= 1'b0; length <= 9'd0; push <= 1'b0; x_q <= 16'sd0; w_q <= 64'd0;
            core_start_c <= 1'b0; eng_run <= 1'b0;
        end else begin
            op_done_o <= 1'b0; op_error_o <= 1'b0;
            rd_en <= 1'b0; q_valid <= 1'b0; start <= 1'b0; core_start_c <= 1'b0;
            if (eng && e_start_i) eng_run <= 1'b1;
            case (state)
                S_IDLE: if (op_valid_i && !eng) begin
                    cur_op <= op_i;
                    if (!stage_ok(op_i) || ((op_i >= OP_DOT) && (len_i == 7'd0 || len_i > 7'd64))) begin
                        op_error_o <= 1'b1; op_done_o <= 1'b1;
                    end else case (op_i)
                        OP_XWR, OP_WWR, OP_BWR: op_done_o <= 1'b1;
                        OP_XRD, OP_WRD: begin rd_en <= 1'b1; rd_addr <= addr_i; state <= S_RDW; end
                        OP_QNT: begin q_valid <= 1'b1; state <= S_QW; end
                        OP_DOT, OP_PE4, OP_DENSE: begin
                            seq_op <= op_i; length <= {2'b00, len_i}; start <= 1'b1; idx <= 6'd0; state <= S_SRD;
                        end
                        OP_CORE: begin core_start_c <= 1'b1; eng_run <= 1'b0; state <= S_CWAIT; end
                        default: begin op_error_o <= 1'b1; op_done_o <= 1'b1; end
                    endcase
                end
                S_RDW: if (x_rv) begin            // BRAM 동기 읽기: rd_valid가 선 클록에 값
                    res_o <= {144'd0, (cur_op == OP_XRD) ? x_rd : w_rd[neuron_i*16 +: 16]};
                    op_done_o <= 1'b1; state <= S_IDLE;
                end
                S_QW: if (q_done) begin
                    res_o <= {152'd0, q_out}; flags_o <= {3'd0, q_ov}; op_done_o <= 1'b1; state <= S_IDLE;
                end
                S_SRD: begin rd_en <= 1'b1; rd_addr <= idx; state <= S_SRDW; end
                S_SRDW: if (x_rv) begin x_q <= x_rd; w_q <= w_rd; push <= 1'b1; state <= S_PUSH; end
                S_PUSH: if (seq_ready) begin      // 이 클록에 모듈이 원소를 받는다
                    push <= 1'b0;
                    if ({1'b0, idx} + 7'd1 == length[6:0]) state <= S_SWAIT;
                    else begin idx <= idx + 6'd1; state <= S_SRD; end
                end
                S_SWAIT: if (seq_done || seq_err) begin
                    res_o <= (seq_op == OP_DOT) ? {120'd0, dot_res} : (seq_op == OP_PE4) ? pe_res : {96'd0, dn_res};
                    flags_o <= (seq_op == OP_DENSE) ? dn_ov : 4'd0;
                    op_error_o <= seq_err; op_done_o <= 1'b1; state <= S_IDLE;
                end
                S_CWAIT: if (core_done || core_err) begin
                    res_o <= {96'd0, core_res}; flags_o <= core_ov;
                    op_error_o <= core_err; op_done_o <= 1'b1; state <= S_IDLE;
                end
                default: state <= S_IDLE;
            endcase
            if (core_done || core_err) eng_run <= 1'b0;
        end
    end
endmodule

#!/usr/bin/env python3
"""2장 4뉴런 NPU 보드 PC 쪽 도구: 명령을 바이트로 바꿔 보내고 응답 한 줄을 받아 출력한다.

  python3 p02/host/p02_send.py --dry X 0 3            # 보낼 바이트만 출력 (보드 없이)
  python3 p02/host/p02_send.py /dev/ttyUSB1 D 4       # X·W0 내적
  python3 p02/host/p02_send.py COM3 2                 # (sw2 올림) 11장 고정 그룹 2 추론
  (pip install pyserial 필요. 115200 8N1)

명령 (p02/rtl/p02_console.v, 11장 uart_npu_command_controller.v와 같은 형식):
  S | X addr v | W n addr v | x addr | w n addr | Q v32 | D len | P len | B n v40 | L len | N len
  11장 (sw2 = 1): S | X | W | B | R len | 0 1 2 3
"""
import struct
import sys

INT = lambda s: int(s, 0)


def encode(args):
    op, a = args[0], [INT(x) for x in args[1:]]
    if op in ("S", "0", "1", "2", "3"):
        return op.encode()
    if op == "X":
        return b"X" + bytes([a[0]]) + struct.pack(">h", a[1])
    if op == "W":
        return b"W" + bytes([a[0], a[1]]) + struct.pack(">h", a[2])
    if op == "x":
        return b"x" + bytes([a[0]])
    if op == "w":
        return b"w" + bytes([a[0], a[1]])
    if op == "Q":
        return b"Q" + struct.pack(">i", a[0])
    if op == "B":
        return b"B" + bytes([a[0]]) + (a[1] & ((1 << 40) - 1)).to_bytes(5, "big")
    if op in ("D", "P", "L", "N", "R"):
        return op.encode() + bytes([a[0]])
    raise SystemExit("모르는 명령: " + op)


def main(argv):
    if len(argv) < 2:
        raise SystemExit(__doc__)
    port, data = argv[0], encode(argv[1:])
    print("TX", data.hex(" "))
    if port == "--dry":
        return 0
    import serial                           # pyserial (보드에서만 필요)
    with serial.Serial(port, 115200, timeout=2) as s:
        s.write(data)
        reply = s.read_until(b"\n")
    print("RX", repr(reply.decode("ascii", "replace")))
    return 0 if reply else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

class Memory:

        self.data = 0

            self.data = 0
            return 0, 0

def quantize(x, width, shift, rounding=1, saturate=1):
    # Python 정수 나눗셈을 사용하여 RTL의 시프트 구현과 독립적으로 계산한다.
    if rounding and shift:
        mag=(abs(x)+(1<<(shift-1)))//(1<<shift)
        y=-mag if x<0 else mag
    else: y=x//(1<<shift)
    lo,hi=-(1<<(width-1)),(1<<(width-1))-1
    ov=int(y<lo or y>hi)
    return (min(hi,max(lo,y)) if saturate else wrap(y,width)),ov

// (추가) UART 명령·모델 적재·추론 결과 직렬화 Controller.
// 단일 문자 '0'~'3'은 2원소 고정 표본. 11장 바이너리 명령은 아래 README 참조.
// 요청/응답은 stop-and-wait: ACK 또는 결과 LF까지 다음 명령 송신 금지.
module uart_npu_command_controller(
    input wire clk,resetn,
    input wire [7:0] rx_data_i,
    input wire rx_valid_i,
    output wire rx_ready_o,
    output wire [7:0] tx_data_o,
    output wire tx_valid_o,
    input wire tx_ready_i,
    output wire npu_start_o,
    output wire [6:0] npu_length_o,
    output wire x_wr_en_o,w_wr_en_o,b_wr_en_o,
    output reg [5:0] wr_addr_o,
    output reg [1:0] neuron_o,
    output reg signed [15:0] wr_data_o,
    output reg signed [39:0] bias_data_o,
    input wire npu_load_ready_i,
    input wire npu_busy_i,npu_done_i,npu_error_i,
    input wire [1:0] npu_class_i,
    input wire [163:0] npu_scores_i,
    output reg protocol_error_o
);
    localparam [2:0] IDLE=0,RECV=1,APPLY=2,PRESET=3,RUN=4,WAIT_NPU=5,SEND=6;
    reg [2:0] state;
    reg [7:0] opcode;
    reg [7:0] payload[0:5];
    reg [2:0] recv_index,expected_bytes;
    reg [1:0] sample_id;
    reg [3:0] preset_index;
    reg [6:0] run_length;
    reg [7:0] response[0:63];
    reg [5:0] send_length,send_index;
    integer i,j;

    assign rx_ready_o=resetn&&(state==IDLE||state==RECV);
    assign tx_valid_o=(state==SEND)&&(send_index<send_length);
    assign tx_data_o=response[send_index];
    assign npu_start_o=(state==RUN);
    assign npu_length_o=run_length;

    // (추가) 필드 형식 거부. 주소 0~63, 뉴런 0~3, 길이 1~64.
    wire accepted_fields =
        (opcode=="X" && payload[0]<64) ||
        (opcode=="W" && payload[0]<4 && payload[1]<64) ||
        (opcode=="B" && payload[0]<4) ||
        (opcode=="R" && payload[0]>=1 && payload[0]<=64);

    assign x_wr_en_o=(state==PRESET && preset_index<2 && npu_load_ready_i) ||
        (state==APPLY && opcode=="X" && accepted_fields && npu_load_ready_i);
    assign w_wr_en_o=(state==PRESET && preset_index>=2 && preset_index<10 &&
                      npu_load_ready_i) ||
        (state==APPLY && opcode=="W" && accepted_fields && npu_load_ready_i);
    assign b_wr_en_o=(state==PRESET && preset_index>=10 && npu_load_ready_i) ||
        (state==APPLY && opcode=="B" && accepted_fields && npu_load_ready_i);

    always @*begin
        wr_addr_o=0;neuron_o=0;wr_data_o=0;bias_data_o=0;
        if(state==PRESET)begin
            if(preset_index<2)begin
                wr_addr_o={2'b00,preset_index};
                // X0=(+1,+1), X1=(+1,-1), X2=(-1,+1), X3=(-1,-1)
                wr_data_o=((preset_index==0 && sample_id[1]) ||
                    (preset_index==1 && sample_id[0])) ? -16'sd1 : 16'sd1;
            end else if(preset_index<10)begin
                neuron_o=(preset_index-2)>>1;
                wr_addr_o=(preset_index-2)&1;
                wr_data_o=((neuron_o[1] && wr_addr_o==0) ||
                    (neuron_o[0] && wr_addr_o==1)) ? -16'sd1 : 16'sd1;
            end else neuron_o=preset_index-10;
        end else if(state==APPLY)begin
            case(opcode)
                "X":begin wr_addr_o=payload[0][5:0];wr_data_o={payload[1],payload[2]};end
                "W":begin neuron_o=payload[0][1:0];wr_addr_o=payload[1][5:0];
                    wr_data_o={payload[2],payload[3]};end
                "B":begin neuron_o=payload[0][1:0];
                    bias_data_o={payload[1],payload[2],payload[3],payload[4],payload[5]};end
                default:begin end
            endcase
        end
    end

    function [7:0] hexdigit(input [3:0] n);
        begin
            hexdigit=(n<10) ? (8'h30+n) : (8'h41+(n-10));
        end
    endfunction
    function [7:0] score_hex(input [40:0] score,input integer position);
        reg [43:0] ext;
        reg [3:0] nib;
        begin
            ext={{3{score[40]}},score};
            nib=(ext>>((10-position)*4)) & 4'hf;
            score_hex=hexdigit(nib);
        end
    endfunction

    always @(posedge clk)begin
        if(!resetn)begin
            state<=IDLE;opcode<=0;recv_index<=0;expected_bytes<=0;
            sample_id<=0;preset_index<=0;run_length<=2;
            send_length<=0;send_index<=0;protocol_error_o<=0;
        end else begin
            protocol_error_o<=0;
            case(state)
                IDLE:if(rx_valid_i && rx_ready_o)begin
                    if(rx_data_i>="0" && rx_data_i<="3")begin
                        sample_id<=rx_data_i[1:0];preset_index<=0;run_length<=2;
                        state<=PRESET;
                    end else if(rx_data_i=="X"||rx_data_i=="W"||
                        rx_data_i=="B"||rx_data_i=="R")begin
                        opcode<=rx_data_i;recv_index<=0;
                        case(rx_data_i)
                            "X":expected_bytes<=3;
                            "W":expected_bytes<=4;
                            "B":expected_bytes<=6;
                            default:expected_bytes<=1;
                        endcase
                        state<=RECV;
                    end else if(rx_data_i=="S")begin
                        response[0]<="I";response[1]<=8'h0d;response[2]<=8'h0a;
                        send_index<=0;send_length<=3;state<=SEND;
                    end else if(rx_data_i!=8'h0d && rx_data_i!=8'h0a)begin
                        protocol_error_o<=1;
                        response[0]<="E";response[1]<="R";response[2]<="R";
                        response[3]<=8'h0d;response[4]<=8'h0a;
                        send_index<=0;send_length<=5;state<=SEND;
                    end
                end
                RECV:if(rx_valid_i && rx_ready_o)begin
                    payload[recv_index]<=rx_data_i;
                    if(recv_index+3'd1==expected_bytes)state<=APPLY;
                    else recv_index<=recv_index+1'b1;
                end
                APPLY:begin
                    if(!accepted_fields || !npu_load_ready_i)begin
                        protocol_error_o<=1;
                        response[0]<="E";response[1]<="R";response[2]<="R";
                        response[3]<=8'h0d;response[4]<=8'h0a;
                        send_index<=0;send_length<=5;state<=SEND;
                    end else if(opcode=="R")begin
                        run_length<=payload[0][6:0];state<=RUN;
                    end else begin
                        response[0]<="O";response[1]<="K";
                        response[2]<=8'h0d;response[3]<=8'h0a;
                        send_index<=0;send_length<=4;state<=SEND;
                    end
                end
                PRESET:if(npu_load_ready_i)begin
                    if(preset_index==13)state<=RUN;
                    else preset_index<=preset_index+1'b1;
                end
                RUN:state<=WAIT_NPU;
                WAIT_NPU:begin
                    if(npu_error_i)begin
                        protocol_error_o<=1;
                        response[0]<="E";response[1]<="R";response[2]<="R";
                        response[3]<=8'h0d;response[4]<=8'h0a;
                        send_index<=0;send_length<=5;state<=SEND;
                    end else if(npu_done_i)begin
                        // Cx: + 4 signed 41bit scores each in 11 hex chars + CRLF.
                        response[0]<="C";response[1]<=8'h30+{6'b0,npu_class_i};
                        response[2]<=":";
                        for(i=0;i<4;i=i+1)begin
                            for(j=0;j<11;j=j+1)
                                response[3+i*12+j]<=score_hex(
                                    npu_scores_i[i*41+:41],j);
                            if(i<3)response[14+i*12]<=",";
                        end
                        response[50]<=8'h0d;response[51]<=8'h0a;
                        send_index<=0;send_length<=52;state<=SEND;
                    end
                end
                SEND:if(tx_valid_o && tx_ready_i)begin
                    if(send_index+6'd1==send_length)begin
                        send_index<=0;state<=IDLE;
                    end else send_index<=send_index+1'b1;
                end
                default:begin state<=IDLE;protocol_error_o<=1;end
            endcase
        end
    end
endmodule

                CMP2:begin

                    done_o<=1;busy_o<=0;state<=IDLE;
                end
                default:begin state<=IDLE;busy_o<=0;error_o<=1;end
            endcase
        end
    end

              @(negedge clk);wwe=0;end

// Empty/Full, 동시 Push/Pop, Ready/Valid, Reset 초기화, 연속 수신 손실, Overrun, Framing, 전송 중단.
// 실제 직렬 비트파형으로 RX를 구동하고 TX 핀을 독립 uart_rx로 복조해 순서·값을 비교한다.
`timescale 1ns/1ps
module tb_uart_core_fifo;
    localparam integer CLK_HZ=1000000,BAUD=100000;
    localparam integer CPB=(CLK_HZ+BAUD/2)/BAUD; // 10 clocks/bit: 시뮬레이션 단축
    reg clk=0,resetn=0,rxd=1;
    wire txd;
    wire [7:0] rx_data;
    wire rx_valid;
    reg rx_ready=0;
    reg [7:0] tx_data=0;
    reg tx_valid=0;
    wire tx_ready;
    wire fe,ov;
    reg mon_resetn=0;
    wire [7:0] mon_data;
    wire mon_valid,mon_fe,mon_ov;
    always #5 clk=~clk;

    uart_core #(.CLK_HZ(CLK_HZ),.BAUD(BAUD)) dut(.clk(clk),.resetn(resetn),
        .uart_rxd_i(rxd),.uart_txd_o(txd),
        .rx_out_data(rx_data),.rx_out_valid(rx_valid),.rx_out_ready(rx_ready),
        .tx_in_data(tx_data),.tx_in_valid(tx_valid),.tx_in_ready(tx_ready),
        .rx_framing_error_o(fe),.rx_overrun_o(ov));
    uart_rx #(.CLK_HZ(CLK_HZ),.BAUD(BAUD)) mon(.clk(clk),.resetn(mon_resetn),
        .rx_i(txd),.out_data_o(mon_data),.out_valid_o(mon_valid),
        .out_ready_i(1'b1),.framing_error_o(mon_fe),.overrun_o(mon_ov));

    // ---------------- scoreboards ----------------
    reg [7:0] txq[0:1023];
    integer tx_head=0,tx_tail=0;
    reg [7:0] rxq[0:1023];
    integer rx_head=0,rx_tail=0;
    integer tx_accepted=0,rx_delivered=0;
    // coverage
    integer cov_rx_full=0,cov_tx_full=0,cov_rx_full_pushpop=0,cov_tx_full_pushpop=0;
    integer cov_rx_pushpop=0,cov_tx_pushpop=0,cov_rx_empty_pop_attempt=0;
    integer cov_tx_stall=0,cov_rx_hold=0;
    reg [7:0] prev_rx_data;
    reg prev_rx_stall=0;
    reg [7:0] prev_tx_data;
    reg prev_tx_stall=0;

    always @(posedge clk)begin
        // TX handshake: 수락된 바이트만 기대 큐에 넣는다.
        if(tx_valid && tx_ready)begin
            txq[tx_tail%1024]=tx_data;tx_tail=tx_tail+1;tx_accepted=tx_accepted+1;
        end
        // RX handshake: 실제 수신 순서 비교
        if(rx_valid && rx_ready)begin
            if(rx_head==rx_tail)$fatal(1,"RX unexpected byte %02h",rx_data);
            if(rx_data!==rxq[rx_head%1024])
                $fatal(1,"RX order/data mismatch idx=%0d got=%02h exp=%02h",
                    rx_head,rx_data,rxq[rx_head%1024]);
            rx_head=rx_head+1;rx_delivered=rx_delivered+1;
        end
        // TX 핀 복조 비교
        if(mon_valid)begin
            if(tx_head==tx_tail)$fatal(1,"TX unexpected byte on pin %02h",mon_data);
            if(mon_data!==txq[tx_head%1024])
                $fatal(1,"TX order/data mismatch idx=%0d got=%02h exp=%02h",
                    tx_head,mon_data,txq[tx_head%1024]);
            tx_head=tx_head+1;
        end
        if(mon_fe)$fatal(1,"TX pin framing error");
        // Ready/Valid 규칙: valid && !ready 동안 rx_out_data 유지
        if(prev_rx_stall && rx_data!==prev_rx_data)$fatal(1,"RX data changed while stalled");
        prev_rx_stall=rx_valid && !rx_ready;prev_rx_data=rx_data;
        if(prev_tx_stall && (!tx_valid || tx_data!==prev_tx_data))
            $fatal(1,"TB violated TX valid/data hold");
        prev_tx_stall=tx_valid && !tx_ready;prev_tx_data=tx_data;
        // FIFO 내부 불변식
        if(resetn)begin
            if(dut.rx_count>16 || dut.tx_count>16)$fatal(1,"FIFO count overflow");
            if(dut.rx_count!=0 && dut.rx_count!=16 &&
               dut.rx_count!=((dut.rx_wptr-dut.rx_rptr)&4'hf))$fatal(1,"RX ptr/count mismatch");
            if(dut.tx_count!=0 && dut.tx_count!=16 &&
               dut.tx_count!=((dut.tx_wptr-dut.tx_rptr)&4'hf))$fatal(1,"TX ptr/count mismatch");
            if((dut.rx_count==0 || dut.rx_count==16) && dut.rx_wptr!=dut.rx_rptr)
                $fatal(1,"RX empty/full pointer mismatch");
            if((dut.tx_count==0 || dut.tx_count==16) && dut.tx_wptr!=dut.tx_rptr)
                $fatal(1,"TX empty/full pointer mismatch");
            if(rx_valid!==(dut.rx_count!=0))$fatal(1,"RX valid != !empty");
            if(dut.tx_count==16 && !dut.tx_pop && tx_ready)$fatal(1,"TX ready while full");
            if(dut.rx_count==16 && dut.rx_push && !dut.rx_pop)$fatal(1,"RX push while full");
            // coverage
            if(dut.rx_count==16)cov_rx_full=cov_rx_full+1;
            if(dut.tx_count==16)cov_tx_full=cov_tx_full+1;
            if(dut.rx_push && dut.rx_pop)cov_rx_pushpop=cov_rx_pushpop+1;
            if(dut.tx_push && dut.tx_pop)cov_tx_pushpop=cov_tx_pushpop+1;
            if(dut.rx_count==16 && dut.rx_push && dut.rx_pop)cov_rx_full_pushpop=cov_rx_full_pushpop+1;
            if(dut.tx_count==16 && dut.tx_push && dut.tx_pop)cov_tx_full_pushpop=cov_tx_full_pushpop+1;
            if(dut.rx_count==0 && rx_ready)cov_rx_empty_pop_attempt=cov_rx_empty_pop_attempt+1;
            if(tx_valid && !tx_ready)cov_tx_stall=cov_tx_stall+1;
            if(rx_valid && !rx_ready)cov_rx_hold=cov_rx_hold+1;
        end
    end

    task automatic send_frame(input [7:0] b,input stop_bit);
        integer k;
        begin
            @(negedge clk);rxd=0;repeat(CPB)@(negedge clk);
            for(k=0;k<8;k=k+1)begin rxd=b[k];repeat(CPB)@(negedge clk);end
            rxd=stop_bit;repeat(CPB)@(negedge clk);
            rxd=1;
        end
    endtask
    task automatic send_byte(input [7:0] b);
        begin rxq[rx_tail%1024]=b;rx_tail=rx_tail+1;send_frame(b,1'b1);end
    endtask
    task automatic send_byte_dropped(input [7:0] b); // 기대 큐에 넣지 않음(overrun 대상)
        begin send_frame(b,1'b1);end
    endtask
    task automatic do_reset(input integer cycles);
        begin
            @(negedge clk);resetn=0;mon_resetn=0;rx_ready=0;tx_valid=0;
            repeat(2)@(negedge clk);
            if(tx_ready!==1'b0)$fatal(1,"tx_ready must be low in reset");
            repeat(cycles)@(negedge clk);
            if(txd!==1'b1)$fatal(1,"TXD not idle in reset");
            if(dut.rx_count!==0 || dut.tx_count!==0)$fatal(1,"FIFO count not cleared");
            if(dut.rx_wptr!==0 || dut.rx_rptr!==0 || dut.tx_wptr!==0 || dut.tx_rptr!==0)
                $fatal(1,"FIFO pointers not cleared");
            if(rx_valid!==1'b0 || fe!==1'b0 || ov!==1'b0)$fatal(1,"status not cleared");
            if(dut.rx.out_valid_o!==1'b0)$fatal(1,"PHY RX hold not cleared");
            // 스코어보드도 초기화(리셋으로 버린 데이터는 기대하지 않는다)
            tx_head=tx_tail;rx_head=rx_tail;
            resetn=1;mon_resetn=1;
            @(negedge clk);
            if(tx_ready!==1'b1)$fatal(1,"tx_ready must rise after reset");
        end
    endtask
    // TX push: negedge에서 데이터 제시, ready를 기다려 다음 posedge에서 수락.
    task automatic tx_push(input [7:0] b);
        begin
            tx_data=b;tx_valid=1;
            while(!tx_ready)@(negedge clk);
            @(negedge clk);
        end
    endtask
    task automatic wait_tx_drain;
        integer guard;
        begin
            guard=0;
            while(tx_head!=tx_tail || dut.tx_count!=0 || dut.tx.sending)begin
                @(negedge clk);guard=guard+1;
                if(guard>CPB*10*80)$fatal(1,"TX drain timeout");
            end
            repeat(CPB*2)@(negedge clk);
        end
    endtask
    task automatic wait_rx_drain;
        integer guard;
        begin
            guard=0;
            while(rx_head!=rx_tail)begin
                @(negedge clk);guard=guard+1;
                if(guard>CPB*10*80)$fatal(1,"RX drain timeout head=%0d tail=%0d",rx_head,rx_tail);
            end
        end
    endtask

    reg rand_ready_en=0;

        $display("T3_NO_VALID_NO_TRANSFER_PASS");

        rx_ready=0;

        if(dut.rx.out_valid_o!==1'b1)$fatal(1,"T4 PHY hold byte missing");

        if(cov_rx_pushpop==0)$fatal(1,"T6 RX simultaneous push/pop not exercised");
        $display("T6_RX_STREAM_RANDOM_READY_PASS bytes=64 pushpop=%0d",cov_rx_pushpop);

        $display("T7_FULL_DUPLEX_PASS rx=20 tx=20");

        send_frame(8'h5A,1'b0);
        repeat(CPB*3)@(negedge clk);

        if(rx_valid || dut.rx_count!=0)$fatal(1,"T8 bad frame was pushed");
        send_byte(8'h3C);rx_ready=1;wait_rx_drain;rx_ready=0;

        @(negedge clk);
        for(i=0;i<12;i=i+1)tx_push(8'h10+i);
        tx_valid=0;
        repeat(CPB*10+CPB*4)@(negedge clk); // 2번째 프레임 중간
        if(!dut.tx.sending || dut.tx_count==0)$fatal(1,"T9 precondition");
        do_reset(5);
        @(negedge clk);
        for(i=0;i<3;i=i+1)tx_push(8'hD0+i);
        tx_valid=0;
        wait_tx_drain;
        $display("T9_RESET_DURING_TX_FLUSH_PASS");

        // ===== T10 RX 프레임 중간 Reset + 송신 중단: 반쪽 바이트 미적재 =====
        fork
            begin : partial
                @(negedge clk);rxd=0;repeat(CPB)@(negedge clk);
                for(t=0;t<4;t=t+1)begin rxd=0;repeat(CPB)@(negedge clk);end
                rxd=1; // 송신 중단(라인 idle)
            end
            begin repeat(CPB*3)@(negedge clk);@(negedge clk);resetn=0;mon_resetn=0;end
        join
        repeat(CPB*12)@(negedge clk);
        if(dut.rx_count!==0 || dut.rx.out_valid_o!==0)$fatal(1,"T10 partial frame stored");
        resetn=1;mon_resetn=1;
        repeat(CPB*12)@(negedge clk);
        if(rx_valid || fe || ov)$fatal(1,"T10 garbage after reset");
        send_byte(8'h99);rx_ready=1;wait_rx_drain;rx_ready=0;
        $display("T10_RESET_DURING_RX_PASS");

        // ===== T11 (관측) Reset 없는 송신 중단: UART 특성상 남은 비트가 1로 읽힌다 =====
        @(negedge clk);rxd=0;repeat(CPB)@(negedge clk);
        for(t=0;t<4;t=t+1)begin rxd=0;repeat(CPB)@(negedge clk);end
        rxd=1;
        rxq[rx_tail%1024]=8'hF0;rx_tail=rx_tail+1; // 하위 4비트 0, 상위 4비트 idle=1
        repeat(CPB*8)@(negedge clk);
        rx_ready=1;wait_rx_drain;rx_ready=0;
        if(fe)$fatal(1,"T11 unexpected framing");
        $display("T11_ABORTED_FRAME_OBSERVED byte=F0 (8N1 has no truncation detection)");

        if(cov_rx_hold==0 || cov_tx_stall==0 || cov_rx_empty_pop_attempt==0)
            $fatal(1,"coverage hole hold=%0d stall=%0d empty=%0d",
                cov_rx_hold,cov_tx_stall,cov_rx_empty_pop_attempt);
        $display("COVERAGE rx_full=%0d tx_full=%0d rx_full_pushpop=%0d tx_full_pushpop=%0d rx_pushpop=%0d tx_pushpop=%0d rx_hold=%0d tx_stall=%0d",
            cov_rx_full,cov_tx_full,cov_rx_full_pushpop,cov_tx_full_pushpop,
            cov_rx_pushpop,cov_tx_pushpop,cov_rx_hold,cov_tx_stall);
        $display("UART_FIFO_ADVERSARIAL_SIM_PASS tests=11");
        $finish;
    end
    initial begin #50000000;$fatal(1,"UART FIFO TB TIMEOUT");end
endmodule

    function automatic signed [40:0] golden(input integer s,input integer n);
        integer x0,x1,w0,w1;
        begin
            x0=(s&2)?-1:1;x1=(s&1)?-1:1;w0=(n&2)?-1:1;w1=(n&1)?-1:1;
            golden=x0*w0+x1*w1;
        end
    endfunction
    function automatic [7:0] gold_hex(input signed [40:0] val,input integer pos);
        reg [43:0] ext;
        begin
            ext={{3{val[40]}},val}; // 부호 비트만 3회 확장(44비트 = 11 hex)
            gold_hex=hexchar((ext>>((10-pos)*4))&4'hf);
        end

#!/usr/bin/env python3
"""2장 4뉴런 NPU 보드 기준 모델.

명령 순서(보낼 바이트, sw[3:2])를 정하고 보드가 돌려줄 응답을 계산해
p02/tb/p02_cases_s{STAGE}.txt (STAGE = 끝낸 서브젝트 수 1~8)로 쓴다. TB는 이 파일만 보고 판정한다.
계산은 서브젝트(구 장) 기준 모델 함수를 그대로 쓴다: 05 quantize, 07 dot, 09 dense.

  python3 p02/reference/p02_model.py
"""
import os
import struct
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
for ch in ("ch01", "ch05", "ch07", "ch09"):
    sys.path.insert(0, str(ROOT / ch / "reference"))
from mac_reference import wrap              # noqa: E402  (1장 구01장)
from fixed_reference import quantize        # noqa: E402  (2-2 구05장)
from dot_reference import dot               # noqa: E402  (2-3 구07장)
from dense_reference import dense           # noqa: E402  (2-5 구09장)

OUT = ROOT / "p02" / "tb" / "p02_cases_s%d.txt"
CRLF = "\r\n"
NEED = {"S": 0, "X": 3, "W": 4, "x": 1, "w": 2, "Q": 4, "D": 1, "P": 1, "B": 6, "L": 1, "N": 1}
MIN_STAGE = {"Q": 2, "D": 3, "P": 4, "B": 5, "L": 5, "N": 6}


def h(v, n):
    return format(v & ((1 << (4 * n)) - 1), "0%dX" % n)


class Board:
    def __init__(self, stage):
        self.stage = stage
        self.x = [None] * 64                     # 안 쓴 칸은 None (RTL에서는 X)
        self.w = [[None] * 64 for _ in range(4)]
        self.bias = [0] * 4
        self.view = 0
        self.x_seen = [False] * 64               # 12장 엔진의 적재 확인 비트
        self.w_seen = [[False] * 64 for _ in range(4)]
        self.cls = 0                             # 마지막 추론 클래스 (STAGE 8 표시)
        self.done = False
        self.err = False

    # ── 디버그 콘솔 (sw[2] = 0) ──
    def console(self, data):
        c = chr(data[0])
        if c in "\r\n":
            return ""
        if c not in NEED or self.stage < MIN_STAGE.get(c, 1):
            return "?" + CRLF
        p = data[1:]
        assert len(p) == NEED[c], (c, data)
        if c == "S":
            return "I" + CRLF
        bad = ((c in "Xx" and p[0] >= 64) or (c in "Ww" and (p[0] >= 4 or p[1] >= 64))
               or (c == "B" and p[0] >= 4) or (c in "DPLN" and not 1 <= p[0] <= 64))
        if bad:
            return "ERR" + CRLF
        s16 = lambda hi, lo: struct.unpack(">h", bytes([hi, lo]))[0]
        if c == "X":
            self.x[p[0]] = s16(p[1], p[2])
            return "OK" + CRLF
        if c == "W":
            self.w[p[0]][p[1]] = s16(p[2], p[3])
            return "OK" + CRLF
        if c == "B":
            self.bias[p[0]] = wrap(int.from_bytes(bytes(p[1:6]), "big"), 40)
            return "OK" + CRLF
        if c in "xw":
            v = self.x[p[0]] if c == "x" else self.w[p[0]][p[1]]
            assert v is not None, "안 쓴 칸을 읽음"
            self.view = v & 0xFFFF
            return c.upper() + ":" + h(v, 4) + CRLF
        if c == "Q":
            v = struct.unpack(">i", bytes(p))[0]
            q, ov = quantize(v, 8, 8, 1, 1)
            self.view = q & 0xFFFF
            return "Q:%s O:%d" % (h(q, 2), ov) + CRLF
        n = p[0]
        xs = self.x[:n]
        ws = [row[:n] for row in self.w]
        assert None not in xs, "안 쓴 X"
        if c == "D":
            assert None not in ws[0]
            r = dot(xs, ws[0], 40)
            self.view = r & 0xFFFF
            return "D:" + h(r, 10) + CRLF
        if c == "P":
            assert all(None not in r for r in ws)
            rs = [dot(xs, r, 40) for r in ws]
            self.view = rs[0] & 0xFFFF
            return "P:" + ",".join(h(r, 10) for r in rs) + CRLF
        assert all(None not in r for r in ws)
        vals, mask = dense(xs, ws, self.bias, 16, 40, 0)      # L·N: Dense(+Bias, ReLU, 16비트)
        self.view = vals[0] & 0xFFFF
        return c + ":" + ",".join(h(v, 4) for v in vals) + " O:%X" % mask + CRLF

    # ── 11장 명령 해석기 (sw[2] = 1) ──
    def write(self, kind, n, a, v):
        if kind == "X":
            self.x[a] = v
            self.x_seen[a] = True
        elif kind == "W":
            self.w[n][a] = v
            self.w_seen[n][a] = True
        else:
            self.bias[n] = v

    def infer(self, length, engine4):
        xs, ws = self.x[:length], [r[:length] for r in self.w]
        if engine4:                              # 12장 엔진: 원시 점수 = 내적 + Bias, 적재 확인
            if not (all(self.x_seen[:length]) and all(all(r[:length]) for r in self.w_seen)):
                self.err = True
                return "ERR" + CRLF
            scores = [wrap(dot(xs, r, 40) + b, 41) for r, b in zip(ws, self.bias)]
        else:                                    # 누적 npu_core: Dense 결과(ReLU, 16비트)를 41비트로
            scores, _ = dense(xs, ws, self.bias, 16, 40, 0)
        cls = 0
        for i in range(1, 4):
            if scores[i] > scores[cls]:
                cls = i
        self.cls, self.done = cls, True
        return "C%d:" % cls + ",".join(h(s, 11) for s in scores) + CRLF

    def engine(self, data, engine4):
        c = chr(data[0])
        if c in "\r\n":
            return ""
        if c in "0123":                          # 고정 그룹: X0·X1, W[n][0·1], Bias 0, 길이 2
            k = int(c)
            self.write("X", 0, 0, -1 if k & 2 else 1)
            self.write("X", 0, 1, -1 if k & 1 else 1)
            for n in range(4):
                self.write("W", n, 0, -1 if n & 2 else 1)
                self.write("W", n, 1, -1 if n & 1 else 1)
            for n in range(4):
                self.write("B", n, 0, 0)
            return self.infer(2, engine4)
        if c == "S":
            return "I" + CRLF
        p = data[1:]
        ok = ((c == "X" and p[0] < 64) or (c == "W" and p[0] < 4 and p[1] < 64)
              or (c == "B" and p[0] < 4) or (c == "R" and 1 <= p[0] <= 64))
        if c not in "XWBR" or not ok:
            self.err = True
            return "ERR" + CRLF
        if c == "R":
            return self.infer(p[0], engine4)
        s16 = struct.unpack(">h", bytes(p[-2:]))[0] if c in "XW" else 0
        if c == "X":
            self.write("X", 0, p[0], s16)
        elif c == "W":
            self.write("W", p[0], p[1], s16)
        else:
            self.write("B", p[0], 0, wrap(int.from_bytes(bytes(p[1:6]), "big"), 40))
        return "OK" + CRLF

    def step(self, sw, data):
        eng = self.stage >= 7 and bool(sw & 1)          # sw 값 = sw[3:2]
        e4 = self.stage >= 8 and bool(sw & 2)
        return (self.engine(data, e4) if eng else self.console(data)).encode()

    def led(self):
        if self.stage < 8:
            return self.view
        return (1 << self.cls) | (int(self.done) << 14) | (int(self.err) << 15)


def cmd(c, *b):
    return [ord(c)] + list(b)


def X(a, v):
    return cmd("X", a, *struct.pack(">h", v))


def W(n, a, v):
    return cmd("W", n, a, *struct.pack(">h", v))


def B(n, v):
    return cmd("B", n, *(v & ((1 << 40) - 1)).to_bytes(5, "big"))


def Q(v):
    return cmd("Q", *struct.pack(">i", v))
