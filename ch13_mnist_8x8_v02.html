ch13/rtl/npu10_input_loader.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 13장 입력 적재기: 8×8 특징 64원소를 순서대로 받아 04장 BRAM 버퍼(8비트×64)에 쓰고,
// 64개가 모두 들어온 뒤에만 14장이 주소로 읽을 수 있게 한다.
// 입력 계약: 원소 i = 8×8 평균 풀링 화소 (행 우선, i = 행*8 + 열),
//            값 = round(평균 화소 / 2), 범위 0~127 (INT8, 부호 비트 0).
module npu10_input_loader (
    input  wire              clk,
    input  wire              resetn,
    input  wire              load_start_i,
    input  wire              in_valid_i,
    input  wire        [7:0] in_data_i,
    input  wire              in_last_i,
    input  wire              rd_en_i,
    input  wire        [5:0] rd_addr_i,
    output wire signed [7:0] rd_data_o,
    output wire              rd_valid_o,
    output wire              loading_o,
    output wire              loaded_o,
    output reg               len_err_o,
    output reg               range_err_o,
    output reg         [6:0] count_o
);
    localparam IDLE=0, LOAD=1, DONE=2, ERR=3;
    reg [1:0] state;

    // LOAD 상태에서 받은 원소의 판정: 부호 비트, 마지막 표시 위치
    wire last_slot  = (count_o == 7'd63);
    wire bad_range  = in_data_i[7];
    wire bad_length = in_last_i != last_slot;
    wire write_fire = (state == LOAD) && in_valid_i && !load_start_i && !bad_range && !bad_length;

    assign loading_o = (state == LOAD);
    assign loaded_o  = (state == DONE);

    bram_sync_buffer #(.DATA_WIDTH(8), .ADDR_WIDTH(6)) input_ram (
        .clk(clk), .resetn(resetn),
        .wr_en(write_fire), .wr_addr(count_o[5:0]), .wr_data(in_data_i),
        .rd_en(rd_en_i && state == DONE), .rd_addr(rd_addr_i),
        .rd_data(rd_data_o), .rd_valid(rd_valid_o));

    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE; count_o <= 0; len_err_o <= 0; range_err_o <= 0;
        end else if (load_start_i) begin
            // 새 샘플: 어느 상태에서든 개수·오류를 지우고 다시 받는다.
            state <= LOAD; count_o <= 0; len_err_o <= 0; range_err_o <= 0;
        end else begin
            case (state)
                IDLE: ;
                LOAD: if (in_valid_i) begin
                    if (bad_range) begin
                        range_err_o <= 1; state <= ERR;
                    end else if (bad_length) begin
                        len_err_o <= 1; state <= ERR;
                    end else begin
                        count_o <= count_o + 7'd1;
                        if (in_last_i) state <= DONE;
                    end
                end
                DONE: if (in_valid_i) begin
                    len_err_o <= 1; state <= ERR;
                end
                ERR: ;
                default: state <= IDLE;
            endcase
        end
    end
endmodule

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. 기준 모델(Python)이 숫자 0~9 합성 영상(5×7 글꼴, 실제 MNIST 아님)에서 64원소 벡터를 만들고, TB는 적재·읽기 결과를 파일로 돌려줍니다.

ch13/tb/tb_npu10_input_loader.sv · systemverilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
// 13장 TB: 64원소 10샘플 적재·임의 순서 읽기, 길이·범위 오류 4종과 복구를 자동 판정한다.
module tb_npu10_input_loader;
    reg clk=0, rn=0, start=0, iv=0, il=0, re=0;
    reg [7:0] id=0; reg [5:0] ra=0;
    wire signed [7:0] rd; wire rv, loading, loaded, len_err, range_err; wire [6:0] count;
    npu10_input_loader dut(.clk(clk), .resetn(rn), .load_start_i(start), .in_valid_i(iv),
        .in_data_i(id), .in_last_i(il), .rd_en_i(re), .rd_addr_i(ra), .rd_data_o(rd),
        .rd_valid_o(rv), .loading_o(loading), .loaded_o(loaded), .len_err_o(len_err),
        .range_err_o(range_err), .count_o(count));
    always #5 clk = ~clk;

    reg [7:0] x [0:63], got [0:63];
    reg [15:0] lfsr = 16'hACE1;
    integer f, o, label, k, n = 0, reads = 0, errs = 0;
    string vp, ap;

    task pulse_start; begin
        @(negedge clk); start = 1; @(negedge clk); start = 0;
        if (!loading || count != 0 || len_err || range_err) $fatal(1, "load_start 후 LOAD·count=0이 아님");
    end endtask

    // 원소 n개를 보낸다. last_at = in_last를 올릴 순번(-1이면 올리지 않음), 사이사이 빈 클록을 넣는다.
    task send(input integer cnt, input integer last_at); integer s; begin
        for (s = 0; s < cnt; s = s + 1) begin
            lfsr = {lfsr[14:0], lfsr[15] ^ lfsr[13] ^ lfsr[12] ^ lfsr[10]};
            if (lfsr[0]) begin @(negedge clk); iv = 0; end
            @(negedge clk); iv = 1; id = x[s]; il = (s == last_at);
            if (s < 64) begin re = 1; ra = s[5:0]; end          // 적재 중 읽기 요청 → 무시돼야 함
        end
        @(negedge clk); iv = 0; il = 0; re = 0;
        if (rv) $fatal(1, "적재 완료 전 읽기가 rd_valid를 냄");
    end endtask

    task read_all; integer a, s; begin
        for (s = 0; s < 64; s = s + 1) begin
            a = (s * 29 + 7) % 64;                                // 주소를 섞어 순차 읽기에 기대지 않는다
            @(negedge clk); re = 1; ra = a;
            @(negedge clk); re = 0;
            if (!rv) $fatal(1, "rd_valid 없음 addr=%0d", a);
            got[a] = rd; reads = reads + 1;
        end
    end endtask

    task expect_error(input string what, input reg want_len, input reg want_range); begin
        @(negedge clk);
        if (len_err !== want_len || range_err !== want_range || loaded)
            $fatal(1, "%s: len_err=%b range_err=%b loaded=%b", what, len_err, range_err, loaded);
        @(negedge clk); re = 1; ra = 0; @(negedge clk); re = 0;
        if (rv) $fatal(1, "%s: 오류 상태에서 읽기가 허용됨", what);
        errs = errs + 1;
    end endtask

    initial begin
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap)) $fatal(1, "경로 인자");
        f = $fopen(vp, "r"); o = $fopen(ap, "w"); if (!f || !o) $fatal(1, "파일 열기");
        repeat (3) @(negedge clk); rn = 1;
        while ($fscanf(f, "%d", label) == 1) begin
            for (k = 0; k < 64; k = k + 1) if ($fscanf(f, "%h", x[k]) != 1) $fatal(1, "벡터 형식");
            pulse_start; send(64, 63);
            if (!loaded || count != 64 || len_err || range_err) $fatal(1, "label=%0d 적재 실패", label);
            read_all;
            $fwrite(o, "%0d", label);
            for (k = 0; k < 64; k = k + 1) $fwrite(o, " %02h", got[k]);
            $fwrite(o, "\n"); n = n + 1;
        end
        // 오류 4종: 마지막 표시가 이름, 마지막 표시 없음, 완료 뒤 추가 원소, 부호 비트 1
        pulse_start; send(10, 9);  expect_error("EARLY_LAST", 1, 0);
        pulse_start; send(64, -1); expect_error("MISSING_LAST", 1, 0);
        pulse_start; send(64, 63); if (!loaded) $fatal(1, "재적재 실패");
        @(negedge clk); iv = 1; id = 8'h01; @(negedge clk); iv = 0;
        expect_error("EXTRA_AFTER_DONE", 1, 0);
        pulse_start; x[5] = 8'h80; send(6, -1); expect_error("NEGATIVE_VALUE", 0, 1);
        // 적재 도중 리셋 → IDLE, 이후 정상 재적재
        x[5] = 8'h05; pulse_start; send(20, -1);
        @(negedge clk); rn = 0; @(negedge clk); rn = 1; @(negedge clk);
        if (loading || loaded || count != 0) $fatal(1, "리셋 후 IDLE 아님");
        pulse_start; send(64, 63); if (!loaded) $fatal(1, "리셋 후 재적재 실패");
        $display("RTL_SIM_PASS INPUT_LOADER samples=%0d reads=%0d error_cases=%0d", n, reads, errs);
        $fclose(o); $finish;
    end
    initial begin #2000000; $fatal(1, "시간 초과"); end
endmodule

#!/usr/bin/env python3
"""13장 기준 모델: 28×28 화소 → 8×8 평균 풀링 → INT8 64원소 (표준 라이브러리만 사용).

시험 영상은 5×7 글꼴로 그린 숫자 0~9 합성 영상이다(실제 MNIST 아님).
실제 MNIST 확인은 --mnist N 옵션 (host_tools/mnist_idx.py가 내려받음, 선택).
"""
from __future__ import annotations
import argparse
import sys
from pathlib import Path

SIZE, OUT = 28, 8
FONT = {  # 5×7 글꼴, 1 = 획
    0: ["01110", "10001", "10011", "10101", "11001", "10001", "01110"],
    1: ["00100", "01100", "00100", "00100", "00100", "00100", "01110"],
    2: ["01110", "10001", "00001", "00010", "00100", "01000", "11111"],
    3: ["11110", "00001", "00001", "01110", "00001", "00001", "11110"],
    4: ["00010", "00110", "01010", "10010", "11111", "00010", "00010"],
    5: ["11111", "10000", "11110", "00001", "00001", "10001", "01110"],
    6: ["00110", "01000", "10000", "11110", "10001", "10001", "01110"],
    7: ["11111", "00001", "00010", "00100", "01000", "01000", "01000"],
    8: ["01110", "10001", "10001", "01110", "10001", "10001", "01110"],
    9: ["01110", "10001", "10001", "01111", "00001", "00010", "01100"],
}


def draw_digit(d: int) -> list[list[int]]:
    """글꼴 1칸을 가로 4 × 세로 3 화소로 키워 28×28 가운데(왼쪽 4, 위 3)에 그린다. 획 255, 바탕 0."""
    img = [[0] * SIZE for _ in range(SIZE)]
    for r, row in enumerate(FONT[d]):
        for c, bit in enumerate(row):
            if bit == "1":
                for y in range(3 + 3 * r, 6 + 3 * r):
                    for x in range(4 + 4 * c, 8 + 4 * c):
                        img[y][x] = 255
    return img


def window(i: int) -> range:
    """PyTorch adaptive_avg_pool2d와 같은 창: [floor(i·28/8), ceil((i+1)·28/8))."""
    return range((i * SIZE) // OUT, -(-((i + 1) * SIZE) // OUT))


def pool_8x8(img: list[list[int]]) -> list[list[int]]:
    """창 평균을 정수 반올림(0.5 올림)한 0~255 값."""
    out = []
    for i in range(OUT):
        row = []
        for j in range(OUT):
            vals = [img[y][x] for y in window(i) for x in window(j)]
            row.append((sum(vals) + len(vals) // 2) // len(vals))
        out.append(row)
    return out


def to_int8(pooled: list[list[int]]) -> list[int]:
    """행 우선 64원소, 값 = round(평균/2) 상한 127 → 부호 비트가 항상 0인 INT8."""
    return [min(127, (p + 1) >> 1) for row in pooled for p in row]


def features(img: list[list[int]]) -> list[int]:
    return to_int8(pool_8x8(img))


def show(x: list[int]) -> str:
    shade = " .:-=+*#%@"
    return "\n".join("".join(shade[min(9, v * 10 // 128)] * 2 for v in x[r * 8:r * 8 + 8]) for r in range(8))


def selftest() -> list[tuple[int, list[int]]]:
    wins = [list(window(i)) for i in range(OUT)]
    assert all(len(w) == 4 for w in wins), wins                  # 28/8 = 3.5 → 창 4화소, 일부 겹침
    assert sorted({p for w in wins for p in w}) == list(range(SIZE))
    assert features([[0] * SIZE for _ in range(SIZE)]) == [0] * 64
    assert features([[255] * SIZE for _ in range(SIZE)]) == [127] * 64
    samples = [(d, features(draw_digit(d))) for d in range(10)]
    for d, x in samples:
        assert len(x) == 64 and all(0 <= v <= 127 for v in x) and any(x), d
    assert len({tuple(x) for _, x in samples}) == 10              # 숫자마다 다른 벡터
    print(f"PYTHON_MNIST8X8_SELF_TEST_PASS samples=10 elements=64 range=0..{max(max(x) for _, x in samples)}")
    return samples


def mnist_check(count: int) -> None:
    """선택: 실제 MNIST 시험 집합 앞 count장으로 라벨·범위와 풀링 반올림 오차(실수 창 평균 대비 ≤ 0.5)를 확인한다."""
    sys.path.insert(0, str(Path(__file__).resolve().parents[2] / "host_tools"))
    from mnist_idx import images, labels
    imgs, labs = images("test"), labels("test")
    worst, seen = 0.0, set()
    for k in range(count):
        img = [list(imgs[k][r * SIZE:(r + 1) * SIZE]) for r in range(SIZE)]
        ours = pool_8x8(img)
        for i in range(OUT):
            for j in range(OUT):
                vals = [img[y][x] for y in window(i) for x in window(j)]
                worst = max(worst, abs(ours[i][j] - sum(vals) / len(vals)))
        x = features(img)
        assert len(x) == 64 and all(0 <= v <= 127 for v in x) and 0 <= labs[k] <= 9
        seen.add(labs[k])
    assert worst <= 0.5, worst
    print(f"MNIST_POOL_CHECK_PASS samples={count} labels={len(seen)} max_round_error={worst:.3f}")


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", help="vectors.txt를 쓸 폴더")
    ap.add_argument("--actual", help="RTL이 읽어 낸 actual.txt (Python 값과 비교)")
    ap.add_argument("--show", type=int, help="숫자 하나의 8×8 특징을 글자 그림으로 출력")
    ap.add_argument("--mnist", type=int, metavar="N", help="선택: 실제 MNIST 시험 집합 앞 N장 확인 (첫 실행 때 내려받음)")
    a = ap.parse_args()
    if a.mnist:
        mnist_check(a.mnist)
        return
    samples = selftest()
    if a.show is not None:
        print(show(samples[a.show][1]))
    if a.out and not a.actual:
        with open(Path(a.out) / "vectors.txt", "w") as f:
            for d, x in samples:
                f.write(f"{d} " + " ".join(f"{v:02x}" for v in x) + "\n")
    if a.actual:
        got = {}
        for ln in open(a.actual):
            p = ln.split()
            got[int(p[0])] = [int(v, 16) for v in p[1:]]
        for d, x in samples:
            if got.get(d) != x:
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL label={d}")
        print(f"PYTHON_RTL_COMPARE_PASS samples={len(samples)} values={64 * len(samples)}")


if __name__ == "__main__":
    main()

07코드 읽기 포인트

위치내용
풀링 창 4화소28÷8 = 3.5이므로 PyTorch adaptive_avg_pool2d와 같이 창 i를 [floor(i·28/8), ceil((i+1)·28/8))로 잡습니다. 창은 모두 4화소이고 3·10·17·24행(열)이 이웃 창과 겹칩니다.
INT8 스케일평균 0~255를 round(평균/2), 상한 127로 바꿉니다. 부호 비트가 늘 0이므로 14장의 부호 있는 가중치와 곱해도 입력이 음수로 해석되지 않습니다. 부호 비트가 1인 값은 range_err_o로 거부합니다.
길이 계약in_last_i는 64번째 원소에만 올립니다. 이르거나 없거나 완료 뒤 원소가 더 오면 len_err_o와 함께 ERR로 가고, 다음 load_start_i까지 읽기를 막습니다.
읽기 잠금BRAM 읽기는 rd_en_i && state == DONE일 때만 요청합니다. 적재 중이거나 오류 상태의 읽기는 rd_valid_o가 오르지 않습니다.
