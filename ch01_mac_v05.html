// 독립 signed MAC: 보드 및 AXI에 의존하지 않는다.
(* use_dsp = "yes" *)
module mac_core #(
    parameter integer DATA_WIDTH = 16,
    parameter integer ACC_WIDTH = 40
) (
    input wire clk,
    input wire resetn,
    input wire clear_i,
    input wire valid_i,
    input wire signed [DATA_WIDTH-1:0] a_i,
    input wire signed [DATA_WIDTH-1:0] b_i,
    output reg signed [ACC_WIDTH-1:0] acc_o,
    output reg valid_o
);
    localparam integer PRODUCT_WIDTH = 2 * DATA_WIDTH;
    wire signed [PRODUCT_WIDTH-1:0] product_w;
    wire signed [ACC_WIDTH-1:0] product_ext_w;

    // 전체 곱을 보존한 뒤 부호 비트를 복제하여 누산 폭으로 확장한다.
    // 지원 조건: DATA_WIDTH >= 2, ACC_WIDTH >= 2*DATA_WIDTH.
    assign product_w = $signed(a_i) * $signed(b_i);
    assign product_ext_w = {{(ACC_WIDTH-PRODUCT_WIDTH){product_w[PRODUCT_WIDTH-1]}}, product_w};

    // 동기식 Active-Low Reset > Clear > Valid 순서.
    // 입력을 수락한 상승 에지 직후 acc_o와 valid_o가 함께 갱신된다.
    // valid_i=0이면 acc_o 유지, valid_o=0. Clear와 겹친 입력은 버린다.
    always @(posedge clk) begin
        if (!resetn) begin
            acc_o <= {ACC_WIDTH{1'b0}};
            valid_o <= 1'b0;
        end else if (clear_i) begin
            acc_o <= {ACC_WIDTH{1'b0}};
            valid_o <= 1'b0;
        end else begin
            valid_o <= valid_i;
            if (valid_i)
                // 범위 초과 시 상위 비트를 버리는 2의 보수 순환 연산이다.
                acc_o <= acc_o + product_ext_w;
        end
    end

    // synthesis translate_off
    initial begin
        if (DATA_WIDTH < 2 || ACC_WIDTH < PRODUCT_WIDTH)
            $fatal(1, "Require DATA_WIDTH>=2 and ACC_WIDTH>=2*DATA_WIDTH");
    end
    // synthesis translate_on
endmodule
`default_nettype wire

module tb_mac_core;
    parameter integer DATA_WIDTH = 16;
    parameter integer ACC_WIDTH = 40;
    reg clk = 0;
    reg resetn = 0, clear_i = 0, valid_i = 0;
    reg signed [DATA_WIDTH-1:0] a_i = 0, b_i = 0;
    wire signed [ACC_WIDTH-1:0] acc_o;
    wire valid_o;
    mac_core #(.DATA_WIDTH(DATA_WIDTH), .ACC_WIDTH(ACC_WIDTH)) dut (.*);
    always #5 clk = ~clk;

    integer vectors, actual, fields, cycle = 0;
    integer rn, cl, vi, ai, bi, expected_valid;
    reg signed [ACC_WIDTH-1:0] expected_acc;
    reg signed [ACC_WIDTH-1:0] previous_acc;
    reg previous_valid;
    longint signed model_acc = 0;
    longint signed model_product;
    reg model_valid;
    string vectors_path, actual_path;

    initial begin
        if (!$value$plusargs("VECTORS=%s", vectors_path) ||
            !$value$plusargs("ACTUAL=%s", actual_path))
            $fatal(1, "Missing VECTORS/ACTUAL path");
        vectors = $fopen(vectors_path, "r");
        actual = $fopen(actual_path, "w");
        if (!vectors || !actual) $fatal(1, "Cannot open vector/trace file");
        // 첫 상승 에지에서 동기식 Reset을 적용하여 알려진 상태를 만든다.
        @(posedge clk); #1;
        if (acc_o !== 0 || valid_o !== 0) $fatal(1, "Initial reset failed");
        while (!$feof(vectors)) begin
            fields = $fscanf(vectors, "%d %d %d %d %d %d %d\n",
                              rn, cl, vi, ai, bi, expected_acc, expected_valid);
            if (fields != 7) $fatal(1, "Malformed vector row");
            @(negedge clk);
            previous_acc = acc_o;
            previous_valid = valid_o;
            resetn = rn; clear_i = cl; valid_i = vi; a_i = ai; b_i = bi;
            #1;
            // 입력 변경만으로 출력이 변하면 동기식 타이밍 위반이다.
            if (acc_o !== previous_acc || valid_o !== previous_valid)
                $fatal(1, "Output changed before rising edge cycle=%0d", cycle+1);
            @(posedge clk); #1; // NBA 갱신 이후 같은 에지의 결과를 검사한다.
            cycle = cycle + 1;
            // Python 파일 외에도 SV의 64비트 정수 산술로 독립 계산한다.
            // 이 TB의 실행 설정은 ACC_WIDTH<=40이므로 곱과 합이 64비트에 들어간다.
            model_product = $signed(a_i);
            model_product = model_product * $signed(b_i);
            model_valid = rn && !cl && vi;
            if (!rn || cl) model_acc = 0;
            else if (vi) model_acc = model_acc + model_product;
            model_acc = $signed(model_acc[ACC_WIDTH-1:0]);
            if (expected_acc !== model_acc[ACC_WIDTH-1:0] || expected_valid[0] !== model_valid)
                $fatal(1, "SV/Python reference mismatch cycle=%0d", cycle);
            if (acc_o !== expected_acc || valid_o !== expected_valid[0])
                $fatal(1, "cycle=%0d a=%0d b=%0d expected=%0d/%0d actual=%0d/%0d",
                       cycle, a_i, b_i, expected_acc, expected_valid, acc_o, valid_o);
            $fdisplay(actual, "%0d %0d %0d", cycle, acc_o, valid_o);
        end
        if (cycle == 0) $fatal(1, "Empty vector file");
        $fclose(vectors); $fclose(actual);
        $display("RTL_SIM_PASS INT%0d/%0d cycles=%0d", DATA_WIDTH, ACC_WIDTH, cycle);
        $finish;
    end
    initial begin
        #10000000;
        $fatal(1, "Simulation timeout");
    end
endmodule

ch01/reference/mac_reference.py · python · 원본 · RTL_SIM 통과 2026-11-18
#!/usr/bin/env python3
"""P3-01 독립 정수 모델, 재현 가능한 자극 생성 및 RTL 전 사이클 대조."""
import argparse
import random
from pathlib import Path


def wrap(value, width):
    """Python 무제한 정수를 signed 2의 보수 범위로 변환한다."""
    modulus = 1 << width
    return (value + modulus // 2) % modulus - modulus // 2


def step(acc, resetn, clear, valid, a, b, acc_width):
    if not resetn or clear:
        return 0, 0
    return (wrap(acc + a * b, acc_width), 1) if valid else (acc, 0)


def self_test():
    # 별도의 상수 정답으로 산술/제어 및 각 설정의 양·음 오버플로를 검증한다.
    for dw, aw in ((8, 32), (16, 40)):
        maximum = (1 << (aw - 1)) - 1
        minimum = -(1 << (aw - 1))
        assert step(0, 1, 0, 1, 3, 4, aw) == (12, 1)
        assert step(0, 1, 0, 1, 3, -4, aw) == (-12, 1)
        assert step(0, 1, 0, 1, -3, -4, aw) == (12, 1)
        assert step(maximum, 1, 0, 1, 1, 1, aw) == (minimum, 1)
        assert step(minimum, 1, 0, 1, 1, -1, aw) == (maximum, 1)
        assert step(7, 1, 0, 0, 3, 4, aw) == (7, 0)
        assert step(7, 1, 1, 1, 3, 4, aw) == (0, 0)
        assert step(7, 0, 1, 1, 3, 4, aw) == (0, 0)
        assert wrap((-(1 << (dw - 1))) ** 2, aw) == 1 << (2 * dw - 2)
    print("PYTHON_SELF_TEST_PASS configurations=INT8/32,INT16/40")


def stimulus(dw, aw):
    lo, hi = -(1 << (dw - 1)), (1 << (dw - 1)) - 1
    yield (0, 0, 1, hi, lo)
    yield (0, 1, 1, lo, lo)
    yield (1, 0, 0, 0, 0)
    # 부호 조합, 0, 입력 경계, 연속 valid 및 idle.
    for a, b in ((3, 4), (3, -4), (-3, -4), (hi, hi),
                 (lo, lo), (lo, hi), (hi, lo), (0, lo), (-1, -1)):
        yield (1, 0, 1, a, b)
    for _ in range(4):
        yield (1, 0, 0, hi, hi)
    yield (1, 1, 1, hi, hi)
    yield (1, 1, 0, lo, lo)
    # 내부 레지스터 force 없이 실제 포트로 최댓값에 도달한다.
    target = (1 << (aw - 1)) - 1
    count, remainder = divmod(target, lo * lo)
    for _ in range(count):
        yield (1, 0, 1, lo, lo)
    count, remainder = divmod(remainder, hi)
    for _ in range(count):
        yield (1, 0, 1, hi, 1)
    if remainder:
        yield (1, 0, 1, remainder, 1)
    yield (1, 0, 0, hi, hi)  # 최댓값 유지
    yield (1, 0, 1, 1, 1)    # max + 1 -> min
    yield (1, 0, 0, hi, hi)  # 최솟값 유지
    yield (1, 0, 1, -1, 1)   # min - 1 -> max
    yield (0, 1, 1, lo, hi)  # Reset/Clear/Valid 동시 입력
    rng = random.Random(301 + dw)
    for _ in range(2000):
        yield (int(rng.randrange(53) != 0), int(rng.randrange(29) == 0),
               int(rng.randrange(4) != 0), rng.randint(lo, hi), rng.randint(lo, hi))
    yield (1, 0, 0, 0, 0)


def generate(dw, aw, vectors, expected):
    acc = 0
    count = 0
    with vectors.open("w") as vf, expected.open("w") as ef:
        for count, row in enumerate(stimulus(dw, aw), 1):
            acc, valid = step(acc, *row, aw)
            vf.write(" ".join(map(str, (*row, acc, valid))) + "\n")
            ef.write(f"{count} {acc} {valid}\n")
    print(f"PYTHON_VECTORS INT{dw}/{aw} cycles={count} seed={301 + dw}")


def compare(expected, actual):
    # 길이도 비교하여 조기 종료/누락 행이 PASS가 되지 않게 한다.
    from itertools import zip_longest
    count = 0
    with expected.open() as ef, actual.open() as af:
        for count, (e, a) in enumerate(zip_longest(ef, af), 1):
            if e is None or a is None or e.split() != a.split():
                raise AssertionError(f"cycle={count}: expected={e!r}, actual={a!r}")
    if count == 0:
        raise AssertionError("Empty comparison")
    print(f"PYTHON_RTL_COMPARE_PASS cycles={count}")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--data-width", type=int, default=16)
    parser.add_argument("--acc-width", type=int, default=40)
    parser.add_argument("--vectors", type=Path)
    parser.add_argument("--expected", type=Path)
    parser.add_argument("--actual", type=Path)
    args = parser.parse_args()
    if args.data_width < 2 or args.acc_width < 2 * args.data_width:
        parser.error("Require DATA_WIDTH>=2 and ACC_WIDTH>=2*DATA_WIDTH")
    self_test()
    if args.actual:
        compare(args.expected, args.actual)
    elif args.vectors:
        generate(args.data_width, args.acc_width, args.vectors, args.expected)

07코드 읽기 포인트

위치내용
부호 확장product_ext_w = {{(ACC_WIDTH-PRODUCT_WIDTH){product_w[PRODUCT_WIDTH-1]}}, product_w}은 32비트 곱의 부호 비트를 40비트까지 복제합니다. 음수 곱도 올바르게 누산됩니다.
Reset > Clear > Valid 우선순위!resetn이 최우선, 그 다음 clear_i, 마지막으로 valid_i를 처리합니다. 이 계약은 02·03·07장도 그대로 따릅니다.
2의 보수 순환ACC_WIDTH가 초과되면 상위 비트를 버리는 wrap-around가 발생합니다. 테스트벤치 SV 모델에서 $signed(model_acc[ACC_WIDTH-1:0])으로 명시적으로 동일하게 처리합니다.
DSP 추론(* use_dsp = "yes" *)는 합성 힌트입니다. 시뮬레이션에서는 무시되며 Vivado 합성 후 Resource 탭에서 DSP48E1 사용 여부를 확인합니다.
