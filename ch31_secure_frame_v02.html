static uint8_t xt(uint8_t x) { return (uint8_t)((x << 1) ^ ((x >> 7) * 0x1b)); }

void aes128_encrypt_block(const uint8_t key[16], const uint8_t in[16], uint8_t out[16])
{
    uint8_t rk[176], s[16], t[16], rcon = 1;
    for (int i = 0; i < 16; i++) rk[i] = key[i];
    for (int i = 16; i < 176; i += 4) {                         /* 키 확장 */
        uint8_t w[4] = {rk[i - 4], rk[i - 3], rk[i - 2], rk[i - 1]};
        if (i % 16 == 0) {
            uint8_t x = w[0];
            w[0] = SBOX[w[1]] ^ rcon; w[1] = SBOX[w[2]]; w[2] = SBOX[w[3]]; w[3] = SBOX[x];
            rcon = xt(rcon);
        }
        for (int k = 0; k < 4; k++) rk[i + k] = rk[i - 16 + k] ^ w[k];
    }
    for (int i = 0; i < 16; i++) s[i] = in[i] ^ rk[i];
    for (int r = 1; r <= 10; r++) {
        for (int i = 0; i < 16; i++) t[i] = SBOX[s[(i + 4 * (i % 4)) % 16]];   /* SubBytes + ShiftRows (열 우선) */
        if (r != 10)
            for (int c = 0; c < 4; c++) {                        /* MixColumns */
                uint8_t *a = t + 4 * c, x = a[0] ^ a[1] ^ a[2] ^ a[3], a0 = a[0];
                a[0] ^= x ^ xt(a[0] ^ a[1]); a[1] ^= x ^ xt(a[1] ^ a[2]);
                a[2] ^= x ^ xt(a[2] ^ a[3]); a[3] ^= x ^ xt(a[3] ^ a0);
            }
        for (int i = 0; i < 16; i++) s[i] = t[i] ^ rk[16 * r + i];
    }
    for (int i = 0; i < 16; i++) out[i] = s[i];
}

void aes128_ctr(const uint8_t key[16], const uint8_t iv[16], const uint8_t *in, uint8_t *out, size_t n)
{
    uint8_t ctr[16], ks[16];
    for (int i = 0; i < 16; i++) ctr[i] = iv[i];
    for (size_t off = 0; off < n; off += 16) {
        aes128_encrypt_block(key, ctr, ks);
        for (size_t i = 0; i < 16 && off + i < n; i++) out[off + i] = in[off + i] ^ ks[i];
        for (int i = 15; i >= 0 && ++ctr[i] == 0; i--) { }      /* 128비트 카운터 +1 */
    }
}

#define ROR(x, n) (((x) >> (n)) | ((x) << (32 - (n))))

static void sha_block(sha256_t *s, const uint8_t *p)
{
    uint32_t w[64], a[8];
    for (int i = 0; i < 16; i++) w[i] = (uint32_t)p[4 * i] << 24 | (uint32_t)p[4 * i + 1] << 16 | (uint32_t)p[4 * i + 2] << 8 | p[4 * i + 3];
    for (int i = 16; i < 64; i++)
        w[i] = w[i - 16] + (ROR(w[i - 15], 7) ^ ROR(w[i - 15], 18) ^ (w[i - 15] >> 3)) + w[i - 7] + (ROR(w[i - 2], 17) ^ ROR(w[i - 2], 19) ^ (w[i - 2] >> 10));
    for (int i = 0; i < 8; i++) a[i] = s->h[i];
    for (int i = 0; i < 64; i++) {
        uint32_t t1 = a[7] + (ROR(a[4], 6) ^ ROR(a[4], 11) ^ ROR(a[4], 25)) + ((a[4] & a[5]) ^ (~a[4] & a[6])) + K[i] + w[i];
        uint32_t t2 = (ROR(a[0], 2) ^ ROR(a[0], 13) ^ ROR(a[0], 22)) + ((a[0] & a[1]) ^ (a[0] & a[2]) ^ (a[1] & a[2]));
        for (int k = 7; k > 0; k--) a[k] = a[k - 1];
        a[4] += t1; a[0] = t1 + t2;
    }
    for (int i = 0; i < 8; i++) s->h[i] += a[i];
}

void sha256_init(sha256_t *s)
{
    static const uint32_t H0[8] = {0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19};
    for (int i = 0; i < 8; i++) s->h[i] = H0[i];
    s->len = 0; s->used = 0;
}

void sha256_update(sha256_t *s, const uint8_t *p, size_t n)
{
    s->len += n;
    while (n--) { s->buf[s->used++] = *p++; if (s->used == 64) { sha_block(s, s->buf); s->used = 0; } }
}

void sha256_final(sha256_t *s, uint8_t out[32])
{
    uint64_t bits = s->len * 8; uint8_t pad = 0x80, z = 0;
    sha256_update(s, &pad, 1);
    while (s->used != 56) sha256_update(s, &z, 1);
    for (int i = 7; i >= 0; i--) { uint8_t b = (uint8_t)(bits >> (8 * i)); sha256_update(s, &b, 1); }
    for (int i = 0; i < 32; i++) out[i] = (uint8_t)(s->h[i / 4] >> (24 - 8 * (i % 4)));
}

void hmac_sha256(const uint8_t *key, size_t klen, const uint8_t *msg, size_t n, uint8_t out[32])
{
    uint8_t k0[64] = {0}, pad[64], inner[32]; sha256_t s;
    if (klen > 64) { sha256_init(&s); sha256_update(&s, key, klen); sha256_final(&s, k0); }
    else for (size_t i = 0; i < klen; i++) k0[i] = key[i];
    for (int i = 0; i < 64; i++) pad[i] = k0[i] ^ 0x36;
    sha256_init(&s); sha256_update(&s, pad, 64); sha256_update(&s, msg, n); sha256_final(&s, inner);
    for (int i = 0; i < 64; i++) pad[i] = k0[i] ^ 0x5c;
    sha256_init(&s); sha256_update(&s, pad, 64); sha256_update(&s, inner, 32); sha256_final(&s, out);
}

int ct_equal(const uint8_t *a, const uint8_t *b, size_t n)
{
    uint8_t d = 0;
    for (size_t i = 0; i < n; i++) d |= a[i] ^ b[i];
    return d == 0;
}

/* 31장 인증 틀 수신: 'S' 형식 1 카운터4 길이2 | 암호문 | 태그16
 *  암호: AES-128-CTR, IV = 세션 Nonce 8바이트 ‖ 카운터 4바이트 ‖ 0 4바이트 (카운터가 겹치지 않으면 IV도 겹치지 않음)
 *  인증: HMAC-SHA256(머리 8바이트 ‖ 암호문)의 앞 16바이트 (암호화 뒤 인증)
 *  순서: 카운터가 마지막으로 받아들인 값보다 커야 함(재전송 거부) → 태그 확인 → 그 뒤에만 카운터 갱신·복호 */
#ifndef SECURE_RX_H
#define SECURE_RX_H
#include <stdint.h>
#include <stddef.h>

#define SF_MODEL_CHUNK  1
#define SF_MODEL_COMMIT 2
#define SF_IMAGE        3
#define SF_MODEL_BYTES  (7840 + 40)          /* INT8 가중치 10×784 + Bias 10×4 (리틀 엔디언) */

typedef enum { SR_OK_CHUNK = 1, SR_OK_COMMIT = 2, SR_OK_IMAGE = 3,
               SR_ERR_FORMAT = -1, SR_ERR_REPLAY = -2, SR_ERR_MAC = -3, SR_ERR_COMMIT = -4 } sr_result_t;

typedef struct {
    const uint8_t *k_enc, *k_mac, *nonce;    /* 키 16·16바이트, Nonce 8바이트 */
    uint32_t last_ctr;
    uint8_t  staging[SF_MODEL_BYTES], active[SF_MODEL_BYTES];
    uint8_t  got[(SF_MODEL_BYTES + 7) / 8];   /* 받은 바이트 표시 */
    uint8_t  image[784];
    uint32_t commits;
} secure_rx_t;

void        srx_init(secure_rx_t *s, const uint8_t *k_enc, const uint8_t *k_mac, const uint8_t *nonce);
sr_result_t srx_frame(secure_rx_t *s, const uint8_t *f, size_t n);

#endif

/* 31장 인증 틀 수신 구현 */
#include "secure_rx.h"
#include "crypto_min.h"

void srx_init(secure_rx_t *s, const uint8_t *k_enc, const uint8_t *k_mac, const uint8_t *nonce)
{
    s->k_enc = k_enc; s->k_mac = k_mac; s->nonce = nonce; s->last_ctr = 0; s->commits = 0;
    for (size_t i = 0; i < SF_MODEL_BYTES; i++) { s->staging[i] = 0; s->active[i] = 0; }
    for (size_t i = 0; i < sizeof s->got; i++) s->got[i] = 0;
}

sr_result_t srx_frame(secure_rx_t *s, const uint8_t *f, size_t n)
{
    static uint8_t plain[1024];
    uint8_t tag[32], iv[16];
    if (n < 8 + 16 || f[0] != 'S') return SR_ERR_FORMAT;
    uint32_t ctr = (uint32_t)f[2] << 24 | (uint32_t)f[3] << 16 | (uint32_t)f[4] << 8 | f[5];
    size_t len = (size_t)f[6] << 8 | f[7];
    if (len > sizeof plain || n != 8 + len + 16) return SR_ERR_FORMAT;
    if (ctr <= s->last_ctr) return SR_ERR_REPLAY;                 /* 재전송·되감기 */
    hmac_sha256(s->k_mac, 16, f, 8 + len, tag);
    if (!ct_equal(tag, f + 8 + len, 16)) return SR_ERR_MAC;       /* 변조·다른 키 */
    s->last_ctr = ctr;                                             /* 인증된 틀만 카운터를 올린다 */
    for (int i = 0; i < 8; i++) iv[i] = s->nonce[i];
    for (int i = 0; i < 4; i++) { iv[8 + i] = f[2 + i]; iv[12 + i] = 0; }
    aes128_ctr(s->k_enc, iv, f + 8, plain, len);
    switch (f[1]) {
    case SF_MODEL_CHUNK: {                                         /* 오프셋 2바이트 + 데이터 */
        size_t off = (size_t)plain[0] << 8 | plain[1];
        if (len < 2 || off + len - 2 > SF_MODEL_BYTES) return SR_ERR_FORMAT;
        for (size_t i = 0; i < len - 2; i++) { s->staging[off + i] = plain[2 + i]; s->got[(off + i) / 8] |= (uint8_t)(1u << ((off + i) % 8)); }
        return SR_OK_CHUNK;
    }
    case SF_MODEL_COMMIT: {                                        /* 전체 모델 SHA-256 32바이트 */
        uint8_t h[32]; sha256_t sh;
        if (len != 32) return SR_ERR_FORMAT;
        for (size_t i = 0; i < SF_MODEL_BYTES; i++) if (!(s->got[i / 8] >> (i % 8) & 1)) return SR_ERR_COMMIT;
        sha256_init(&sh); sha256_update(&sh, s->staging, SF_MODEL_BYTES); sha256_final(&sh, h);
        if (!ct_equal(h, plain, 32)) return SR_ERR_COMMIT;          /* 활성 모델은 그대로 */
        for (size_t i = 0; i < SF_MODEL_BYTES; i++) s->active[i] = s->staging[i];
        s->commits++;
        return SR_OK_COMMIT;
    }
    case SF_IMAGE:
        if (len != 784) return SR_ERR_FORMAT;
        for (size_t i = 0; i < 784; i++) s->image[i] = plain[i];
        return SR_OK_IMAGE;
    default:
        return SR_ERR_FORMAT;
    }
}

/* 31장 시연용 키 — 공개된 시험 값이므로 실제 제품에 쓰면 안 된다.
 * 실제 키는 펌웨어 소스·비트스트림에 평문으로 두지 않는다 (보관 방법은 8절, ⚠ 미검증). */
#ifndef DEMO_KEYS_H

def xt(x):
    return ((x << 1) ^ (0x1B if x & 0x80 else 0)) & 0xFF


def aes_block(key, blk):
    rk, rcon = list(key), 1
    while len(rk) < 176:
        w = rk[-4:]
        if len(rk) % 16 == 0:
            w = [SB[w[1]] ^ rcon, SB[w[2]], SB[w[3]], SB[w[0]]]; rcon = xt(rcon)
        rk += [rk[-16 + k] ^ w[k] for k in range(4)]
    s = [b ^ k for b, k in zip(blk, rk[:16])]
    for r in range(1, 11):
        t = [SB[s[(i + 4 * (i % 4)) % 16]] for i in range(16)]
        if r != 10:
            for c in range(4):
                a = t[4 * c:4 * c + 4]; x = a[0] ^ a[1] ^ a[2] ^ a[3]
                t[4 * c:4 * c + 4] = [a[0] ^ x ^ xt(a[0] ^ a[1]), a[1] ^ x ^ xt(a[1] ^ a[2]), a[2] ^ x ^ xt(a[2] ^ a[3]), a[3] ^ x ^ xt(a[3] ^ a[0])]
        s = [b ^ k for b, k in zip(t, rk[16 * r:16 * r + 16])]
    return bytes(s)


def ctr(key, iv, data):
    out, c = bytearray(), int.from_bytes(iv, "big")
    for off in range(0, len(data), 16):
        ks = aes_block(key, c.to_bytes(16, "big")); c = (c + 1) % (1 << 128)
        out += bytes(a ^ b for a, b in zip(data[off:off + 16], ks))
    return bytes(out)

def frame(typ, counter, plain, k_enc=K_ENC, k_mac=K_MAC):
    head = b"S" + bytes([typ]) + struct.pack(">IH", counter, len(plain))
    ct = ctr(k_enc, NONCE + struct.pack(">I", counter) + bytes(4), plain)
    return head + ct + hmac.new(k_mac, head + ct, hashlib.sha256).digest()[:16]

07코드 읽기 포인트

위치내용
틀 구조'S' 형식 카운터(4) 길이(2) ‖ 암호문 ‖ 태그(16). 태그는 머리와 암호문 전체에 대한 HMAC-SHA256의 앞 16바이트입니다(암호화 뒤 인증).
확인 순서카운터가 마지막으로 받아들인 값보다 큰지 → 태그 → 그다음에만 카운터를 올리고 복호합니다. 태그가 틀린 틀로 카운터를 올릴 수 있으면 공격자가 정상 틀을 막을 수 있기 때문입니다.
IV 재사용 금지CTR의 IV = 세션 Nonce ‖ 카운터 ‖ 0. 카운터가 겹치지 않으면 키 흐름도 겹치지 않습니다. 같은 IV를 두 번 쓰면 두 평문의 XOR가 드러납니다.
모델 커밋조각은 준비 영역에만 쓰고, 모든 바이트가 왔고 전체 SHA-256이 커밋 틀의 값과 같을 때만 활성 모델로 복사합니다. 중간에 끊겨도 NPU는 이전 모델을 계속 씁니다.
시간 일정 비교태그는 ct_equal()로 모든 바이트를 다 본 뒤 판정합니다. 첫 다른 바이트에서 멈추면 걸린 시간으로 태그를 한 바이트씩 맞혀 볼 수 있습니다.
