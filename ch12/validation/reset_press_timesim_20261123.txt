## 시험 TB 원문
`timescale 1ns/1ps
module tb_press;
  reg clk=0,btnC=1,rxd=1; integer k;
  wire txd,dp; wire [15:0] led; wire [6:0] seg; wire [3:0] an;
  basys3_npu_top dut(.clk(clk),.btnC(btnC),.uart_rxd(rxd),.uart_txd(txd),.led(led),.seg(seg),.an(an),.dp(dp));
  always #5 clk=~clk;
  initial begin
    #400 btnC=0; #600;
    for(k=0;k<200;k=k+1) begin
      @(posedge clk); #(k*0.050) btnC=1;
      #100 @(posedge clk); #(k*0.050) btnC=0;
      #150;
    end
    $display("PRESS_DONE presses=%0d",k); $finish;
  end
endmodule
