제4장 · 784 가속기 설계 공간

28×28 손글씨 → 13타일 누산 → INT8/INT16 차이 → PE 수와 클록. 서브젝트 4개.

00 장 목표·진행 위치

장 시연 (보드): PC가 28×28 원본 손글씨를 타일 13개로 나눠 보내면 보드가 3장 코어로 784 입력을 누산해 분류한다. 같은 그림이 INT8에서는 4, INT16에서는 9(정답)로 갈리고, 1PE·4PE·8PE 클록 수가 보드에서 나온다.

편성: 1일 · 서브젝트 4개 · 서브젝트마다 보드 확인 1번 + 장 시연.

  1. 4-1INT8·INT16 정밀도 비교 (구 06장 · 오전)
    같은 32비트 값을 두 정밀도로: 768 → Q:03,0030, 65536 → INT8은 7F로 포화(넘침 1), INT16은 1000으로 정확.    시뮬 통과 · 보드 ⚠ 미검증
  2. 4-2784 입력 13타일 누산 (INT8) (구 17장 · 오전)
    시험 #92(숫자 9)를 타일 13개로: 타일마다 가중치 640개 + x 64개를 싣고 T → T0…TB, 마지막에 C4 (INT8은 4로 틀림). 시작 전 T는 E1.    시뮬 통과 · 보드 ⚠ 미검증
  3. 4-3INT8·INT16 MNIST 정밀도 (구 18장 · 오후)
    같은 그림을 INT16 엔진으로: 가중치·x가 2바이트라 보낼 양이 두 배, 결과 C9 (정답). LED가 4번에서 9번으로 옮겨 간다.    시뮬 통과 · 보드 ⚠ 미검증
  4. 4-4PE 병렬도·메모리 구조 (구 19장 · 오후)
    같은 타일 하나를 4PE K4:00CE(206클록), 8PE K8:008A(138), 1PE는 class마다 가중치를 다시 실어 K1:0046(70) × 10. 세 구성의 점수는 같다.    시뮬 통과 · 보드 ⚠ 미검증

    PC가 28×28 원본 손글씨를 타일 13개로 나눠 보내면 보드가 3장 코어로 784 입력을 누산해 분류한다. 같은 그림이 INT8에서는 4, INT16에서는 9(정답)로 갈리고, 1PE·4PE·8PE 클록 수가 보드에서 나온다.    보드 ⚠ 미실시

// 4장 디버그 콘솔: 784 가속기 서브젝트를 UART 명령으로 확인한다. 1~3장과 같은 형식 (1바이트 명령 + 바이너리).
// p = 엔진 (0 INT8, 1 INT16), 원소 하나 = INT8 1바이트 / INT16 2바이트(빅엔디언). g = PE 구성 1·4·8.
//
//   명령 페이로드                         응답                                       서브젝트 (STAGE)
//   S    없음                             "I"                                        모두
//   Q    int32 4바이트                     "Q:" INT8 2자리 "," INT16 4자리 " O:" 넘침 2자리   4-1 (1)
//   B    p class b4..b0 (signed40)         "OK"                                       4-2 (2)   IDLE에서만
//   V    p class n + 원소 n개              "OK"   W[class][0..n-1] 쓰기                4-2 (2)
//   A    p                                "OK"   784 추론 시작 (외부 누산 Clear)       4-2 (2)
//   X    p n(=64) + 원소 64개               "OK"   x 타일 (마지막 타일은 16 + 0 48개)     4-2 (2)
//   T    p 타일번호                         "T" 번호 / 마지막 타일 "C" 클래스 ":" 점수 11자리 ×10   4-2 (2)
//   U    g class n + 바이트 n개             "OK"   PE 구성 g의 가중치 (PE1은 class 무시)   4-4 (4)
//   K    g                                "K" g ":" 클록 4자리 " " 점수 7자리 (PE1 1개, 그 밖 10개)  4-4 (4)
//   엔진 오류 "E" 코드 (17장 err_code: 1 상태, 2 타일 중복, 3 타일 건너뜀, 4 프레임 없음, 5 쓰기)
//   모르는 명령 "?", 범위 밖 "ERR". 끝은 모두 \r\n.  p = 1은 STAGE 3부터.
`timescale 1ns/1ps
`default_nettype none
module p04_console #(
    parameter integer STAGE = 4
) (
    input  wire               clk,
    input  wire               resetn,
    input  wire [7:0]         rx_data_i,
    input  wire               rx_valid_i,
    output wire               rx_ready_o,
    output reg  [7:0]         tx_data_o,
    output reg                tx_valid_o,
    input  wire               tx_ready_i,
    // 묶음으로
    output reg                q_valid_o,
    output reg  signed [31:0] q_x_o,
    input  wire               q_done_i,
    input  wire signed [7:0]  q8_i,
    input  wire signed [15:0] q16_i,
    input  wire               q_ov8_i,
    input  wire               q_ov16_i,
    output wire [7:0]         s_data_o,
    output wire               s_valid_o,
    output wire               s_last_o,
    input  wire               s_ready_i,
    input  wire               held8_i,
    output reg                ack8_o,
    output reg                x16_we_o,
    output reg  [5:0]         x16_addr_o,
    output reg  signed [15:0] x16_data_o,
    output reg                x16_set_o,
    input  wire               held16_i,
    output reg                p_o,
    output reg                start_o,
    output reg                tile_go_o,
    output reg  [3:0]         tile_id_o,
    output reg                w_wr_o,
    output reg                b_wr_o,
    output reg  [3:0]         class_o,
    output reg  [5:0]         addr_o,
    output reg  signed [15:0] w_o,
    output reg  signed [39:0] b_o,
    input  wire               tile_done_i,
    input  wire               done_i,
    input  wire               error_i,
    input  wire [2:0]         err_code_i,
    input  wire               ovf_i,
    input  wire [3:0]         class_i,
    input  wire [419:0]       scores_i,
    output reg  [3:0]         g_o,
    output reg                pe_wr_o,
    output reg                pe_run_o,
    input  wire               pe_done_i,
    input  wire               pe_err_i,
    input  wire [15:0]        pe_cycles_i,
    input  wire [249:0]       pe_scores_i,
    // 표시
    output reg  [15:0]        view_o,
    output reg  [3:0]         cls_o,
    output reg                cls_ok_o
);
    localparam [3:0] S_IDLE = 4'd0, S_HDR = 4'd1, S_ISSUE = 4'd2, S_DATA = 4'd3, S_DONE = 4'd4,
                     S_QW = 4'd5, S_TW = 4'd6, S_KW = 4'd7, S_LOAD = 4'd8, S_SEND = 4'd9, S_ACK = 4'd10;
    localparam integer MSG_MAX = 128;

    reg [3:0] state;
    reg [7:0] opc;
    reg [7:0] hd [0:6];
    reg [3:0] need, got;
    reg [1:0] dly;
    reg [7:0] n_elem;                 // 남은 원소 수
    reg [5:0] eidx;                   // 원소 주소
    reg       hi_half;                // INT16 원소의 위 바이트를 받았음
    reg [7:0] hi_byte;
    reg       bad;                    // 데이터는 받되 결과는 ERR
    reg       eng_err;
    reg [2:0] eng_code;
    reg [7:0] msg [0:MSG_MAX-1];
    reg [6:0] msg_len, idx;
    integer   k, p;

    function [7:0] hex(input [3:0] n);
        hex = (n < 4'd10) ? (8'h30 + {4'd0, n}) : (8'h37 + {4'd0, n});
    endfunction

    function [3:0] plen(input [7:0] c);    // 고정 머리 바이트 수, 15 = 모르는 명령
        case (c)
            "S":     plen = 4'd0;
            "Q":     plen = 4'd4;
            "B":     plen = (STAGE >= 2) ? 4'd7 : 4'd15;
            "V":     plen = (STAGE >= 2) ? 4'd3 : 4'd15;
            "A":     plen = (STAGE >= 2) ? 4'd1 : 4'd15;
            "X":     plen = (STAGE >= 2) ? 4'd2 : 4'd15;
            "T":     plen = (STAGE >= 2) ? 4'd2 : 4'd15;
            "U":     plen = (STAGE >= 4) ? 4'd3 : 4'd15;
            "K":     plen = (STAGE >= 4) ? 4'd1 : 4'd15;
            default: plen = 4'd15;
        endcase
    endfunction

    wire is_eng   = (opc == "B" || opc == "V" || opc == "A" || opc == "X" || opc == "T");
    wire p_ok     = (hd[0] == 8'd0) || (hd[0] == 8'd1 && STAGE >= 3);
    wire g_ok     = (hd[0] == 8'd1) || (hd[0] == 8'd4) || (hd[0] == 8'd8);
    wire wide     = (opc == "V" || opc == "X") && hd[0][0];       // INT16 원소
    wire [7:0] n_hdr = (opc == "X") ? hd[1] : hd[2];
    wire fields_ok =
        (opc == "B") ? (p_ok && hd[1] < 8'd10) :
        (opc == "V") ? (p_ok && hd[1] < 8'd10 && hd[2] >= 8'd1 && hd[2] <= 8'd64) :
        (opc == "A") ? p_ok :
        (opc == "X") ? (p_ok && hd[1] == 8'd64) :
        (opc == "T") ? (p_ok && hd[1] < 8'd13) :
        (opc == "U") ? (g_ok && hd[1] < 8'd10 && hd[2] >= 8'd1 && hd[2] <= 8'd64) :
        (opc == "K") ? g_ok : 1'b1;

    // INT8 x 타일 바이트는 3-1 버퍼로 바로 흘린다
    wire stream8 = (state == S_DATA) && (opc == "X") && !hd[0][0] && !bad;
    assign s_data_o   = rx_data_i;
    assign s_valid_o  = stream8 && rx_valid_i;
    assign s_last_o   = (n_elem == 8'd1);
    assign rx_ready_o = (state == S_IDLE) || (state == S_HDR) || (state == S_DATA && (!stream8 || s_ready_i));
    wire   rx_take    = rx_valid_i && rx_ready_o;

    task put3(input [7:0] a, input [7:0] b, input [7:0] c, input [6:0] n);
        begin
            msg[0] <= a; msg[1] <= b; msg[2] <= c; msg[n - 2] <= 8'h0D; msg[n - 1] <= 8'h0A; msg_len <= n;
        end
    endtask

    always @(posedge clk) begin
        if (!resetn) begin
            state <= S_IDLE; opc <= 8'd0; need <= 4'd0; got <= 4'd0; dly <= 2'd0; n_elem <= 8'd0; eidx <= 6'd0;
            hi_half <= 1'b0; hi_byte <= 8'd0; bad <= 1'b0; eng_err <= 1'b0; eng_code <= 3'd0;
            msg_len <= 7'd0; idx <= 7'd0; tx_data_o <= 8'd0; tx_valid_o <= 1'b0;
            q_valid_o <= 1'b0; q_x_o <= 32'sd0; ack8_o <= 1'b0; x16_we_o <= 1'b0; x16_addr_o <= 6'd0;
            x16_data_o <= 16'sd0; x16_set_o <= 1'b0; p_o <= 1'b0; start_o <= 1'b0; tile_go_o <= 1'b0;
            tile_id_o <= 4'd0; w_wr_o <= 1'b0; b_wr_o <= 1'b0; class_o <= 4'd0; addr_o <= 6'd0;
            w_o <= 16'sd0; b_o <= 40'sd0; g_o <= 4'd0; pe_wr_o <= 1'b0; pe_run_o <= 1'b0;
            view_o <= 16'd0; cls_o <= 4'd0; cls_ok_o <= 1'b0;
        end else begin
            q_valid_o <= 1'b0; ack8_o <= 1'b0; x16_we_o <= 1'b0; x16_set_o <= 1'b0; start_o <= 1'b0;
            tile_go_o <= 1'b0; w_wr_o <= 1'b0; b_wr_o <= 1'b0; pe_wr_o <= 1'b0; pe_run_o <= 1'b0;
            if (error_i) begin eng_err <= 1'b1; eng_code <= err_code_i; end      // 엔진 오류는 명령 동안 모은다
            case (state)
                S_IDLE: if (rx_take && rx_data_i != 8'h0D && rx_data_i != 8'h0A) begin
                    opc <= rx_data_i; got <= 4'd0; need <= plen(rx_data_i); eng_err <= 1'b0; bad <= 1'b0; dly <= 2'd0;
                    if (plen(rx_data_i) == 4'd15) begin put3("?", 8'h0D, 8'h0A, 7'd3); state <= S_LOAD; end
                    else if (plen(rx_data_i) == 4'd0) begin put3("I", 8'h0D, 8'h0A, 7'd3); state <= S_LOAD; end
                    else state <= S_HDR;
                end
                S_HDR: if (rx_take) begin
                    hd[got] <= rx_data_i;
                    if (got + 4'd1 == need) state <= S_ISSUE;
                    else got <= got + 4'd1;
                end
                S_ISSUE: begin
                    p_o <= hd[0][0];
                    if (opc == "V" || opc == "X" || opc == "U") begin   // 데이터가 따라온다: 범위가 틀려도 받아서 버린다
                        bad <= !fields_ok; n_elem <= n_hdr; eidx <= 6'd0; hi_half <= 1'b0;
                        class_o <= hd[1][3:0]; g_o <= hd[0][3:0];
                        // INT8 x는 들고 있는 타일을 먼저 놓는다
                        if (opc == "X" && fields_ok && !hd[0][0] && held8_i) begin ack8_o <= 1'b1; state <= S_ACK; end
                        else state <= S_DATA;
                    end else if (!fields_ok) begin
                        put3("E", "R", "R", 7'd5); msg[3] <= 8'h0D; msg[4] <= 8'h0A; state <= S_LOAD;
                    end else case (opc)
                        "Q": begin q_x_o <= {hd[0], hd[1], hd[2], hd[3]}; q_valid_o <= 1'b1; state <= S_QW; end
                        "B": begin class_o <= hd[1][3:0]; b_o <= {hd[2], hd[3], hd[4], hd[5], hd[6]}; b_wr_o <= 1'b1; state <= S_DONE; end
                        "A": begin start_o <= 1'b1; state <= S_DONE; end
                        "T": begin tile_id_o <= hd[1][3:0]; tile_go_o <= 1'b1; state <= S_TW; end
                        default: begin g_o <= hd[0][3:0]; pe_run_o <= 1'b1; state <= S_KW; end   // "K"
                    endcase
                end
                S_ACK: state <= S_DATA;
                S_DATA: if (rx_take) begin
                    if (wide && !hi_half) begin
                        hi_byte <= rx_data_i; hi_half <= 1'b1;
                    end else begin
                        hi_half <= 1'b0;
                        if (!bad) begin
                            if (opc == "V") begin
                                addr_o <= eidx; w_o <= wide ? {hi_byte, rx_data_i} : {{8{rx_data_i[7]}}, rx_data_i}; w_wr_o <= 1'b1;
                            end else if (opc == "U") begin
                                addr_o <= eidx; w_o <= {{8{rx_data_i[7]}}, rx_data_i}; pe_wr_o <= 1'b1;
                            end else if (wide) begin                  // INT16 x
                                x16_addr_o <= eidx; x16_data_o <= {hi_byte, rx_data_i}; x16_we_o <= 1'b1;
                            end                                       // INT8 x는 스트림으로 이미 들어감
                        end
                        eidx <= eidx + 6'd1;
                        n_elem <= n_elem - 8'd1;
                        if (n_elem == 8'd1) begin
                            if (opc == "X" && wide && !bad) x16_set_o <= 1'b1;
                            state <= S_DONE;
                        end
                    end
                end
                S_DONE: if (dly != 2'd3) dly <= dly + 2'd1;   // 쓰기·시작 오류가 엔진에서 나올 때까지 기다린다
                else begin
                    if (bad) begin
                        put3("E", "R", "R", 7'd5); msg[3] <= 8'h0D; msg[4] <= 8'h0A;
                    end else if (eng_err || error_i) begin
                        put3("E", hex({1'b0, error_i ? err_code_i : eng_code}), 8'h0D, 7'd4); msg[3] <= 8'h0A;
                    end else begin
                        put3("O", "K", 8'h0D, 7'd4); msg[3] <= 8'h0A;
                    end
                    state <= S_LOAD;
                end
                S_QW: if (q_done_i) begin

                    end else begin

                    end
                end
                default: state <= S_IDLE;
            endcase
        end

"""4장 784 가속기 보드 PC 쪽 도구: 명령을 바이트로 바꿔 보내고 응답을 출력한다.

  python3 p04/host/p04_send.py /dev/ttyUSB1 Q 65536     # 4-1 INT8·INT16 양자화
  python3 p04/host/p04_send.py /dev/ttyUSB1 digit 92 int8    # 4-2 Bias 10 → A → 13타일(V 10·X·T) → 클래스
  python3 p04/host/p04_send.py COM3 digit 92 int16      # 4-3 같은 그림을 INT16으로
  python3 p04/host/p04_send.py COM3 pe 6                # 4-4 타일 6으로 PE4·PE8·PE1 클록 비교
  (pip install pyserial 필요. 115200 8N1. 그림은 part03/ch18/reference/mnist_rtl_samples.txt의 표본 16개)

명령 형식은 p04/rtl/p04_console.v 머리말. 'digit'·'pe'는 p04/reference/p04_model.py와 같은 순서로 보낸다.
"""
import json
import struct
import sys
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))
INT = lambda s: int(s, 0)


def elems(vals, wide):
    out = b""
    for v in vals:
        out += struct.pack(">h", v) if wide else bytes([v & 0xFF])
    return out


def quant(bits):
    from part03.ch18.reference.precision_reference import quantize_model
    m = json.loads((ROOT / "part03/ch18/reference/mnist_linear_f32.json").read_text(encoding="utf-8"))
    return quantize_model(m, bits)


def sample(idx):
    for ln in (ROOT / "part03/ch18/reference/mnist_rtl_samples.txt").read_text(encoding="utf-8").splitlines():
        f = ln.split()
        if f and f[0] == str(idx):
            return int(f[1]), [int(v) for v in f[5:]]
    raise SystemExit(f"표본에 시험 #{idx}가 없다 (part03/ch18/reference/mnist_rtl_samples.txt)")


def digit(idx, prec):
    from part03.ch17.reference.tile784_reference import tile_ranges
    from part03.ch18.reference.precision_reference import quantize_x
    p, bits = (1, 16) if prec == "int16" else (0, 8)
    qm, (label, px) = quant(bits), sample(idx)
    xq, wide = quantize_x(px, bits), p == 1
    cmds = [(b"B" + bytes([p, c]) + (qm["b"][c] & ((1 << 40) - 1)).to_bytes(5, "big"), "B") for c in range(10)]
    cmds.append((b"A" + bytes([p]), "A"))
    for t, (s, n) in enumerate(tile_ranges()):
        cmds += [(b"V" + bytes([p, c, n]) + elems(qm["w"][c][s:s + n], wide), "V") for c in range(10)]
        cmds.append((b"X" + bytes([p, 64]) + elems(xq[s:s + n] + [0] * (64 - n), wide), "X"))
        cmds.append((b"T" + bytes([p, t]), f"T {t}"))
    return cmds, f"시험 #{idx}, 숫자 {label}, {prec.upper()}"


def pe(tile):
    from part03.ch17.reference.tile784_reference import tile_ranges
    from part03.ch18.reference.precision_reference import quantize_x
    qm, (_, px) = quant(8), sample(92)
    s, _ = tile_ranges()[tile]
    cmds = [(b"U" + bytes([g, c, 64]) + elems(qm["w"][c][s:s + 64], False), "U") for g in (4, 8) for c in range(10)]
    cmds.append((b"X" + bytes([0, 64]) + elems(quantize_x(px, 8)[s:s + 64], False), "X"))
    cmds += [(b"K\x04", "K 4"), (b"K\x08", "K 8")]
    for c in range(10):
        cmds += [(b"U" + bytes([1, c, 64]) + elems(qm["w"][c][s:s + 64], False), "U"), (b"K\x01", f"K 1 (class {c})")]
    return cmds, f"타일 {tile} (시험 #92 INT8)"


def encode(args):
    op, a = args[0], args[1:]
    if op == "S":
        return [(b"S", "S")], ""
    if op == "Q":
        return [(b"Q" + struct.pack(">i", INT(a[0])), "Q")], ""
    if op == "digit":
        return digit(INT(a[0]), a[1] if len(a) > 1 else "int8")
    if op == "pe":
        return pe(INT(a[0]) if a else 6)
    raise SystemExit("모르는 명령: " + op)


def main(argv):
    if len(argv) < 2:
        raise SystemExit(__doc__)
    port, (cmds, what) = argv[0], encode(argv[1:])
    total = sum(len(d) for d, _ in cmds)
    if port == "--dry":

        return 0
    import serial                           # pyserial (보드에서만 필요)
    bad, t0 = 0, time.time()

            reply = s.read_until(b"\n").decode("ascii", "replace")
            if note not in ("B", "V", "X", "U", "A") or not reply.startswith("OK"):

    return 1 if bad else 0

// 14장 p3_group10_core(64→10, 4PE 4+4+2 그룹)를 변경 없이 타일마다 한 번 실행한다.
// 코어는 그룹마다 내부 누산기를 지우므로, 타일 결과(부분합)를 클래스별 외부 누산기 acc[c]에 더한다.
//   score[c] = sum_t part[t][c] + bias[c]
// Bias 중복 방지: 코어 Bias 쓰기 포트를 0으로 고정 → 코어 Bias는 Reset 값 0 유지 → 코어 점수 = 순수 부분합.
//   실제 Bias는 외부 레지스터에 두고 13타일 누산이 끝난 뒤 클래스별 1회만 더한다.
// 마지막 타일(12)은 16원소만 유효. 코어는 IN_LEN=64로 64클록을 돌므로 주소 16~63의 x를 0으로 가려
//   프레임 버퍼 잔여값과 이전 타일 가중치가 누산에 들어가지 않게 한다.
// 순서: start → [W 적재 + x 프레임 + tile_go(t)] × 13 → done.
//   W 쓰기는 IDLE·LOAD에서만 받는다. 다음 타일 프레임은 이전 타일의 4+4+2 그룹 계산과
//   결과 수락(외부 누산, frame_ack)이 끝나야 프레임 버퍼가 비워져 들어올 수 있다.
// (변경) 18장: x·W 데이터 폭을 파라미터로 둔다. 기본 DATA_W 8 / CORE_ACC_W 24 / PART_W 25가 17장 INT8 동작이다.
//   INT16: DATA_W 16 / CORE_ACC_W 38(64항 최악 2^36) / PART_W 39(코어 점수 = max(38, 코어 Bias 24) + 1).

    parameter integer ACC_W = 41,   // 외부 누산 폭. INT16 784항 최악 841,813,590,016 → signed 41
    parameter integer BIAS_W = 40,  // 외부 Bias 폭
    parameter integer SCORE_W = 42, // 최종 점수 폭 ≥ max(ACC_W, BIAS_W) + 1
    parameter integer DATA_W = 8,   // x·W signed 폭
    parameter integer CORE_ACC_W = 24, // 14장 코어 lane 누산 폭(타일 64항)
    parameter integer PART_W = 25   // 14장 코어 점수(부분합) 폭
)(
    input wire clk,
    input wire resetn,
    input wire start_i,                 // 784 추론 시작: 외부 누산기 Clear, 타일 0 대기
    input wire abort_i,                 // LOAD 중 취소: 보관 프레임 해제 후 IDLE
    input wire tile_go_i,               // 현재 프레임·가중치로 타일 tile_id_i 실행
    input wire [3:0] tile_id_i,
    input wire x_valid_i,               // 13장 프레임 버퍼 frame_valid
    output wire x_rd_en_o,
    output wire [5:0] x_rd_addr_o,
    input wire signed [DATA_W-1:0] x_rd_data_i,
    output wire frame_ack_o,
    input wire w_wr_en_i,
    input wire b_wr_en_i,
    input wire [3:0] wr_class_i,
    input wire [5:0] wr_addr_i,
    input wire signed [DATA_W-1:0] w_data_i,
    input wire signed [BIAS_W-1:0] b_data_i,
    output wire load_ready_o,           // W 쓰기 가능(IDLE·LOAD)
    output reg busy_o,
    output reg done_o,
    output reg error_o,
    output reg [2:0] err_code_o,        // 마지막 오류 원인
    output reg ovf_o,                   // 외부 누산 overflow(이번 추론 동안 유지)
    output wire [3:0] tile_idx_o,
    output reg tile_done_o,             // 타일 결과 수락 1클록. 이때 part_scores_o = 해당 타일 부분합

            end
            // 허용 상태 밖의 명령: IDLE은 start만, LOAD는 tile_go·abort만
            if ((start_i && state != IDLE) || ((tile_go_i || abort_i) && state != LOAD)) begin
                error_o <= 1'b1;
                err_code_o <= ERR_STATE;
            end
            case (state)
                IDLE: if (start_i) begin
                    busy_o <= 1'b1;
                    tile_q <= 4'd0;
                    ovf_o <= 1'b0;
                    for (k = 0; k < N_CLASS; k = k + 1)
                        acc[k] <= {ACC_W{1'b0}};
                    state <= LOAD;
                end
                LOAD: if (abort_i) begin
                    state <= DROP;
                end else if (tile_go_i) begin
                    // 타일 번호는 0..12 순서대로 한 번씩만: 이미 처리한 번호는 중복, 앞선 번호는 누락
                    if (tile_id_i < tile_q) begin
                        error_o <= 1'b1;
                        err_code_o <= ERR_TILE_DUP;
                    end else if (tile_id_i > tile_q) begin
                        error_o <= 1'b1;
                        err_code_o <= ERR_TILE_SKIP;
                    end else if (!x_valid_i) begin
                        error_o <= 1'b1;
                        err_code_o <= ERR_NO_FRAME;
                    end else begin
                        core_start_q <= 1'b1;
                        state <= RUN;
                    end
                end
                // 코어 done 시 10개 부분합은 고정되어 있다 → 외부 누산기에 한 번에 더한다
                RUN: if (core_done) begin
                    for (k = 0; k < N_CLASS; k = k + 1)
                        acc[k] <= sum_w[k][ACC_W-1:0];
                    if (~&acc_fit)
                        ovf_o <= 1'b1;
                    tile_done_o <= 1'b1;
                    state <= ACK;
                end
                // frame_ack_o = 1: 현재 타일 프레임 해제 → 다음 타일 프레임 수신 가능
                // 마지막 타일: 외부 누산이 직전 에지에서 끝났으므로 여기서 Bias를 더해 점수 확정
                ACK: if (tile_q == LAST_TILE) begin
                    for (k = 0; k < N_CLASS; k = k + 1)
                        score_q[k] <= final_w[k];
                    state <= SCORE;
                end else begin
                    tile_q <= tile_q + 4'd1;
                    state <= LOAD;
                end
                // argmax 1·2단 결과 저장
                SCORE: begin
                    a2s0_q <= t2s0; a2c0_q <= t2c0;
                    a2s1_q <= t2s1; a2c1_q <= t2c1;
                    a2s2_q <= t1s[4]; a2c2_q <= t1c[4];
                    state <= FINISH;
                end
                FINISH: begin
                    class_o <= best_class;
                    done_o <= 1'b1;
                    busy_o <= 1'b0;
                    state <= IDLE;
                end
                // 취소: frame_ack_o = 1로 보관 프레임 해제
                DROP: begin
                    busy_o <= 1'b0;
                    state <= IDLE;
                end
                default: begin
                    busy_o <= 1'b0;
                    state <= IDLE;
                end
            endcase
            // 컨트롤러가 쓰기·start를 거른 뒤이므로 코어 오류는 발생하면 안 된다(진단용)
            if (core_err) begin
                error_o <= 1'b1;
                err_code_o <= ERR_CORE;
            end
        end
    end
    // synthesis translate_off
    initial begin
        if (ACC_W < 2 || BIAS_W < 2 || SCORE_W < ((ACC_W > BIAS_W) ? ACC_W : BIAS_W) + 1)
            $fatal(1, "Require SCORE_W >= max(ACC_W, BIAS_W) + 1");
    end
    // synthesis translate_on
endmodule

    task automatic load_tile_w(input integer t, input integer len);

        end

    task automatic send_tile_x(input integer t, input integer len);
        integer j;
        begin
            if (DATA_W == 8) begin
                for (j = 0; j < 64; j = j + 1) begin
                    @(negedge clk);
                    s_data = (j < len) ? X[64*t + j] : PAD[j - len];
                    s_valid = 1; s_last = (j == 63);

                @(negedge clk);
                s_valid = 0; s_last = 0;
            end else begin
                while (!s_ready) @(negedge clk);
                for (j = 0; j < 64; j = j + 1) begin
                    @(negedge clk);
                    x_wr = 1; x_wr_addr = j;
                    x_wr_data = (j < len) ? X[64*t + j] : PAD[j - len];
                end
                @(negedge clk);
                x_wr = 0; x_commit = 1;
                @(negedge clk);
                x_commit = 0;
            end

    task automatic do_reset;

            @(negedge clk);
            resetn = 0;
            repeat (3) @(negedge clk);
            resetn = 1;
            @(negedge clk);
            if (busy || frame_valid || ovf || tile_idx != 0 || !load_ready || dut.state != 0)
                $fatal(1, "reset did not return controller to IDLE");

    task automatic run_inference(input integer mode);
        integer t, len, j;
        reg cut;
        begin
            cut = 0;
            load_bias;
            pulse_start;
            if (!busy) $fatal(1, "start not accepted");
            for (t = 0; t < 13 && !cut; t = t + 1) begin
                len = (t == 12) ? 16 : 64;
                // 결과 수락(ACK) 1클록 뒤 LOAD로 돌아온다
                j = 0;
                while (!load_ready) begin
                    @(negedge clk);
                    j = j + 1;
                    if (j > 2) $fatal(1, "load_ready not restored before tile %0d", t);
                end
                if (tile_idx != t || frame_valid) $fatal(1, "not waiting for tile %0d", t);
                load_tile_w(t, len);
                if (mode == MODE_ERRORS && t == 2) begin
                    pulse_go(2);
                    expect_error(ERR_NO_FRAME, "tile_go without frame");
                end
                if (mode == MODE_ERRORS && t == 4) begin
                    // 추론 중 Bias 변경·범위 밖 class·중복 start는 거부(결과 불변으로 확인)
                    write_b(3, 12345);
                    expect_error(ERR_WRITE, "bias write during inference");
                    write_w(10, 0, 7);
                    expect_error(ERR_WRITE, "class 10 weight write");
                    pulse_start;
                    expect_error(ERR_STATE, "start while active");
                end
                send_tile_x(t, len);
                if (mode == MODE_RESET_LOAD && t == 8) begin
                    do_reset;
                    cut = 1;
                end else if (mode == MODE_ABORT && t == 7) begin
                    abort_expected = 1;
                    pulse_abort;
                    @(negedge clk);
                    abort_expected = 0;
                    if (busy || frame_valid || !s_ready || !load_ready) $fatal(1, "abort did not release");
                    if (n_error != err_before) $fatal(1, "abort raised error");
                    cut = 1;
                end else begin
                    if (mode == MODE_ERRORS && t == 1) begin
                        pulse_go(0);
                        expect_error(ERR_TILE_DUP, "duplicate tile 0");
                        pulse_go(2);
                        expect_error(ERR_TILE_SKIP, "skip tile 1");
                        pulse_go(13);
                        expect_error(ERR_TILE_SKIP, "tile id 13");
                    end
                    pulse_go(t);
                    if (mode == MODE_ERRORS && t == 3) begin
                        // 계산 중: W(마지막 그룹이 늦게 읽는 class 9, 주소 63)·Bias 쓰기, 명령 모두 거부
                        write_w(9, 63, 99);
                        expect_error(ERR_WRITE, "weight write during compute");
                        write_b(0, 1);
                        expect_error(ERR_WRITE, "bias write during compute");
                        pulse_start;
                        expect_error(ERR_STATE, "start during compute");
                        pulse_go(3);
                        expect_error(ERR_STATE, "tile_go during compute");
                        pulse_abort;
                        expect_error(ERR_STATE, "abort during compute");
                        if (s_ready) $fatal(1, "next frame accepted during compute");
                    end
                    if (mode == MODE_RESET_RUN && t == 5) begin
                        repeat (50) @(negedge clk);
                        if (!dut.core.busy_o) $fatal(1, "core not running before reset");
                        do_reset;
                        cut = 1;
                    end else begin
                        j = 0;
                        while (!tile_done) begin
                            @(negedge clk);
                            j = j + 1;
                            if (j > 1000) $fatal(1, "tile %0d timeout", t);
                        end
                        if (tile_idx != t) $fatal(1, "tile_done index %0d != %0d", tile_idx, t);
                        if (!busy || done) $fatal(1, "busy/done wrong after tile %0d", t);
                        for (c = 0; c < 10; c = c + 1)
                            P[t][c] = $signed(parts[c*PART_W +: PART_W]);
                    end
                end
            end
            if (!cut) begin
                j = 0;
                while (!done) begin
                    @(negedge clk);
                    j = j + 1;
                    if (j > 20) $fatal(1, "done timeout");
                end
                if (busy || !load_ready) $fatal(1, "busy after done");
                for (t = 0; t < 13; t = t + 1) begin
                    $fwrite(af, "T%0d", t);
                    for (c = 0; c < 10; c = c + 1)
                        $fwrite(af, " %0d", P[t][c]);
                    $fwrite(af, "\n");
                end
                $fwrite(af, "S");
                for (c = 0; c < 10; c = c + 1)
                    $fwrite(af, " %0d", $signed(scores[c*SCORE_W +: SCORE_W]));
                $fwrite(af, " C%0d O%0d\n", cls, ovf);
                if (n_error != err_before) $fatal(1, "unexpected error pulse (code %0d)", last_code);
            end
        end
    endtask

    initial begin
        if (!$value$plusargs("VECTORS=%s", vpath)) $fatal(1, "+VECTORS required");
        if (!$value$plusargs("ACTUAL=%s", apath)) $fatal(1, "+ACTUAL required");
        full = $test$plusargs("FULL");
        vf = $fopen(vpath, "r");
        af = $fopen(apath, "w");
        rc = $fscanf(vf, "%d", ncase);
        repeat (4) @(negedge clk);

#!/usr/bin/env python3
"""17장: 784입력 13타일 외부 부분합 누산 기준모델·시험 벡터 생성·RTL 결과 비교.

타일 부분합은 host_tools의 dot_signed, 784 전체는 dot_tiled_64·dot_signed로 계산해 교차 확인한다.
Bias는 13타일 누산을 마친 뒤 클래스별로 1회만 더한다. 외부 누산 폭(acc_w)이 부족하면
RTL과 같은 2의 보수 순환값과 overflow 플래그를 기대값으로 만든다(폭 검증용).
"""
import argparse
import random
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[3]))
from part03.host_tools.p3_integer_reference import (  # noqa: E402
    argmax_lowest, dot_signed, dot_tiled_64,
)

N_IN, TILE, N_CLASS = 784, 64, 10
N_TILE = (N_IN + TILE - 1) // TILE          # 13
LAST_LEN = N_IN - TILE * (N_TILE - 1)       # 16
CORE_BIAS_W = 24                            # 14장 코어 Bias 폭(17장에서는 0으로 고정)


def core_part_w(data_w):
    """14장 코어 점수(부분합) 폭: INT8은 기존 25비트, 그 밖은 lane 누산 2N+6비트 + 1."""
    core_acc = 24 if data_w == 8 else 2 * data_w + 6
    return max(core_acc, CORE_BIAS_W) + 1


def tile_ranges():
    """(시작 주소, 유효 길이) 13개: 64×12 + 16."""
    return [(t * TILE, min(TILE, N_IN - t * TILE)) for t in range(N_TILE)]


def signed_bits(lo, hi):
    """lo..hi를 담는 최소 signed 비트 수."""
    n = 1
    while not (-(1 << (n - 1)) <= lo and hi <= (1 << (n - 1)) - 1):
        n += 1
    return n


def width_plan(data_w, bias_w):
    """signed data_w 입력·가중치, signed bias_w Bias일 때 필요한 폭."""
    lo, hi = -(1 << (data_w - 1)), (1 << (data_w - 1)) - 1
    p_max, p_min = lo * lo, lo * hi                  # 최대 양의 곱, 최소 음의 곱
    b_lo, b_hi = -(1 << (bias_w - 1)), (1 << (bias_w - 1)) - 1
    return {
        "tile": signed_bits(TILE * p_min, TILE * p_max),
        "acc": signed_bits(N_IN * p_min, N_IN * p_max),
        "score": signed_bits(N_IN * p_min + b_lo, N_IN * p_max + b_hi),
        "acc_max": N_IN * p_max,
        "score_max": N_IN * p_max + b_hi,
    }


def fits(v, w):
    return -(1 << (w - 1)) <= v <= (1 << (w - 1)) - 1


def wrap(v, w):
    v &= (1 << w) - 1
    return v - (1 << w) if v >> (w - 1) else v


def model(x, w, b, acc_w, bias_w, data_w=8):
    """반환: 타일별 부분합[13][10], 최종 점수[10], class, overflow."""
    assert len(x) == N_IN and len(w) == N_CLASS and len(b) == N_CLASS
    for v in x + [e for row in w for e in row]:
        if not fits(v, data_w):
            raise ValueError(f"x/w must be signed {data_w}-bit (14장 코어 DATA_W 계약)")
    for v in b:
        if not fits(v, bias_w):
            raise ValueError(f"bias must be signed {bias_w}-bit")
    parts = [[dot_signed(x[s:s + n], w[c][s:s + n]) for c in range(N_CLASS)]
             for s, n in tile_ranges()]
    for row in parts:
        for p in row:
            assert fits(p, width_plan(data_w, CORE_BIAS_W)["tile"]) and fits(p, core_part_w(data_w))
    acc, ovf = [0] * N_CLASS, False
    for row in parts:
        for c in range(N_CLASS):
            v = acc[c] + row[c]
            ovf |= not fits(v, acc_w)
            acc[c] = wrap(v, acc_w)
    full = [dot_tiled_64(x, w[c]) for c in range(N_CLASS)]
    assert full == [dot_signed(x, w[c]) for c in range(N_CLASS)]
    if not ovf:
        assert acc == full
    scores = [a + bias for a, bias in zip(acc, b)]   # Bias는 누산 완료 후 1회
    return parts, scores, argmax_lowest(scores), ovf


def build_cases(suite, bias_w, rng, data_w=8):
    b_lo, b_hi = -(1 << (bias_w - 1)), (1 << (bias_w - 1)) - 1
    lo, hi = -(1 << (data_w - 1)), (1 << (data_w - 1)) - 1

    def rand_case():
        return ([rng.randint(lo, hi) for _ in range(N_IN)],
                [[rng.randint(lo, hi) for _ in range(N_IN)] for _ in range(N_CLASS)],
                [rng.randint(b_lo, b_hi) for _ in range(N_CLASS)])

    # 짝수 class = 최대 양수 방향, 홀수 class = 최대 음수 방향 누산
    w_ext = [[lo] * N_IN if c % 2 == 0 else [hi] * N_IN for c in range(N_CLASS)]
    b_ext = [b_hi, b_lo, 0, 0, b_hi, b_lo, -1, 1, b_hi, b_lo]
    extremes = [([lo] * N_IN, w_ext, b_ext), ([hi] * N_IN, w_ext, b_ext)]
    if suite == "width":
        return extremes + [rand_case() for _ in range(2)]

    cases = [rand_case() for _ in range(3)]          # 0: 오류 주입, 1: Reset, 2: 취소
    cases += extremes                                # 3·4: 누산 극값 + Bias 극값
    # 5: 마지막 타일만 유효 데이터(앞 768개 x=0). 패딩·이전 타일 가중치가 섞이면 틀린다.
    x = [0] * 768 + [rng.choice([lo, hi, rng.randint(lo, hi)]) for _ in range(LAST_LEN)]
    cases.append((x, rand_case()[1], [0] * N_CLASS))
    # 6: 타일 위치 판별: class c는 타일 12-c에만 가중치 1, x = 타일번호+1
    x = [i // TILE + 1 for i in range(N_IN)]
    w = [[int(i // TILE == 12 - c) for i in range(N_IN)] for c in range(N_CLASS)]
    cases.append((x, w, [0] * N_CLASS))
    # 7: 전 점수 음수·타일 경계 class 동점(3·7 → 3)
    cases.append(([1] * N_IN, [[-1] * N_IN for _ in range(N_CLASS)],
                  [-9, -8, -7, -1, -6, -5, -4, -1, -3, -2]))
    # 8: x=0 → 점수 = Bias 그대로(13번 중복 적용이면 즉시 드러남)
    cases.append(([0] * N_IN, [[rng.randint(lo, hi) for _ in range(N_IN)]
                               for _ in range(N_CLASS)], b_ext))
    # 9: 작은 Bias 1000*(c+1): 타일마다 중복 적용되면 점수가 12000*(c+1)만큼 달라진다
    x = [rng.randint(-2, 2) for _ in range(N_IN)]
    w = [[rng.randint(-2, 2) for _ in range(N_IN)] for _ in range(N_CLASS)]
    cases.append((x, w, [1000 * (c + 1) for c in range(N_CLASS)]))
    cases += [rand_case() for _ in range(12)]
    # (추가) 19장 argmax 트리: 1단 쌍(0·1, 2·3)·2단(1 vs 2, 5 vs 6) 동점 → 낮은 번호. x=0이라 점수 = Bias
    z = [[0] * N_IN for _ in range(N_CLASS)]
    for tie in ((0, 1), (2, 3), (1, 2), (5, 6)):
        b = [-100 - c for c in range(N_CLASS)]
        b[tie[0]] = b[tie[1]] = 77
        cases.append(([0] * N_IN, z, b))
    return cases


def padding(rng, data_w=8):
    """마지막 타일 주소 16~63에 보내는 쓰레기 값(0이 아니어야 누락 검출 가능)."""
    lo, hi = -(1 << (data_w - 1)), (1 << (data_w - 1)) - 1
    return [rng.choice([lo, hi, 55, -77]) for _ in range(TILE - LAST_LEN)]


def selftest():
    assert [n for _, n in tile_ranges()] == [64] * 12 + [16]
    assert sum(n for _, n in tile_ranges()) == N_IN
    p8, p16 = width_plan(8, 24), width_plan(16, 40)
    assert (p8["tile"], p8["acc"], p8["score"]) == (22, 25, 26)
    assert p8["acc_max"] == 12_845_056
    assert (p16["tile"], p16["acc"], p16["score"]) == (38, 41, 42)
    assert p16["acc_max"] == 841_813_590_016 and p16["score_max"] == 1_391_569_403_903
    assert not fits(p16["acc_max"], 40)              # 40비트 누산은 INT16 784항에 부족
    assert wrap(1 << 23, 24) == -(1 << 23) and wrap(-1, 24) == -1
    # 24비트 외부 누산은 INT8 최악값에서 넘친다, 25비트는 넘치지 않는다
    x, w = [-128] * N_IN, [[-128] * N_IN] * N_CLASS
    assert model(x, w, [0] * N_CLASS, 24, 24)[3] is True
    _, s, _, o = model(x, w, [0] * N_CLASS, 25, 24)
    assert o is False and s[0] == 12_845_056
    # 위치 판별 case: score[c] = 타일(12-c) 길이 × (13-c)
    x = [i // TILE + 1 for i in range(N_IN)]
    w = [[int(i // TILE == 12 - c) for i in range(N_IN)] for c in range(N_CLASS)]
    assert model(x, w, [0] * N_CLASS, 41, 40)[1] == [16 * 13] + [64 * (13 - c) for c in range(1, 10)]
    return p8, p16


def write_vectors(out, cases, pads, acc_w, bias_w, data_w=8):
    """TB 입력(vectors.txt)과 기대값(expected.txt)을 쓴다. 반환: overflow 사례 수."""
    n_ovf = 0
    with (out / "vectors.txt").open("w") as vf, (out / "expected.txt").open("w") as ef:
        vf.write(f"{len(cases)}\n")
        for (x, w, b), pad in zip(cases, pads):
            parts, scores, cls, ovf = model(x, w, b, acc_w, bias_w, data_w)
            n_ovf += ovf
            vf.write(" ".join(map(str, x)) + "\n" + " ".join(map(str, pad)) + "\n")
            for row in w:
                vf.write(" ".join(map(str, row)) + "\n")
            vf.write(" ".join(map(str, b)) + "\n")
            for t, row in enumerate(parts):
                ef.write(f"T{t} " + " ".join(map(str, row)) + "\n")
            ef.write("S " + " ".join(map(str, scores)) + f" C{cls} O{int(ovf)}\n")
    return n_ovf


def generate(out, suite, acc_w, bias_w, data_w=8):
    p8, p16 = selftest()
    rng = random.Random(1717)
    cases = build_cases(suite, bias_w, rng, data_w)
    pads = [padding(rng, data_w) for _ in cases]
    n_ovf = write_vectors(out, cases, pads, acc_w, bias_w, data_w)
    dw = "" if data_w == 8 else f" data_w={data_w}"
    print(f"PYTHON_TILE784_SELF_TEST_PASS suite={suite} cases={len(cases)} tiles=13 "
          f"last_len={LAST_LEN} acc_w={acc_w} bias_w={bias_w} overflow_cases={n_ovf} seed=1717{dw}")
    print(f"WIDTH_PLAN INT8: tile={p8['tile']} acc={p8['acc']} score(+bias24)={p8['score']} | "
          f"INT16: tile={p16['tile']} acc={p16['acc']} score(+bias40)={p16['score']}")


def compare(out, actual):
    exp = (out / "expected.txt").read_text().splitlines()
    got = actual.read_text().splitlines()
    if exp != got:
        for i, (e, g) in enumerate(zip(exp, got)):
            if e != g:
                print(f"FAIL line {i}:\n  expected={e}\n  got     ={g}")
                break
        print(f"lines expected={len(exp)} got={len(got)}")
        raise SystemExit("PYTHON_RTL_COMPARE_FAIL")
    n_case = sum(line.startswith("S ") for line in exp)
    print(f"PYTHON_RTL_COMPARE_PASS cases={n_case} tile_partials={n_case * N_TILE * N_CLASS} "
          f"scores=10 class=1 overflow=1 bit_exact=1")

"""18장: MNIST 784→10 Float·INT8·INT16 정밀도 비교 기준모델(표준 라이브러리만 사용).

같은 Float32 가중치(mnist_linear_f32.json)를 INT8·INT16으로 양자화하고, 같은 시험셋·전처리로 비교한다.
양자화 계약(대칭, zero-point 0, N = 8 또는 16, Q = 2^(N-1) - 1):
  x_q = round_half_away(pixel * Q / 255)                 0..Q      (s_x = 1/Q,  x ≈ x_q·s_x)
  w_q = sat_N(round_half_away(W / s_w)),  s_w = max|W|/Q  -Q..Q
  b_q = sat_40(round_half_away(b / (s_x·s_w)))           누산과 같은 단위
  z_q[c] = Σ x_q·w_q[c] + b_q[c]  (17장 13타일 RTL과 같은 정수, 반올림·포화 없음)  → argmax(동점 낮은 번호)
반올림·포화는 PC 양자화 단계에서 한 번만 일어난다. RTL은 넓은 폭으로 정확히 누산한다(ovf=0 확인).

def qmax(bits):
    return (1 << (bits - 1)) - 1


def sat_signed(v, bits):
    return max(-(1 << (bits - 1)), min((1 << (bits - 1)) - 1, v))


def round_half_away(v):
    """실수 반올림: 절반은 0에서 먼 쪽(Python round()의 짝수 반올림과 다름)."""
    return int(math.copysign(math.floor(abs(v) + 0.5), v))


def div_round_away(n, d):
    """정수 n/d (d > 0) 반올림, 절반은 0에서 먼 쪽."""
    q = (2 * abs(n) + d) // (2 * d)
    return -q if n < 0 else q


def quantize_model(m, bits):
    q = qmax(bits)
    max_w = max(abs(v) for row in m["weight"] for v in row)
    s_x, s_w = 1.0 / q, max_w / q
    w_q = [[sat_signed(round_half_away(v / s_w), bits) for v in row] for row in m["weight"]]
    b_q = [sat_signed(round_half_away(v / (s_x * s_w)), BIAS_W) for v in m["bias"]]
    return {"bits": bits, "q": q, "s_x": s_x, "s_w": s_w, "max_w": max_w, "w": w_q, "b": b_q}


def quantize_x(pixels, bits):
    return [div_round_away(p * qmax(bits), 255) for p in pixels]


def int_logits(x_q, qm):
    return [dot_tiled_64(x_q, qm["w"][c]) + qm["b"][c] for c in range(N_CLASS)]


def float_logits(pixels, m):
    x = [p / 255.0 for p in pixels]
    return [sum(a * w for a, w in zip(x, row)) + b for row, b in zip(m["weight"], m["bias"])]


def read_idx(path):
    raw = path.read_bytes()
    ndim = raw[3]
    dims = [int.from_bytes(raw[4 + 4 * i:8 + 4 * i], "big") for i in range(ndim)]
    body = raw[4 + 4 * ndim:]
    if ndim == 1:
        return list(body)
    step = dims[1] * dims[2]
    return [list(body[i * step:(i + 1) * step]) for i in range(dims[0])]


def selftest():
    assert [round_half_away(v) for v in (-2.5, -1.5, -0.5, 0.5, 1.5, 2.5)] == [-3, -2, -1, 1, 2, 3]
    assert [round(v) for v in (0.5, 1.5, 2.5)] == [0, 2, 2]   # Python round()는 짝수 반올림
    assert [div_round_away(n, 2) for n in (-5, -3, -1, 1, 3, 5)] == [-3, -2, -1, 1, 2, 3]
    assert div_round_away(255 * 127, 255) == 127 and div_round_away(1 * 127, 255) == 0
    assert div_round_away(2 * 127, 255) == 1 and div_round_away(128 * 127, 255) == 64
    for v in (-999, -129, -128, -1, 0, 127, 128, 999):
        assert sat_signed(v, 8) == sat_int8(v)
    assert sat_signed(40000, 16) == 32767 and sat_signed(-40000, 16) == -32768
    assert quantize_x([0, 255], 8) == [0, 127] and quantize_x([0, 255], 16) == [0, 32767]
    toy = {"weight": [[0.5, -1.0]], "bias": [0.25]}
    t8 = quantize_model(toy, 8)
    assert t8["w"] == [[64, -127]]          # 0.5·127 = 63.5 → 64 (절반 0에서 먼 쪽)
    assert t8["b"] == [round_half_away(0.25 * 127 * 127)]


def evaluate(mnist_dir, m, out_dir):

    ys = read_idx(mnist_dir / "t10k-labels-idx1-ubyte")
    qms = {b: quantize_model(m, b) for b in (8, 16)}
    names = ("float", "int8", "int16")
    conf = {n: [[0] * N_CLASS for _ in range(N_CLASS)] for n in names}
    err = {8: [0.0, 0.0], 16: [0.0, 0.0]}     # 역양자화 Logit 오차 [최대, 합]

"""18장: MNIST 784→10 Float32 선형 분류기 학습(PC 전용, PyTorch·NumPy 필요).

전처리: x = pixel / 255 (0..1, 평균·표준편차 정규화 없음 → 정수화 zero-point 0).
결과: 같은 폴더의 mnist_linear_f32.json (float32 가중치·Bias를 정확한 10진 표현으로 저장,
      학습 설정·데이터 파일 SHA-256·PyTorch Float32 시험 정확도 포함). 데이터셋 경로는 기록하지 않는다.
실행: python train_mnist_linear.py --mnist-dir <MNIST raw 폴더(idx 파일 4개)>
"""
import argparse
import hashlib
import json
from pathlib import Path

import numpy as np
import torch

FILES = {
    "train_x": "train-images-idx3-ubyte", "train_y": "train-labels-idx1-ubyte",
    "test_x": "t10k-images-idx3-ubyte", "test_y": "t10k-labels-idx1-ubyte",
}
SEED, EPOCHS, BATCH, LR = 1818, 10, 128, 1e-3


def read_idx(path):
    raw = path.read_bytes()
    ndim = raw[3]
    dims = [int.from_bytes(raw[4 + 4 * i:8 + 4 * i], "big") for i in range(ndim)]
    return np.frombuffer(raw, dtype=np.uint8, offset=4 + 4 * ndim).reshape(dims)


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--mnist-dir", type=Path, required=True)
    p.add_argument("--out", type=Path, default=Path(__file__).with_name("mnist_linear_f32.json"))
    a = p.parse_args()
    torch.manual_seed(SEED)
    torch.use_deterministic_algorithms(True)
    torch.set_num_threads(1)
    d = {k: read_idx(a.mnist_dir / f) for k, f in FILES.items()}
    sha = {f: hashlib.sha256((a.mnist_dir / f).read_bytes()).hexdigest() for f in FILES.values()}
    xtr = torch.from_numpy(d["train_x"].reshape(-1, 784).astype(np.float32) / 255.0)
    ytr = torch.from_numpy(d["train_y"].astype(np.int64))
    xte = torch.from_numpy(d["test_x"].reshape(-1, 784).astype(np.float32) / 255.0)
    yte = torch.from_numpy(d["test_y"].astype(np.int64))
    model = torch.nn.Linear(784, 10)
    opt = torch.optim.Adam(model.parameters(), lr=LR)
    gen = torch.Generator().manual_seed(SEED)
    for epoch in range(EPOCHS):
        perm = torch.randperm(len(xtr), generator=gen)
        for i in range(0, len(xtr), BATCH):
            idx = perm[i:i + BATCH]
            opt.zero_grad()
            loss = torch.nn.functional.cross_entropy(model(xtr[idx]), ytr[idx])
            loss.backward()
            opt.step()
        with torch.no_grad():
            acc = (model(xte).argmax(1) == yte).float().mean().item()
        print(f"epoch={epoch + 1} loss={loss.item():.4f} test_acc_f32={acc:.4f}")
    with torch.no_grad():
        logits = model(xte)
        pred = logits.argmax(1)
        correct = int((pred == yte).sum())
    w = model.weight.detach().numpy().astype(np.float32)
    b = model.bias.detach().numpy().astype(np.float32)
    doc = {
        "model": "mnist_linear_784x10_f32",
        "preprocess": "x = pixel / 255 (uint8 0..255 -> float 0..1), no mean/std normalization",
        "train": {"seed": SEED, "epochs": EPOCHS, "batch": BATCH, "optimizer": "Adam", "lr": LR,
                  "loss": "cross_entropy", "torch": torch.__version__, "numpy": np.__version__},
        "dataset_sha256": sha,
        "torch_f32_test": {"correct": correct, "total": len(yte)},
        # float32 값을 Python float(정확한 binary64 확장)로 저장 → JSON 왕복 시 값이 바뀌지 않음
        "weight": [[float(v) for v in row] for row in w],
        "bias": [float(v) for v in b],
    }
    a.out.write_text(json.dumps(doc) + "\n")
    print(f"TRAIN_DONE torch_f32_test_correct={correct}/{len(yte)} out={a.out.name}")


if __name__ == "__main__":
    main()

    """이론 하한: MAC을 모든 PE에 완벽히 나눈 발행 클록(메모리·제어 지연 없음)."""
    return math.ceil(macs / pe)


def groups(out_n, pe):
    return math.ceil(out_n / pe)


def schedule_issue(in_len, out_n, pe):
    """출력 고정 그룹 스케줄의 발행 클록: 그룹마다 IN개 입력을 모든 lane에 방송."""
    return groups(out_n, pe) * in_len


def core_cycles(in_len, n_groups):
    return n_groups * (in_len + GROUP_OVERHEAD) + RUN_OVERHEAD


def stall_issue(in_len, out_n, pe, w_ports):
    """가중치 읽기 포트가 w_ports개(각 1단어/클록)일 때 발행 클록. x는 1단어 방송으로 충분."""
    total = 0
    for g in range(groups(out_n, pe)):
        active = min(pe, out_n - g * pe)
        total += in_len * math.ceil(active / w_ports)
    return total


def config_cycles():
    """RTL에서 측정할 구성별 기대 사이클(타일 1개)."""
    pe1_pass = core_cycles(IN_LEN, groups(1, 4))
    pe8a, pe8b = core_cycles(IN_LEN, groups(6, 4)), core_cycles(IN_LEN, groups(4, 4))
    return {"PE1_pass": pe1_pass, "PE1": N_CLASS * pe1_pass,
            "PE4": core_cycles(IN_LEN, groups(N_CLASS, 4)),
            "PE8A": pe8a, "PE8B": pe8b, "PE8": max(pe8a, pe8b),
            "LOAD_W": N_CLASS * IN_LEN}


def selftest():
    macs = IN_LEN * N_CLASS
    assert [issue_lower_bound(macs, p) for p in (1, 4, 8)] == [640, 160, 80]
    assert [schedule_issue(IN_LEN, N_CLASS, p) for p in (1, 4, 8)] == [640, 192, 128]
    cc = config_cycles()
    assert cc["PE4"] == 206                       # 14장 측정값과 같은 식
    assert core_cycles(8, 3) == 38 and core_cycles(64, 2) == 138   # 16장 2층·1층 측정값
    assert cc["PE8"] == core_cycles(IN_LEN, groups(N_CLASS, 8))     # 두 코어 = 8PE 스케줄
    assert stall_issue(IN_LEN, N_CLASS, 8, 8) == 128 and stall_issue(IN_LEN, N_CLASS, 8, 2) == 320
    assert stall_issue(IN_LEN, N_CLASS, 4, 4) == 192 and stall_issue(IN_LEN, N_CLASS, 1, 1) == 640
    assert sorted(PE8_MAP) == list(range(N_CLASS))
    return cc


def load_runs(n_samples):
    m = json.loads(MODEL.read_text())
    qm = quantize_model(m, 8)
    runs, samples = [], []
    for line in SAMPLES.read_text().splitlines():
        if line.startswith("#"):
            continue
        v = list(map(int, line.split()))
        samples.append((v[0], v[3], quantize_x(v[5:], 8)))     # index, INT8 예측, x_q
        if len(samples) == n_samples:
            break
    for si, (idx, _, x) in enumerate(samples):
        for t, (s, n) in enumerate(tile_ranges()):
            xt = x[s:s + n] + [0] * (IN_LEN - n)
            wt = [qm["w"][c][s:s + n] + [0] * (IN_LEN - n) for c in range(N_CLASS)]
            runs.append((si, t, xt, wt))
    return qm, samples, runs


def generate(out, n_samples):
    cc = selftest()
    _, samples, runs = load_runs(n_samples)
    with (out / "vectors.txt").open("w") as vf, (out / "expected.txt").open("w") as ef:
        vf.write(f"{len(runs)}\n")
        for r, (_, _, xt, wt) in enumerate(runs):
            vf.write(" ".join(map(str, xt)) + "\n")
            for row in wt:
                vf.write(" ".join(map(str, row)) + "\n")
            part = " ".join(str(dot_signed(xt, row)) for row in wt)
            for cfg in ("PE1", "PE4", "PE8"):
                ef.write(f"R{r} {cfg} {part}\n")
        ef.write(" ".join(f"{k}={cc[k]}" for k in ("PE1_pass", "PE1", "PE4", "PE8A", "PE8B", "PE8"))
                 + f" LOAD_W_PE1={cc['LOAD_W']} LOAD_W_PE4={cc['LOAD_W']} LOAD_W_PE8={cc['LOAD_W']}\n")
    print(f"PYTHON_ARCH_SELF_TEST_PASS samples={len(samples)} tile_runs={len(runs)} "
          f"cycles PE1={cc['PE1']} PE4={cc['PE4']} PE8={cc['PE8']} load_w={cc['LOAD_W']}")


def compare(out, actual, n_samples):
    exp = (out / "expected.txt").read_text().splitlines()
    got = actual.read_text().splitlines()
    if exp != got:
        for i, (e, g) in enumerate(zip(exp, got)):
            if e != g:
                print(f"FAIL line {i}:\n  expected={e}\n  got     ={g}")
                break
        print(f"lines expected={len(exp)} got={len(got)}")
        raise SystemExit("PYTHON_RTL_COMPARE_FAIL")
    # RTL 타일 부분합(PE4 줄)을 13타일 합 + Bias → 18장 INT8 평가의 Class와 같아야 한다
    qm, samples, runs = load_runs(n_samples)
    sums = [[0] * N_CLASS for _ in samples]
    for line in got:
        f = line.split()
        if len(f) == 12 and f[1] == "PE4":
            r = int(f[0][1:])
            for c in range(N_CLASS):
                sums[runs[r][0]][c] += int(f[2 + c])
    for (idx, pred8, _), acc in zip(samples, sums):
        cls = argmax_lowest([a + b for a, b in zip(acc, qm["b"])])
        if cls != pred8:
            raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL class sample {idx}: rtl={cls} eval={pred8}")
    print(f"PYTHON_RTL_COMPARE_PASS tile_runs={len(runs)} configs=PE1,PE4,PE8 partials_bit_exact=1 "
          f"cycles_match_model=1 mnist_samples={len(samples)} class_match_ch18_int8=1")


def report():
    """구성별 사이클·포트 표(Markdown). 시뮬레이션 사이클 모델이며 실측 시간이 아니다."""
    macs, cc = IN_LEN * N_CLASS, selftest()
    tiles = len(tile_ranges())
    print("| PE | 이론 하한(640 MAC) | 그룹 스케줄 발행 | lane 이용률 | 코어 사이클(타일 1개) | "
          "784 추론 연산 사이클(13타일) | 가중치 읽기 포트 |")
    print("|---:|---:|---:|---:|---:|---:|---:|")
    for pe, key in ((1, "PE1"), (4, "PE4"), (8, "PE8")):
        issue = schedule_issue(IN_LEN, N_CLASS, pe)
        print(f"| {pe} | {issue_lower_bound(macs, pe)} | {issue} | {100 * macs / (pe * issue):.1f}% | "
              f"{cc[key]} | {tiles * cc[key]} | {pe} |")
    print("\n가중치 읽기 포트 수에 따른 발행 클록(타일 1개, x 방송 1포트):\n")
    print("| PE \\ 포트 | 1 | 2 | 4 | 8 |")
    print("|---:|---:|---:|---:|---:|")
    for pe in (1, 4, 8):
        print(f"| {pe} | " + " | ".join(str(stall_issue(IN_LEN, N_CLASS, pe, p)) for p in (1, 2, 4, 8)) + " |")

def h(v, n):
    return format(v & ((1 << (4 * n)) - 1), "0%dX" % n)


def sample(idx):
    for ln in SAMPLES.read_text(encoding="utf-8").splitlines():
        f = ln.split()
        if f and f[0] == str(idx):
            return int(f[1]), [int(v) for v in f[2:5]], [int(v) for v in f[5:]]

class Engine:
    """p3_tile784_10 한 개의 바깥 동작 (17장 상태 표 중 이 장에서 쓰는 부분)."""

    def __init__(self, bits):
        self.bits = bits
        self.state = "IDLE"
        self.bank = [[None] * 64 for _ in range(10)]
        self.w = [[None] * 784 for _ in range(10)]      # 이번 추론에 실제로 쓰인 가중치
        self.b = [0] * 10
        self.x = [None] * 784
        self.next = 0
        self.frame = None


class Board:
    def __init__(self, stage):
        self.stage = stage
        self.eng = [Engine(8), Engine(16)]
        self.pe = {1: [None] * 64, 4: [[None] * 64 for _ in range(10)], 8: [[None] * 64 for _ in range(10)]}
        self.view, self.cls, self.cls_ok = 0, 0, False

    def held8(self):
        return self.eng[0].frame is not None

    def step(self, data):
        c = chr(data[0])
        if c not in NEED or self.stage < MIN_STAGE.get(c, 1):
            return "?" + CRLF
        hd, body = data[1:1 + NEED[c]], data[1 + NEED[c]:]
        if c == "S":
            return "I" + CRLF
        if c == "Q":
            v = struct.unpack(">i", bytes(hd))[0]
            q8, o8 = quantize(v, 8, 8, 1, 1)
            q16, o16 = quantize(v, 16, 4, 1, 1)
            self.view = q16 & 0xFFFF
            return "Q:%s,%s O:%d%d" % (h(q8, 2), h(q16, 4), o8, o16) + CRLF
        if c in "UK":
            g = hd[0]
            if g not in (1, 4, 8) or (c == "U" and not (hd[1] < 10 and 1 <= hd[2] <= 64)):
                return "ERR" + CRLF
            if c == "U":
                vals = [struct.unpack("b", bytes([v]))[0] for v in body]
                for i, v in enumerate(vals):
                    if g == 1:
                        self.pe[1][i] = v
                    else:
                        self.pe[g][hd[1]][i] = v
                return "OK" + CRLF
            x = self.eng[0].frame
            assert x is not None
            cyc = config_cycles()
            if g == 1:
                scores = [dot_signed(x, self.pe[1])]
                n = cyc["PE1_pass"]
            else:
                scores = [dot_signed(x, row) for row in self.pe[g]]
                n = cyc["PE%d" % g]
            self.view = n
            return "K%d:%s " % (g, h(n, 4)) + ",".join(h(s, 7) for s in scores) + CRLF
        p = hd[0]
        if p > 1 or (p == 1 and self.stage < 3):
            return "ERR" + CRLF
        e = self.eng[p]
        wide = p == 1
        if c == "B":
            if hd[1] >= 10:
                return "ERR" + CRLF
            assert e.state == "IDLE"
            v = int.from_bytes(bytes(hd[2:7]), "big")
            e.b[hd[1]] = v - (1 << 40) if v >> 39 else v
            return "OK" + CRLF
        if c in "VX":
            if wide:
                vals = [struct.unpack(">h", bytes(body[2 * i:2 * i + 2]))[0] for i in range(len(body) // 2)]
            else:
                vals = [struct.unpack("b", bytes([v]))[0] for v in body]
            if c == "V":
                if not (hd[1] < 10 and 1 <= hd[2] <= 64):
                    return "ERR" + CRLF
                for i, v in enumerate(vals):
                    e.bank[hd[1]][i] = v
                return "OK" + CRLF
            if hd[1] != 64:
                return "ERR" + CRLF
            e.frame = vals
            return "OK" + CRLF
        if c == "A":
            assert e.state == "IDLE"
            e.state, e.next = "LOAD", 0
            return "OK" + CRLF
        # T
        t = hd[1]
        if t >= 13:
            return "ERR" + CRLF
        if e.state != "LOAD":
            return "E1" + CRLF                                  # 17장 ERR_STATE
        assert t == e.next and e.frame is not None
        s, n = tile_ranges()[t]
        for cl in range(10):
            e.w[cl][s:s + n] = e.bank[cl][:n]
        e.x[s:s + n] = e.frame[:n]
        e.frame = None                                          # 타일이 끝나면 프레임을 놓는다
        e.next += 1
        if t < 12:
            self.view = t
            return "T%X" % t + CRLF
        _, scores, cls, ovf = model(e.x, e.w, e.b, ACC_W, BIAS_W, e.bits)
        e.state = "IDLE"
        self.view, self.cls, self.cls_ok = cls, cls, True
        return "C%X:" % cls + ",".join(h(v, 11) for v in scores) + (" V" if ovf else "") + CRLF

    def led(self):
        onehot = (1 << self.cls) if self.cls_ok else 0
        return onehot | (int(self.eng[1].frame is not None) << 14) | (int(self.held8()) << 15)


def elems(vals, wide):
    out = []
    for v in vals:
        out += list(struct.pack(">h", v)) if wide else [v & 0xFF]
    return out


def run_784(p, qm, xq, add, tag):
    """784 추론 한 번: Bias 10 → A → 타일마다 V 10 · X · T (PC 도구 'digit'과 같은 순서)."""
    wide = p == 1
    for cl in range(10):
        add([ord("B"), p, cl] + list((qm["b"][cl] & ((1 << 40) - 1)).to_bytes(5, "big")), f"load Bias {tag}")
    add([ord("A"), p], "")
    for t, (s, n) in enumerate(tile_ranges()):
        for cl in range(10):
            add([ord("V"), p, cl, n] + elems(qm["w"][cl][s:s + n], wide), f"load W {tag} 타일 {t}")
        xt = xq[s:s + n] + [0] * (64 - n)
        add([ord("X"), p, 64] + elems(xt, wide), f"X {p} 64 ({tag} 타일 {t} x{', 16 + 0 48개' if n < 64 else ''})")
        add([ord("T"), p, t], "")
