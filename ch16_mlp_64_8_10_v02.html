03RTL 작성

아래 파일을 경로 그대로 만들고 전체 코드를 입력합니다.

ch16/rtl/npu_argmax.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 16장 순차 argmax: start 뒤 한 클록에 점수 하나씩 비교한다. 같으면 앞 번호를 유지한다 (Python argmax_lowest).
module npu_argmax #(
    parameter integer N = 10,              // 2~16
    parameter integer W = 32
) (
    input  wire           clk,
    input  wire           resetn,
    input  wire           start_i,
    input  wire [N*W-1:0] scores_i,
    output reg            busy_o,
    output reg            done_o,
    output reg      [3:0] class_o
);
    reg [3:0] k_q;
    wire signed [W-1:0] cand = scores_i[k_q*W +: W];
    wire signed [W-1:0] best = scores_i[class_o*W +: W];
    always @(posedge clk) begin
        if (!resetn) begin
            busy_o <= 0; done_o <= 0; class_o <= 0; k_q <= 0;
        end else begin
            done_o <= 0;
            if (!busy_o) begin
                if (start_i) begin busy_o <= 1; class_o <= 0; k_q <= 1; end
            end else begin
                if (cand > best) class_o <= k_q;
                if (k_q == N - 1) begin busy_o <= 0; done_o <= 1; end
                else k_q <= k_q + 4'd1;
            end
        end
    end
endmodule
`default_nettype wire
ch16/rtl/npu_mlp_64_8_10.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 16장 2층 신경망 64→8→10: 15장 Dense 두 개 + ReLU·정수 양자화(05장) + 은닉 버퍼 + argmax.
// 단계: L1(64→8) → QUANT(은닉 8개를 하나씩 ReLU·반올림 시프트·INT8 포화) → L2(8→10) → ARGMAX.
// 각 단계는 앞 단계의 done 펄스로 시작하고, 은닉값·점수는 단계 완료 펄스와 함께 밖에서 볼 수 있다.
module npu_mlp_64_8_10 #(
    parameter integer ACC_WIDTH = 32,
    parameter integer SHIFT = 10           // 은닉 = sat_int8(round(ReLU(점수) / 2^SHIFT))
) (
    input  wire                     clk,
    input  wire                     resetn,
    input  wire                     model_clear_i,
    input  wire                     w_layer_i,          // 0: 은닉층 64→8, 1: 출력층 8→10
    input  wire                     w_wr_en_i,
    input  wire               [3:0] w_neuron_i,
    input  wire               [5:0] w_addr_i,
    input  wire               [7:0] w_data_i,
    input  wire                     b_wr_en_i,
    input  wire               [3:0] b_neuron_i,
    input  wire     [ACC_WIDTH-1:0] b_data_i,
    output wire              [31:0] model_sum_o,        // 두 층 검증합의 합
    output wire                     model_err_o,
    output wire                     x_rd_en_o,
    output wire               [5:0] x_rd_idx_o,
    input  wire signed        [7:0] x_data_i,
    input  wire                     x_valid_i,
    input  wire                     start_i,
    output wire                     busy_o,
    output reg                      hidden_valid_o,     // 은닉 8개 확정 1클록 펄스
    output wire              [63:0] hidden_o,           // 은닉 h = [h*8 +: 8], INT8
    output reg                      hidden_sat_o,       // 이번 실행에서 은닉 포화가 있었음
    output wire [10*ACC_WIDTH-1:0]  scores_o,
    output reg                      done_o,
    output reg                      error_o,
    output wire               [3:0] class_o
);
    localparam IDLE=0, L1=1, QUANT=2, L2=3, ARGMAX=4;
    reg [2:0] state;
    reg [3:0] h_q;                                       // 양자화에 넣을 은닉 번호
    reg signed [7:0] hid_q [0:7];
    integer i;

    wire [31:0] sum1, sum2; wire err1, err2;
    assign model_sum_o = sum1 + sum2;
    assign model_err_o = err1 | err2;
    assign busy_o = (state != IDLE);

    // ── 은닉층 64→8
    wire l1_busy, l1_done, l1_err;
    wire [8*ACC_WIDTH-1:0] l1_scores;
    npu_dense #(.N_IN(64), .N_OUT(8), .ACC_WIDTH(ACC_WIDTH)) layer1 (
        .clk(clk), .resetn(resetn), .model_clear_i(model_clear_i),
        .w_wr_en_i(w_wr_en_i && !w_layer_i && !busy_o), .w_neuron_i(w_neuron_i), .w_addr_i(w_addr_i), .w_data_i(w_data_i),
        .b_wr_en_i(b_wr_en_i && !w_layer_i && !busy_o), .b_neuron_i(b_neuron_i), .b_data_i(b_data_i),
        .model_sum_o(sum1), .model_err_o(err1),
        .x_rd_en_o(x_rd_en_o), .x_rd_idx_o(x_rd_idx_o), .x_data_i(x_data_i), .x_valid_i(x_valid_i),
        .start_i(state == IDLE && start_i), .busy_o(l1_busy), .done_o(l1_done), .error_o(l1_err), .scores_o(l1_scores));

    // ── ReLU → 05장 양자화기 (반올림 0에서 먼 쪽, INT8 포화)
    wire signed [ACC_WIDTH-1:0] pick = l1_scores[h_q[2:0]*ACC_WIDTH +: ACC_WIDTH];
    wire signed [ACC_WIDTH-1:0] relu = pick[ACC_WIDTH-1] ? {ACC_WIDTH{1'b0}} : pick;
    wire signed [7:0] q; wire q_valid, q_ov;
    fixed_quantizer #(.IN_WIDTH(ACC_WIDTH), .OUT_WIDTH(8), .SHIFT(SHIFT), .ROUND(1), .SATURATE(1)) quant (
        .clk(clk), .resetn(resetn), .valid_i(state == QUANT && h_q < 8), .x_i(relu),
        .q_o(q), .valid_o(q_valid), .overflow_o(q_ov));
    reg [2:0] q_idx;                                     // 양자화기 출력(1클록 뒤)이 들어갈 칸

    // ── 출력층 8→10 (입력 = 은닉 버퍼, 1클록 지연 읽기)
    wire l2_x_en; wire [5:0] l2_x_idx; reg signed [7:0] l2_x; reg l2_xv;
    wire l2_busy, l2_done, l2_err;
    npu_dense #(.N_IN(8), .N_OUT(10), .ACC_WIDTH(ACC_WIDTH)) layer2 (
        .clk(clk), .resetn(resetn), .model_clear_i(model_clear_i),
        .w_wr_en_i(w_wr_en_i && w_layer_i && !busy_o), .w_neuron_i(w_neuron_i), .w_addr_i(w_addr_i), .w_data_i(w_data_i),
        .b_wr_en_i(b_wr_en_i && w_layer_i && !busy_o), .b_neuron_i(b_neuron_i), .b_data_i(b_data_i),
        .model_sum_o(sum2), .model_err_o(err2),
        .x_rd_en_o(l2_x_en), .x_rd_idx_o(l2_x_idx), .x_data_i(l2_x), .x_valid_i(l2_xv),
        .start_i(state == QUANT && h_q == 9), .busy_o(l2_busy), .done_o(l2_done), .error_o(l2_err), .scores_o(scores_o));
    always @(posedge clk) begin
        l2_xv <= resetn && l2_x_en;
        if (l2_x_en) l2_x <= hid_q[l2_x_idx[2:0]];
    end

    wire am_busy, am_done;
    npu_argmax #(.N(10), .W(ACC_WIDTH)) argmax (
        .clk(clk), .resetn(resetn), .start_i(l2_done), .scores_i(scores_o),
        .busy_o(am_busy), .done_o(am_done), .class_o(class_o));

    genvar g;
    generate for (g = 0; g < 8; g = g + 1) begin : h
        assign hidden_o[g*8 +: 8] = hid_q[g];
    end endgenerate

    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE; h_q <= 0; q_idx <= 0; hidden_valid_o <= 0; hidden_sat_o <= 0; done_o <= 0; error_o <= 0;
            for (i = 0; i < 8; i = i + 1) hid_q[i] <= 0;
        end else begin
            hidden_valid_o <= 0; done_o <= 0; error_o <= 0;
            if (q_valid) begin
                hid_q[q_idx] <= q;
                if (q_ov) hidden_sat_o <= 1;
            end
            case (state)
                IDLE: if (start_i) begin state <= L1; hidden_sat_o <= 0; end
                L1: if (l1_err) begin state <= IDLE; error_o <= 1; end
                    else if (l1_done) begin state <= QUANT; h_q <= 0; end
                QUANT: begin                                  // h_q 0~7: 양자화기 입력, 8: 마지막 출력 저장, 9: L2 시작
                    q_idx <= h_q[2:0];
                    if (h_q == 8) hidden_valid_o <= 1;
                    if (h_q == 9) state <= L2;
                    else h_q <= h_q + 4'd1;
                end
                L2: if (l2_err) begin state <= IDLE; error_o <= 1; end
                    else if (l2_done) state <= ARGMAX;
                ARGMAX: if (am_done) begin state <= IDLE; done_o <= 1; end
                default: begin state <= IDLE; error_o <= 1; end
            endcase
        end
    end
endmodule
`default_nettype wire

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. 기준 모델(Python)이 11사례를 만듭니다: 숫자 0~7 입력과 무작위 가중치 8개, 은닉 포화, ReLU 뒤 은닉 전부 0, 반올림 0.5(512/1024 → 1). 가중치는 무작위이며 학습한 모델이 아닙니다.

ch16/tb/tb_npu_mlp.sv · systemverilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
// 16장 TB: 11사례(숫자 8개, 포화, ReLU 전부 0, 반올림 0.5)로 은닉 8개·점수 10개·클래스·포화 표시를 돌려준다.
module tb_npu_mlp;
    localparam AW = 32;
    reg clk=0, rn=0, mc=0, wl=0, wwe=0, bwe=0, start=0, drop=0;
    reg [3:0] wn=0, bn=0; reg [5:0] wa=0; reg [7:0] wd=0; reg [AW-1:0] bd=0;
    reg signed [7:0] xd=0; reg xv=0;
    wire xre, busy, hv, hsat, done, err, merr; wire [5:0] xi; wire [31:0] msum;
    wire [63:0] hid; wire [10*AW-1:0] scores; wire [3:0] cls;
    npu_mlp_64_8_10 #(.ACC_WIDTH(AW), .SHIFT(10)) dut(.clk(clk), .resetn(rn), .model_clear_i(mc), .w_layer_i(wl),
        .w_wr_en_i(wwe), .w_neuron_i(wn), .w_addr_i(wa), .w_data_i(wd), .b_wr_en_i(bwe), .b_neuron_i(bn), .b_data_i(bd),
        .model_sum_o(msum), .model_err_o(merr), .x_rd_en_o(xre), .x_rd_idx_o(xi), .x_data_i(xd), .x_valid_i(xv),
        .start_i(start), .busy_o(busy), .hidden_valid_o(hv), .hidden_o(hid), .hidden_sat_o(hsat),
        .scores_o(scores), .done_o(done), .error_o(err), .class_o(cls));
    always #5 clk = ~clk;

    integer x [0:63]; integer w1 [0:7][0:63]; integer b1 [0:7]; integer w2 [0:9][0:7]; integer b2 [0:9];
    integer f, o, n, k, c, v, cases = 0, guards = 0, hv_count;
    reg [8*16-1:0] name; string vp, ap;
    reg [63:0] hid_at_valid;

    always @(posedge clk) begin xv <= xre && !drop; if (xre) xd <= x[xi]; end
    always @(posedge clk) if (hv) begin hv_count = hv_count + 1; hid_at_valid = hid; end

    task rdv(output integer r); begin if ($fscanf(f, "%d", r) != 1) $fatal(1, "벡터 형식"); end endtask

    task wr_w(input l, input [3:0] nn, input [5:0] aa, input integer vv); begin
        @(negedge clk); bwe = 0; wl = l; wwe = 1; wn = nn; wa = aa; wd = vv; end endtask
    task wr_b(input l, input [3:0] nn, input integer vv); begin
        @(negedge clk); wwe = 0; wl = l; bwe = 1; bn = nn; bd = vv; end endtask

    task load_model; integer cc, s; begin
        @(negedge clk); mc = 1; @(negedge clk); mc = 0;
        for (cc = 0; cc < 8; cc = cc + 1) begin for (s = 0; s < 64; s = s + 1) wr_w(0, cc, s, w1[cc][s]); wr_b(0, cc, b1[cc]); end
        for (cc = 0; cc < 10; cc = cc + 1) begin for (s = 0; s < 8; s = s + 1) wr_w(1, cc, s, w2[cc][s]); wr_b(1, cc, b2[cc]); end
        @(negedge clk); wwe = 0; bwe = 0;
        if (merr) $fatal(1, "정상 적재에서 model_err");
    end endtask

    initial begin
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap)) $fatal(1, "경로 인자");
        f = $fopen(vp, "r"); o = $fopen(ap, "w"); if (!f || !o) $fatal(1, "파일 열기");
        repeat (3) @(negedge clk); rn = 1;
        while ($fscanf(f, "%d %s", n, name) == 2) begin
            for (k = 0; k < 64; k = k + 1) rdv(x[k]);
            for (c = 0; c < 8; c = c + 1) for (k = 0; k < 64; k = k + 1) begin rdv(v); w1[c][k] = v; end
            for (c = 0; c < 8; c = c + 1) rdv(b1[c]);
            for (c = 0; c < 10; c = c + 1) for (k = 0; k < 8; k = k + 1) begin rdv(v); w2[c][k] = v; end
            for (c = 0; c < 10; c = c + 1) rdv(b2[c]);
            load_model;
            hv_count = 0;
            @(negedge clk); start = 1; @(negedge clk); start = 0;
            if (!busy) $fatal(1, "start 후 busy 아님");
            wr_w(1, 0, 0, 99); @(negedge clk); wwe = 0;                 // 실행 중 쓰기 → 무시
            while (!done) begin @(negedge clk); if (err) $fatal(1, "%s: error_o", name); end
            if (hv_count != 1 || hid_at_valid !== hid) $fatal(1, "%s: hidden_valid 펄스 %0d회", name, hv_count);
            $fwrite(o, "%0d", n);
            for (c = 0; c < 8; c = c + 1) $fwrite(o, " %0d", $signed(hid[c*8 +: 8]));
            for (c = 0; c < 10; c = c + 1) $fwrite(o, " %0d", $signed(scores[c*AW +: AW]));
            $fwrite(o, " %0d %0d %0d\n", cls, hsat, msum);
            cases = cases + 1;
        end
        // 입력 응답 없음 → 1층 오류가 error_o로 올라오고 IDLE
        drop = 1; @(negedge clk); start = 1; @(negedge clk); start = 0;
        k = 0; while (!err && k < 10) begin @(negedge clk); k = k + 1; end
        drop = 0; @(negedge clk);
        if (k >= 10 || busy) $fatal(1, "1층 오류 전달 실패"); guards = guards + 1;
        $display("RTL_SIM_PASS MLP_64_8_10 cases=%0d guards=%0d", cases, guards);
        $fclose(o); $finish;
    end
    initial begin #50000000; $fatal(1, "시간 초과"); end
endmodule

#!/usr/bin/env python3
"""16장 기준 모델: 64→8(ReLU·반올림 시프트·INT8 포화)→10, argmax(동점은 낮은 번호).

최종 점수는 host_tools의 mlp_64_8_10, 은닉값은 같은 식(sat_int8·round_shift_away)으로 따로 계산해 비교한다.
"""
from __future__ import annotations
import argparse
import random
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT / "ch13" / "reference"), str(ROOT / "host_tools"), str(ROOT / "ch15" / "reference")]
from mnist8x8_reference import draw_digit, features            # noqa: E402
from p3_integer_reference import argmax_lowest, dot_signed, mlp_64_8_10, round_shift_away, sat_int8  # noqa: E402
from dense_reference import model_sum                           # noqa: E402

SHIFT = 10


def hidden(x, w1, b1, shift=SHIFT):
    return [sat_int8(round_shift_away(max(0, dot_signed(x, w) + b), shift)) for w, b in zip(w1, b1)]


def cases():
    rng = random.Random(1616)
    out = []
    for k in range(8):
        x = features(draw_digit(k))
        w1 = [[rng.randint(-32, 32) for _ in range(64)] for _ in range(8)]
        b1 = [rng.randint(-4096, 4096) for _ in range(8)]
        w2 = [[rng.randint(-128, 127) for _ in range(8)] for _ in range(10)]
        b2 = [rng.randint(-2000, 2000) for _ in range(10)]
        out.append((f"digit_{k}", x, w1, b1, w2, b2))
    x = features(draw_digit(8))
    out.append(("saturate", x, [[127] * 64] * 4 + [[-128] * 64] * 4, [0] * 8,
                [[1] * 8] * 10, list(range(10))))
    out.append(("relu_all_zero", x, [[-5] * 64] * 8, [-1] * 8, [[3] * 8] * 10, [7] * 10))
    out.append(("round_half", [1] * 64, [[8] * 64] + [[0] * 64] * 7, [0] * 8,
                [[1] + [0] * 7] * 10, [0] * 10))
    return out


def selftest():
    cs = cases()
    seen_sat = seen_zero = False
    for name, x, w1, b1, w2, b2 in cs:
        h = hidden(x, w1, b1)
        s = mlp_64_8_10(x, w1, b1, w2, b2, SHIFT)
        assert s == [dot_signed(h, w) + b for w, b in zip(w2, b2)], name
        seen_sat |= any(abs(v) >= 127 for v in h); seen_zero |= not any(h)
    assert seen_sat and seen_zero
    assert hidden([1] * 64, [[8] * 64] + [[0] * 64] * 7, [0] * 8)[0] == 1   # 512/1024 = 0.5 → 1 (0에서 먼 쪽)
    print(f"PYTHON_MLP_SELF_TEST_PASS cases={len(cs)} shift={SHIFT} saturate=1 relu_zero=1 round_half=1")
    return cs


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--out"); ap.add_argument("--actual")
    a = ap.parse_args()
    cs = selftest()
    if a.out and not a.actual:
        with open(Path(a.out) / "vectors.txt", "w") as f:
            for n, (name, x, w1, b1, w2, b2) in enumerate(cs):
                f.write(f"{n} {name}\n" + " ".join(map(str, x)) + "\n")
                for blk in (w1, [b1], w2, [b2]):
                    f.writelines(" ".join(map(str, r)) + "\n" for r in blk)
    if a.actual:
        rows = {int(p[0]): [int(v) for v in p[1:]] for p in (ln.split() for ln in open(a.actual))}
        for n, (name, x, w1, b1, w2, b2) in enumerate(cs):
            s = mlp_64_8_10(x, w1, b1, w2, b2, SHIFT)
            sat = int(any(round_shift_away(max(0, dot_signed(x, w) + b), SHIFT) > 127 for w, b in zip(w1, b1)))
            want = hidden(x, w1, b1) + s + [argmax_lowest(s), sat, (model_sum(w1, b1) + model_sum(w2, b2)) & 0xFFFFFFFF]
            if rows.get(n) != want:
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL case={name}\n rtl={rows.get(n)}\n py ={want}")
        print(f"PYTHON_RTL_COMPARE_PASS cases={len(cs)} hidden={8 * len(cs)} scores={10 * len(cs)} classes={len(cs)}")


if __name__ == "__main__":
    main()

07코드 읽기 포인트

위치내용
은닉 계산식은닉 = sat_int8(round(ReLU(점수) / 2^SHIFT)), SHIFT=10. 반올림은 0에서 먼 쪽(05장 fixed_quantizer, Python round_shift_away)입니다. ReLU 뒤라 은닉은 0~127입니다.
QUANT 단계 타이밍h_q 0~7에 양자화기 입력을 넣고, 1클록 뒤 출력을 q_idx 칸에 저장합니다. h_q==8에서 마지막 저장과 함께 hidden_valid_o를, h_q==9에서 출력층을 시작합니다.
출력층 입력출력층(8→10)의 입력 읽기 포트는 은닉 버퍼 hid_q에 연결되고 1클록 지연으로 응답합니다. 15장 코드를 바꾸지 않고 입력원만 바꿉니다.
층 선택 쓰기w_layer_i로 층을 고릅니다. 실행 중에는 상위에서 쓰기를 막습니다. 출력층 자체는 1층이 도는 동안 쉬고 있어 15장 잠금만으로는 막히지 않기 때문입니다.
argmax 분리npu_argmax는 점수 N개를 한 클록에 하나씩 비교합니다(동점 → 낮은 번호). 17장과 27장이 같은 모듈을 씁니다.
