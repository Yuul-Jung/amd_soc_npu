#!/usr/bin/env python3
"""28장 PC 전처리: 손글씨 영상(PGM 또는 28×28 숫자 표) → MNIST와 같은 모양(20×20 상자에 맞추고 무게중심을 28×28 가운데로)
→ 784 INT8(13·18장 계약) 또는 8×8 64 INT8(13장). 표준 라이브러리만 쓴다.

검사(--selftest): 실제 MNIST 시험 그림을 일부러 옮기고 키운 뒤(바깥 64×64 캔버스) 전처리로 되돌려
18장 INT8 모델 정확도가 원래 그림과 얼마나 같은지 본다."""
from __future__ import annotations
import argparse
import random
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT / "host_tools"), str(ROOT / "ch13" / "reference"), str(ROOT / "ch18" / "reference")]


def read_pgm(path):
    data = Path(path).read_bytes()
    tok, i = [], 0
    while len(tok) < 4:                                  # P2/P5, 너비, 높이, 최대값 (주석 # 무시)
        while data[i:i + 1].isspace(): i += 1
        if data[i:i + 1] == b"#":
            while data[i:i + 1] not in (b"\n", b""): i += 1
            continue
        j = i
        while not data[j:j + 1].isspace(): j += 1
        tok.append(data[i:j].decode()); i = j
    kind, w, h, mx = tok[0], int(tok[1]), int(tok[2]), int(tok[3])
    vals = list(data[i + 1:i + 1 + w * h]) if kind == "P5" else list(map(int, data[i:].split()))[:w * h]
    img = [[vals[r * w + c] * 255 // mx for c in range(w)] for r in range(h)]
    if sum(map(sum, img)) > 127 * w * h:                 # 흰 바탕에 검은 글씨면 뒤집는다 (MNIST는 검은 바탕·흰 글씨)
        img = [[255 - v for v in row] for row in img]
    return img


def resize(img, nw, nh):
    """면적 평균 축소·확대 (정수 좌표, 단순 박스 필터)."""
    h, w = len(img), len(img[0])
    out = []
    for r in range(nh):
        y0, y1 = r * h // nh, max(r * h // nh + 1, (r + 1) * h // nh)
        row = []
        for c in range(nw):
            x0, x1 = c * w // nw, max(c * w // nw + 1, (c + 1) * w // nw)
            vals = [img[y][x] for y in range(y0, y1) for x in range(x0, x1)]
            row.append(sum(vals) // len(vals))
        out.append(row)
    return out


def mnist_like(img):
    """글씨 상자를 잘라 긴 변 20화소로 맞추고, 무게중심이 28×28 가운데(14,14)에 오게 놓는다."""
    ys = [y for y, row in enumerate(img) if any(v > 30 for v in row)]
    xs = [x for x in range(len(img[0])) if any(img[y][x] > 30 for y in range(len(img)))]
    if not ys:
        return [[0] * 28 for _ in range(28)]
    crop = [row[xs[0]:xs[-1] + 1] for row in img[ys[0]:ys[-1] + 1]]
    h, w = len(crop), len(crop[0])
    s = 20 / max(h, w)
    nh, nw = max(1, round(h * s)), max(1, round(w * s))
    small = resize(crop, nw, nh)
    tot = sum(map(sum, small)) or 1
    cy = sum(y * v for y, row in enumerate(small) for v in row) / tot
    cx = sum(x * v for row in small for x, v in enumerate(row)) / tot
    oy, ox = round(14 - cy - 0.5), round(14 - cx - 0.5)
    out = [[0] * 28 for _ in range(28)]
    for y, row in enumerate(small):
        for x, v in enumerate(row):
            if 0 <= y + oy < 28 and 0 <= x + ox < 28:
                out[y + oy][x + ox] = v
    return out


def to_x784(img28):
    return [min(127, (p + 1) >> 1) for row in img28 for p in row]


def selftest(n):
    from mnist_idx import images, labels
    from precision_rtl import load, scores
    from p3_integer_reference import argmax_lowest
    w, b, _ = load(8)
    imgs, labs = images("test"), labels("test")
    rng = random.Random(28)
    base = moved = 0
    for k in range(n):
        img = [list(imgs[k][r * 28:(r + 1) * 28]) for r in range(28)]
        base += argmax_lowest(scores(to_x784(img), w, b)) == labs[k]
        big = resize(img, 28 * 2, 28 * 2)                # 두 배로 키워
        canvas = [[0] * 64 for _ in range(64)]
        oy, ox = rng.randint(0, 8), rng.randint(0, 8)    # 아무 데나 놓는다
        for y in range(56):
            for x in range(56):
                canvas[y + oy][x + ox] = big[y][x]
        moved += argmax_lowest(scores(to_x784(mnist_like(canvas)), w, b)) == labs[k]
    print(f"PREPROCESS_CHECK original={100 * base / n:.1f}% moved_scaled_then_preprocessed={100 * moved / n:.1f}% images={n}")
    ok = moved >= base * 0.9
    print("PREPROCESS_HOST_PASS" if ok else "PREPROCESS_HOST_FAIL")
    return ok


if __name__ == "__main__":
    ap = argparse.ArgumentParser(); ap.add_argument("--selftest", type=int); ap.add_argument("pgm", nargs="?")
    a = ap.parse_args()

        sys.exit(0 if selftest(a.selftest) else 1)
    x = to_x784(mnist_like(read_pgm(a.pgm)))
    print(" ".join(map(str, x)))

/* 28장 UART 그림 틀: 'N' + 784바이트(INT8 0~127) + 검사합(784바이트 합의 하위 8비트) = 786바이트 */
#ifndef PROTO_H
#define PROTO_H
#include <stdint.h>

#define PROTO_N 784
typedef enum { PROTO_BUSY = 0, PROTO_FRAME = 1, PROTO_ERR_SUM = -1, PROTO_ERR_RANGE = -2 } proto_ev_t;

typedef struct {
    uint32_t pos;          /* 0: 'N' 기다림, 1~784: 받은 화소 수 + 1 */
    uint8_t  sum, bad;
    uint8_t  x[PROTO_N];
} proto_t;

void       proto_reset(proto_t *p);
proto_ev_t proto_feed(proto_t *p, uint8_t c);

#endif
ch28/sw/proto.c · c · 원본 · PC 시험 통과 2026-11-19 · 보드 ⚠ 미검증
/* 28장 틀 해석: 바이트 하나씩 넣는다. 'N' 전의 바이트는 버리고(잡음·줄바꿈), 틀이 끝나면 검사합·범위를 판정한다. */
#include "proto.h"

void proto_reset(proto_t *p) { p->pos = 0; p->sum = 0; p->bad = 0; }

proto_ev_t proto_feed(proto_t *p, uint8_t c)
{
    if (p->pos == 0) {
        if (c == 'N') { p->pos = 1; p->sum = 0; p->bad = 0; }
        return PROTO_BUSY;
    }
    if (p->pos <= PROTO_N) {
        p->x[p->pos - 1] = c;
        p->sum = (uint8_t)(p->sum + c);
        if (c > 127) p->bad = 1;
        p->pos++;
        return PROTO_BUSY;
    }
    p->pos = 0;                                     /* 검사합 바이트 */
    if (c != p->sum) return PROTO_ERR_SUM;
    if (p->bad) return PROTO_ERR_RANGE;
    return PROTO_FRAME;
}

/* 28장 AXI UART Lite 직접 접근 (PG142: RX FIFO 0x00, TX FIFO 0x04, STAT 0x08 — bit0 RX 있음, bit3 TX 가득) */
#include "uart_io.h"
#include "xil_io.h"

#ifndef UART_BASE
#define UART_BASE 0x40600000u   /* ⚠ Vivado 기본값. xparameters.h의 UART Lite 기준 주소로 바꾼다 */
#endif

int uart_getc(void)
{
    if ((Xil_In32(UART_BASE + 0x08u) & 0x01u) == 0u) return -1;
    return (int)(Xil_In32(UART_BASE + 0x00u) & 0xFFu);
}

void uart_puts(const char *s)
{
    while (*s) {
        while (Xil_In32(UART_BASE + 0x08u) & 0x08u) { }
        Xil_Out32(UART_BASE + 0x04u, (uint8_t)*s++);
    }
}
ch28/sw/main.c · c · 원본 · PC 시험 통과 2026-11-19 · 보드 ⚠ 미검증
/* 28장 손글씨 분류기: PC가 보낸 784화소 틀을 받아 NPU로 분류하고, UART로 한 줄 응답 + LED·7세그먼트 표시.
 * 응답: "CLASS c S s0 s1 … s9" / "ERR SUM" / "ERR RANGE" / "ERR NPU n" */
#include "npu_drv.h"
#include "board_io.h"
#include "proto.h"
#include "uart_io.h"
#include "model_int8.h"

static char *put_int(char *p, int32_t v)
{
    char t[12]; int n = 0; uint32_t u = v < 0 ? (uint32_t)(-(int64_t)v) : (uint32_t)v;
    if (v < 0) *p++ = '-';
    do { t[n++] = (char)('0' + u % 10u); u /= 10u; } while (u);
    while (n) *p++ = t[--n];
    return p;
}

static void answer(const uint8_t x[PROTO_N])
{
    npu_result_t r; char line[160], *p = line;
    npu_status_t s = npu_load_input(x);
    if (s == NPU_OK) s = npu_run(&r, 100000u);
    if (s != NPU_OK) {
        led_write(led_word(0, 1));
        p = put_int(line + 8, (int32_t)s); *p = 0;
        for (int i = 0; i < 8; i++) line[i] = "ERR NPU "[i];
        uart_puts(line); uart_puts("\r\n"); return;
    }
    led_write(led_word(r.cls, 0));
    *p++ = 'C'; *p++ = 'L'; *p++ = 'A'; *p++ = 'S'; *p++ = 'S'; *p++ = ' ';
    p = put_int(p, (int32_t)r.cls); *p++ = ' '; *p++ = 'S';
    for (int k = 0; k < NPU_N_OUT; k++) { *p++ = ' '; p = put_int(p, r.score[k]); }
    *p++ = '\r'; *p++ = '\n'; *p = 0;
    uart_puts(line);
}

int main(void)
{
    static proto_t pr;
    uart_puts("=== ch28 digit classifier: send 'N' + 784 bytes + sum ===\r\n");
    if (npu_init() != NPU_OK) { led_write(led_word(0, 1)); uart_puts("ERR NPU ID\r\n"); return 1; }
    npu_load_model(MODEL_W, MODEL_B);
    led_write(0);
    proto_reset(&pr);
    for (;;) {
        int c = uart_getc();
        if (c == -2) return 0;                      /* PC 시험: 입력 끝 */
        if (c < 0) continue;
        proto_ev_t ev = proto_feed(&pr, (uint8_t)c);
        if (ev == PROTO_FRAME) answer(pr.x);
        else if (ev == PROTO_ERR_SUM) { led_write(led_word(0, 1)); uart_puts("ERR SUM\r\n"); }
        else if (ev == PROTO_ERR_RANGE) { led_write(led_word(0, 1)); uart_puts("ERR RANGE\r\n"); }
    }
}

#!/usr/bin/env python3
"""28장 PC 쪽 프로그램: 손글씨 PGM을 전처리해 보드로 보내고 응답을 PC 정수 계산과 비교한다.
  보드:  python3 ch28/host/digit_client.py --port /dev/ttyUSB1 그림.pgm   (pyserial 필요)
  시험:  --frames 파일  : 보낼 바이트를 파일로 쓴다 (05절 시뮬레이션 입력)
         --check 응답파일 : 응답 줄을 PC 계산과 비교한다"""
from __future__ import annotations
import argparse
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT / "ch28" / "reference"), str(ROOT / "ch18" / "reference")]
from preprocess import mnist_like, read_pgm, to_x784     # noqa: E402
from precision_rtl import load, scores                    # noqa: E402
from p3_integer_reference import argmax_lowest            # noqa: E402


def frame(x):
    assert len(x) == 784
    return b"N" + bytes(x) + bytes([sum(x) & 0xFF])


def expect(x):
    w, b, _ = load(8)
    s = scores(x, w, b)
    return f"CLASS {argmax_lowest(s)} S " + " ".join(map(str, s))


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("pgm", nargs="*"); ap.add_argument("--port"); ap.add_argument("--frames"); ap.add_argument("--check")
    a = ap.parse_args()
    xs = [to_x784(mnist_like(read_pgm(p))) for p in a.pgm]
    if a.frames:                                          # 05절 시험 틀: 정상 틀들, 검사합 틀림, 범위 밖, 잡음 뒤 정상
        bad_sum = bytearray(frame(xs[0])); bad_sum[-1] ^= 0xFF
        bad_rng = bytearray(frame(xs[1])); bad_rng[5] = 200; bad_rng[-1] = (sum(bad_rng[1:-1])) & 0xFF
        data = b"".join(frame(x) for x in xs) + bytes(bad_sum) + bytes(bad_rng) + b"\r\n?junk" + frame(xs[-1])
        Path(a.frames).write_bytes(data)
        Path(a.frames + ".expect").write_text("\n".join([expect(x) for x in xs] + ["ERR SUM", "ERR RANGE", expect(xs[-1])]) + "\n")
        print(f"CLIENT_FRAMES bytes={len(data)} frames={len(xs) + 3} (정상 {len(xs)}, 검사합 틀림 1, 범위 밖 1, 잡음 뒤 정상 1)")
    if a.port:
        import serial                                     # pyserial (보드 실행 때만)
        with serial.Serial(a.port, 115200, timeout=5) as s:
            for p, x in zip(a.pgm, xs):
                s.write(frame(x))
                got = s.readline().decode().strip()
                print(f"{p}: board='{got[:40]}…' pc='{expect(x)[:40]}…' {'SAME' if got == expect(x) else 'DIFF'}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""28장: 펌웨어 응답(PC 실행) + RTL 재생 읽기 값으로 실제 보드가 보낼 응답 줄을 만들어 PC 기대 줄과 비교한다."""
import sys

actual, host_out, expect = sys.argv[1:4]
runs, cur = [], None
for op, off, val in (l.split() for l in open(actual)):
    off, val = int(off), int(val)
    if op == "P":
        cur = {"s": []}; runs.append(cur)
    elif cur is not None and 0x40 <= off <= 0x64:
        cur["s"].append(val - (1 << 32) if val >= 1 << 31 else val)
    elif cur is not None and off == 0x08:
        cur["c"] = val
lines = [l.strip() for l in open(host_out) if l.strip() and not l.startswith("===")]
it = iter(runs)
board = [(lambda r: f"CLASS {r['c']} S " + " ".join(map(str, r["s"])))(next(it)) if l.startswith("CLASS") else l for l in lines]
want = [l.strip() for l in open(expect) if l.strip()]
for k, (g, w) in enumerate(zip(board, want)):
    print(f"REPLY {k}: {g[:60]}{'…' if len(g) > 60 else ''} {'OK' if g == w else 'DIFF'}")
ok = board == want
print(f"{'PROTOCOL_RTL_COMPARE_PASS' if ok else 'PROTOCOL_RTL_COMPARE_FAIL'} replies={len(board)} inferences={len(runs)}")
sys.exit(0 if ok else 1)

07코드 읽기 포인트

위치내용
MNIST 모양 맞추기학습 그림과 같은 모양이어야 정확도가 유지됩니다. 글씨 상자를 잘라 긴 변을 20화소로 줄이고, 무게중심을 28×28의 (14,14)에 놓습니다. 흰 바탕이면 먼저 뒤집습니다.
UART 틀'N' + 784바이트 + 검사합(합의 하위 8비트). 'N' 앞의 바이트(줄바꿈·잡음)는 버리므로 틀이 어긋나도 다음 'N'에서 다시 맞춰집니다. 값이 127을 넘으면 13장 계약 위반으로 ERR RANGE를 보냅니다.
보드 = PC보드 응답은 PC 정수 계산과 글자 하나까지 같아야 합니다. 다르면 전처리가 아니라 전송·하드웨어 문제입니다.
printf 없이응답 줄은 작은 정수 변환 함수로 직접 만들어 xil_printf 없이 UART Lite로 보냅니다. 같은 코드가 PC에서는 표준 출력으로 갑니다.
