03RTL 작성

아래 파일을 경로 그대로 만들고 전체 코드를 입력합니다.

ch15/rtl/npu_dense_core.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 15장 Dense 연산 코어: 08장 4PE로 뉴런 4개씩 그룹을 돌려 내적만 계산한다 (Bias는 바깥에서 더한다).
// 입력·가중치는 바깥 메모리에서 1클록 지연 읽기 포트로 받는다 → 16장(2층)·17장(타일)이 주소만 바꿔 재사용.
// 뉴런 n → 그룹 n/4, 차선 n%4. N_OUT을 넘는 차선은 가중치 0을 곱하고 결과를 버린다.
module npu_dense_core #(
    parameter integer N_OUT = 10,          // 1~16
    parameter integer ACC_WIDTH = 32
) (
    input  wire                        clk,
    input  wire                        resetn,
    input  wire                        start_i,
    input  wire                  [6:0] len_i,         // 이번 실행 원소 수 1~64
    output wire                        x_rd_en_o,
    output wire                  [5:0] x_rd_idx_o,
    input  wire signed           [7:0] x_data_i,
    input  wire                        x_valid_i,
    output wire                        w_rd_en_o,
    output wire                  [1:0] w_rd_group_o,
    output wire                  [5:0] w_rd_idx_o,
    input  wire                 [31:0] w_data_i,      // 차선 0~3 × INT8
    input  wire                  [3:0] w_valid_i,
    output reg                         busy_o,
    output reg                         done_o,
    output reg                         error_o,
    output wire [N_OUT*ACC_WIDTH-1:0]  dots_o
);
    localparam integer GROUPS = (N_OUT + 3) / 4;
    localparam IDLE=0, LAUNCH=1, READ=2, WAIT_READ=3, SEND=4, WAIT_PE=5;
    reg [2:0] state;
    reg [1:0] group_q;
    reg [5:0] idx_q;
    reg [6:0] len_q;
    reg signed [ACC_WIDTH-1:0] dot_q [0:N_OUT-1];
    integer i;

    assign x_rd_en_o = (state == READ);
    assign x_rd_idx_o = idx_q;
    assign w_rd_en_o = (state == READ);
    assign w_rd_group_o = group_q;
    assign w_rd_idx_o = idx_q;

    wire [31:0] w_bus;
    genvar g;
    generate for (g = 0; g < 4; g = g + 1) begin : lane
        // 마지막 그룹에서 뉴런 번호가 N_OUT 이상인 차선은 0을 곱한다.
        assign w_bus[g*8 +: 8] = (group_q * 4 + g >= N_OUT) ? 8'd0 : w_data_i[g*8 +: 8];
    end endgenerate
    generate for (g = 0; g < N_OUT; g = g + 1) begin : out
        assign dots_o[g*ACC_WIDTH +: ACC_WIDTH] = dot_q[g];
    end endgenerate

    wire [3:0] pe_ready;
    wire pe_done;
    wire [4*ACC_WIDTH-1:0] pe_res;
    pe_array4 #(.DATA_WIDTH(8), .ACC_WIDTH(ACC_WIDTH)) pe (
        .clk(clk), .resetn(resetn), .start_i(state == LAUNCH), .length_i({2'b00, len_q}),
        .valid_i({4{state == SEND}}), .a_i({4{x_data_i}}), .b_i(w_bus),
        .ready_o(pe_ready), .busy_o(), .done_o(pe_done), .error_o(), .pe_results_o(pe_res), .sum_o());

    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE; busy_o <= 0; done_o <= 0; error_o <= 0;
            group_q <= 0; idx_q <= 0; len_q <= 0;
            for (i = 0; i < N_OUT; i = i + 1) dot_q[i] <= 0;
        end else begin
            done_o <= 0; error_o <= 0;
            case (state)
                IDLE: if (start_i) begin
                    if (len_i == 0 || len_i > 7'd64) error_o <= 1;
                    else begin busy_o <= 1; len_q <= len_i; group_q <= 0; idx_q <= 0; state <= LAUNCH; end
                end
                LAUNCH: state <= READ;
                READ: state <= WAIT_READ;
                WAIT_READ: if (x_valid_i && (&w_valid_i)) state <= SEND;
                    else begin state <= IDLE; busy_o <= 0; error_o <= 1; end
                SEND: if (&pe_ready) begin
                    if (idx_q == len_q - 7'd1) state <= WAIT_PE;
                    else begin idx_q <= idx_q + 6'd1; state <= READ; end
                end
                WAIT_PE: if (pe_done) begin
                    for (i = 0; i < N_OUT; i = i + 1)
                        if (i / 4 == group_q) dot_q[i] <= $signed(pe_res[(i % 4)*ACC_WIDTH +: ACC_WIDTH]);
                    idx_q <= 0;
                    if (group_q == GROUPS - 1) begin state <= IDLE; busy_o <= 0; done_o <= 1; end
                    else begin group_q <= group_q + 2'd1; state <= LAUNCH; end
                end
                default: begin state <= IDLE; busy_o <= 0; error_o <= 1; end
            endcase
        end
    end
endmodule
`default_nettype wire
ch15/rtl/npu_dense_store.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 15장 Dense 모델 저장소: 가중치(차선별 04장 BRAM 4개)·Bias 레지스터·적재 검증합.
// 가중치 주소 = {그룹, 원소}. 검증합 = 받아들인 가중치(부호 확장)와 Bias의 합 (mod 2^32) → PC 값과 비교.
module npu_dense_store #(
    parameter integer N_IN = 64,           // 2의 거듭제곱, 2~64
    parameter integer N_OUT = 10,          // 1~16
    parameter integer ACC_WIDTH = 32
) (
    input  wire                        clk,
    input  wire                        resetn,
    input  wire                        clear_i,      // 검증합·오류 지움
    input  wire                        wr_lock_i,    // 1이면 쓰기 무시 (연산 중)
    input  wire                        w_wr_en_i,
    input  wire                  [3:0] w_neuron_i,
    input  wire                  [5:0] w_addr_i,
    input  wire                  [7:0] w_data_i,
    input  wire                        b_wr_en_i,
    input  wire                  [3:0] b_neuron_i,
    input  wire        [ACC_WIDTH-1:0] b_data_i,
    input  wire                        w_rd_en_i,
    input  wire                  [1:0] w_rd_group_i,
    input  wire                  [5:0] w_rd_idx_i,
    output wire                 [31:0] w_data_o,
    output wire                  [3:0] w_valid_o,
    output wire [N_OUT*ACC_WIDTH-1:0]  bias_o,
    output reg                  [31:0] sum_o,
    output reg                         wr_err_o      // 범위 밖 주소·뉴런 쓰기 (무시하고 표시)
);
    localparam integer AW = $clog2(N_IN);
    reg signed [ACC_WIDTH-1:0] bias_q [0:N_OUT-1];
    integer i;
    wire w_in_range = (w_neuron_i < N_OUT) && (w_addr_i < N_IN);
    wire b_in_range = (b_neuron_i < N_OUT);
    wire w_fire = w_wr_en_i && !wr_lock_i && w_in_range;
    wire b_fire = b_wr_en_i && !wr_lock_i && b_in_range;

    genvar g;
    generate for (g = 0; g < 4; g = g + 1) begin : lane
        bram_sync_buffer #(.DATA_WIDTH(8), .ADDR_WIDTH(2 + AW)) weight_ram (
            .clk(clk), .resetn(resetn),
            .wr_en(w_fire && w_neuron_i[1:0] == g), .wr_addr({w_neuron_i[3:2], w_addr_i[AW-1:0]}), .wr_data(w_data_i),
            .rd_en(w_rd_en_i), .rd_addr({w_rd_group_i, w_rd_idx_i[AW-1:0]}),
            .rd_data(w_data_o[g*8 +: 8]), .rd_valid(w_valid_o[g]));
    end endgenerate
    generate for (g = 0; g < N_OUT; g = g + 1) begin : b
        assign bias_o[g*ACC_WIDTH +: ACC_WIDTH] = bias_q[g];
    end endgenerate

    always @(posedge clk) begin
        if (!resetn) begin
            sum_o <= 0; wr_err_o <= 0;
            for (i = 0; i < N_OUT; i = i + 1) bias_q[i] <= 0;
        end else begin
            if (clear_i) begin
                sum_o <= 0; wr_err_o <= 0;
            end else begin
                if ((w_wr_en_i && !w_in_range) || (b_wr_en_i && !b_in_range)) wr_err_o <= 1;
                sum_o <= sum_o + (w_fire ? {{24{w_data_i[7]}}, w_data_i} : 32'd0)
                               + (b_fire ? b_data_i[31:0] : 32'd0);
            end
            if (b_fire) bias_q[b_neuron_i] <= b_data_i;
        end
    end
endmodule
`default_nettype wire
ch15/rtl/npu_dense.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 15장 Dense 층 N_IN→N_OUT: 모델 저장소 + 연산 코어 + Bias 더하기(1클록).
// 입력 x는 바깥 메모리에서 읽는다 (16장: 은닉 버퍼, 13장: 입력 적재기).
module npu_dense #(
    parameter integer N_IN = 64,
    parameter integer N_OUT = 10,
    parameter integer ACC_WIDTH = 32
) (
    input  wire                        clk,
    input  wire                        resetn,
    input  wire                        model_clear_i,
    input  wire                        w_wr_en_i,
    input  wire                  [3:0] w_neuron_i,
    input  wire                  [5:0] w_addr_i,
    input  wire                  [7:0] w_data_i,
    input  wire                        b_wr_en_i,
    input  wire                  [3:0] b_neuron_i,
    input  wire        [ACC_WIDTH-1:0] b_data_i,
    output wire                 [31:0] model_sum_o,
    output wire                        model_err_o,
    output wire                        x_rd_en_o,
    output wire                  [5:0] x_rd_idx_o,
    input  wire signed           [7:0] x_data_i,
    input  wire                        x_valid_i,
    input  wire                        start_i,
    output wire                        busy_o,
    output reg                         done_o,
    output wire                        error_o,
    output reg  [N_OUT*ACC_WIDTH-1:0]  scores_o
);
    localparam [6:0] LEN = N_IN;          // 이 층은 늘 N_IN개 전체를 쓴다 (17장 타일은 코어 len_i를 직접 바꾼다)
    wire w_rd_en; wire [1:0] w_rd_group; wire [5:0] w_rd_idx;
    wire [31:0] w_data; wire [3:0] w_valid;
    wire [N_OUT*ACC_WIDTH-1:0] bias, dots;
    wire core_busy, core_done;
    reg  add_q;                            // 코어 완료 후 Bias 더하는 1클록
    integer i;

    assign busy_o = core_busy || add_q;

    npu_dense_store #(.N_IN(N_IN), .N_OUT(N_OUT), .ACC_WIDTH(ACC_WIDTH)) store (
        .clk(clk), .resetn(resetn), .clear_i(model_clear_i), .wr_lock_i(busy_o),
        .w_wr_en_i(w_wr_en_i), .w_neuron_i(w_neuron_i), .w_addr_i(w_addr_i), .w_data_i(w_data_i),
        .b_wr_en_i(b_wr_en_i), .b_neuron_i(b_neuron_i), .b_data_i(b_data_i),
        .w_rd_en_i(w_rd_en), .w_rd_group_i(w_rd_group), .w_rd_idx_i(w_rd_idx),
        .w_data_o(w_data), .w_valid_o(w_valid), .bias_o(bias), .sum_o(model_sum_o), .wr_err_o(model_err_o));

    npu_dense_core #(.N_OUT(N_OUT), .ACC_WIDTH(ACC_WIDTH)) core (
        .clk(clk), .resetn(resetn), .start_i(start_i && !busy_o), .len_i(LEN),
        .x_rd_en_o(x_rd_en_o), .x_rd_idx_o(x_rd_idx_o), .x_data_i(x_data_i), .x_valid_i(x_valid_i),
        .w_rd_en_o(w_rd_en), .w_rd_group_o(w_rd_group), .w_rd_idx_o(w_rd_idx), .w_data_i(w_data), .w_valid_i(w_valid),
        .busy_o(core_busy), .done_o(core_done), .error_o(error_o), .dots_o(dots));

    always @(posedge clk) begin
        if (!resetn) begin
            add_q <= 0; done_o <= 0; scores_o <= 0;
        end else begin
            add_q <= core_done; done_o <= add_q;
            if (core_done)
                for (i = 0; i < N_OUT; i = i + 1)
                    scores_o[i*ACC_WIDTH +: ACC_WIDTH] <= $signed(dots[i*ACC_WIDTH +: ACC_WIDTH]) + $signed(bias[i*ACC_WIDTH +: ACC_WIDTH]);
        end
    end
endmodule
`default_nettype wire

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. 같은 TB를 -P tb_npu_dense.N_IN=… -P tb_npu_dense.N_OUT=…로 세 번 컴파일합니다. 크기마다 무작위 가중치·Bias 6사례와 극단값 1사례(최대 크기 곱, Bias 224)를 넣고, 점수와 적재 검증합을 Python과 비교합니다.

ch15/tb/tb_npu_dense.sv · systemverilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
// 15장 TB: N_IN·N_OUT을 바꿔 (64→10, 64→8, 8→10) 같은 Dense 층을 시험한다.
// 입력 x는 TB 메모리(1클록 지연)가 공급하고, 점수와 적재 검증합을 파일로 돌려준다.
module tb_npu_dense;
    parameter integer N_IN = 64, N_OUT = 10;
    localparam AW = 32;
    reg clk=0, rn=0, mc=0, wwe=0, bwe=0, start=0, drop=0;
    reg [3:0] wn=0, bn=0; reg [5:0] wa=0; reg [7:0] wd=0; reg [AW-1:0] bd=0;
    reg signed [7:0] xd=0; reg xv=0;
    wire xre, busy, done, err, merr; wire [5:0] xi; wire [31:0] msum; wire [N_OUT*AW-1:0] scores;
    npu_dense #(.N_IN(N_IN), .N_OUT(N_OUT), .ACC_WIDTH(AW)) dut(.clk(clk), .resetn(rn), .model_clear_i(mc),
        .w_wr_en_i(wwe), .w_neuron_i(wn), .w_addr_i(wa), .w_data_i(wd), .b_wr_en_i(bwe), .b_neuron_i(bn), .b_data_i(bd),

        $fclose(o); $finish;
    end

#!/usr/bin/env python3
"""15장 기준 모델: Dense N_IN→N_OUT 점수 = 내적 + Bias, 적재 검증합 = Σ가중치 + ΣBias (mod 2^32).

64→10은 host_tools의 linear_64x10, 64→8·8→10(16장용)은 같은 dot_signed로 계산한다.
"""
from __future__ import annotations
import argparse
import random
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT / "ch13" / "reference"), str(ROOT / "host_tools")]
from mnist8x8_reference import draw_digit, features   # noqa: E402
from p3_integer_reference import dot_signed, linear_64x10  # noqa: E402

CONFIGS = [(64, 10), (64, 8), (8, 10)]
BIAS_LIMIT = 1 << 24


def dense(x: list[int], w: list[list[int]], b: list[int]) -> list[int]:
    if len(x) == 64 and len(w) == 10:
        return linear_64x10(x, w, b)
    return [dot_signed(x, row) + bias for row, bias in zip(w, b)]


def model_sum(w: list[list[int]], b: list[int]) -> int:
    return (sum(map(sum, w)) + sum(b)) & 0xFFFFFFFF


def cases(n_in: int, n_out: int) -> list:
    rng = random.Random(1500 + 100 * n_in + n_out)
    out = []
    for k in range(6):
        if n_in == 64:
            x = features(draw_digit(k))
        else:  # 16장 은닉층 출력 범위 (INT8, ReLU 뒤라 0 이상이지만 부호 검사를 위해 음수도 섞음)
            x = [rng.randint(-128, 127) for _ in range(n_in)]
        w = [[rng.randint(-128, 127) for _ in range(n_in)] for _ in range(n_out)]
        b = [rng.randint(-BIAS_LIMIT, BIAS_LIMIT) for _ in range(n_out)]
        out.append((f"random_{k}", x, w, b))
    ext = 127 if n_in == 64 else -128
    out.append(("extreme", [ext] * n_in, [[-128] * n_in for _ in range(n_out)], [BIAS_LIMIT] * n_out))
    return out


def selftest() -> None:
    for n_in, n_out in CONFIGS:
        for name, x, w, b in cases(n_in, n_out):
            s = dense(x, w, b)
            assert len(s) == n_out and all(-(1 << 31) <= v < (1 << 31) for v in s), (n_in, n_out, name)
    x = [1] * 8
    assert dense(x, [[1] * 8] * 10, [0] * 10) == [8] * 10
    print(f"PYTHON_DENSE_SELF_TEST_PASS configs={'/'.join(f'{a}x{b}' for a, b in CONFIGS)} cases_each=7")


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--cfg", help="N_IN,N_OUT")
    ap.add_argument("--out", help="vectors.txt 파일 경로")
    ap.add_argument("--actual", help="RTL actual.txt (Python 값과 비교)")
    a = ap.parse_args()
    if not a.cfg:
        selftest()
        return
    n_in, n_out = map(int, a.cfg.split(","))
    cs = cases(n_in, n_out)
    if a.out:
        with open(a.out, "w") as f:
            for n, (name, x, w, b) in enumerate(cs):
                f.write(f"{n} {name}\n" + " ".join(map(str, x)) + "\n")
                f.writelines(" ".join(map(str, row)) + "\n" for row in w)
                f.write(" ".join(map(str, b)) + "\n")
    if a.actual:
        rows = {int(p[0]): [int(v) for v in p[1:]] for p in (ln.split() for ln in open(a.actual))}
        for n, (name, x, w, b) in enumerate(cs):
            want = dense(x, w, b) + [model_sum(w, b)]
            if rows.get(n) != want:
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL cfg={n_in}x{n_out} case={name} rtl={rows.get(n)} python={want}")
        print(f"PYTHON_RTL_COMPARE_PASS cfg={n_in}x{n_out} cases={len(cs)} scores={n_out * len(cs)} model_sum=ok")


if __name__ == "__main__":
    main()

07코드 읽기 포인트

위치내용
코어와 저장소 분리npu_dense_core는 내적만 계산하고 입력·가중치를 *_rd_* 포트로 읽습니다. 16장은 은닉 버퍼를, 17장은 타일 주소를 이 포트에 연결합니다. 4PE·MAC는 01·07·08장 파일을 그대로 씁니다.
len_i코어는 실행마다 원소 수(1~64)를 받습니다. 이 층(npu_dense)은 늘 N_IN을 넘기고, 17장은 마지막 타일에 16을 넘깁니다.
적재 검증합받아들인 가중치(8→32비트 부호 확장)와 Bias를 모두 더한 값(mod 232)입니다. PC가 같은 합을 계산해 비교하면 빠진 쓰기·중복 쓰기를 찾습니다. 이 TB도 Bias 쓰기 신호를 한 클록 더 올린 TB 실수를 검증합 불일치로 찾았습니다.
쓰기 잠금busy_o 동안 저장소 쓰기를 막습니다(wr_lock_i). 범위 밖 뉴런·주소 쓰기는 무시하고 model_err_o를 올립니다.
Bias는 마지막에 한 번코어 완료 다음 클록에 점수 = 내적 + Bias를 만들고 그다음 클록에 done_o를 냅니다.
