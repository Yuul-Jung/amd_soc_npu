/* 22장 NPU 레지스터 맵 — 20장 npu_axi_lite.v 머리말 표와 같아야 한다. */
#ifndef NPU_REGS_H
#define NPU_REGS_H

#define NPU_CTRL      0x00u   /* W  bit0 START, bit1 CLR */
#define NPU_STATUS    0x04u   /* R  bit0 BUSY, bit1 DONE, bit2 ERROR */
#define NPU_CLASS     0x08u   /* R  [3:0] */
#define NPU_CYCLES    0x0Cu   /* R  [15:0] */
#define NPU_X_ADDR    0x10u   /* RW [9:0] */
#define NPU_X_DATA    0x14u   /* W  [7:0], X_ADDR +1 */
#define NPU_W_ADDR    0x18u   /* RW [19:16] 클래스, [9:0] 원소 */
#define NPU_W_DATA    0x1Cu   /* W  [7:0], 원소 +1 */
#define NPU_B_SEL     0x20u   /* RW [3:0] */
#define NPU_B_DATA    0x24u   /* W  [31:0], B_SEL +1 */
#define NPU_ID        0x28u   /* R  0x4E505531 */
#define NPU_LENGTH    0x2Cu   /* R  784 */
#define NPU_SCORE(k)  (0x40u + 4u * (unsigned)(k))   /* R  k = 0~9, signed 32비트 */

#define NPU_CTRL_START   (1u << 0)
#define NPU_CTRL_CLR     (1u << 1)
#define NPU_ST_BUSY      (1u << 0)
#define NPU_ST_DONE      (1u << 1)
#define NPU_ST_ERROR     (1u << 2)
#define NPU_ID_VALUE     0x4E505531u
#define NPU_N_IN         784
#define NPU_N_OUT        10

#endif

/* 22장 MicroBlaze V용 레지스터 접근. NPU 기준 주소는 21장 Address Editor 값이며, BSP xparameters.h의 매크로로 바꾼다. */
#include "npu_hal.h"
#include "xil_io.h"
#include "xparameters.h"

#ifndef NPU_BASE
/* ⚠ 모듈 참조 셀의 매크로 이름은 BSP마다 다르다. xparameters.h에서 0x44A00000을 찾아 그 이름으로 바꾼다. */
#define NPU_BASE 0x44A00000u
#endif

void npu_wr(uint32_t off, uint32_t value) { Xil_Out32(NPU_BASE + off, value); }

uint32_t npu_rd(uint32_t off) { return Xil_In32(NPU_BASE + off); }

uint32_t npu_poll(uint32_t off, uint32_t mask, uint32_t limit)
{
    while (limit-- != 0u) {
        uint32_t v = Xil_In32(NPU_BASE + off);
        if ((v & mask) != 0u) return v;
    }
    return 0u;
}

/* 22장 NPU 드라이버: 모델·입력 적재, 시작, 완료 대기(시간 제한), 결과 읽기 */

/* 22장 NPU 드라이버 — 레지스터 순서는 20장 TB와 같다. */
#include "npu_drv.h"
#include "npu_hal.h"

npu_status_t npu_init(void)
{
    if (npu_rd(NPU_ID) != NPU_ID_VALUE) return NPU_ERR_ID;
    npu_wr(NPU_CTRL, NPU_CTRL_CLR);                 /* 이전 DONE·ERROR 지움 */
    return NPU_OK;
}

void npu_load_model(const int8_t w[NPU_N_OUT][NPU_N_IN], const int32_t b[NPU_N_OUT])
{
    for (uint32_t c = 0; c < NPU_N_OUT; c++) {
        npu_wr(NPU_W_ADDR, c << 16);                /* 클래스 c, 원소 0 */
        for (uint32_t i = 0; i < NPU_N_IN; i++)
            npu_wr(NPU_W_DATA, (uint32_t)(uint8_t)w[c][i]);   /* 원소 주소는 하드웨어가 +1 */
    }
    npu_wr(NPU_B_SEL, 0);
    for (uint32_t c = 0; c < NPU_N_OUT; c++)
        npu_wr(NPU_B_DATA, (uint32_t)b[c]);
}

npu_status_t npu_load_input(const uint8_t x[NPU_N_IN])
{
    if (npu_rd(NPU_STATUS) & NPU_ST_BUSY) return NPU_ERR_BUSY;   /* 실행 중 쓰기는 하드웨어가 무시한다 */
    npu_wr(NPU_X_ADDR, 0);
    for (uint32_t i = 0; i < NPU_N_IN; i++)
        npu_wr(NPU_X_DATA, x[i]);
    return NPU_OK;
}

npu_status_t npu_run(npu_result_t *out, uint32_t poll_limit)
{
    if (npu_rd(NPU_STATUS) & NPU_ST_BUSY) return NPU_ERR_BUSY;
    npu_wr(NPU_CTRL, NPU_CTRL_START);
    uint32_t st = npu_poll(NPU_STATUS, NPU_ST_DONE | NPU_ST_ERROR, poll_limit);
    if (st == 0u) return NPU_ERR_TIMEOUT;
    if (st & NPU_ST_ERROR) return NPU_ERR_HW;
    for (uint32_t k = 0; k < NPU_N_OUT; k++)
        out->score[k] = (int32_t)npu_rd(NPU_SCORE(k));
    out->cls = npu_rd(NPU_CLASS) & 0xFu;
    out->cycles = npu_rd(NPU_CYCLES) & 0xFFFFu;
    npu_wr(NPU_CTRL, NPU_CTRL_CLR);
    return NPU_OK;
}

/* 22장 Vitis 애플리케이션: 18장 실제 MNIST INT8 모델과 시험 5장으로 NPU 추론, UART로 점수·클래스 출력 */
#include "npu_drv.h"
#include "model_int8.h"
#ifdef NPU_HOST
#include <stdio.h>
#define PRINT printf
#else
#include "xil_printf.h"
#define PRINT xil_printf
#endif

#define POLL_LIMIT 100000u        /* 추론 7,185클록 동안 STATUS 읽기는 수백 번 → 넉넉한 상한 */

int main(void)
{
    npu_result_t r;
    PRINT("=== ch22 NPU driver (MNIST INT8 784x10) ===\r\n");
    if (npu_init() != NPU_OK) { PRINT("[FAIL] NPU ID mismatch\r\n"); return 1; }
    npu_load_model(MODEL_W, MODEL_B);
    int right = 0;
    for (int n = 0; n < TEST_N; n++) {
        npu_status_t s = npu_load_input(TEST_X[n]);
        if (s == NPU_OK) s = npu_run(&r, POLL_LIMIT);
        if (s != NPU_OK) { PRINT("[FAIL] image %d status %d\r\n", n, (int)s); return 1; }
        PRINT("image %d label %d class %d cycles %d score", n, (int)TEST_LABEL[n], (int)r.cls, (int)r.cycles);
        for (int k = 0; k < NPU_N_OUT; k++) PRINT(" %d", (int)r.score[k]);
        PRINT("\r\n");
        right += (r.cls == (uint32_t)TEST_LABEL[n]);
    }
    PRINT("[PASS] %d/%d correct\r\n", right, TEST_N);
    return 0;
}

/* 22장 PC용 레지스터 접근: 실제 버스 대신 접근 순서를 파일에 기록한다 (RTL 재생용).
 * 형식: W 주소 값 / R 주소 / P 주소 마스크 상한. 읽기 값은 재생할 때 RTL이 정하므로 여기서는
 * 드라이버가 다음으로 넘어갈 수 있는 값만 돌려준다 (ID = 0x4E505531, STATUS = 0, 폴링 = 성공). */
#include <stdio.h>
#include <stdlib.h>
#include "npu_hal.h"
#include "npu_regs.h"

static FILE *trace_fp(void)
{
    static FILE *fp;
    if (!fp) {
        const char *p = getenv("NPU_TRACE");
        fp = p ? fopen(p, "w") : stdout;
        if (!fp) { perror("NPU_TRACE"); exit(2); }
    }
    return fp;
}

void npu_wr(uint32_t off, uint32_t value) { fprintf(trace_fp(), "W %u %u\n", off, value); }

uint32_t npu_rd(uint32_t off)
{
    fprintf(trace_fp(), "R %u\n", off);
    return off == NPU_ID ? NPU_ID_VALUE : 0u;
}

uint32_t npu_poll(uint32_t off, uint32_t mask, uint32_t limit)
{
    fprintf(trace_fp(), "P %u %u %u\n", off, mask, limit);
    return mask & NPU_ST_DONE;
}

{
    fake_reads++;
    return off == NPU_ID ? fake_id : off == NPU_STATUS ? fake_status : 0u;
}

#include <stdio.h>
#include "npu_drv.h"

extern uint32_t fake_id, fake_status, fake_reads;
static int fails;
#define CHECK(c, m) do { if (!(c)) { printf("  FAIL %s\n", m); fails++; } else printf("  ok   %s\n", m); } while (0)

int main(void)
{
    static const uint8_t x[NPU_N_IN];
    npu_result_t r;
    fake_id = 0x12345678u;            CHECK(npu_init() == NPU_ERR_ID, "ID가 다르면 NPU_ERR_ID");
    fake_id = NPU_ID_VALUE;           CHECK(npu_init() == NPU_OK, "ID가 맞으면 NPU_OK");
    fake_status = NPU_ST_BUSY;        CHECK(npu_load_input(x) == NPU_ERR_BUSY, "BUSY 중 입력 적재 거부");
                                      CHECK(npu_run(&r, 10) == NPU_ERR_BUSY, "BUSY 중 시작 거부");

#!/usr/bin/env python3
"""22장: 드라이버 기록을 RTL에서 재생한 결과(actual)에서 그림마다 점수 10개·클래스를 뽑아 18장 기준 모델과 비교한다."""
from __future__ import annotations
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "ch18" / "reference"))
from precision_rtl import load, scores                    # noqa: E402
from p3_integer_reference import argmax_lowest            # noqa: E402


def main(actual, n):
    w, b, cases = load(8)
    reads = [ln.split() for ln in open(actual)]
    images, cur = [], None
    for op, off, val in reads:
        off, val = int(off), int(val)
        if op == "P":
            cur = {"poll": val}; images.append(cur)
        elif cur is not None and 0x40 <= off <= 0x64:
            cur.setdefault("score", []).append(val - (1 << 32) if val >= 1 << 31 else val)
        elif cur is not None and off == 0x08:
            cur["class"] = val
    if len(images) != n:
        raise SystemExit(f"DRIVER_RTL_COMPARE_FAIL images={len(images)} expected={n}")
    for k, img in enumerate(images):
        s = scores(cases[k][2], w, b)
        if img.get("score") != s or img.get("class") != argmax_lowest(s) or not img["poll"] & 2:
            raise SystemExit(f"DRIVER_RTL_COMPARE_FAIL image={k} rtl={img} python={s}")
    print(f"DRIVER_RTL_COMPARE_PASS images={n} scores={10 * n} classes={n}")


if __name__ == "__main__":
    main(sys.argv[1], int(sys.argv[2]))

07코드 읽기 포인트

위치내용
HAL 분리드라이버(npu_drv.c)는 npu_wr·npu_rd·npu_poll만 부릅니다. 보드에서는 Xil_Out32·Xil_In32, PC에서는 기록 또는 가짜 NPU로 바꿔 끼웁니다. 그래서 보드에 올릴 코드와 시험한 코드가 같습니다.
기록 재생PC 실행은 하드웨어 값을 모르므로 읽기는 '다음으로 넘어갈 수 있는 값'만 돌려주고 순서만 남깁니다. 실제 값은 RTL 재생이 정하고, 폴링(P)은 RTL에서 DONE이 올 때까지 다시 읽습니다.
시간 제한npu_poll은 상한(POLL_LIMIT 100,000번)까지만 읽습니다. 추론 7,185클록 동안 STATUS 읽기는 수백 번이므로 정상이라면 상한에 닿지 않고, 클록·리셋이 빠졌을 때 무한 대기 대신 NPU_ERR_TIMEOUT을 돌려줍니다.
주소 매크로모듈 참조 셀의 xparameters.h 매크로 이름은 미리 알 수 없어 NPU_BASE 기본값을 0x44A0_0000으로 두었습니다. 08절 2번에서 실제 이름으로 바꿉니다.
부호 복원점수 레지스터는 32비트 2의 보수입니다. C에서는 (int32_t)로, Python 비교에서는 231 이상이면 232을 빼서 되돌립니다.
