03RTL 작성

아래 파일을 경로 그대로 만들고 전체 코드를 입력합니다.

ch08/rtl/pe_array4.v · verilog · 원본 · RTL_SIM 통과 2026-11-18
`timescale 1ns/1ps
module pe_array4 #(parameter integer DATA_WIDTH=16,ACC_WIDTH=40)(
    input wire clk,resetn,start_i,
    input wire [8:0] length_i,
    input wire [3:0] valid_i,
    input wire [4*DATA_WIDTH-1:0] a_i,b_i,
    output wire [3:0] ready_o,
    output reg busy_o,done_o,error_o,
    output wire [4*ACC_WIDTH-1:0] pe_results_o,
    output reg signed [ACC_WIDTH+1:0] sum_o
);
    wire launch=start_i && !busy_o && length_i<=256;
    wire [3:0] lane_ready,lane_done;
    wire signed [ACC_WIDTH-1:0] lane_result[0:3];
    reg signed [ACC_WIDTH-1:0] saved[0:3];
    reg [3:0] finished;
    wire signed [ACC_WIDTH+1:0] extended[0:3];
    wire signed [ACC_WIDTH+1:0] left_sum,right_sum;
    assign left_sum=extended[0]+extended[1];
    assign right_sum=extended[2]+extended[3];
    assign ready_o=lane_ready & {4{busy_o}};
    genvar g;
    generate for(g=0;g<4;g=g+1)begin: pe
        assign pe_results_o[g*ACC_WIDTH+:ACC_WIDTH]=saved[g];
        assign extended[g]={{2{saved[g][ACC_WIDTH-1]}},saved[g]};
        vector_dot_core #(.DATA_WIDTH(DATA_WIDTH),.ACC_WIDTH(ACC_WIDTH)) dot
          (.clk(clk),.resetn(resetn),.start_i(launch),.length_i(length_i),
           .valid_i(valid_i[g] && busy_o),.a_i(a_i[g*DATA_WIDTH+:DATA_WIDTH]),.b_i(b_i[g*DATA_WIDTH+:DATA_WIDTH]),
           .ready_o(lane_ready[g]),.busy_o(),.done_o(lane_done[g]),.error_o(),.result_o(lane_result[g]));
    end endgenerate
    integer i;
    always @(posedge clk)begin
        if(!resetn)begin
            busy_o<=0;done_o<=0;error_o<=0;sum_o<=0;finished<=0;
            for(i=0;i<4;i=i+1)saved[i]<=0;
        end else begin
            done_o<=0;error_o<=0;
            if(start_i && !busy_o)begin
                if(length_i>256)error_o<=1;
                else begin busy_o<=1;finished<=0;end
            end else if(busy_o)begin
                for(i=0;i<4;i=i+1)if(lane_done[i])begin saved[i]<=lane_result[i];finished[i]<=1;end
                // 서로 다른 완료 에지의 결과를 모은 후 정렬된 합을 등록한다.
                if(&finished)begin sum_o<=left_sum+right_sum;done_o<=1;busy_o<=0;end
            end
        end
    end
endmodule

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. 기준 모델(Python)은 시험 벡터와 정답을 만듭니다.

ch08/tb/tb_pe_array4.sv · systemverilog · 원본 · RTL_SIM 통과 2026-11-18
`timescale 1ns/1ps
module tb_pe_array4;
    parameter DW=16,AW=40;
    reg clk=0,rn=0,start=0;
    reg [8:0] len=0;
    reg [3:0] vi=0,fire;
    reg [4*DW-1:0] a=0,b=0;
    wire [3:0] ready;wire busy,done,error;
    wire [4*AW-1:0] pe;wire signed [AW+1:0] sum;
    pe_array4 #(.DATA_WIDTH(DW),.ACC_WIDTH(AW)) dut(.clk(clk),.resetn(rn),.start_i(start),.length_i(len),
      .valid_i(vi),.a_i(a),.b_i(b),.ready_o(ready),.busy_o(busy),.done_o(done),.error_o(error),
      .pe_results_o(pe),.sum_o(sum));
    always #5 clk=~clk;
    integer f,o,n,c=0,i,k,t,fields,idx[0:3],x[0:3],y[0:3];
    reg signed [DW-1:0] av[0:3][0:255],bv[0:3][0:255];
    reg signed [AW-1:0] expected[0:3];reg signed [AW+1:0] total;
    string vp,ap;
    initial begin
        if(!$value$plusargs("VECTORS=%s",vp)||!$value$plusargs("ACTUAL=%s",ap))$fatal(1,"Paths");
        f=$fopen(vp,"r");o=$fopen(ap,"w");if(!f||!o)$fatal(1,"Open");
        repeat(2)@(negedge clk);rn=1;start=1;len=4;
        @(negedge clk);start=0;vi=15;
        @(negedge clk);rn=0;
        @(posedge clk);#1;if(busy||done||ready!==0)$fatal(1,"Reset");
        @(negedge clk);rn=1;vi=0;
        while(!$feof(f))begin
            fields=$fscanf(f,"%d %d %d %d %d %d\n",n,expected[0],expected[1],expected[2],expected[3],total);
            if(fields!=6)$fatal(1,"Header");
            for(k=0;k<n;k=k+1)begin
                if($fscanf(f,"%d %d %d %d %d %d %d %d\n",x[0],y[0],x[1],y[1],x[2],y[2],x[3],y[3])!=8)$fatal(1,"Pair");
                for(i=0;i<4;i=i+1)begin av[i][k]=x[i];bv[i][k]=y[i];end
            end
            @(negedge clk);start=1;len=n;vi=0;
            @(posedge clk);#1;if(!busy||done)$fatal(1,"Start");
            for(i=0;i<4;i=i+1)idx[i]=0;
            t=0;
            while(!done)begin
                @(negedge clk);start=(t==2);vi=0;
                for(i=0;i<4;i=i+1)if(idx[i]<n && (t+i)%5!=0 && t%11!=0)begin
                    vi[i]=1;a[i*DW+:DW]=av[i][idx[i]];b[i*DW+:DW]=bv[i][idx[i]];
                end
                #1;fire=vi & ready;
                @(posedge clk);#1;
                for(i=0;i<4;i=i+1)if(fire[i])idx[i]=idx[i]+1;
                t=t+1;if(t>2000)$fatal(1,"Timeout transaction");
            end
            if(busy||error||sum!==total)$fatal(1,"Sum case=%0d",c+1);
            for(i=0;i<4;i=i+1)if(idx[i]!=n||$signed(pe[i*AW+:AW])!==expected[i])$fatal(1,"PE %0d",i);
            c=c+1;$fdisplay(o,"%0d %0d %0d %0d %0d %0d",c,
              $signed(pe[0*AW+:AW]),$signed(pe[1*AW+:AW]),$signed(pe[2*AW+:AW]),$signed(pe[3*AW+:AW]),sum);
            @(negedge clk);vi=0;start=0;
            @(posedge clk);#1;if(done||busy)$fatal(1,"Pulse");
        end
        if(!c)$fatal(1,"Empty");$fclose(o);$display("RTL_SIM_PASS PE4 INT%0d/%0d cases=%0d",DW,AW,c);$finish;
    end
    initial begin #10000000;$fatal(1,"Timeout");end
endmodule

import argparse
import random
import sys
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[2]/'ch07/reference'))
from dot_reference import dot,compare


def generate(dw,aw,out):
    rng=random.Random(308);lo,hi=-(1<<(dw-1)),(1<<(dw-1))-1
    sizes=[0,1,4,16,64,256]+[rng.randrange(65) for _ in range(20)]
    with (out/'vectors.txt').open('w') as vf,(out/'expected.txt').open('w') as ef:
        for c,n in enumerate(sizes,1):
            a=[[rng.randint(lo,hi) for _ in range(n)] for _ in range(4)]
            b=[[rng.randint(lo,hi) for _ in range(n)] for _ in range(4)]
            if c==2:
                a=[[lo],[lo],[hi],[0]];b=[[lo],[hi],[hi],[lo]]
            pe=[dot(x,y,aw) for x,y in zip(a,b)];ans=sum(pe)
            vf.write(' '.join(map(str,(n,*pe,ans)))+'\n')
            for k in range(n):vf.write(' '.join(str(v) for i in range(4) for v in (a[i][k],b[i][k]))+'\n')
            ef.write(' '.join(map(str,(c,*pe,ans)))+'\n')
    assert dot([1,-2],[3,4],aw)==-5
    print(f'PYTHON_PE_SELF_TEST_PASS INT{dw}/{aw} cases={len(sizes)} seed=308')


if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--dw',type=int,default=16);p.add_argument('--aw',type=int,default=40)
    p.add_argument('--out',type=Path);p.add_argument('--actual',type=Path);a=p.parse_args()
    if a.actual:compare(a.out/'expected.txt',a.actual)
    else:generate(a.dw,a.aw,a.out)

07코드 읽기 포인트

위치내용
generate 블록4개 vector_dot_core를 genvar g로 반복 생성합니다. 슬라이스 인덱스 a_i[g*DATA_WIDTH+:DATA_WIDTH]로 비트 필드를 추출합니다.
비동기 완료 처리4개 PE는 같은 길이를 처리하더라도 valid 타이밍이 달라 done이 다른 클록에 올 수 있습니다. saved[g]에 각 결과를 보관하고 finished[g]를 세팅합니다. &finished에서 sum을 등록합니다.
2비트 확장 합산extended[g] = {{2{saved[g][ACC_WIDTH-1]}}, saved[g]}으로 (ACC_WIDTH+2)비트로 확장한 뒤 더합니다. 4개 값을 더하면 올림이 최대 2비트까지 생기므로 보호 비트 2개를 둡니다.
