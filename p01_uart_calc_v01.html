// 장 Top 공용 리셋: btnC(Active-High)를 2단 동기화해 동기 resetn을 만든다 (12장 npu_reset_controller와 같은 동작).
// (변경 2026-11-22) 비동기 assert 제거 — 비동기 clear되는 플립플롭 출력이 RAMB18 제어 핀으로 들어가면
// 버튼을 누르는 순간 BRAM 내용이 깨질 수 있다 (Vivado DRC REQP-1840). 모든 플립플롭이 clk에만 반응한다.
`timescale 1ns/1ps
`default_nettype none
module p01_reset_sync (
    input  wire clk,
    input  wire btnC,
    output wire resetn
);
    (* ASYNC_REG = "TRUE" *) reg [1:0] sync_q = 2'b11;
    reg resetn_q = 1'b0;
    always @(posedge clk) begin
        sync_q   <= {sync_q[0], btnC};
        resetn_q <= ~sync_q[1];
    end
    assign resetn = resetn_q;
endmodule

p01/rtl/p01_calc_controller.v · verilog · 원본 · RTL_SIM 통과 2026-11-23
// 1장 UART 계산기 명령 해석기. 11장과 같은 형식: 1바이트 명령 + 바이너리 페이로드(빅엔디언), stop-and-wait.
//
//   명령   페이로드                     응답
//   'M'    a_hi a_lo b_hi b_lo (signed16)  "A:" + 누산값 16진수 10자리 + " L:" + 지연 클록 1자리 + "\r\n"
//   'C'    없음                          "OK\r\n"   (누산값 0으로)
//   'S'    없음                          "I\r\n"    (준비 상태)
//   그 밖  —                             받은 바이트를 그대로 돌려준다 (1-1 에코)
//
// 지연(L)은 valid를 낸 클록부터 고른 코어의 valid가 보일 때까지의 클록 수다 (1-2 = 1, 1-3·1-4 = 2).
// STAGE = 지금까지 끝낸 서브젝트 수. STAGE 1(1-1)에서는 MAC 코어가 없으므로 'M'·'C'도 에코로 돌려준다.
// 응답을 보내는 동안에는 새 바이트를 받지 않는다 (rx_ready = 0, uart_core FIFO가 보관).
`timescale 1ns/1ps
`default_nettype none
module p01_calc_controller #(
    parameter integer ACC_WIDTH = 40,
    parameter integer STAGE     = 4
) (
    input  wire                        clk,
    input  wire                        resetn,
    // UART 수신
    input  wire [7:0]                  rx_data_i,
    input  wire                        rx_valid_i,
    output wire                        rx_ready_o,
    // UART 송신
    output reg  [7:0]                  tx_data_o,
    output reg                         tx_valid_o,
    input  wire                        tx_ready_i,
    // MAC 묶음
    output reg                         clear_o,
    output reg                         valid_o,
    output reg  signed [15:0]          a_o,
    output reg  signed [15:0]          b_o,
    input  wire signed [ACC_WIDTH-1:0] acc_i,
    input  wire                        acc_valid_i
);
    localparam [2:0] S_IDLE = 3'd0, S_ARG = 3'd1, S_WAIT = 3'd2, S_LOAD = 3'd3, S_SEND = 3'd4;
    localparam integer MSG_MAX = 18;            // "A:" 2 + 10 + " L:" 3 + 1 + "\r\n" 2
    localparam integer NIB = (ACC_WIDTH + 3) / 4;

    reg [2:0]  state;
    reg [1:0]  arg_cnt;
    reg [7:0]  arg [0:2];                       // 앞 세 바이트 (네 번째는 받는 클록에 바로 씀)
    reg [3:0]  lat;
    reg [7:0]  msg [0:MSG_MAX-1];
    reg [4:0]  msg_len, idx;
    integer    k;

    function [7:0] hex(input [3:0] n);
        hex = (n < 4'd10) ? (8'h30 + {4'd0, n}) : (8'h37 + {4'd0, n});   // '0'~'9', 'A'~'F'
    endfunction

    assign rx_ready_o = (state == S_IDLE) || (state == S_ARG);
    wire rx_take = rx_valid_i && rx_ready_o;

    always @(posedge clk) begin
        if (!resetn) begin
            state      <= S_IDLE;
            arg_cnt    <= 2'd0;
            lat        <= 4'd0;
            msg_len    <= 5'd0;
            idx        <= 5'd0;
            tx_data_o  <= 8'd0;
            tx_valid_o <= 1'b0;
            clear_o    <= 1'b0;
            valid_o    <= 1'b0;
            a_o        <= 16'sd0;
            b_o        <= 16'sd0;
        end else begin
            clear_o <= 1'b0;
            valid_o <= 1'b0;
            case (state)
                S_IDLE: if (rx_take) begin
                    case ((STAGE >= 2 || rx_data_i == "S") ? rx_data_i : 8'h00)
                        "M": begin arg_cnt <= 2'd0; state <= S_ARG; end
                        "C": begin
                            clear_o <= 1'b1;
                            msg[0] <= "O"; msg[1] <= "K"; msg[2] <= 8'h0D; msg[3] <= 8'h0A;
                            msg_len <= 5'd4; state <= S_LOAD;
                        end
                        "S": begin
                            msg[0] <= "I"; msg[1] <= 8'h0D; msg[2] <= 8'h0A;
                            msg_len <= 5'd3; state <= S_LOAD;
                        end
                        default: begin                   // 1-1 에코 (STAGE 1에서는 M·C 포함)
                            msg[0] <= rx_data_i;
                            msg_len <= 5'd1; state <= S_LOAD;
                        end
                    endcase
                end
                S_ARG: if (rx_take) begin
                    if (arg_cnt == 2'd3) begin
                        a_o     <= {arg[0], arg[1]};
                        b_o     <= {arg[2], rx_data_i};
                        valid_o <= 1'b1;
                        lat     <= 4'd0;
                        state   <= S_WAIT;
                    end else begin
                        arg[arg_cnt] <= rx_data_i;
                        arg_cnt <= arg_cnt + 2'd1;
                    end
                end
                S_WAIT: begin
                    if (acc_valid_i) begin
                        msg[0] <= "A"; msg[1] <= ":";
                        for (k = 0; k < NIB; k = k + 1)
                            msg[2 + k] <= hex(acc_i[4*(NIB-1-k) +: 4]);
                        msg[2 + NIB] <= " "; msg[3 + NIB] <= "L"; msg[4 + NIB] <= ":";
                        msg[5 + NIB] <= hex(lat);
                        msg[6 + NIB] <= 8'h0D; msg[7 + NIB] <= 8'h0A;
                        msg_len <= NIB + 8;
                        state <= S_LOAD;
                    end else if (lat != 4'hF) begin
                        lat <= lat + 4'd1;
                    end
                end
                S_LOAD: begin
                    tx_data_o  <= msg[0];
                    tx_valid_o <= 1'b1;
                    idx        <= 5'd0;
                    state      <= S_SEND;
                end
                S_SEND: if (tx_valid_o && tx_ready_i) begin
                    if (idx + 5'd1 == msg_len) begin
                        tx_valid_o <= 1'b0;
                        state      <= S_IDLE;
                    end else begin
                        tx_data_o <= msg[idx + 5'd1];
                        idx       <= idx + 5'd1;
                    end
                end
                default: state <= S_IDLE;
            endcase
        end
    end

    initial begin
        if (NIB + 8 > MSG_MAX)
            $fatal(1, "ACC_WIDTH too wide for response buffer");
    end
endmodule
`default_nettype wire

// 1장 MAC 묶음: 1-2·1-3·1-4 서브젝트의 세 코어에 같은 입력을 넣고, sw로 고른 코어의 결과만 내보낸다.
// STAGE = 지금까지 끝낸 서브젝트 수 (1~4). 아직 만들지 않은 코어는 넣지 않고, 그 sw 값은 앞 코어로 대신한다.
//   sel 0 = 1-2 mac_core (01장)            지연 1클록
//   sel 1 = 1-3 dsp_multiplier (02장) + 누산 지연 2클록 (곱 레지스터 1 + 누산 레지스터 1)
//   sel 2·3 = 1-4 mac_pipeline_core (03장) 지연 2클록
// 세 코어가 같은 입력을 늘 함께 받으므로 누산값은 항상 같다. sw를 바꿔도 값은 그대로이고 지연만 달라진다.
`timescale 1ns/1ps
`default_nettype none
module p01_mac_bank #(
    parameter integer DATA_WIDTH = 16,
    parameter integer ACC_WIDTH  = 40,
    parameter integer STAGE      = 4
) (
    input  wire                         clk,
    input  wire                         resetn,
    input  wire                         clear_i,
    input  wire                         valid_i,
    input  wire signed [DATA_WIDTH-1:0] a_i,
    input  wire signed [DATA_WIDTH-1:0] b_i,
    input  wire [1:0]                   sel_i,      // 보드 스위치 (비동기 입력)
    output reg  signed [ACC_WIDTH-1:0]  acc_o,
    output reg                          valid_o,
    output wire [1:0]                   sel_o       // 동기화한 선택값
);
    localparam integer PRODUCT_WIDTH = 2 * DATA_WIDTH;

    // 스위치 2FF 동기화
    reg [1:0] sel_meta, sel_sync;
    always @(posedge clk) begin
        if (!resetn) begin
            sel_meta <= 2'd0;
            sel_sync <= 2'd0;
        end else begin
            sel_meta <= sel_i;
            sel_sync <= sel_meta;
        end
    end
    assign sel_o = sel_sync;

    // 1-2: 01장 mac_core
    wire signed [ACC_WIDTH-1:0] acc_mac;
    wire                        valid_mac;
    generate if (STAGE >= 2) begin : g_mac
        mac_core #(.DATA_WIDTH(DATA_WIDTH), .ACC_WIDTH(ACC_WIDTH)) u_mac (
            .clk(clk), .resetn(resetn), .clear_i(clear_i), .valid_i(valid_i),
            .a_i(a_i), .b_i(b_i), .acc_o(acc_mac), .valid_o(valid_mac));
    end else begin : g_no_mac
        assign acc_mac = {ACC_WIDTH{1'b0}};
        assign valid_mac = 1'b0;
    end endgenerate

    // 1-3: 02장 dsp_multiplier + 누산 레지스터
    wire signed [ACC_WIDTH-1:0] acc_dsp;
    wire                        valid_dsp;
    generate if (STAGE >= 3) begin : g_dsp
        wire signed [PRODUCT_WIDTH-1:0] prod_dsp;
        wire                            prod_valid_dsp;
        dsp_multiplier #(.A_WIDTH(DATA_WIDTH), .B_WIDTH(DATA_WIDTH)) u_dsp (
            .clk(clk), .resetn(resetn), .clear_i(clear_i), .valid_i(valid_i),
            .a_i(a_i), .b_i(b_i), .product_o(prod_dsp), .valid_o(prod_valid_dsp));

        reg signed [ACC_WIDTH-1:0] acc_q;
        reg                        valid_q;
        wire signed [ACC_WIDTH-1:0] prod_ext = {{(ACC_WIDTH-PRODUCT_WIDTH){prod_dsp[PRODUCT_WIDTH-1]}}, prod_dsp};
        always @(posedge clk) begin
            if (!resetn || clear_i) begin
                acc_q   <= {ACC_WIDTH{1'b0}};
                valid_q <= 1'b0;
            end else begin
                valid_q <= prod_valid_dsp;
                if (prod_valid_dsp)
                    acc_q <= acc_q + prod_ext;
            end
        end
        assign acc_dsp = acc_q;
        assign valid_dsp = valid_q;
    end else begin : g_no_dsp
        assign acc_dsp = acc_mac;              // 아직 없음: 1-2 코어로 대신
        assign valid_dsp = valid_mac;
    end endgenerate

    // 1-4: 03장 mac_pipeline_core
    wire signed [ACC_WIDTH-1:0] acc_pipe;
    wire                        valid_pipe;
    generate if (STAGE >= 4) begin : g_pipe
        mac_pipeline_core #(.DATA_WIDTH(DATA_WIDTH), .ACC_WIDTH(ACC_WIDTH)) u_pipe (
            .clk(clk), .resetn(resetn), .clear_i(clear_i), .valid_i(valid_i),
            .a_i(a_i), .b_i(b_i), .acc_o(acc_pipe), .valid_o(valid_pipe));
    end else begin : g_no_pipe
        assign acc_pipe = acc_dsp;             // 아직 없음: 앞 코어로 대신
        assign valid_pipe = valid_dsp;
    end endgenerate

    // 고른 코어의 결과
    always @(*) begin
        case (sel_sync)
            2'd0:    begin acc_o = acc_mac;  valid_o = valid_mac;  end
            2'd1:    begin acc_o = acc_dsp;  valid_o = valid_dsp;  end
            default: begin acc_o = acc_pipe; valid_o = valid_pipe; end
        endcase
    end
    initial begin
        if (STAGE < 1 || STAGE > 4)
            $fatal(1, "STAGE must be 1..4");
    end
endmodule
`default_nettype wire

// 1장 7세그먼트: 16비트 값을 16진수 4자리로 띄운다 (보드 seg·an·dp는 active-low, seg = {g,f,e,d,c,b,a}).
// 자리 전환은 100 MHz / 2^(REFRESH_BITS-2) 마다 (기본 18 → 약 1.5 kHz, 네 자리 한 바퀴 약 381 Hz).
// 5장 SoC의 표준 SEG7 IP(ip_seg7)와 같은 표시 규칙이다.
`timescale 1ns/1ps
`default_nettype none
module p01_hex4_display #(
    parameter integer REFRESH_BITS = 18
) (
    input  wire        clk,
    input  wire        resetn,
    input  wire [15:0] value_i,
    output reg  [6:0]  seg_o,
    output reg  [3:0]  an_o,
    output wire        dp_o
);
    reg [REFRESH_BITS-1:0] cnt;
    always @(posedge clk) begin
        if (!resetn) cnt <= {REFRESH_BITS{1'b0}};
        else         cnt <= cnt + 1'b1;
    end

    wire [1:0] digit = cnt[REFRESH_BITS-1 -: 2];
    wire [3:0] nib   = value_i[4*digit +: 4];

    reg [6:0] on;                               // active-high, [0] a … [6] g
    always @(*) begin
        case (nib)
            4'h0: on = 7'b0111111;  4'h1: on = 7'b0000110;
            4'h2: on = 7'b1011011;  4'h3: on = 7'b1001111;
            4'h4: on = 7'b1100110;  4'h5: on = 7'b1101101;
            4'h6: on = 7'b1111101;  4'h7: on = 7'b0000111;
            4'h8: on = 7'b1111111;  4'h9: on = 7'b1101111;
            4'hA: on = 7'b1110111;  4'hB: on = 7'b1111100;
            4'hC: on = 7'b0111001;  4'hD: on = 7'b1011110;
            4'hE: on = 7'b1111001;  default: on = 7'b1110001;
        endcase
        seg_o = ~on;
        an_o  = ~(4'b0001 << digit);
    end
    assign dp_o = 1'b1;                         // 소수점 끔
endmodule
`default_nettype wire

"""1장 UART 계산기 PC 쪽 도구: 명령을 바이트로 바꿔 보내고 응답 한 줄을 받아 출력한다.

  python3 p01/host/p01_send.py --dry M 3 4          # 보낼 바이트만 출력 (보드 없이)
  python3 p01/host/p01_send.py /dev/ttyUSB1 C       # 누산값 0  → OK
  python3 p01/host/p01_send.py /dev/ttyUSB1 M 3 4   # 3×4 누산  → A:000000000C L:1
  (Windows는 COM3 같은 포트 이름. pip install pyserial 필요)

명령 형식은 p01/rtl/p01_calc_controller.v 머리말과 같다. 'M' 뒤 두 수는 signed16(-32768~32767).
"""
import struct
import sys


def encode(args):
    op = args[0]
    if op == "M":
        a, b = int(args[1], 0), int(args[2], 0)
        return b"M" + struct.pack(">hh", a, b)
    if len(op) == 1:
        return op.encode("ascii")          # C, S, 또는 에코할 글자
    raise SystemExit("명령은 M a b, C, S, 또는 글자 하나")


def main(argv):
    if len(argv) < 2:
        raise SystemExit(__doc__)
    port, args = argv[0], argv[1:]
    data = encode(args)
    print("TX", data.hex(" "))
    if port == "--dry":
        return 0
    import serial                           # pyserial (보드에서만 필요)
    with serial.Serial(port, 115200, timeout=1) as s:
        s.write(data)
        echo_only = len(data) == 1 and data not in (b"C", b"S")
        reply = s.read(1) if echo_only else s.read_until(b"\n")
    print("RX", repr(reply.decode("ascii", "replace")))
    return 0 if reply else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

// (추가) 8N1 수신. RX 2FF, start midpoint/bit center, hold/overrun.
module uart_rx #(
    parameter integer CLK_HZ=100000000, BAUD=115200
)(
    input wire clk,resetn,rx_i,
    output reg [7:0] out_data_o,
    output reg out_valid_o,
    input wire out_ready_i,
    output reg framing_error_o,overrun_o
);
    localparam integer CPB=(CLK_HZ+BAUD/2)/BAUD;
    localparam integer HALF=CPB/2;
    localparam [1:0] IDLE=0,START=1,DATA=2,STOP=3;
    reg [1:0] state;
    reg rx_meta,rx_sync;
    reg [31:0] count_q;
    reg [2:0] bit_q;
    reg [7:0] shift_q;
    always @(posedge clk)begin
        if(!resetn)begin
            state<=IDLE;rx_meta<=1;rx_sync<=1;count_q<=0;bit_q<=0;shift_q<=0;
            out_data_o<=0;out_valid_o<=0;framing_error_o<=0;overrun_o<=0;
        end else begin
            rx_meta<=rx_i;rx_sync<=rx_meta;
            if(out_valid_o && out_ready_i)out_valid_o<=0;
            case(state)
                IDLE:if(!rx_sync)begin count_q<=HALF-1;state<=START;end
                START:if(count_q==0)begin
                    if(!rx_sync)begin count_q<=CPB-1;bit_q<=0;state<=DATA;end
                    else state<=IDLE;
                end else count_q<=count_q-1'b1;
                DATA:if(count_q==0)begin
                    shift_q[bit_q]<=rx_sync;count_q<=CPB-1;
                    if(bit_q==7)state<=STOP;
                    else bit_q<=bit_q+1'b1;
                end else count_q<=count_q-1'b1;
                STOP:if(count_q==0)begin
                    state<=IDLE;
                    if(!rx_sync)framing_error_o<=1;
                    else if(!out_valid_o || out_ready_i)begin
                        out_data_o<=shift_q;out_valid_o<=1;
                    end else overrun_o<=1;
                end else count_q<=count_q-1'b1;
                default:state<=IDLE;
            endcase
        end
    end
    // synthesis translate_off
    initial if(CPB<4)$fatal(1,"Unsupported UART clock/baud ratio");
    // synthesis translate_on
endmodule

// (추가) 8N1 송신. valid/ready handshake, 10bit start/data/stop.
module uart_tx #(
    parameter integer CLK_HZ=100000000, BAUD=115200
)(
    input wire clk,resetn,
    input wire [7:0] in_data_i,
    input wire in_valid_i,
    output wire in_ready_o,
    output reg tx_o
);
    localparam integer CPB=(CLK_HZ+BAUD/2)/BAUD;
    reg sending;
    reg [31:0] tick_q;
    reg [3:0] bit_q;
    reg [9:0] frame_q;
    assign in_ready_o=resetn && !sending;
    always @(posedge clk)begin
        if(!resetn)begin
            sending<=0;tick_q<=0;bit_q<=0;frame_q<=10'h3ff;tx_o<=1;
        end else if(!sending)begin
            tx_o<=1;
            if(in_valid_i)begin
                sending<=1;tick_q<=CPB-1;bit_q<=0;
                frame_q<={1'b1,in_data_i,1'b0};tx_o<=0;
            end
        end else if(tick_q!=0)tick_q<=tick_q-1'b1;

            if(bit_q==9)begin sending<=0;tx_o<=1;end
            else begin bit_q<=bit_q+1'b1;tx_o<=frame_q[1];end
        end
    end
    // synthesis translate_off
    initial if(CPB<4)$fatal(1,"Unsupported UART clock/baud ratio");
    // synthesis translate_on
endmodule

)(
    input wire clk,resetn,uart_rxd_i,
    output wire uart_txd_o,
    output wire [7:0] rx_out_data,
    output wire rx_out_valid,
    input wire rx_out_ready,

        .clk(clk),.resetn(resetn),.rx_i(uart_rxd_i),
        .out_data_o(phy_rx_data),.out_valid_o(phy_rx_valid),
        .out_ready_i(phy_rx_ready),.framing_error_o(rx_framing_error_o),
        .overrun_o(rx_overrun_o));
    uart_tx #(.CLK_HZ(CLK_HZ),.BAUD(BAUD)) tx(
        .clk(clk),.resetn(resetn),.in_data_i(phy_tx_data),
        .in_valid_i(phy_tx_valid),.in_ready_o(phy_tx_ready),.tx_o(uart_txd_o));

    reg [7:0] rx_fifo[0:15];
    reg [3:0] rx_wptr,rx_rptr;
    reg [4:0] rx_count;
    wire rx_pop=rx_out_valid && rx_out_ready;
    wire rx_push=phy_rx_valid && phy_rx_ready;
    assign rx_out_valid=(rx_count!=0);
    assign rx_out_data=rx_fifo[rx_rptr];
    assign phy_rx_ready=resetn && ((rx_count<16) || rx_pop);

    reg [7:0] tx_fifo[0:15];
    reg [3:0] tx_wptr,tx_rptr;
    reg [4:0] tx_count;
    wire tx_pop=phy_tx_ready && phy_tx_valid;
    wire tx_push=tx_in_valid && tx_in_ready;
    assign tx_in_ready=resetn && ((tx_count<16) || tx_pop);
    assign phy_tx_valid=(tx_count!=0);
    assign phy_tx_data=tx_fifo[tx_rptr];

    always @(posedge clk)begin
        if(!resetn)begin
            rx_wptr<=0;rx_rptr<=0;rx_count<=0;
            tx_wptr<=0;tx_rptr<=0;tx_count<=0;
        end else begin
            if(rx_push)begin
                rx_fifo[rx_wptr]<=phy_rx_data;
                rx_wptr<=rx_wptr+1'b1;
            end
            if(rx_pop)rx_rptr<=rx_rptr+1'b1;
            case({rx_push,rx_pop})
                2'b10:rx_count<=rx_count+1'b1;

                default:begin end
            endcase

                tx_fifo[tx_wptr]<=tx_in_data;
                tx_wptr<=tx_wptr+1'b1;
            end
            if(tx_pop)tx_rptr<=tx_rptr+1'b1;
            case({tx_push,tx_pop})
                2'b10:tx_count<=tx_count+1'b1;
                2'b01:tx_count<=tx_count-1'b1;
                default:begin end
            endcase
        end
    end
endmodule

module mac_core #(
    parameter integer DATA_WIDTH = 16,
    parameter integer ACC_WIDTH = 40
) (
    input wire clk,
    input wire resetn,
    input wire clear_i,
    input wire valid_i,
    input wire signed [DATA_WIDTH-1:0] a_i,
    input wire signed [DATA_WIDTH-1:0] b_i,
    output reg signed [ACC_WIDTH-1:0] acc_o,
    output reg valid_o
);
    localparam integer PRODUCT_WIDTH = 2 * DATA_WIDTH;
    wire signed [PRODUCT_WIDTH-1:0] product_w;
    wire signed [ACC_WIDTH-1:0] product_ext_w;

    // 전체 곱을 보존한 뒤 부호 비트를 복제하여 누산 폭으로 확장한다.
    // 지원 조건: DATA_WIDTH >= 2, ACC_WIDTH >= 2*DATA_WIDTH.
    assign product_w = $signed(a_i) * $signed(b_i);
    assign product_ext_w = {{(ACC_WIDTH-PRODUCT_WIDTH){product_w[PRODUCT_WIDTH-1]}}, product_w};

    // 동기식 Active-Low Reset > Clear > Valid 순서.
    // 입력을 수락한 상승 에지 직후 acc_o와 valid_o가 함께 갱신된다.
    // valid_i=0이면 acc_o 유지, valid_o=0. Clear와 겹친 입력은 버린다.
    always @(posedge clk) begin
        if (!resetn) begin
            acc_o <= {ACC_WIDTH{1'b0}};
            valid_o <= 1'b0;
        end else if (clear_i) begin
            acc_o <= {ACC_WIDTH{1'b0}};
            valid_o <= 1'b0;
        end else begin
            valid_o <= valid_i;
            if (valid_i)
                // 범위 초과 시 상위 비트를 버리는 2의 보수 순환 연산이다.
                acc_o <= acc_o + product_ext_w;
        end
    end

    // synthesis translate_off
    initial begin
        if (DATA_WIDTH < 2 || ACC_WIDTH < PRODUCT_WIDTH)
            $fatal(1, "Require DATA_WIDTH>=2 and ACC_WIDTH>=2*DATA_WIDTH");
    end
    // synthesis translate_on
endmodule

"""P3-01 독립 정수 모델, 재현 가능한 자극 생성 및 RTL 전 사이클 대조."""
import argparse
import random
from pathlib import Path


def wrap(value, width):
    """Python 무제한 정수를 signed 2의 보수 범위로 변환한다."""
    modulus = 1 << width
    return (value + modulus // 2) % modulus - modulus // 2


def step(acc, resetn, clear, valid, a, b, acc_width):
    if not resetn or clear:
        return 0, 0
    return (wrap(acc + a * b, acc_width), 1) if valid else (acc, 0)


def self_test():
    # 별도의 상수 정답으로 산술/제어 및 각 설정의 양·음 오버플로를 검증한다.
    for dw, aw in ((8, 32), (16, 40)):
        maximum = (1 << (aw - 1)) - 1
        minimum = -(1 << (aw - 1))
        assert step(0, 1, 0, 1, 3, 4, aw) == (12, 1)
        assert step(0, 1, 0, 1, 3, -4, aw) == (-12, 1)
        assert step(0, 1, 0, 1, -3, -4, aw) == (12, 1)
        assert step(maximum, 1, 0, 1, 1, 1, aw) == (minimum, 1)
        assert step(minimum, 1, 0, 1, 1, -1, aw) == (maximum, 1)
        assert step(7, 1, 0, 0, 3, 4, aw) == (7, 0)
        assert step(7, 1, 1, 1, 3, 4, aw) == (0, 0)
        assert step(7, 0, 1, 1, 3, 4, aw) == (0, 0)
        assert wrap((-(1 << (dw - 1))) ** 2, aw) == 1 << (2 * dw - 2)
    print("PYTHON_SELF_TEST_PASS configurations=INT8/32,INT16/40")


def stimulus(dw, aw):
    lo, hi = -(1 << (dw - 1)), (1 << (dw - 1)) - 1
    yield (0, 0, 1, hi, lo)
    yield (0, 1, 1, lo, lo)
    yield (1, 0, 0, 0, 0)
    # 부호 조합, 0, 입력 경계, 연속 valid 및 idle.
    for a, b in ((3, 4), (3, -4), (-3, -4), (hi, hi),
                 (lo, lo), (lo, hi), (hi, lo), (0, lo), (-1, -1)):
        yield (1, 0, 1, a, b)
    for _ in range(4):
        yield (1, 0, 0, hi, hi)
    yield (1, 1, 1, hi, hi)
    yield (1, 1, 0, lo, lo)
    # 내부 레지스터 force 없이 실제 포트로 최댓값에 도달한다.
    target = (1 << (aw - 1)) - 1
    count, remainder = divmod(target, lo * lo)
    for _ in range(count):
        yield (1, 0, 1, lo, lo)
    count, remainder = divmod(remainder, hi)
    for _ in range(count):
        yield (1, 0, 1, hi, 1)
    if remainder:
        yield (1, 0, 1, remainder, 1)
    yield (1, 0, 0, hi, hi)  # 최댓값 유지
    yield (1, 0, 1, 1, 1)    # max + 1 -> min
    yield (1, 0, 0, hi, hi)  # 최솟값 유지
    yield (1, 0, 1, -1, 1)   # min - 1 -> max
    yield (0, 1, 1, lo, hi)  # Reset/Clear/Valid 동시 입력
    rng = random.Random(301 + dw)
    for _ in range(2000):
        yield (int(rng.randrange(53) != 0), int(rng.randrange(29) == 0),
               int(rng.randrange(4) != 0), rng.randint(lo, hi), rng.randint(lo, hi))
    yield (1, 0, 0, 0, 0)


def generate(dw, aw, vectors, expected):
    acc = 0
    count = 0
    with vectors.open("w") as vf, expected.open("w") as ef:
        for count, row in enumerate(stimulus(dw, aw), 1):
            acc, valid = step(acc, *row, aw)
            vf.write(" ".join(map(str, (*row, acc, valid))) + "\n")
            ef.write(f"{count} {acc} {valid}\n")

def compare(expected, actual):

    if count == 0:
        raise AssertionError("Empty comparison")

def vectors(a_width, b_width, acc_width):
    # 대칭 INT16/INT8은 기존 누산 오버플로 자극도 그대로 사용한다.
    if a_width == b_width:
        yield from stimulus(a_width, acc_width)
    else:
        yield (0, 1, 1, 1, -1)
        yield (1, 0, 0, 0, 0)
    amin, amax = -(1 << (a_width - 1)), (1 << (a_width - 1)) - 1
    bmin, bmax = -(1 << (b_width - 1)), (1 << (b_width - 1)) - 1
    yield (1, 1, 1, amax, bmax)
    for a in (amin, amin + 1, -1, 0, 1, amax - 1, amax):
        for b in (bmin, bmin + 1, -1, 0, 1, bmax - 1, bmax):
            yield (1, 0, 1, a, b)
    # INT8의 65536개 입력 쌍은 전수 대조한다(제어 상태 전수 검증은 아님).
    if (a_width, b_width) == (8, 8):
        for a in range(amin, amax + 1):
            for b in range(bmin, bmax + 1):
                yield (1, 0, 1, a, b)
    rng = random.Random(a_width * 1000 + b_width)
    for _ in range(2000):
        yield (int(rng.randrange(43) != 0), int(rng.randrange(31) == 0),
               int(rng.randrange(4) != 0), rng.randint(amin, amax), rng.randint(bmin, bmax))
    yield (0, 0, 1, amin, bmax)
    yield (0, 1, 1, amin, bmin)
    yield (1, 0, 1, amax, bmin)
    yield (1, 0, 0, amin, bmax)
    yield (1, 1, 0, 1, 1)
    yield (1, 0, 0, 0, 0)


def generate(a_width, b_width, acc_width, vector_path, expected_path):
    product, acc = 0, 0
    count = 0
    with vector_path.open("w") as vf, expected_path.open("w") as ef:
        for count, row in enumerate(vectors(a_width, b_width, acc_width), 1):
            rn, clear, valid, a, b = row
            if not rn or clear:
                product = 0
            elif valid:
                product = a * b  # Python의 무제한 정수 곱, RTL 폭 변환에 의존하지 않는다.
            acc, out_valid = step(acc, *row, acc_width)
            if not -(1 << (a_width+b_width-1)) <= product < (1 << (a_width+b_width-1)):
                raise AssertionError("Product outside full signed result width")
            vf.write(" ".join(map(str, (*row, product, acc, out_valid))) + "\n")
            ef.write(f"{count} {product} {acc} {out_valid}\n")
    print(f"PYTHON_VECTORS A={a_width} B={b_width} ACC={acc_width} cycles={count}")

module mac_pipeline_core #(
    parameter integer DATA_WIDTH = 16,
    parameter integer ACC_WIDTH = 40
) (
    input wire clk,
    input wire resetn,
    input wire clear_i,
    input wire valid_i,
    input wire signed [DATA_WIDTH-1:0] a_i,
    input wire signed [DATA_WIDTH-1:0] b_i,
    output reg signed [ACC_WIDTH-1:0] acc_o,
    output reg valid_o
);
    localparam integer PRODUCT_WIDTH = 2 * DATA_WIDTH;
    wire signed [PRODUCT_WIDTH-1:0] product_w;
    reg signed [PRODUCT_WIDTH-1:0] product_q;
    reg product_valid_q;
    wire signed [ACC_WIDTH-1:0] product_ext_w;

    assign product_w = $signed(a_i) * $signed(b_i);
    assign product_ext_w = {{(ACC_WIDTH-PRODUCT_WIDTH){product_q[PRODUCT_WIDTH-1]}}, product_q};

    always @(posedge clk) begin
        if (!resetn) begin
            product_q <= {PRODUCT_WIDTH{1'b0}};
            product_valid_q <= 1'b0;
            acc_o <= {ACC_WIDTH{1'b0}};
            valid_o <= 1'b0;
        end else if (clear_i) begin
            // 현재 입력과 미누산 곱을 모두 버려 지연 Valid가 남지 않게 한다.
            product_q <= {PRODUCT_WIDTH{1'b0}};
            product_valid_q <= 1'b0;
            acc_o <= {ACC_WIDTH{1'b0}};
            valid_o <= 1'b0;
        end else begin
            // 1단계: 곱과 같은 에지의 Valid를 함께 저장한다.
            product_valid_q <= valid_i;
            if (valid_i) begin
                product_q <= product_w;
            end
            // 2단계: 비차단 대입은 이전 에지에 저장된 곱과 Valid를 참조한다.
            valid_o <= product_valid_q;
            if (product_valid_q) begin
                // 부호 확장 후 더하며 범위 초과는 P3-01과 같은 2의 보수 순환이다.
                acc_o <= acc_o + product_ext_w;
            end
        end
    end

    // synthesis translate_off
    initial begin
        if (DATA_WIDTH < 2 || ACC_WIDTH < PRODUCT_WIDTH)
            $fatal(1, "Require DATA_WIDTH>=2 and ACC_WIDTH>=2*DATA_WIDTH");
    end
    // synthesis translate_on

class PipelineReference:
    def __init__(self, acc_width):
        self.acc_width = acc_width
        self.pending = deque()
        self.product = 0
        self.acc = 0
        self.edge = 0

    def tick(self, resetn, clear, valid, a, b):
        self.edge += 1
        source_edge = 0
        out_valid = 0
        if not resetn or clear:
            self.pending.clear()
            self.product = 0
            self.acc = 0
        else:
            # 먼저 이전 입력을 배출한 다음 현재 입력을 큐에 넣는다.
            if self.pending:
                product, source_edge = self.pending.popleft()
                if source_edge != self.edge - 1:
                    raise AssertionError("Pipeline latency must be one edge")
                self.acc = wrap(self.acc + product, self.acc_width)
                out_valid = 1
            if valid:
                self.product = a * b
                self.pending.append((self.product, self.edge))
        return self.product, int(bool(self.pending)), self.acc, out_valid, source_edge

ACC_WIDTH = 40
CORE = {0: 2, 1: 3, 2: 4, 3: 4}  # sw 값 → 서브젝트 번호 (1-2 mac_core, 1-3 DSP+누산, 1-4 파이프라인)
LAT = {2: 1, 3: 2, 4: 2}         # 서브젝트별 지연 클록: 1-2 = 1, 1-3 = 곱 레지스터 + 누산 레지스터 = 2, 1-4 = 2

def wrap(v):
    """40비트 2의 보수로 접는다 (RTL 누산기와 같음)."""
    v &= (1 << ACC_WIDTH) - 1
    return v - (1 << ACC_WIDTH) if v >> (ACC_WIDTH - 1) else v


class Calc:
    def __init__(self, stage=4):
        self.acc = 0
        self.stage = stage

    def step(self, sel, data):
        """보낼 바이트 → 응답 바이트."""
        op = data[0]
        if self.stage < 2 and op != ord("S"):
            return bytes([op])      # 1-1까지: MAC 코어가 없어 'M'·'C'도 에코
        if op == ord("M"):
            a, b = struct.unpack(">hh", bytes(data[1:5]))
            self.acc = wrap(self.acc + a * b)
            core = min(CORE[sel], self.stage)           # 아직 없는 코어는 앞 코어가 대신
            text = "A:%010X L:%X\r\n" % (self.acc & ((1 << ACC_WIDTH) - 1), LAT[core])
            return text.encode()
        if op == ord("C"):
            self.acc = 0
            return b"OK\r\n"
        if op == ord("S"):
            return b"I\r\n"
        return bytes([op])          # 1-1 에코


def m(a, b):
    return [ord("M")] + list(struct.pack(">hh", a, b))
