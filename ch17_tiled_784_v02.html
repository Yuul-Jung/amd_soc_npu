제17장 · 784원소 13타일 실행 (코어 밖 부분합 누산)

28×28 입력 784원소를 64원소 타일 13개(64×12 + 16)로 나눠 15장 Dense 코어를 13번 실행합니다. 타일별 부분합은 코어 밖 누산기에서 더하고, Bias는 13타일이 모두 끝난 뒤 한 번만 더합니다.

ch17/rtl/npu_tiled_784.v · verilog · 원본 · RTL_SIM 통과 2026-11-19
`timescale 1ns/1ps
`default_nettype none
// 17장 784원소 10분류: 28×28 입력을 64원소 타일 13개(64×12 + 16)로 나눠 15장 Dense 코어를 13번 돌린다.
// 타일별 부분합은 코어 밖 누산기에서 더하고, Bias는 13타일이 끝난 뒤 한 번만 더한 다음 16장 argmax로 클래스를 낸다.
module npu_tiled_784 #(
    parameter integer ACC_WIDTH = 32
) (
    input  wire                     clk,
    input  wire                     resetn,
    input  wire                     x_wr_en_i,          // 입력 784원소 쓰기 (INT8, 0~127)
    input  wire               [9:0] x_addr_i,
    input  wire               [7:0] x_data_i,
    input  wire                     w_wr_en_i,          // 가중치 10×784 쓰기
    input  wire               [3:0] w_neuron_i,
    input  wire               [9:0] w_addr_i,
    input  wire               [7:0] w_data_i,
    input  wire                     b_wr_en_i,
    input  wire               [3:0] b_neuron_i,
    input  wire     [ACC_WIDTH-1:0] b_data_i,
    input  wire                     start_i,
    output wire                     busy_o,
    output reg                      tile_done_o,        // 타일 하나 누산 완료 1클록 펄스
    output reg                [3:0] tile_o,             // 지금 실행 중인 타일 번호 0~12
    output reg                [3:0] tile_done_num_o,    // tile_done_o가 가리키는 (방금 끝난) 타일 번호
    output wire               [6:0] tile_len_o,         // 지금 타일 원소 수 (0~11: 64, 12: 16)
    output wire [10*ACC_WIDTH-1:0]  acc_o,              // 타일 부분합 누계 (Bias 없음)
    output wire [10*ACC_WIDTH-1:0]  scores_o,           // 누계 + Bias (13타일 뒤)
    output reg                      done_o,
    output reg                      error_o,
    output wire               [3:0] class_o
);
    localparam integer TILES = 13, LAST_LEN = 784 - 64 * 12;
    localparam IDLE=0, TILE_START=1, TILE_RUN=2, BIAS=3, AM_GO=4, AM_WAIT=5;
    reg [2:0] state;
    reg signed [ACC_WIDTH-1:0] acc_q [0:9];
    reg signed [ACC_WIDTH-1:0] score_q [0:9];
    reg signed [ACC_WIDTH-1:0] bias_q [0:9];
    integer i;

    assign busy_o = (state != IDLE);
    assign tile_len_o = (tile_o == TILES - 1) ? LAST_LEN[6:0] : 7'd64;

    // ── 15장 코어: 읽기 포트를 타일 주소로 바꿔 연결
    wire x_rd_en, w_rd_en; wire [5:0] x_idx, w_idx; wire [1:0] w_group;
    wire signed [7:0] x_rd; wire x_rv; wire [31:0] w_rd; wire [3:0] w_rv;
    wire core_done, core_err; wire [10*ACC_WIDTH-1:0] dots;
    npu_dense_core #(.N_OUT(10), .ACC_WIDTH(ACC_WIDTH)) core (
        .clk(clk), .resetn(resetn), .start_i(state == TILE_START), .len_i(tile_len_o),
        .x_rd_en_o(x_rd_en), .x_rd_idx_o(x_idx), .x_data_i(x_rd), .x_valid_i(x_rv),
        .w_rd_en_o(w_rd_en), .w_rd_group_o(w_group), .w_rd_idx_o(w_idx), .w_data_i(w_rd), .w_valid_i(w_rv),
        .busy_o(), .done_o(core_done), .error_o(core_err), .dots_o(dots));

    wire [9:0] x_tile_addr = {tile_o, 6'd0} + x_idx;      // 타일 t의 원소 i = 전체 주소 t*64 + i
    wire [9:0] w_tile_addr = {tile_o, 6'd0} + w_idx;
    bram_sync_buffer #(.DATA_WIDTH(8), .ADDR_WIDTH(10)) x_ram (
        .clk(clk), .resetn(resetn), .wr_en(x_wr_en_i && !busy_o), .wr_addr(x_addr_i), .wr_data(x_data_i),
        .rd_en(x_rd_en), .rd_addr(x_tile_addr), .rd_data(x_rd), .rd_valid(x_rv));
    genvar g;
    generate for (g = 0; g < 4; g = g + 1) begin : lane
        bram_sync_buffer #(.DATA_WIDTH(8), .ADDR_WIDTH(12)) w_ram (
            .clk(clk), .resetn(resetn),
            .wr_en(w_wr_en_i && !busy_o && w_neuron_i <= 9 && w_neuron_i[1:0] == g),
            .wr_addr({w_neuron_i[3:2], w_addr_i}), .wr_data(w_data_i),
            .rd_en(w_rd_en), .rd_addr({w_group, w_tile_addr}), .rd_data(w_rd[g*8 +: 8]), .rd_valid(w_rv[g]));
    end endgenerate
    generate for (g = 0; g < 10; g = g + 1) begin : o
        assign acc_o[g*ACC_WIDTH +: ACC_WIDTH] = acc_q[g];
        assign scores_o[g*ACC_WIDTH +: ACC_WIDTH] = score_q[g];
    end endgenerate

    wire am_done;
    npu_argmax #(.N(10), .W(ACC_WIDTH)) argmax (
        .clk(clk), .resetn(resetn), .start_i(state == AM_GO), .scores_i(scores_o),
        .busy_o(), .done_o(am_done), .class_o(class_o));

    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE; tile_o <= 0; tile_done_num_o <= 0; tile_done_o <= 0; done_o <= 0; error_o <= 0;
            for (i = 0; i < 10; i = i + 1) begin acc_q[i] <= 0; score_q[i] <= 0; bias_q[i] <= 0; end
        end else begin
            tile_done_o <= 0; done_o <= 0; error_o <= 0;
            if (b_wr_en_i && !busy_o && b_neuron_i <= 9) bias_q[b_neuron_i] <= b_data_i;
            case (state)
                IDLE: if (start_i) begin
                    tile_o <= 0; state <= TILE_START;
                    for (i = 0; i < 10; i = i + 1) acc_q[i] <= 0;
                end
                TILE_START: state <= TILE_RUN;
                TILE_RUN: if (core_err) begin state <= IDLE; error_o <= 1; end
                    else if (core_done) begin                  // 코어 밖에서 타일 부분합 누산
                        for (i = 0; i < 10; i = i + 1) acc_q[i] <= acc_q[i] + $signed(dots[i*ACC_WIDTH +: ACC_WIDTH]);
                        tile_done_o <= 1; tile_done_num_o <= tile_o;
                        if (tile_o == TILES - 1) state <= BIAS;
                        else begin tile_o <= tile_o + 4'd1; state <= TILE_START; end
                    end
                BIAS: begin                                     // Bias는 13타일 누산 뒤 한 번
                    for (i = 0; i < 10; i = i + 1) score_q[i] <= acc_q[i] + bias_q[i];
                    state <= AM_GO;
                end
                AM_GO: state <= AM_WAIT;
                AM_WAIT: if (am_done) begin state <= IDLE; done_o <= 1; end
                default: begin state <= IDLE; error_o <= 1; end
            endcase
        end
    end
endmodule

04테스트벤치·기준 모델 작성

테스트벤치는 출력을 자동 판정합니다. 기준 모델(Python)이 6사례를 만듭니다: 28×28 형판 가중치로 숫자 0·4·9, 무작위 가중치 2개, 마지막 타일 16원소만 0이 아닌 사례. 입력 스케일은 13장과 같습니다(화소/2, 0~127). TB는 타일이 끝날 때마다 번호와 누계를 기록합니다.

// 17장 TB: 6사례에서 타일 13개의 순서·길이(64×12, 16)·누계와 최종 점수·클래스를 파일로 돌려준다.
module tb_npu_tiled_784;
    localparam AW = 32;
    reg clk=0, rn=0, xwe=0, wwe=0, bwe=0, start=0;
    reg [9:0] xa=0, wa=0; reg [7:0] xd=0, wd=0; reg [3:0] wn=0, bn=0; reg [AW-1:0] bd=0;
    wire busy, td, done, err; wire [3:0] tile, tdn, cls; wire [6:0] tlen;
    wire [10*AW-1:0] acc, scores;
    npu_tiled_784 #(.ACC_WIDTH(AW)) dut(.clk(clk), .resetn(rn), .x_wr_en_i(xwe), .x_addr_i(xa), .x_data_i(xd),
        .w_wr_en_i(wwe), .w_neuron_i(wn), .w_addr_i(wa), .w_data_i(wd), .b_wr_en_i(bwe), .b_neuron_i(bn), .b_data_i(bd),
        .start_i(start), .busy_o(busy), .tile_done_o(td), .tile_o(tile), .tile_done_num_o(tdn), .tile_len_o(tlen),
        .acc_o(acc), .scores_o(scores), .done_o(done), .error_o(err), .class_o(cls));
    always #5 clk = ~clk;

    integer f, o, n, k, c, v, cases = 0, tiles_seen, expect_tile, len_sum;
    reg [8*16-1:0] name; string vp, ap;

    // 타일 완료마다 번호·누계 기록, 번호는 0부터 1씩 늘어야 한다
    always @(posedge clk) if (td) begin
        if (tdn != expect_tile) $fatal(1, "타일 순서 %0d, 기대 %0d", tdn, expect_tile);
        #1 $fwrite(o, "%0d T%0d %0d", n, tdn, tdn);
        for (c = 0; c < 10; c = c + 1) $fwrite(o, " %0d", $signed(acc[c*AW +: AW]));
        $fwrite(o, "\n"); expect_tile = expect_tile + 1; tiles_seen = tiles_seen + 1;
    end
    // 코어가 받은 길이 합 = 784 (타일 시작 시점의 tile_len_o 누적)
    always @(posedge clk) if (dut.state == 1) len_sum = len_sum + tlen;

    initial begin
        if (!$value$plusargs("VECTORS=%s", vp) || !$value$plusargs("ACTUAL=%s", ap)) $fatal(1, "경로 인자");
        f = $fopen(vp, "r"); o = $fopen(ap, "w"); if (!f || !o) $fatal(1, "파일 열기");
        repeat (3) @(negedge clk); rn = 1;
        while ($fscanf(f, "%d %s", n, name) == 2) begin
            for (k = 0; k < 784; k = k + 1) begin
                if ($fscanf(f, "%d", v) != 1) $fatal(1, "x"); @(negedge clk); xwe = 1; xa = k; xd = v; end
            @(negedge clk); xwe = 0;
            for (c = 0; c < 10; c = c + 1) for (k = 0; k < 784; k = k + 1) begin
                if ($fscanf(f, "%d", v) != 1) $fatal(1, "w"); @(negedge clk); wwe = 1; wn = c; wa = k; wd = v; end
            @(negedge clk); wwe = 0;
            for (c = 0; c < 10; c = c + 1) begin
                if ($fscanf(f, "%d", v) != 1) $fatal(1, "b"); @(negedge clk); bwe = 1; bn = c; bd = v; end
            @(negedge clk); bwe = 0;
            tiles_seen = 0; expect_tile = 0; len_sum = 0;
            @(negedge clk); start = 1; @(negedge clk); start = 0;
            @(negedge clk); xwe = 1; xa = 770; xd = 8'h7f; @(negedge clk); xwe = 0;   // 실행 중 쓰기 → 무시
            while (!done) begin @(negedge clk); if (err) $fatal(1, "%s: error_o", name); end
            if (tiles_seen != 13 || len_sum != 784) $fatal(1, "%s: 타일 %0d개, 길이 합 %0d", name, tiles_seen, len_sum);
            $fwrite(o, "%0d S", n);
            for (c = 0; c < 10; c = c + 1) $fwrite(o, " %0d", $signed(scores[c*AW +: AW]));
            $fwrite(o, " %0d\n", cls);
            cases = cases + 1;
        end
        $display("RTL_SIM_PASS TILED784 cases=%0d tiles=13 len_sum=784", cases);
        $fclose(o); $finish;
    end
    initial begin #200000000; $fatal(1, "시간 초과"); end
endmodule

#!/usr/bin/env python3
"""17장 기준 모델: 784원소(28×28) × 10클래스를 64원소 타일 13개로 나눠 누산, Bias는 끝에 한 번.

입력 스케일은 13장과 같다: 화소 p(0~255) → min(127, (p+1)>>1).
점수는 host_tools의 dot_tiled_64, 타일 누계(부분합)는 같은 타일 경계로 따로 계산해 RTL 타일 순서를 확인한다.
"""
from __future__ import annotations
import argparse
import random
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT / "ch13" / "reference"), str(ROOT / "host_tools")]
from mnist8x8_reference import draw_digit               # noqa: E402
from p3_integer_reference import argmax_lowest, dot_signed, dot_tiled_64  # noqa: E402

N, TILE = 784, 64
TILE_LENS = [TILE] * 12 + [N - TILE * 12]               # 64 × 12 + 16


def x784(img):
    return [min(127, (p + 1) >> 1) for row in img for p in row]


def scores(x, w, b):
    return [dot_tiled_64(x, row) + bias for row, bias in zip(w, b)]


def tile_prefix(x, w):
    """타일 t까지의 누계 (Bias 없음), t = 0~12."""
    out, acc, start = [], [0] * len(w), 0
    for ln in TILE_LENS:
        acc = [a + dot_signed(x[start:start + ln], row[start:start + ln]) for a, row in zip(acc, w)]
        out.append(acc)
        start += ln
    return out


def cases():
    tmpl = [x784(draw_digit(d)) for d in range(10)]
    tw, tb = tmpl, [-(sum(v * v for v in t) // 2) for t in tmpl]       # 최근접 형판 (학습 아님)
    out = [(f"template_{d}", tmpl[d], tw, tb) for d in (0, 4, 9)]
    rng = random.Random(1717)
    for k in range(2):
        w = [[rng.randint(-128, 127) for _ in range(N)] for _ in range(10)]
        out.append((f"random_{k}", tmpl[k + 2], w, [rng.randint(-(1 << 20), 1 << 20) for _ in range(10)]))
    # 마지막 타일(768~783) 16원소만 0이 아니고, 그 구간 가중치만 클래스 6을 고르게 한다
    x = [0] * 768 + [100] * 16
    w = [[0] * N for _ in range(10)]
    for c in range(10):
        for i in range(768, N):
            w[c][i] = 5 if c == 6 else -3
    out.append(("last_tile_only", x, w, [0] * 10))
    return out


def selftest():
    cs = cases()
    assert sum(TILE_LENS) == N and len(TILE_LENS) == 13
    for name, x, w, b in cs:
        s = scores(x, w, b)
        assert s == [dot_signed(x, row) + bb for row, bb in zip(w, b)], name
        assert tile_prefix(x, w)[-1] == [v - bb for v, bb in zip(s, b)], name
        assert all(-(1 << 31) <= v < (1 << 31) for v in s), name
    cls = {n: argmax_lowest(scores(x, w, b)) for n, x, w, b in cs}
    assert cls["template_0"] == 0 and cls["template_4"] == 4 and cls["template_9"] == 9 and cls["last_tile_only"] == 6
    print(f"PYTHON_TILED784_SELF_TEST_PASS cases={len(cs)} tiles=13 lens=64x12+16")
    return cs


def main():
    ap = argparse.ArgumentParser(); ap.add_argument("--out"); ap.add_argument("--actual")
    a = ap.parse_args()
    cs = selftest()
    if a.out and not a.actual:
        with open(Path(a.out) / "vectors.txt", "w") as f:
            for n, (name, x, w, b) in enumerate(cs):
                f.write(f"{n} {name}\n" + " ".join(map(str, x)) + "\n")
                f.writelines(" ".join(map(str, r)) + "\n" for r in w)
                f.write(" ".join(map(str, b)) + "\n")
    if a.actual:
        got = {}
        for ln in open(a.actual):
            p = ln.split()
            try:
                got.setdefault(int(p[0]), {})[p[1]] = [int(v) for v in p[2:]]
            except ValueError:                                  # RTL이 x·z(정의되지 않은 값)를 냈음
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL 정수가 아닌 RTL 출력: {ln.strip()[:80]}")
        for n, (name, x, w, b) in enumerate(cs):
            g = got.get(n, {})
            s = scores(x, w, b)
            for t, acc in enumerate(tile_prefix(x, w)):
                if g.get(f"T{t}") != [t] + acc:
                    raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL case={name} tile={t} rtl={g.get(f'T{t}')}")
            if g.get("S") != s + [argmax_lowest(s)]:
                raise SystemExit(f"PYTHON_RTL_COMPARE_FAIL case={name} scores rtl={g.get('S')}")
        print(f"PYTHON_RTL_COMPARE_PASS cases={len(cs)} tile_prefix={13 * 10 * len(cs)} scores={10 * len(cs)}")


if __name__ == "__main__":
    main()

07코드 읽기 포인트

위치내용
타일 주소코어는 타일 안 번호 0~63만 압니다. 상위가 {tile_o, 6'd0} + idx로 전체 주소를 만들어 입력·가중치 RAM에 줍니다.
마지막 16원소tile_len_o는 타일 12에서 16입니다. 코어 len_i가 16이면 4PE가 16개만 누산하고 끝나므로 784를 넘는 주소를 읽지 않습니다.
코어 밖 누산타일마다 코어는 내적을 처음부터 다시 계산하고, 상위 acc_q가 더해 갑니다. 코어 누산기를 이어 쓰지 않으므로 15장 코어를 바꾸지 않습니다.
Bias 한 번BIAS 상태에서 점수 = 누계 + Bias를 한 번 만듭니다. 타일마다 Bias를 더하면 13배가 되며, TB가 이 실수를 타일 누계 비교로 잡습니다.
누산 폭INT8 기준 |내적| ≤ 784×127×128 ≈ 1.27×107 < 224이므로 32비트로 충분합니다. INT16과 40비트 Bias의 최악 범위는 18장에서 다룹니다.
